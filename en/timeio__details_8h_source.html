358 inline static const prefix_tree<char, zone_ref>& s_timezone_tree =
359 *new prefix_tree<char, zone_ref>([]()
360 {
361 prefix_tree<char, zone_ref> res;
362
363 // Registered ahead of the tzdb walk so that it is present even when the walk
364 // throws and leaves the rest of the trie empty: the token the put side writes
365 // for a nameless zone has to parse back regardless of the database. Its text is
366 // empty on purpose -- see s_unknown_zone for why that is not the same as absent.
367 res.add(s_unknown_zone.begin(), s_unknown_zone.end(),
368 zone_ref{.text = std::string{}, .is_name = false, .is_abbrev = true});
369
370 try
371 {
372 const auto& tzdb = std::chrono::get_tzdb();
373
374 // Classify every key before touching the trie. Two keys can name the same
375 // string from different sources -- CET, EET, EST, GMT, HST, MST, UTC and WET
376 // are each both a link name and an abbreviation -- and going through a map
377 // merges those into one entry with both flags set, instead of calling add()
378 // twice on one key and having it throw on the mismatched values.
379 std::map<std::string, zone_ref> entries;
380
381 // Every abbreviation each zone passes through.
382 {
383 using namespace std::chrono;
384 const sys_seconds projected =
385 time_point_cast<seconds>(system_clock::now()) + years{10};
386 const sys_seconds floor{sys_days{2038y / January / 1}};
387 const sys_seconds horizon = (projected > floor) ? projected : floor;
388
389 // Not sys_seconds::min(): libstdc++ starts every zone's first period at
390 // year::min(), and an earlier instant on a period with rules throws
391 // (CST6CDT and co. since tzdata 2026d). Nothing is lost by starting here.
392 const sys_seconds start{sys_days{year::min() / January / 1}};
393
394 for (const auto& zone : tzdb.zones)
395 {
396 try
397 {
398 sys_seconds t = start;
399 while (t < horizon)
400 {
401 const sys_info info = zone.get_info(t);
402 if (!info.abbrev.empty()) entries[info.abbrev].is_abbrev = true;
403 if (info.end <= t) break;
404 t = time_point_cast<seconds>(info.end);
405 }
406 }
407 catch (...) {} // NOLINT(bugprone-empty-catch)
408 }
409 }
410
411 // Every identifier locate_zone() accepts. Links belong here just as much as
412 // zones do: a link is a full name, merely not the canonical one, and nothing
413 // downstream needs the canonical form -- locate_zone normalizes on its own.
414 for (const auto& zone : tzdb.zones)
415 entries[std::string{zone.name()}].is_name = true;
416 for (const auto& link : tzdb.links)
417 entries[std::string{link.name()}].is_name = true;
418
419 // The unknown-zone token has to parse back to "no zone at all", so it cannot
420 // share a key with a real one. That it does not is a documented invariant of
421 // s_unknown_zone; if some future tzdata breaks it, refuse to answer rather
422 // than quietly get that one entry wrong. Thrown here -- before a single tzdb
423 // entry reaches the trie, and inside the catch below -- so the trie ends up
424 // holding only the token and every real %Z reports a catchable stream_error.
425 // Letting it escape the static initializer instead would call std::terminate
426 // in every program that includes this header, %Z user or not.
427 if (entries.contains(std::string{s_unknown_zone}))
428 throw io_error(
429 "timeio: the tz database has a zone or abbreviation named '"
430 + std::string{s_unknown_zone} + "', colliding with the unknown-zone token");
431
432 for (const auto& [key, ident] : entries)
433 res.add(key.begin(), key.end(),
434 zone_ref{.text = key, .is_name = ident.is_name, .is_abbrev = ident.is_abbrev});
435 }
436 catch (...) // NOLINT(bugprone-empty-catch)
437 {
438 // tz database unavailable, malformed, or colliding with the unknown-zone
439 // token at static-init time: degrade to a trie holding nothing but that
440 // token, instead of letting the exception escape this static initializer
441 // (which would call std::terminate). %Z then simply fails to match and
442 // reports a catchable stream_error at parse time, matching the defensive
443 // behaviour of time_zone_parse_helper.
444 }
445 return res;
446 }());
447 };

1225 static std::optional<std::string> normalize_time_format(const std::string& fmt)
1226 {
1227 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
1228
1229 std::string out;
1230 out.reserve(fmt.size());
1231
1232 for (std::size_t i = 0; i < fmt.size(); ++i)
1233 {
1234 if (fmt[i] != '%') { out += fmt[i]; continue; }
1235
1236 std::size_t j = i + 1;
1237 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
1238 || fmt[j] == '^' || fmt[j] == '#'))
1239 ++j;
1240 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
1241
1242 char modifier = '\0';
1243 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
1244
1245 if (j >= fmt.size()) { out.append(fmt, i, std::string::npos); break; }
1246
1247 char spec = fmt[j];
1248 if (spec == 'l') spec = 'I';
1249 else if (spec == 'k') spec = 'H';
1250 else if (spec == 'P') spec = 'p';
1251
1252 if (supported.find(spec) == std::string_view::npos) return std::nullopt;
1253
1254 out += '%';
1255 if (modifier != '\0') out += modifier;
1256 out += spec;
1257 i = j;
1258 }
1259
1260 return out;
1261 }

1366 static std::vector<era_entry> parse_glibc_era_entries()
1367 {
1368 struct era_wire_header
1369 {
1370 int32_t direction_marker;
1371 int32_t offset;
1372 int32_t from_year;
1373 int32_t from_month;
1374 int32_t from_day;
1375 int32_t to_year;
1376 int32_t to_month;
1377 int32_t to_day;
1378 };
1379 static_assert(sizeof(era_wire_header) == 8 * sizeof(int32_t));
1380
1381 std::vector<era_entry> items;
1382
1383 const auto record_count = static_cast<int32_t>(
1384 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
1385 if (record_count <= 0)
1386 return items;
1387
1388 items.reserve(static_cast<std::size_t>(record_count));
1389 const char* cursor = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
1390 for (int32_t index = 0; index < record_count; ++index)
1391 {
1392 const char* const record = cursor;
1393 era_wire_header header{};
1394 std::memcpy(&header, cursor, sizeof header);
1395 cursor += sizeof header;
1396
1397 era_entry value{};
1398 constexpr int32_t epoch = 1900;
1399 constexpr int32_t last_relative_year = std::numeric_limits<int32_t>::max() - epoch;
1400 value.from_year = header.from_year > last_relative_year
1401 ? std::numeric_limits<int32_t>::max()
1402 : header.from_year + epoch;
1403 value.from_month = static_cast<uint8_t>(header.from_month + 1);
1404 value.from_day = static_cast<uint8_t>(header.from_day);
1405
1406 const bool open_ended = header.to_year > last_relative_year;
1407 value.to_year = open_ended
1408 ? std::numeric_limits<int32_t>::max()
1409 : header.to_year + epoch;
1410 value.to_month = open_ended ? uint8_t{12}
1411 : static_cast<uint8_t>(header.to_month + 1);
1412 value.to_day = open_ended ? uint8_t{31}
1413 : static_cast<uint8_t>(header.to_day);
1414 value.offset = header.offset;
1415
1416 const int8_t marked_direction = header.direction_marker == static_cast<int32_t>('+')
1417 ? int8_t{1} : int8_t{-1};
1418 const bool runs_forward = TimeioHelper::era_small_or_equal(
1419 value.from_year, value.from_month, value.from_day,
1420 value.to_year, value.to_month, value.to_day);
1421 value.direction = runs_forward
1422 ? marked_direction : static_cast<int8_t>(-marked_direction);
1423
1424 value.name = cursor;
1425 cursor += std::strlen(cursor) + 1;
1426 value.format = cursor;
1427 cursor += std::strlen(cursor) + 1;
1428
1429 constexpr std::size_t wire_alignment = sizeof(int32_t);
1430 const auto narrow_size = static_cast<std::size_t>(cursor - record);
1431 cursor += (wire_alignment - (narrow_size % wire_alignment)) % wire_alignment;
1432
1433 const auto* wide_name = reinterpret_cast<const wchar_t*>(cursor);
1434 cursor += (std::wcslen(wide_name) + 1) * sizeof(wchar_t);
1435 const auto* wide_format = reinterpret_cast<const wchar_t*>(cursor);
1436 cursor += (std::wcslen(wide_format) + 1) * sizeof(wchar_t);
1437
1438 items.push_back(std::move(value));
1439 }
1440 return items;
1441 }
