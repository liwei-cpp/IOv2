350 inline static const prefix_tree<char, zone_ref>& s_timezone_tree =
351 *new prefix_tree<char, zone_ref>([]()
352 {
353 prefix_tree<char, zone_ref> res;
354
355 // Registered ahead of the tzdb walk so that it is present even when the walk
356 // throws and leaves the rest of the trie empty: the token the put side writes
357 // for a nameless zone has to parse back regardless of the database. Its text is
358 // empty on purpose -- see s_unknown_zone for why that is not the same as absent.
359 res.add(s_unknown_zone.begin(), s_unknown_zone.end(),
360 zone_ref{.text = std::string{}, .is_name = false, .is_abbrev = true});
361
362 try
363 {
364 const auto& tzdb = std::chrono::get_tzdb();
365
366 // Classify every key before touching the trie. Two keys can name the same
367 // string from different sources -- CET, EET, EST, GMT, HST, MST, UTC and WET
368 // are each both a link name and an abbreviation -- and going through a map
369 // merges those into one entry with both flags set, instead of calling add()
370 // twice on one key and having it throw on the mismatched values.
371 std::map<std::string, zone_ref> entries;
372
373 // Every abbreviation each zone passes through.
374 {
375 using namespace std::chrono;
376 const sys_seconds projected =
377 time_point_cast<seconds>(system_clock::now()) + years{10};
378 const sys_seconds floor{sys_days{2038y / January / 1}};
379 const sys_seconds horizon = (projected > floor) ? projected : floor;
380
381 for (const auto& zone : tzdb.zones)
382 {
383 sys_seconds t = sys_seconds::min();
384 while (t < horizon)
385 {
386 const sys_info info = zone.get_info(t);
387 if (!info.abbrev.empty()) entries[info.abbrev].is_abbrev = true;
388 if (info.end <= t) break;
389 t = time_point_cast<seconds>(info.end);
390 }
391 }
392 }
393
394 // Every identifier locate_zone() accepts. Links belong here just as much as
395 // zones do: a link is a full name, merely not the canonical one, and nothing
396 // downstream needs the canonical form -- locate_zone normalizes on its own.
397 for (const auto& zone : tzdb.zones)
398 entries[std::string{zone.name()}].is_name = true;
399 for (const auto& link : tzdb.links)
400 entries[std::string{link.name()}].is_name = true;
401
402 // The unknown-zone token has to parse back to "no zone at all", so it cannot
403 // share a key with a real one. That it does not is a documented invariant of
404 // s_unknown_zone; if some future tzdata breaks it, refuse to answer rather
405 // than quietly get that one entry wrong. Thrown here -- before a single tzdb
406 // entry reaches the trie, and inside the catch below -- so the trie ends up
407 // holding only the token and every real %Z reports a catchable stream_error.
408 // Letting it escape the static initializer instead would call std::terminate
409 // in every program that includes this header, %Z user or not.
410 if (entries.contains(std::string{s_unknown_zone}))
411 throw std::runtime_error(
412 "timeio: the tz database has a zone or abbreviation named '"
413 + std::string{s_unknown_zone} + "', colliding with the unknown-zone token");
414
415 for (const auto& [key, ident] : entries)
416 res.add(key.begin(), key.end(),
417 zone_ref{.text = key, .is_name = ident.is_name, .is_abbrev = ident.is_abbrev});
418 }
419 catch (...) // NOLINT(bugprone-empty-catch)
420 {
421 // tz database unavailable, malformed, or colliding with the unknown-zone
422 // token at static-init time: degrade to a trie holding nothing but that
423 // token, instead of letting the exception escape this static initializer
424 // (which would call std::terminate). %Z then simply fails to match and
425 // reports a catchable stream_error at parse time, matching the defensive
426 // behaviour of time_zone_parse_helper.
427 }
428 return res;
429 }());

1208 static std::optional<std::string> normalize_time_format(const std::string& fmt)
1209 {
1210 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
1211
1212 std::string out;
1213 out.reserve(fmt.size());
1214
1215 for (std::size_t i = 0; i < fmt.size(); ++i)
1216 {
1217 if (fmt[i] != '%') { out += fmt[i]; continue; }
1218
1219 std::size_t j = i + 1;
1220 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
1221 || fmt[j] == '^' || fmt[j] == '#'))
1222 ++j;
1223 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
1224
1225 char modifier = '\0';
1226 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
1227
1228 if (j >= fmt.size()) { out.append(fmt, i, std::string::npos); break; }
1229
1230 char spec = fmt[j];
1231 if (spec == 'l') spec = 'I';
1232 else if (spec == 'k') spec = 'H';
1233 else if (spec == 'P') spec = 'p';
1234
1235 if (supported.find(spec) == std::string_view::npos) return std::nullopt;
1236
1237 out += '%';
1238 if (modifier != '\0') out += modifier;
1239 out += spec;
1240 i = j;
1241 }
1242
1243 return out;
1244 }

1349 static std::vector<era_entry> parse_glibc_era_entries()
1350 {
1351 struct era_wire_header
1352 {
1353 int32_t direction_marker;
1354 int32_t offset;
1355 int32_t from_year;
1356 int32_t from_month;
1357 int32_t from_day;
1358 int32_t to_year;
1359 int32_t to_month;
1360 int32_t to_day;
1361 };
1362 static_assert(sizeof(era_wire_header) == 8 * sizeof(int32_t));
1363
1364 std::vector<era_entry> items;
1365
1366 const auto record_count = static_cast<int32_t>(
1367 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
1368 if (record_count <= 0)
1369 return items;
1370
1371 items.reserve(static_cast<std::size_t>(record_count));
1372 const char* cursor = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
1373 for (int32_t index = 0; index < record_count; ++index)
1374 {
1375 const char* const record = cursor;
1376 era_wire_header header{};
1377 std::memcpy(&header, cursor, sizeof header);
1378 cursor += sizeof header;
1379
1380 era_entry value{};
1381 constexpr int32_t epoch = 1900;
1382 constexpr int32_t last_relative_year = std::numeric_limits<int32_t>::max() - epoch;
1383 value.from_year = header.from_year > last_relative_year
1384 ? std::numeric_limits<int32_t>::max()
1385 : header.from_year + epoch;
1386 value.from_month = static_cast<uint8_t>(header.from_month + 1);
1387 value.from_day = static_cast<uint8_t>(header.from_day);
1388
1389 const bool open_ended = header.to_year > last_relative_year;
1390 value.to_year = open_ended
1391 ? std::numeric_limits<int32_t>::max()
1392 : header.to_year + epoch;
1393 value.to_month = open_ended ? uint8_t{12}
1394 : static_cast<uint8_t>(header.to_month + 1);
1395 value.to_day = open_ended ? uint8_t{31}
1396 : static_cast<uint8_t>(header.to_day);
1397 value.offset = header.offset;
1398
1399 const int8_t marked_direction = header.direction_marker == static_cast<int32_t>('+')
1400 ? int8_t{1} : int8_t{-1};
1401 const bool runs_forward = TimeioHelper::era_small_or_equal(
1402 value.from_year, value.from_month, value.from_day,
1403 value.to_year, value.to_month, value.to_day);
1404 value.direction = runs_forward
1405 ? marked_direction : static_cast<int8_t>(-marked_direction);
1406
1407 value.name = cursor;
1408 cursor += std::strlen(cursor) + 1;
1409 value.format = cursor;
1410 cursor += std::strlen(cursor) + 1;
1411
1412 constexpr std::size_t wire_alignment = sizeof(int32_t);
1413 const auto narrow_size = static_cast<std::size_t>(cursor - record);
1414 cursor += (wire_alignment - (narrow_size % wire_alignment)) % wire_alignment;
1415
1416 const auto* wide_name = reinterpret_cast<const wchar_t*>(cursor);
1417 cursor += (std::wcslen(wide_name) + 1) * sizeof(wchar_t);
1418 const auto* wide_format = reinterpret_cast<const wchar_t*>(cursor);
1419 cursor += (std::wcslen(wide_format) + 1) * sizeof(wchar_t);
1420
1421 items.push_back(std::move(value));
1422 }
1423 return items;
1424 }
