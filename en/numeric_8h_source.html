128 template <shared_ptr_to<numeric_conf<CharT>> TConfPtr,
129 shared_ptr_to<ctype<CharT>> TCtypePtr>
130 numeric(TConfPtr p_obj, TCtypePtr p_ctype) : m_ctype(p_ctype)
131 {
132 if (!p_obj || !p_ctype) throw stream_error("shared_ptr is empty");
133 m_decimal_point = p_obj->decimal_point();
134 m_thousands_sep = p_obj->thousands_sep();
135 m_true_name = p_obj->truename();
136 m_false_name = p_obj->falsename();
137 // grouping() is contracted to return the INTERNAL convention
138 // (1–255 = group size, 0 = stop, last element implicitly repeats).
139 // numeric_conf and any user-derived override are both expected to
140 // satisfy this contract — POSIX-style normalisation is done at the
141 // POSIX boundary in numeric_conf, not here.
142 m_grouping = p_obj->grouping();
143
144 // Each alphabet is widened whole and every other atom is widened by
145 // name, so nothing here depends on a character sitting at a
146 // remembered offset. string_view carries the length, with no trailing
147 // '\0' counted; static_assert keeps literal and destination in step.
148 constexpr std::string_view hex_lower = "0123456789abcdef";
149 constexpr std::string_view hex_upper = "0123456789ABCDEF";
150 constexpr std::string_view upper_hex_letters = "ABCDEF";
151 static_assert(hex_lower.size() == std::tuple_size_v<decltype(out_atoms::lower)>);
152 static_assert(hex_upper.size() == std::tuple_size_v<decltype(out_atoms::upper)>);
153
154 m_ctype->widen_seq(hex_lower.data(), hex_lower.data() + hex_lower.size(),
155 m_out.lower.data());
156 m_ctype->widen_seq(hex_upper.data(), hex_upper.data() + hex_upper.size(),
157 m_out.upper.data());
158 m_out.minus = m_ctype->widen('-');
159 m_out.plus = m_ctype->widen('+');
160 m_out.lower_x = m_ctype->widen('x');
161 m_out.upper_x = m_ctype->widen('X');
162
163 // The parser reads a digit's value off the position it matched at, so
164 // the two letter cases are laid end to end after the decimal digits:
165 // '0'..'9', then 'a'..'f', then 'A'..'F' at s_upper_hex.
166 m_ctype->widen_seq(hex_lower.data(), hex_lower.data() + hex_lower.size(),
167 m_in.digits.data());
168 m_ctype->widen_seq(upper_hex_letters.data(),
169 upper_hex_letters.data() + upper_hex_letters.size(),
170 m_in.digits.data() + s_upper_hex);
171 m_in.minus = m_out.minus;
172 m_in.plus = m_out.plus;
173 m_in.lower_x = m_out.lower_x;
174 m_in.upper_x = m_out.upper_x;
175 m_in.lower_e = m_ctype->widen('e');
176 m_in.upper_e = m_ctype->widen('E');
177
178 if constexpr (std::is_same_v<CharT, wchar_t> ||
179 std::is_same_v<CharT, char32_t>)
180 {
181 // Distinctness is a property of the whole set a stage works with,
182 // so each side is checked as one run: the marks followed by every
183 // glyph that side can emit or match.
184 std::array<CharT, 26> in_view{m_in.minus, m_in.plus,
185 m_in.lower_x, m_in.upper_x};
186 std::copy(m_in.digits.begin(), m_in.digits.end(), in_view.begin() + 4);
187 assert(atoms_pairwise_distinct(in_view.data(), in_view.size()));
188
189 std::array<CharT, 26> out_view{m_out.minus, m_out.plus,
190 m_out.lower_x, m_out.upper_x};
191 std::copy(m_out.lower.begin(), m_out.lower.end(), out_view.begin() + 4);
192 std::copy(m_out.upper.begin() + s_hex_letters, m_out.upper.end(),
193 out_view.begin() + 20);
194 assert(atoms_pairwise_distinct(out_view.data(), out_view.size()));
195 }

304 template <typename TIter>
305 TIter put(TIter s, ios_base<char_type>& io, bool v) const
306 {
307 const auto flags = io.flags();
308 if ((flags & ios_defs::boolalpha) == 0)
309 {
310 return insert_int(s, io, static_cast<long>(v));
311 }
312
313 const auto& name = v ? m_true_name : m_false_name;
314 std::size_t len = name.size();
315
316 const auto w = io.width();
317 io.width(0);
318 if (w > static_cast<decltype(w)>(len))
319 {
320 const auto plen = w - len;
321 if (plen > ios_defs::max_pad_count)
322 throw stream_error("numeric put fail: fill count exceeds max_pad_count");
323
324 // No fill_alters_reading check here: the padded content is truename() /
325 // falsename(), not a number, so no fill character can make the field read
326 // as a different value. Rejecting digits here would only break the ordinary
327 // setfill('0') << setw(n) << boolalpha spelling.
328 if ((flags & ios_defs::adjustfield) == ios_defs::left)
329 {
330 s = std::copy(name.begin(), name.end(), s);
331 s = std::fill_n(s, plen, io.fill());
332 }
333 else
334 {
335 s = std::fill_n(s, plen, io.fill());
336 s = std::copy(name.begin(), name.end(), s);
337 }
338 return s;
339 }
340
341 return std::copy(name.begin(), name.end(), s);
342 }

486 template <typename TIter, std::sentinel_for<TIter> TSent>
487 TIter get(TIter beg, TSent end, ios_base<char_type>& io, bool& v) const
488 {
489 bool success = true;
490
491 if (!(io.flags() & ios_defs::boolalpha))
492 {
493 // Numeric form: read the field as a `long` and accept only 0/1.
494 long l = -1;
495 std::tie(success, beg) = extract_int(beg, end, io, l);
496
497 if (l == 0 || l == 1) v = bool(l);
498 else
499 {
500 // LWG 23 resolved that a field which does not fit the target
501 // still stores a value rather than leaving it untouched: the
502 // most positive representable one, which for bool is `true`.
503 v = true;
504 success = false;
505 }
506 }
507 else
508 {
509 // `boolalpha` form: match the input against the two locale names.
510 //
511 // Both names race for the input. A name stays in the running only
512 // while every character read so far is a prefix of it, and only
513 // while it still has a character left to offer; extraction stops
514 // the moment no name can take the next character, which is what
515 // keeps the facet from consuming more than it needs.
516 const std::basic_string<CharT>& fname = m_false_name;
517 const std::basic_string<CharT>& tname = m_true_name;
518
519 bool f_live = true;
520 bool t_live = true;
521 std::size_t n = 0;
522
523 while (beg != end)
524 {
525 const bool f_open = f_live && n < fname.size();
526 const bool t_open = t_live && n < tname.size();
527 if (!f_open && !t_open)
528 break;
529
530 const CharT c = *beg;
531 const bool f_takes = f_open && fname[n] == c;
532 const bool t_takes = t_open && tname[n] == c;
533 if (!f_takes && !t_takes)
534 break;
535
536 f_live = f_takes;
537 t_live = t_takes;
538 ++n;
539 ++beg;
540 }
541
542 // A name won only if it is still live and came out exactly
543 // exhausted; `n != 0` keeps an empty name from matching nothing.
544 const bool got_false = f_live && n == fname.size() && n != 0;
545 const bool got_true = t_live && n == tname.size() && n != 0;
546
547 // Exactly one name may win. Both winning means the two are
548 // indistinguishable on this input; neither winning means nothing
549 // matched. The standard asks for `false` plus a failure either way.
550 v = got_true && !got_false;
551 if (got_false == got_true)
552 success = false;
553 }
554
555 if (!success) throw stream_error("numeric::get fail: parse boolean fail");
556 return beg;
557 }

763 template <typename TIter, typename TValue>
764 TIter insert_float(TIter s, ios_base<char_type>& io, TValue v, char mod) const
765 {
766 // precision() is now a bounded uint8_t (0-255); there is no negative /
767 // out-of-range sentinel to normalise here.
768 const std::streamsize prec = io.precision();
769 const int max_digits = std::numeric_limits<TValue>::digits10;
770
771 // Consume the field width up front, before any allocation or conversion
772 // that can throw: width() is one-shot and a stale value must not survive
773 // onto the stream if we leave by an exception. Used for padding below.
774 const std::size_t w = io.width();
775 io.width(0);
776
777 // Build the printf conversion spec, then let the C library render the
778 // value into a narrow buffer; everything after this is localisation.
779 std::size_t len = 0;
780 std::array<char, 16> fbuf{};
781 format_float(io.flags(), fbuf.data(), mod);
782
783 const ios_defs::fmtflags fltfield = io.flags() & ios_defs::floatfield;
784
785 // First guess at the buffer. Outside `fixed`, three times the decimal
786 // digit count leaves room for sign, point, exponent and slack; `fixed`
787 // is the notation that can run long, so it gets the full integer part.
788 std::size_t cs_size = static_cast<std::size_t>(max_digits) * 3 + 32;
789 if (fltfield == ios_defs::fixed)
790 cs_size = static_cast<std::size_t>(std::numeric_limits<TValue>::max_exponent10) + static_cast<std::size_t>(prec) + 32;
791
792 // Cap initial allocation to a reasonable size (e.g., 2048) to avoid huge initial pressure.
793 if (cs_size > 2048) cs_size = 2048;
794
795 std::vector<char> vec_cs(cs_size);
796
797 {
798 clocale_wrapper inter_locale("C");
799 clocale_user guard(inter_locale);
800
801 auto do_snprintf = [&](char* buf, std::size_t size) {
802 if (fltfield == (ios_defs::fixed | ios_defs::scientific))
803 return snprintf(buf, size, fbuf.data(), v);
804 else
805 return snprintf(buf, size, fbuf.data(), static_cast<int>(prec), v);
806 };
807
808 // snprintf returns int. Trap negative (encoding error) before
809 // promoting to size_t — otherwise -1 becomes SIZE_MAX and
810 // bypasses every downstream bound check.
811 int n = do_snprintf(vec_cs.data(), cs_size);
812 if (n < 0)
813 throw stream_error("numeric::put fail: floating-point conversion failed");
814 len = static_cast<std::size_t>(n);
815
816 // If buffer was too small, snprintf returns required length.
817 if (len >= cs_size)
818 {
819 cs_size = len + 1;
820 vec_cs.resize(cs_size);
821 n = do_snprintf(vec_cs.data(), cs_size);
822 if (n < 0)
823 throw stream_error("numeric::put fail: floating-point conversion failed");
824 len = static_cast<std::size_t>(n);
825 }
826 }
827
828 // len == 0 is also treated as failure: every well-formed numeric
829 // print (including 0, "inf", "nan") emits at least one character,
830 // and the downstream path assumes vec_ws(len) has at least one
831 // element before dereferencing vec_ws.data().
832 if (len == 0 || len >= cs_size)
833 throw stream_error("numeric::put fail: floating-point conversion failed");
834
835 const char* cs = vec_cs.data();
836
837 // Widen into char_type, then apply the two things the C locale could
838 // not know about: this facet's decimal point, and its grouping.
839 std::vector<CharT> vec_ws(len);
840 CharT* ws = vec_ws.data();
841 m_ctype->widen_seq(cs, cs + len, ws);
842
843 const char* p = std::find(cs, cs + len, '.');
844 CharT* wp = nullptr;
845 if (p != cs + len)
846 {
847 wp = ws + (p - cs);
848 *wp = m_decimal_point;
849 }
850
851 // Only a real number gets grouped -- "inf", "nan" and exponent-only forms
852 // like 2e20 must come through untouched. A decimal point settles it
853 // outright; failing that, text too short to be one of those words is a
854 // number, and so is text whose second and third characters are digits.
855 const auto is_digit = [](char c) { return c >= '0' && c <= '9'; };
856 const bool numeric_text =
857 wp || len < 3u || (is_digit(cs[1]) && is_digit(cs[2]));
858
859 if (!m_grouping.empty() && numeric_text)
860 {
861 // Worst case one separator per digit, so twice the length is ample.
862 std::vector<CharT> vec_ws2(len * 2);
863 CharT* ws2 = vec_ws2.data();
864
865 // A leading sign is not part of the number being grouped. Copy it
866 // across on its own and hand the grouper only what follows, so the
867 // digit count it works from never has to be unwound afterwards.
868 const std::size_t sign_len = (cs[0] == '-' || cs[0] == '+') ? 1u : 0u;
869 if (sign_len != 0)
870 ws2[0] = ws[0];
871
872 // LWG 282: grouping applies to the integer part only, so the split
873 // happens here and the fraction is handed over to be copied through.
874 const std::size_t digit_len = len - sign_len;
875 const std::size_t int_len = wp ? static_cast<std::size_t>(wp - (ws + sign_len))
876 : digit_len;
877 len = sign_len + group_float(m_grouping, m_thousands_sep, ws2 + sign_len,
878 ws + sign_len, int_len, digit_len - int_len);
879 std::swap(vec_ws, vec_ws2);
880 ws = vec_ws.data();
881 }
882
883 // Pad.
884 if (std::cmp_greater(w, len))
885 {
886 if (w - len > ios_defs::max_pad_count)
887 throw stream_error("numeric put fail: fill count exceeds max_pad_count");
888
889 std::vector<CharT> vec_ws3(w);
890 CharT* ws3 = vec_ws3.data();
891 bool startSign = (ws[0] == m_out.minus) || (ws[0] == m_out.plus);
892 bool start0x = (ws[0] == m_out.lower[0]) && (len > 1u) &&
893 ((ws[1] == m_out.lower_x) || (ws[1] == m_out.upper_x));
894 const ios_defs::fmtflags adjust = io.flags() & ios_defs::adjustfield;
895 pad(io.fill(), w, adjust, io.flags() & ios_defs::basefield,
896 ws3, ws, len, startSign, start0x);
897
898 std::swap(vec_ws, vec_ws3);
899 ws = vec_ws.data();
900 }
901
902 // Nothing is left to decide; hand the finished field to the caller.
903 return std::copy(ws, ws + len, s);
904 }

937 template <typename TIter, typename TValue>
938 TIter insert_int(TIter s, ios_base<char_type>& io, TValue v) const
939 {
940 using unsigned_type = std::make_unsigned_t<TValue>;
941
942 const ios_defs::fmtflags flags = io.flags();
943 const std::size_t w = io.width();
944 io.width(0);
945
946 const ios_defs::fmtflags basefield = flags & ios_defs::basefield;
947 const bool dec = (basefield != ios_defs::oct && basefield != ios_defs::hex);
948
949 auto u = static_cast<unsigned_type>(v);
950 if constexpr (std::is_signed_v<TValue>)
951 {
952 if (dec && v < 0)
953 u = -u;
954 }
955
956 // Even base two would need no more than one character per value bit. Two
957 // fixed buffers therefore cover both raw and maximally grouped output while
958 // keeping every ordinary integer conversion off the heap.
959 constexpr std::size_t digit_capacity = std::numeric_limits<unsigned_type>::digits;
960 std::array<char_type, digit_capacity> raw_digits{};
961 char_type* const raw_end = raw_digits.data() + raw_digits.size();
962 char_type* raw_begin = raw_end;
963
964 if (dec)
965 {
966 do // NOLINT(cppcoreguidelines-avoid-do-while)
967 {
968 *--raw_begin = m_out.lower[u % 10];
969 u /= 10;
970 } while (u != 0);
971 }
972 else
973 {
974 const unsigned shift = (basefield == ios_defs::oct) ? 3u : 4u;
975 const unsigned_type mask = (unsigned_type{1} << shift) - 1;
976 const auto& alphabet = (basefield == ios_defs::hex && (flags & ios_defs::uppercase))
977 ? m_out.upper : m_out.lower;
978 do // NOLINT(cppcoreguidelines-avoid-do-while)
979 {
980 *--raw_begin = alphabet[static_cast<std::size_t>(u & mask)];
981 u >>= shift;
982 } while (u != 0);
983 }
984
985 const char_type* digits = raw_begin;
986 auto digit_len = static_cast<std::size_t>(raw_end - raw_begin);
987 std::array<char_type, digit_capacity * 2> grouped_digits{};
988
989 if (!m_grouping.empty())
990 {
991 char_type* const grouped_end = FacetHelper::add_grouping(
992 grouped_digits.data(), m_thousands_sep, m_grouping, raw_begin, raw_end);
993 digits = grouped_digits.data();
994 digit_len = static_cast<std::size_t>(grouped_end - digits);
995 }
996
997 std::array<char_type, 2> prefix{};
998 std::size_t prefix_len = 0;
999 if (dec)
1000 {
1001 if constexpr (std::is_signed_v<TValue>)
1002 {
1003 if (v < 0)
1004 prefix[prefix_len++] = m_out.minus;
1005 else if (flags & ios_defs::showpos)
1006 prefix[prefix_len++] = m_out.plus;
1007 }
1008 }
1009 else if (bool(flags & ios_defs::showbase) && v)
1010 {
1011 prefix[prefix_len++] = m_out.lower[0];
1012 if (basefield == ios_defs::hex)
1013 prefix[prefix_len++] = (flags & ios_defs::uppercase) ? m_out.upper_x : m_out.lower_x;
1014 }
1015
1016 const std::size_t len = prefix_len + digit_len;
1017 if (w <= len)
1018 {
1019 s = std::copy_n(prefix.data(), prefix_len, s);
1020 return std::copy_n(digits, digit_len, s);
1021 }
1022
1023 const std::size_t pad_len = w - len;
1024 if (pad_len > ios_defs::max_pad_count)
1025 throw stream_error("numeric put fail: fill count exceeds max_pad_count");
1026
1027 const bool start_sign = prefix_len == 1
1028 && (prefix[0] == m_out.minus || prefix[0] == m_out.plus);
1029 const bool start_0x = prefix_len == 2
1030 && prefix[0] == m_out.lower[0]
1031 && (prefix[1] == m_out.lower_x || prefix[1] == m_out.upper_x);
1032 const ios_defs::fmtflags adjust = flags & ios_defs::adjustfield;
1033 const bool trails_value = adjust == ios_defs::left;
1034 const bool leads_digits = !trails_value
1035 && (adjust == ios_defs::internal || !(start_sign || start_0x));
1036 if (fill_alters_reading(io.fill(), basefield, leads_digits, trails_value,
1037 prefix_len != 0 && prefix[0] == m_out.minus))
1038 throw stream_error("numeric put fail: fill would change the value the field reads as");
1039
1040 if (trails_value)
1041 {
1042 s = std::copy_n(prefix.data(), prefix_len, s);
1043 s = std::copy_n(digits, digit_len, s);
1044 return std::fill_n(s, pad_len, io.fill());
1045 }
1046
1047 if (adjust == ios_defs::internal && (start_sign || start_0x))
1048 {
1049 s = std::copy_n(prefix.data(), prefix_len, s);
1050 s = std::fill_n(s, pad_len, io.fill());

1172 bool leads_digits, bool trails_value,

1229 void pad(char_type fill, std::size_t w, ios_defs::fmtflags adjust, // NOLINT(bugprone-easily-swappable-parameters)
1230 ios_defs::fmtflags basefield,
1231 char_type* new_buf, const char_type* cs, std::size_t& len,
1232 bool startSign, bool start0x) const
1233 {
1234 // Fill is about to be written, so this is where it gets vetted: a fill character
1235 // that would change the number this field reads as is rejected outright. Where
1236 // the run lands decides what is safe, and `pad_impl` below fixes that: `left`
1237 // appends it after the value; `internal`, and `right` on content that starts with
1238 // a digit, put it directly in front of the digits; `right` on content that starts
1239 // with a sign or `0x` leaves that prefix between the fill and the digits.
1240 // The check belongs here rather than at the top of the insert functions because
1241 // `fill` is sticky stream state: a stream carrying setfill('1') must keep working
1242 // for every output whose width leaves nothing to pad.
1243 const bool trails_value = (adjust == ios_defs::left);
1244 const bool leads_digits = !trails_value
1245 && (adjust == ios_defs::internal || !(startSign || start0x));
1246 if (fill_alters_reading(fill, basefield, leads_digits, trails_value,
1247 cs[0] == m_out.minus))
1248 throw stream_error("numeric put fail: fill would change the value the field reads as");
1249
1250 // The field is short of `w`, so the fill run gets placed.
1251 pad_impl(adjust, fill, new_buf, cs, w, len, startSign, start0x);
1252 len = w;
1253 }

1293 void pad_impl(ios_defs::fmtflags adjust, char_type fill,
1294 char_type* news, const char_type* olds,
1295 std::size_t newlen, std::size_t oldlen,
1296 bool startSign, bool start0x) const
1297 {
1298
1299 const std::size_t plen = newlen - oldlen;
1300
1301 if (adjust == ios_defs::left)
1302 {
1303 // Fill trails the value.
1304 std::copy_n(olds, oldlen, news);
1305 std::fill_n(news + oldlen, plen, fill);
1306 return;
1307 }
1308
1309 // Every other adjustment puts the fill ahead of the digits, and the two
1310 // that remain differ only in what gets held back at the left edge:
1311 // `internal` pins whatever announces the number -- a sign, or a `0x` /
1312 // `0X` base marker -- and lets the fill slot in behind it, while `right`
1313 // pins nothing at all. A pinned run of length zero makes that one case,
1314 // so both are served by a single three-step write.
1315 std::size_t pinned = 0;
1316 if (adjust == ios_defs::internal)
1317 {
1318 // At most two characters, so they go across by hand: handing a
1319 // length of 0-2 to a range algorithm costs a memmove call that
1320 // dwarfs the copy itself, and `right` would pay it for nothing.
1321 if (startSign)
1322 {
1323 news[0] = olds[0];
1324 pinned = 1;
1325 }
1326 else if (start0x)
1327 {
1328 news[0] = olds[0];
1329 news[1] = olds[1];
1330 pinned = 2;
1331 }
1332 }
1333
1334 std::fill_n(news + pinned, plen, fill);
1335 std::copy_n(olds + pinned, oldlen - pinned, news + pinned + plen);
1336 }

1361 void format_float(ios_defs::fmtflags flags, char* fptr, char mod) const noexcept
1362 {
1363 // The notation picks the conversion letter, and with it the one thing
1364 // that is not a free choice: hexfloat carries no explicit precision,
1365 // because `%a` already prints exactly as many hex digits as the value
1366 // needs and pinning a precision on it would round the result away.
1367 // Every other notation takes its precision as a `*` argument, which is
1368 // what LWG 231 asked for -- unconditionally, rather than only under
1369 // `fixed` or a non-zero precision as an earlier reading had it.
1370 const ios_defs::fmtflags notation = flags & ios_defs::floatfield;
1371 const bool hexfloat = (notation == (ios_defs::fixed | ios_defs::scientific));
1372
1373 char conv = 'g';
1374 if (notation == ios_defs::fixed) conv = 'f';
1375 else if (notation == ios_defs::scientific) conv = 'e';
1376 else if (hexfloat) conv = 'a';
1377
1378 // `f` has no upper-case spelling; the other three do, and they sit a
1379 // fixed case-distance apart in ASCII.
1380 if (conv != 'f' && (flags & ios_defs::uppercase))
1381 conv = static_cast<char>(conv - ('a' - 'A'));
1382
1383 // Assemble in the order printf's own grammar fixes: introducer, flags,
1384 // precision, length modifier, conversion.
1385 *fptr++ = '%';
1386 if (flags & ios_defs::showpos) *fptr++ = '+';
1387 if (flags & ios_defs::showpoint) *fptr++ = '#';
1388 if (!hexfloat) { *fptr++ = '.'; *fptr++ = '*'; }
1389 if (mod) *fptr++ = mod;
1390 *fptr++ = conv;
1391 *fptr = '\0';
1392 }

1509 template <typename TIter, std::sentinel_for<TIter> TSent, typename TValue>
1510 std::pair<bool, TIter> extract_int(TIter beg, TSent end, ios_base<char_type>& io, TValue& v) const
1511 {
1512 using unsigned_type = std::make_unsigned_t<TValue>;
1513
1514 const ios_defs::fmtflags requested_base = io.flags() & ios_defs::basefield;
1515 const bool infer_base = requested_base == 0;
1516 int radix = requested_base == ios_defs::hex ? 16
1517 : requested_base == ios_defs::oct ? 8
1518 : 10;
1519
1520 bool negative = false;
1521 if (beg != end)
1522 {
1523 const char_type first = *beg;
1524 const bool has_sign = first == m_in.minus
1525 || first == m_in.plus;
1526 if (has_sign && !is_punct(first))
1527 {
1528 negative = first == m_in.minus;
1529 ++beg;
1530 }
1531 }
1532
1533 // Consume one possible prefix zero up front. Any further zero is handled
1534 // by the ordinary digit loop, which keeps prefix handling independent of
1535 // the length of the field and avoids a second digit-scanning phase.
1536 bool has_digit = false;
1537 int digits_in_group = 0;
1538 if (beg != end && *beg == m_in.digits[0] && !is_punct(*beg))
1539 {
1540 has_digit = true;
1541 digits_in_group = 1;
1542 ++beg;
1543
1544 if (infer_base)
1545 radix = 8;
1546 if (radix == 8)
1547 digits_in_group = 0;
1548
1549 if (beg != end && !is_punct(*beg)
1550 && (*beg == m_in.lower_x || *beg == m_in.upper_x)
1551 && (infer_base || radix == 16))
1552 {
1553 radix = 16;
1554 has_digit = false;
1555 digits_in_group = 0;
1556 ++beg;
1557 }
1558 }
1559
1560 const unsigned_type magnitude_limit =
1561 negative && std::is_signed_v<TValue>
1562 ? -static_cast<unsigned_type>(std::numeric_limits<TValue>::min())
1563 : static_cast<unsigned_type>(std::numeric_limits<TValue>::max());
1564 const auto radix_value = static_cast<unsigned_type>(radix);
1565 const unsigned_type cutoff = magnitude_limit / radix_value;
1566 const unsigned_type last_digit_limit = magnitude_limit % radix_value;
1567
1568 unsigned_type magnitude = 0;
1569 bool out_of_range = false;
1570 bool invalid_grouping = false;
1571 std::vector<uint8_t> group_widths;
1572 if (!m_grouping.empty())
1573 group_widths.reserve(32);
1574
1575 const char_type* const digit_atoms = m_in.digits.data();
1576 const char_type* const primary_end = digit_atoms + radix;
1577 const char_type* const uppercase_hex = digit_atoms + s_upper_hex;
1578
1579 while (beg != end)
1580 {
1581 const char_type current = *beg;
1582
1583 // Locale punctuation takes precedence even if widening happened to
1584 // give it the same representation as a sign or digit.
1585 if (!m_grouping.empty() && current == m_thousands_sep)
1586 {
1587 if (digits_in_group == 0
1588 || std::cmp_greater(digits_in_group,
1589 std::numeric_limits<uint8_t>::max()))
1590 {
1591 invalid_grouping = true;
1592 break;
1593 }
1594
1595 group_widths.push_back(static_cast<uint8_t>(digits_in_group));
1596 digits_in_group = 0;
1597 ++beg;
1598 continue;
1599 }
1600 if (current == m_decimal_point)
1601 break;
1602
1603 int digit = -1;
1604 const char_type* match = std::find(digit_atoms, primary_end, current);
1605 if (match != primary_end)
1606 digit = static_cast<int>(match - digit_atoms);
1607 else if (radix == 16)
1608 {
1609 const char_type* upper = std::find(uppercase_hex,
1610 uppercase_hex + 6, current);
1611 if (upper != uppercase_hex + 6)
1612 digit = static_cast<int>(upper - uppercase_hex) + 10;
1613 }
1614 if (digit < 0)
1615 break;
1616
1617 has_digit = true;
1618 if (!out_of_range)
1619 {
1620 const auto next = static_cast<unsigned_type>(digit);
1621 if (magnitude > cutoff)
1622 out_of_range = true;
1623 else if (magnitude == cutoff && next > last_digit_limit)
1624 {
1625 // Preserve the established counter semantics: when the
1626 // previous magnitude is exactly at the cutoff, this digit
1627 // is counted before the group width freezes.
1628 out_of_range = true;
1629 ++digits_in_group;
1630 }
1631 else
1632 {
1633 magnitude = magnitude * radix_value + next;
1634 ++digits_in_group;
1635 }
1636 }
1637 ++beg;
1638 }
1639
1640 bool success = true;
1641 if (!group_widths.empty())
1642 {
1643 if (std::cmp_greater(digits_in_group,
1644 std::numeric_limits<uint8_t>::max()))
1645 invalid_grouping = true;
1646 else
1647 {
1648 group_widths.push_back(static_cast<uint8_t>(digits_in_group));
1649 success = FacetHelper::verify_grouping(m_grouping, group_widths);
1650 }
1651 }
1652
1653 // LWG 23 (Num_get overflow result): when the field overflows, v must be
1654 // set to numeric_limits::max() / min() with failbit.
1655 //
1656 // `out_of_range` is tested BEFORE `invalid_grouping` by design. Once the
1657 // digit loop sets `out_of_range`, ++digits_in_group is skipped for every
1658 // later digit, so the counter freezes. A subsequent, otherwise
1659 // well-formed thousands_sep then fails the grouping check and sets
1660 // `invalid_grouping` -- but that is a side effect of the overflow
1661 // short-circuit, not an independent structural error in the input.
1662 //
1663 // Letting `invalid_grouping` win in that overlap would map a
1664 // structurally valid, numerically out-of-range field (say
1665 // "12,345,678,901,234,567" read into uint32_t under grouping "\3") to
1666 // v = 0 instead of v = max, contradicting LWG 23. We diverge from
1667 // libstdc++'s ordering here -- it has the same latent issue -- to keep
1668 // the overflow signal dominant whenever it fires.
1669 if (out_of_range)
1670 {
1671 v = negative && std::is_signed_v<TValue>
1672 ? std::numeric_limits<TValue>::min()
1673 : std::numeric_limits<TValue>::max();
1674 success = false;
1675 }
1676 else if (!has_digit || invalid_grouping)
1677 {
1678 v = 0;
1679 success = false;
1680 }
1681 else
1682 v = negative ? -magnitude : magnitude;
1683
1684 return {success, beg};
1685 }

1723 template <typename TIter, std::sentinel_for<TIter> TSent>
1724 std::pair<bool, TIter> extract_float(TIter beg, TSent end, ios_base<char_type>& io, std::string& xtrc) const
1725 {
1726 char_type c = char_type();
1727 bool at_end = beg == end;
1728
1729 std::vector<uint8_t> groups_seen;
1730 if (!m_grouping.empty())
1731 groups_seen.reserve(32);
1732
1733 // A sign is recognised in two places -- in front of the mantissa and in
1734 // front of the exponent -- and behaves identically in both, so the test
1735 // and the transcription live here rather than being spelled twice.
1736 const auto take_sign = [&](char_type ch) -> bool
1737 {
1738 const bool plus = ch == m_in.plus;
1739 if ((plus || ch == m_in.minus) && !is_punct(ch))
1740 {
1741 xtrc += plus ? '+' : '-';
1742 return true;
1743 }
1744 return false;
1745 };
1746
1747 // Closing a group records how many digits it held. Widths are stored as
1748 // bytes, so one too wide to record abandons the field rather than
1749 // wrapping silently; the caller breaks out on a false return.
1750 int group_len = 0;
1751 const auto close_group = [&]() -> bool
1752 {
1753 if (std::cmp_greater(group_len, std::numeric_limits<uint8_t>::max()))
1754 return false;
1755 groups_seen.push_back(static_cast<uint8_t>(group_len));
1756 return true;
1757 };
1758
1759 const auto advance = [&]()
1760 {
1761 if (++beg != end) c = *beg;
1762 else at_end = true;
1763 };
1764
1765 // -- mantissa sign -------------------------------------------------
1766 if (!at_end)
1767 {
1768 c = *beg;
1769 if (take_sign(c))
1770 advance();
1771 }
1772
1773 // -- leading zeros -------------------------------------------------
1774 // They all count towards the first group, but only one reaches `xtrc`:
1775 // strtod does not care how many were written, and keeping them would
1776 // just make the buffer longer.
1777 bool saw_digit = false;
1778 while (!at_end && !is_punct(c) && c == m_in.digits[0])
1779 {
1780 if (!saw_digit)
1781 {
1782 xtrc += '0';
1783 saw_digit = true;
1784 }
1785 ++group_len;
1786 advance();
1787 }
1788
1789 // -- mantissa, decimal point and exponent --------------------------
1790 bool saw_point = false;
1791 bool saw_exp = false;
1792 const char_type* const digit_table = m_in.digits.data();
1793
1794 while (!at_end)
1795 {
1796 // Punctuation is tested before the digit table, because a locale
1797 // may well have nominated a character that appears in both.
1798 if (!m_grouping.empty() && c == m_thousands_sep)
1799 {
1800 // Separators belong to the integer part alone; past a point or
1801 // an exponent one simply ends the field. A separator also has
1802 // to close a group that holds digits, which rules out a leading
1803 // separator and two in a row alike.
1804 if (saw_point || saw_exp)
1805 break;
1806 if (group_len == 0 || !close_group())
1807 {
1808 // Leaving `xtrc` empty is what tells convert_to_v to fail.
1809 xtrc.clear();
1810 break;
1811 }
1812 group_len = 0;
1813 }
1814 else if (c == m_decimal_point)
1815 {
1816 if (saw_point || saw_exp)
1817 break;
1818 // A grouping check only happens if the input used grouping at
1819 // all, so the run of digits before the point is worth recording
1820 // only when some separator already opened the tally.
1821 if (!groups_seen.empty() && !close_group())
1822 {
1823 xtrc.clear();
1824 break;
1825 }
1826 xtrc += '.';
1827 saw_point = true;
1828 }
1829 else
1830 {
1831 const char_type* q = std::find(digit_table, digit_table + 10, c);
1832 if (q != digit_table + 10)
1833 {
1834 xtrc += static_cast<char>('0' + (q - digit_table));
1835 saw_digit = true;
1836 ++group_len;
1837 }
1838 else if ((c == m_in.lower_e || c == m_in.upper_e)
1839 && !saw_exp && saw_digit)
1840 {
1841 // The exponent ends the integer part too, so an open tally
1842 // is closed here for the same reason as at a decimal point
1843 // -- unless a point already did it.
1844 if (!groups_seen.empty() && !saw_point && !close_group())
1845 {
1846 xtrc.clear();
1847 break;
1848 }
1849 xtrc += 'e';
1850 saw_exp = true;
1851
1852 if (++beg == end)
1853 {
1854 at_end = true;
1855 break;
1856 }
1857 c = *beg;
1858 if (!take_sign(c))
1859 // No exponent sign after all, so `c` is already the next
1860 // character to judge: go round again without advancing.
1861 continue;
1862 }
1863 else
1864 break;
1865 }
1866
1867 advance();
1868 }
1869
1870 bool success = true;
1871 // A grouping check only applies to input that actually used grouping.
1872 // When it did, the final group still has to be closed -- unless a point
1873 // or an exponent closed it already.
1874 if (!groups_seen.empty())
1875 {
1876 if (!saw_point && !saw_exp && !close_group())
1877 success = false;
1878 else
1879 success = FacetHelper::verify_grouping(m_grouping, groups_seen);
1880 }
1881
1882 return std::pair(success, beg);
1883 }

1918 {

1945 v = std::signbit(parsed) ? -limit : limit;
1946 return false;
1947 }
1948
1949 v = parsed;
1950 return true;
