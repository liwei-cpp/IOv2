128 template <shared_ptr_to<numeric_conf<CharT>> TConfPtr,
129 shared_ptr_to<ctype<CharT>> TCtypePtr>
130 numeric(TConfPtr p_obj, TCtypePtr p_ctype) : m_ctype(p_ctype)
131 {
132 if (!p_obj || !p_ctype) throw io_error("shared_ptr is empty");
133 m_decimal_point = p_obj->decimal_point();
134 m_thousands_sep = p_obj->thousands_sep();
135 m_true_name = p_obj->truename();
136 m_false_name = p_obj->falsename();
137 // grouping() is contracted to return the INTERNAL convention
138 // (1–255 = group size, 0 = stop, last element implicitly repeats).
139 // numeric_conf and any user-derived override are both expected to
140 // satisfy this contract — POSIX-style normalisation is done at the
141 // POSIX boundary in numeric_conf, not here.
142 m_grouping = p_obj->grouping();
143
144 // Each alphabet is widened whole and every other atom is widened by
145 // name, so nothing here depends on a character sitting at a
146 // remembered offset. string_view carries the length, with no trailing
147 // '\0' counted; static_assert keeps literal and destination in step.
148 constexpr std::string_view hex_lower = "0123456789abcdef";
149 constexpr std::string_view hex_upper = "0123456789ABCDEF";
150 constexpr std::string_view upper_hex_letters = "ABCDEF";
151 static_assert(hex_lower.size() == std::tuple_size_v<decltype(out_atoms::lower)>);
152 static_assert(hex_upper.size() == std::tuple_size_v<decltype(out_atoms::upper)>);
153
154 m_ctype->widen_seq(hex_lower.data(), hex_lower.data() + hex_lower.size(),
155 m_out.lower.data());
156 m_ctype->widen_seq(hex_upper.data(), hex_upper.data() + hex_upper.size(),
157 m_out.upper.data());
158 m_out.minus = m_ctype->widen('-');
159 m_out.plus = m_ctype->widen('+');
160 m_out.lower_x = m_ctype->widen('x');
161 m_out.upper_x = m_ctype->widen('X');
162
163 // The parser reads a digit's value off the position it matched at, so
164 // the two letter cases are laid end to end after the decimal digits:
165 // '0'..'9', then 'a'..'f', then 'A'..'F' at s_upper_hex.
166 m_ctype->widen_seq(hex_lower.data(), hex_lower.data() + hex_lower.size(),
167 m_in.digits.data());
168 m_ctype->widen_seq(upper_hex_letters.data(),
169 upper_hex_letters.data() + upper_hex_letters.size(),
170 m_in.digits.data() + s_upper_hex);
171 m_in.minus = m_out.minus;
172 m_in.plus = m_out.plus;
173 m_in.lower_x = m_out.lower_x;
174 m_in.upper_x = m_out.upper_x;
175 m_in.lower_e = m_ctype->widen('e');
176 m_in.upper_e = m_ctype->widen('E');
177
178 if constexpr (std::is_same_v<CharT, wchar_t> ||
179 std::is_same_v<CharT, char32_t>)
180 {
181 // Distinctness is a property of the whole set a stage works with,
182 // so each side is checked as one run: the marks followed by every
183 // glyph that side can emit or match.
184 std::array<CharT, 26> in_view{m_in.minus, m_in.plus,
185 m_in.lower_x, m_in.upper_x};
186 std::copy(m_in.digits.begin(), m_in.digits.end(), in_view.begin() + 4);
187 assert(atoms_pairwise_distinct(in_view.data(), in_view.size()));
188
189 std::array<CharT, 26> out_view{m_out.minus, m_out.plus,
190 m_out.lower_x, m_out.upper_x};
191 std::copy(m_out.lower.begin(), m_out.lower.end(), out_view.begin() + 4);
192 std::copy(m_out.upper.begin() + s_hex_letters, m_out.upper.end(),
193 out_view.begin() + 20);
194 assert(atoms_pairwise_distinct(out_view.data(), out_view.size()));
195 }

304 template <typename TIter>
305 TIter put(TIter s, ios_base<char_type>& io, bool v) const
306 {
307 const auto flags = io.flags();
308 if ((flags & ios_defs::boolalpha) == 0)
309 {
310 return insert_int(s, io, static_cast<long>(v));
311 }
312
313 const auto& name = v ? m_true_name : m_false_name;
314 std::size_t len = name.size();
315
316 const auto w = io.width(0);
317 if (w > static_cast<decltype(w)>(len))
318 {
319 const auto plen = w - len;
320 if (plen > ios_defs::max_pad_count)
321 throw stream_error("numeric put fail: fill count exceeds max_pad_count");
322
323 // Not fill_alters_reading: the content is a name, not a number, so digits are
324 // harmless (setfill('0') << boolalpha is ordinary). What is not is a fill in
325 // front that starts either name: "tttrue" would read as a different word.
326 const char_type fill = io.fill();
327 if ((flags & ios_defs::adjustfield) != ios_defs::left
328 && ((!m_true_name.empty() && m_true_name[0] == fill)
329 || (!m_false_name.empty() && m_false_name[0] == fill)))
330 throw stream_error("numeric put fail: fill would change the value the field reads as");
331
332 if ((flags & ios_defs::adjustfield) == ios_defs::left)
333 {
334 s = std::copy(name.begin(), name.end(), s);
335 s = std::fill_n(s, plen, fill);
336 }
337 else
338 {
339 s = std::fill_n(s, plen, fill);
340 s = std::copy(name.begin(), name.end(), s);
341 }
342 return s;
343 }
344
345 return std::copy(name.begin(), name.end(), s);
346 }

503 template <typename TIter, std::sentinel_for<TIter> TSent>
504 TIter get(TIter beg, TSent end, ios_base<char_type>& io, bool& v) const
505 {
506 bool success = true;
507
508 if (!(io.flags() & ios_defs::boolalpha))
509 {
510 // Numeric form: read the field as a `long` and accept only 0/1.
511 long l = -1;
512 std::tie(success, beg) = extract_int(beg, end, io, l);
513
514 if (l == 0 || l == 1) v = bool(l);
515 else
516 {
517 // LWG 23 resolved that a field which does not fit the target
518 // still stores a value rather than leaving it untouched: the
519 // most positive representable one, which for bool is `true`.
520 v = true;
521 success = false;
522 }
523 }
524 else
525 {
526 // `boolalpha` form: match the input against the two locale names.
527 //
528 // Both names race for the input. A name stays in the running only
529 // while every character read so far is a prefix of it, and only
530 // while it still has a character left to offer; extraction stops
531 // the moment no name can take the next character, which is what
532 // keeps the facet from consuming more than it needs.
533 const std::basic_string<CharT>& fname = m_false_name;
534 const std::basic_string<CharT>& tname = m_true_name;
535
536 // Fill in front of the name is `right` padding. The writer refuses a fill
537 // that starts either name, so one that does is left to the match instead.
538 const CharT fill = io.fill();
539 if ((fname.empty() || fname[0] != fill) && (tname.empty() || tname[0] != fill))
540 while (beg != end && *beg == fill)
541 ++beg;
542
543 bool f_live = true;
544 bool t_live = true;
545 std::size_t n = 0;
546
547 while (beg != end)
548 {
549 const bool f_open = f_live && n < fname.size();
550 const bool t_open = t_live && n < tname.size();
551 if (!f_open && !t_open)
552 break;
553
554 const CharT c = *beg;
555 const bool f_takes = f_open && fname[n] == c;
556 const bool t_takes = t_open && tname[n] == c;
557 if (!f_takes && !t_takes)
558 break;
559
560 f_live = f_takes;
561 t_live = t_takes;
562 ++n;
563 ++beg;
564 }
565
566 // A name won only if it is still live and came out exactly
567 // exhausted; `n != 0` keeps an empty name from matching nothing.
568 const bool got_false = f_live && n == fname.size() && n != 0;
569 const bool got_true = t_live && n == tname.size() && n != 0;
570
571 // Exactly one name may win. Both winning means the two are
572 // indistinguishable on this input; neither winning means nothing
573 // matched. The standard asks for `false` plus a failure either way.
574 v = got_true && !got_false;
575 if (got_false == got_true)
576 success = false;
577 }
578
579 if (!success) throw stream_error("numeric::get fail: parse boolean fail");
580 return beg;
581 }

829 template <typename TIter, typename TValue>
830 TIter insert_float(TIter s, ios_base<char_type>& io, TValue v, char mod) const
831 {
832 // precision() is now a bounded uint8_t (0-255); there is no negative /
833 // out-of-range sentinel to normalise here.
834 const std::streamsize prec = io.precision();
835 const int max_digits = std::numeric_limits<TValue>::digits10;
836
837 const std::size_t w = io.width(0);
838
839 // Build the printf conversion spec, then let the C library render the
840 // value into a narrow buffer; everything after this is localisation.
841 std::size_t len = 0;
842 std::array<char, 16> fbuf{};
843 format_float(io.flags(), fbuf.data(), mod);
844
845 const ios_defs::fmtflags fltfield = io.flags() & ios_defs::floatfield;
846
847 // First guess at the buffer. Outside `fixed`, three times the decimal
848 // digit count leaves room for sign, point, exponent and slack; `fixed`
849 // is the notation that can run long, so it gets the full integer part.
850 std::size_t cs_size = static_cast<std::size_t>(max_digits) * 3 + 32;
851 if (fltfield == ios_defs::fixed)
852 cs_size = static_cast<std::size_t>(std::numeric_limits<TValue>::max_exponent10) + static_cast<std::size_t>(prec) + 32;
853
854 // Cap initial allocation to a reasonable size (e.g., 2048) to avoid huge initial pressure.
855 if (cs_size > 2048) cs_size = 2048;
856
857 std::vector<char> vec_cs(cs_size);
858
859 {
860 clocale_wrapper inter_locale("C");
861 clocale_user guard(inter_locale);
862
863 auto do_snprintf = [&](char* buf, std::size_t size) {
864 if (fltfield == (ios_defs::fixed | ios_defs::scientific))
865 return snprintf(buf, size, fbuf.data(), v);
866 else
867 return snprintf(buf, size, fbuf.data(), static_cast<int>(prec), v);
868 };
869
870 // snprintf returns int. Trap negative (encoding error) before
871 // promoting to size_t — otherwise -1 becomes SIZE_MAX and
872 // bypasses every downstream bound check.
873 int n = do_snprintf(vec_cs.data(), cs_size);
874 if (n < 0)
875 throw stream_error("numeric::put fail: floating-point conversion failed");
876 len = static_cast<std::size_t>(n);
877
878 // If buffer was too small, snprintf returns required length.
879 if (len >= cs_size)
880 {
881 cs_size = len + 1;
882 vec_cs.resize(cs_size);
883 n = do_snprintf(vec_cs.data(), cs_size);
884 if (n < 0)
885 throw stream_error("numeric::put fail: floating-point conversion failed");
886 len = static_cast<std::size_t>(n);
887 }
888 }
889
890 // len == 0 is also treated as failure: every well-formed numeric
891 // print (including 0, "inf", "nan") emits at least one character,
892 // and the downstream path assumes vec_ws(len) has at least one
893 // element before dereferencing vec_ws.data().
894 if (len == 0 || len >= cs_size)
895 throw stream_error("numeric::put fail: floating-point conversion failed");
896
897 const char* cs = vec_cs.data();
898
899 // Widen into char_type, then apply the two things the C locale could
900 // not know about: this facet's decimal point, and its grouping.
901 std::vector<CharT> vec_ws(len);
902 CharT* ws = vec_ws.data();
903 m_ctype->widen_seq(cs, cs + len, ws);
904
905 const char* p = std::find(cs, cs + len, '.');
906 CharT* wp = nullptr;
907 if (p != cs + len)
908 {
909 wp = ws + (p - cs);
910 *wp = m_decimal_point;
911 }
912
913 // Only a real number gets grouped -- "inf", "nan" and exponent-only forms
914 // like 2e20 must come through untouched. A decimal point settles it
915 // outright; failing that, text too short to be one of those words is a
916 // number, and so is text whose second and third characters are digits.
917 const auto is_digit = [](char c) { return c >= '0' && c <= '9'; };
918 const bool numeric_text =
919 wp || len < 3u || (is_digit(cs[1]) && is_digit(cs[2]));
920
921 if (!m_grouping.empty() && numeric_text)
922 {
923 // Worst case one separator per digit, so twice the length is ample.
924 std::vector<CharT> vec_ws2(len * 2);
925 CharT* ws2 = vec_ws2.data();
926
927 // A leading sign is not part of the number being grouped. Copy it
928 // across on its own and hand the grouper only what follows, so the
929 // digit count it works from never has to be unwound afterwards.
930 const std::size_t sign_len = (cs[0] == '-' || cs[0] == '+') ? 1u : 0u;
931 if (sign_len != 0)
932 ws2[0] = ws[0];
933
934 // LWG 282: grouping applies to the integer part only, so the split
935 // happens here and the fraction is handed over to be copied through.
936 const std::size_t digit_len = len - sign_len;
937 const std::size_t int_len = wp ? static_cast<std::size_t>(wp - (ws + sign_len))
938 : digit_len;
939 len = sign_len + group_float(m_grouping, m_thousands_sep, ws2 + sign_len,
940 ws + sign_len, int_len, digit_len - int_len);
941 std::swap(vec_ws, vec_ws2);
942 ws = vec_ws.data();
943 }
944
945 // Pad.
946 if (std::cmp_greater(w, len))
947 {
948 if (w - len > ios_defs::max_pad_count)
949 throw stream_error("numeric put fail: fill count exceeds max_pad_count");
950
951 std::vector<CharT> vec_ws3(w);
952 CharT* ws3 = vec_ws3.data();
953 bool startSign = (ws[0] == m_out.minus) || (ws[0] == m_out.plus);
954 bool start0x = (ws[0] == m_out.lower[0]) && (len > 1u) &&
955 ((ws[1] == m_out.lower_x) || (ws[1] == m_out.upper_x));
956 const ios_defs::fmtflags adjust = io.flags() & ios_defs::adjustfield;
957 pad(io.fill(), w, adjust, io.flags() & ios_defs::basefield,
958 ws3, ws, len, startSign, start0x);
959
960 std::swap(vec_ws, vec_ws3);
961 ws = vec_ws.data();
962 }
963
964 // Nothing is left to decide; hand the finished field to the caller.
965 return std::copy(ws, ws + len, s);
966 }

999 template <typename TIter, typename TValue>
1000 TIter insert_int(TIter s, ios_base<char_type>& io, TValue v) const
1001 {
1002 using unsigned_type = std::make_unsigned_t<TValue>;
1003
1004 const ios_defs::fmtflags flags = io.flags();
1005 const std::size_t w = io.width(0);
1006
1007 const ios_defs::fmtflags basefield = flags & ios_defs::basefield;
1008 const bool dec = (basefield != ios_defs::oct && basefield != ios_defs::hex);
1009
1010 auto u = static_cast<unsigned_type>(v);
1011 if constexpr (std::is_signed_v<TValue>)
1012 {
1013 if (dec && v < 0)
1014 u = -u;
1015 }
1016
1017 // Even base two would need no more than one character per value bit. Two
1018 // fixed buffers therefore cover both raw and maximally grouped output while
1019 // keeping every ordinary integer conversion off the heap.
1020 constexpr std::size_t digit_capacity = std::numeric_limits<unsigned_type>::digits;
1021 std::array<char_type, digit_capacity> raw_digits{};
1022 char_type* const raw_end = raw_digits.data() + raw_digits.size();
1023 char_type* raw_begin = raw_end;
1024
1025 if (dec)
1026 {
1027 do // NOLINT(cppcoreguidelines-avoid-do-while)
1028 {
1029 *--raw_begin = m_out.lower[u % 10];
1030 u /= 10;
1031 } while (u != 0);
1032 }
1033 else
1034 {
1035 const unsigned shift = (basefield == ios_defs::oct) ? 3u : 4u;
1036 const unsigned_type mask = (unsigned_type{1} << shift) - 1;
1037 const auto& alphabet = (basefield == ios_defs::hex && (flags & ios_defs::uppercase))
1038 ? m_out.upper : m_out.lower;
1039 do // NOLINT(cppcoreguidelines-avoid-do-while)
1040 {
1041 *--raw_begin = alphabet[static_cast<std::size_t>(u & mask)];
1042 u >>= shift;
1043 } while (u != 0);
1044 }
1045
1046 const char_type* digits = raw_begin;
1047 auto digit_len = static_cast<std::size_t>(raw_end - raw_begin);
1048 std::array<char_type, digit_capacity * 2> grouped_digits{};
1049
1050 if (!m_grouping.empty())
1051 {
1052 char_type* const grouped_end = FacetHelper::add_grouping(
1053 grouped_digits.data(), m_thousands_sep, m_grouping, raw_begin, raw_end);
1054 digits = grouped_digits.data();
1055 digit_len = static_cast<std::size_t>(grouped_end - digits);
1056 }
1057
1058 std::array<char_type, 2> prefix{};
1059 std::size_t prefix_len = 0;
1060 if (dec)
1061 {
1062 if constexpr (std::is_signed_v<TValue>)
1063 {
1064 if (v < 0)
1065 prefix[prefix_len++] = m_out.minus;
1066 else if (flags & ios_defs::showpos)
1067 prefix[prefix_len++] = m_out.plus;
1068 }
1069 }
1070 else if (bool(flags & ios_defs::showbase) && v)
1071 {
1072 prefix[prefix_len++] = m_out.lower[0];
1073 if (basefield == ios_defs::hex)
1074 prefix[prefix_len++] = (flags & ios_defs::uppercase) ? m_out.upper_x : m_out.lower_x;
1075 }
1076
1077 const std::size_t len = prefix_len + digit_len;
1078 if (w <= len)
1079 {
1080 s = std::copy_n(prefix.data(), prefix_len, s);
1081 return std::copy_n(digits, digit_len, s);
1082 }
1083
1084 const std::size_t pad_len = w - len;
1085 if (pad_len > ios_defs::max_pad_count)
1086 throw stream_error("numeric put fail: fill count exceeds max_pad_count");
1087
1088 const bool start_sign = prefix_len == 1
1089 && (prefix[0] == m_out.minus || prefix[0] == m_out.plus);
1090 const bool start_0x = prefix_len == 2
1091 && prefix[0] == m_out.lower[0]
1092 && (prefix[1] == m_out.lower_x || prefix[1] == m_out.upper_x);
1093 const ios_defs::fmtflags adjust = flags & ios_defs::adjustfield;
1094 const bool trails_value = adjust == ios_defs::left;
1095 // Leading zeros would join the first group and break it, so a field that
1096 // carries a separator is not one a '0' fill may lead.
1097 const bool separated = digit_len != static_cast<std::size_t>(raw_end - raw_begin);
1098 const bool leads_digits = !trails_value && !separated
1099 && (adjust == ios_defs::internal || !(start_sign || start_0x));
1100 if (fill_alters_reading(io.fill(), basefield, leads_digits, trails_value,
1101 prefix_len != 0 && prefix[0] == m_out.minus, false))
1102 throw stream_error("numeric put fail: fill would change the value the field reads as");
1103
1104 if (trails_value)
1105 {
1106 s = std::copy_n(prefix.data(), prefix_len, s);
1107 s = std::copy_n(digits, digit_len, s);
1108 return std::fill_n(s, pad_len, io.fill());
1109 }
1110
1111 if (adjust == ios_defs::internal && (start_sign || start_0x))
1112 {
1113 s = std::copy_n(prefix.data(), prefix_len, s);
1114 s = std::fill_n(s, pad_len, io.fill());
1115 return std::copy_n(digits, digit_len, s);
1116 }
1117
1118 s = std::fill_n(s, pad_len, io.fill());
1119 s = std::copy_n(prefix.data(), prefix_len, s);
1120 return std::copy_n(digits, digit_len, s);
1121 }

1260 [[nodiscard]] bool fill_alters_reading(char_type fill, ios_defs::fmtflags basefield,
1261 bool leads_digits, bool trails_value,
1262 bool negative, bool floating) const
1263 {
1264 const int radix = basefield == ios_defs::hex ? 16 : 10;
1265 if (is_digit_glyph(fill, radix))
1266 return !(fill == m_out.lower[0] && leads_digits);
1267
1268 // Past the value only what the parser would take as a continuation matters.
1269 if (trails_value)
1270 return (!m_grouping.empty() && fill == m_thousands_sep)
1271 || (floating ? (fill == m_in.lower_e || fill == m_in.upper_e)
1272 : (radix == 16 && (fill == m_in.lower_x || fill == m_in.upper_x)));
1273
1274 if (fill == m_decimal_point)
1275 return true;
1276 if (fill == m_out.minus)
1277 return !negative;
1278 if (fill == m_out.plus)
1279 return negative;
1280 return false;
1281 }

1317 void pad(char_type fill, std::size_t w, ios_defs::fmtflags adjust, // NOLINT(bugprone-easily-swappable-parameters)
1318 ios_defs::fmtflags basefield,
1319 char_type* new_buf, const char_type* cs, std::size_t& len,
1320 bool startSign, bool start0x) const
1321 {
1322 // Fill is about to be written, so this is where it gets vetted: a fill character
1323 // that would change the number this field reads as is rejected outright. Where
1324 // the run lands decides what is safe, and `pad_impl` below fixes that: `left`
1325 // appends it after the value; `internal`, and `right` on content that starts with
1326 // a digit, put it directly in front of the digits; `right` on content that starts
1327 // with a sign or `0x` leaves that prefix between the fill and the digits.
1328 // The check belongs here rather than at the top of the insert functions because
1329 // `fill` is sticky stream state: a stream carrying setfill('1') must keep working
1330 // for every output whose width leaves nothing to pad.
1331 const bool trails_value = (adjust == ios_defs::left);
1332 // Leading zeros would join the first group and break it, so a field that
1333 // carries a separator is not one a '0' fill may lead.
1334 const bool separated = !m_grouping.empty()
1335 && std::find(cs, cs + len, m_thousands_sep) != cs + len;
1336 const bool leads_digits = !trails_value && !separated
1337 && (adjust == ios_defs::internal || !(startSign || start0x));
1338 if (fill_alters_reading(fill, basefield, leads_digits, trails_value,
1339 cs[0] == m_out.minus, true))
1340 throw stream_error("numeric put fail: fill would change the value the field reads as");
1341
1342 // The field is short of `w`, so the fill run gets placed.
1343 pad_impl(adjust, fill, new_buf, cs, w, len, startSign, start0x);
1344 len = w;
1345 }
1346
1385 void pad_impl(ios_defs::fmtflags adjust, char_type fill,
1386 char_type* news, const char_type* olds,
1387 std::size_t newlen, std::size_t oldlen,
1388 bool startSign, bool start0x) const
1389 {
1390
1391 const std::size_t plen = newlen - oldlen;
1392
1393 if (adjust == ios_defs::left)
1394 {
1395 // Fill trails the value.
1396 std::copy_n(olds, oldlen, news);
1397 std::fill_n(news + oldlen, plen, fill);
1398 return;
1399 }
1400
1401 // Every other adjustment puts the fill ahead of the digits, and the two
1402 // that remain differ only in what gets held back at the left edge:
1403 // `internal` pins whatever announces the number -- a sign, or a `0x` /
1404 // `0X` base marker -- and lets the fill slot in behind it, while `right`
1405 // pins nothing at all. A pinned run of length zero makes that one case,
1406 // so both are served by a single three-step write.
1407 std::size_t pinned = 0;
1408 if (adjust == ios_defs::internal)
1409 {
1410 // At most two characters, so they go across by hand: handing a
1411 // length of 0-2 to a range algorithm costs a memmove call that
1412 // dwarfs the copy itself, and `right` would pay it for nothing.
1413 if (startSign)
1414 {
1415 news[0] = olds[0];
1416 pinned = 1;
1417 }
1418 else if (start0x)
1419 {
1420 news[0] = olds[0];
1421 news[1] = olds[1];
1422 pinned = 2;
1423 }
1424 }
1425
1426 std::fill_n(news + pinned, plen, fill);
1427 std::copy_n(olds + pinned, oldlen - pinned, news + pinned + plen);
1428 }

1453 void format_float(ios_defs::fmtflags flags, char* fptr, char mod) const noexcept
1454 {
1455 // The notation picks the conversion letter, and with it the one thing
1456 // that is not a free choice: hexfloat carries no explicit precision,
1457 // because `%a` already prints exactly as many hex digits as the value
1458 // needs and pinning a precision on it would round the result away.
1459 // Every other notation takes its precision as a `*` argument, which is
1460 // what LWG 231 asked for -- unconditionally, rather than only under
1461 // `fixed` or a non-zero precision as an earlier reading had it.
1462 const ios_defs::fmtflags notation = flags & ios_defs::floatfield;
1463 const bool hexfloat = (notation == (ios_defs::fixed | ios_defs::scientific));
1464
1465 char conv = 'g';
1466 if (notation == ios_defs::fixed) conv = 'f';
1467 else if (notation == ios_defs::scientific) conv = 'e';
1468 else if (hexfloat) conv = 'a';
1469
1470 // `f` has no upper-case spelling; the other three do, and they sit a
1471 // fixed case-distance apart in ASCII.
1472 if (conv != 'f' && (flags & ios_defs::uppercase))
1473 conv = static_cast<char>(conv - ('a' - 'A'));
1474
1475 // Assemble in the order printf's own grammar fixes: introducer, flags,
1476 // precision, length modifier, conversion.
1477 *fptr++ = '%';
1478 if (flags & ios_defs::showpos) *fptr++ = '+';
1479 if (flags & ios_defs::showpoint) *fptr++ = '#';
1480 if (!hexfloat) { *fptr++ = '.'; *fptr++ = '*'; }
1481 if (mod) *fptr++ = mod;
1482 *fptr++ = conv;
1483 *fptr = '\0';
1484 }

1706 template <typename TIter, std::sentinel_for<TIter> TSent, typename TValue>
1707 std::pair<bool, TIter> extract_int(TIter beg, TSent end, ios_base<char_type>& io, TValue& v) const
1708 {
1709 using unsigned_type = std::make_unsigned_t<TValue>;
1710
1711 const ios_defs::fmtflags requested_base = io.flags() & ios_defs::basefield;
1712 const bool infer_base = requested_base == 0;
1713 int radix = requested_base == ios_defs::hex ? 16
1714 : requested_base == ios_defs::oct ? 8
1715 : 10;
1716
1717 // Fill in front of the field is `right` padding. A sign-shaped fill got past the
1718 // writer only by agreeing with the value, so a run of it is the sign.
1719 const char_type fill = io.fill();
1720 const bool sign_fill = (fill == m_in.minus || fill == m_in.plus) && !is_punct(fill);
1721 bool negative = false;
1722 if (skip_fill(beg, end, fill, radix) != 0 && sign_fill)
1723 negative = fill == m_in.minus;
1724 else if (beg != end)
1725 {
1726 const char_type first = *beg;
1727 const bool has_sign = first == m_in.minus
1728 || first == m_in.plus;
1729 if (has_sign && !is_punct(first))
1730 {
1731 negative = first == m_in.minus;
1732 ++beg;
1733 skip_fill(beg, end, fill, radix); // `internal` padding
1734 }
1735 }
1736
1737 // Consume one possible prefix zero up front. Any further zero is handled
1738 // by the ordinary digit loop, which keeps prefix handling independent of
1739 // the length of the field and avoids a second digit-scanning phase.
1740 bool has_digit = false;
1741 int digits_in_group = 0;
1742 if (beg != end && *beg == m_in.digits[0] && !is_punct(*beg))
1743 {
1744 has_digit = true;
1745 digits_in_group = 1;
1746 ++beg;
1747
1748 if (infer_base)
1749 radix = 8;
1750 if (radix == 8)
1751 digits_in_group = 0;
1752
1753 if (beg != end && !is_punct(*beg)
1754 && (*beg == m_in.lower_x || *beg == m_in.upper_x)
1755 && (infer_base || radix == 16))
1756 {
1757 radix = 16;
1758 has_digit = false;
1759 digits_in_group = 0;
1760 ++beg;
1761 skip_fill(beg, end, fill, radix); // `internal` padding
1762 }
1763 }
1764
1765 const unsigned_type magnitude_limit =
1766 negative && std::is_signed_v<TValue>
1767 ? -static_cast<unsigned_type>(std::numeric_limits<TValue>::min())
1768 : static_cast<unsigned_type>(std::numeric_limits<TValue>::max());
1769 const auto radix_value = static_cast<unsigned_type>(radix);
1770 const unsigned_type cutoff = magnitude_limit / radix_value;
1771 const unsigned_type last_digit_limit = magnitude_limit % radix_value;
1772
1773 unsigned_type magnitude = 0;
1774 bool out_of_range = false;
1775 bool invalid_grouping = false;
1776 std::vector<uint8_t> group_widths;
1777 if (!m_grouping.empty())
1778 group_widths.reserve(32);
1779
1780 const char_type* const digit_atoms = m_in.digits.data();
1781 const char_type* const primary_end = digit_atoms + radix;
1782 const char_type* const uppercase_hex = digit_atoms + s_upper_hex;
1783
1784 while (beg != end)
1785 {
1786 const char_type current = *beg;
1787
1788 // Locale punctuation takes precedence even if widening happened to
1789 // give it the same representation as a sign or digit.
1790 if (!m_grouping.empty() && current == m_thousands_sep)
1791 {
1792 if (digits_in_group == 0
1793 || std::cmp_greater(digits_in_group,
1794 std::numeric_limits<uint8_t>::max()))
1795 {
1796 invalid_grouping = true;
1797 break;
1798 }
1799
1800 group_widths.push_back(static_cast<uint8_t>(digits_in_group));
1801 digits_in_group = 0;
1802 ++beg;
1803 continue;
1804 }
1805 if (current == m_decimal_point)
1806 break;
1807
1808 int digit = -1;
1809 const char_type* match = std::find(digit_atoms, primary_end, current);
1810 if (match != primary_end)
1811 digit = static_cast<int>(match - digit_atoms);
1812 else if (radix == 16)
1813 {
1814 const char_type* upper = std::find(uppercase_hex,
1815 uppercase_hex + 6, current);
1816 if (upper != uppercase_hex + 6)
1817 digit = static_cast<int>(upper - uppercase_hex) + 10;
1818 }
1819 if (digit < 0)
1820 break;
1821
1822 has_digit = true;
1823 if (!out_of_range)
1824 {
1825 const auto next = static_cast<unsigned_type>(digit);
1826 if (magnitude > cutoff)
1827 out_of_range = true;
1828 else if (magnitude == cutoff && next > last_digit_limit)
1829 {
1830 // Preserve the established counter semantics: when the
1831 // previous magnitude is exactly at the cutoff, this digit
1832 // is counted before the group width freezes.
1833 out_of_range = true;
1834 ++digits_in_group;
1835 }
1836 else
1837 {
1838 magnitude = magnitude * radix_value + next;
1839 ++digits_in_group;
1840 }
1841 }
1842 ++beg;
1843 }
1844
1845 bool success = true;
1846 if (!group_widths.empty())
1847 {
1848 if (std::cmp_greater(digits_in_group,
1849 std::numeric_limits<uint8_t>::max()))
1850 invalid_grouping = true;
1851 else
1852 {
1853 group_widths.push_back(static_cast<uint8_t>(digits_in_group));
1854 success = FacetHelper::verify_grouping(m_grouping, group_widths);
1855 }
1856 }
1857
1858 // LWG 23 (Num_get overflow result): when the field overflows, v must be
1859 // set to numeric_limits::max() / min() with failbit.
1860 //
1861 // `out_of_range` is tested BEFORE `invalid_grouping` by design. Once the
1862 // digit loop sets `out_of_range`, ++digits_in_group is skipped for every
1863 // later digit, so the counter freezes. A subsequent, otherwise
1864 // well-formed thousands_sep then fails the grouping check and sets
1865 // `invalid_grouping` -- but that is a side effect of the overflow
1866 // short-circuit, not an independent structural error in the input.
1867 //
1868 // Letting `invalid_grouping` win in that overlap would map a
1869 // structurally valid, numerically out-of-range field (say
1870 // "12,345,678,901,234,567" read into uint32_t under grouping "\3") to
1871 // v = 0 instead of v = max, contradicting LWG 23. We diverge from
1872 // libstdc++'s ordering here -- it has the same latent issue -- to keep
1873 // the overflow signal dominant whenever it fires.
1874 if (out_of_range)
1875 {
1876 v = negative && std::is_signed_v<TValue>
1877 ? std::numeric_limits<TValue>::min()
1878 : std::numeric_limits<TValue>::max();
1879 success = false;
1880 }
1881 else if (!has_digit || invalid_grouping)
1882 {
1883 v = 0;
1884 success = false;
1885 }
1886 else
1887 v = negative ? -magnitude : magnitude;
1888
1889 return {success, beg};
1890 }

1930 template <typename TIter, std::sentinel_for<TIter> TSent>
1931 std::pair<bool, TIter> extract_float(TIter beg, TSent end, ios_base<char_type>& io, std::string& xtrc) const
1932 {
1933 char_type c = char_type();
1934 bool at_end = beg == end;
1935
1936 std::vector<uint8_t> groups_seen;
1937 if (!m_grouping.empty())
1938 groups_seen.reserve(32);
1939
1940 // A sign is recognised in two places -- in front of the mantissa and in
1941 // front of the exponent -- and behaves identically in both, so the test
1942 // and the transcription live here rather than being spelled twice.
1943 const auto take_sign = [&](char_type ch) -> bool
1944 {
1945 const bool plus = ch == m_in.plus;
1946 if ((plus || ch == m_in.minus) && !is_punct(ch))
1947 {
1948 xtrc += plus ? '+' : '-';
1949 return true;
1950 }
1951 return false;
1952 };
1953
1954 // Closing a group records how many digits it held. Widths are stored as
1955 // bytes, so one too wide to record abandons the field rather than
1956 // wrapping silently; the caller breaks out on a false return.
1957 int group_len = 0;
1958 const auto close_group = [&]() -> bool
1959 {
1960 if (std::cmp_greater(group_len, std::numeric_limits<uint8_t>::max()))
1961 return false;
1962 groups_seen.push_back(static_cast<uint8_t>(group_len));
1963 return true;
1964 };
1965
1966 const auto advance = [&]()
1967 {
1968 if (++beg != end) c = *beg;
1969 else at_end = true;
1970 };
1971
1972 // -- padding and mantissa sign -------------------------------------
1973 // As in extract_int: fill in front is `right` padding, a sign-shaped run of it is
1974 // the sign, and `internal` padding follows a sign.
1975 const char_type fill = io.fill();
1976 const bool sign_fill = (fill == m_in.minus || fill == m_in.plus) && !is_punct(fill);
1977 if (skip_fill(beg, end, fill, 10) != 0 && sign_fill)
1978 xtrc += fill == m_in.plus ? '+' : '-';
1979 else if (beg != end && take_sign(*beg))
1980 {
1981 ++beg;
1982 skip_fill(beg, end, fill, 10);
1983 }
1984 at_end = beg == end;
1985 if (!at_end)
1986 c = *beg;
1987
1988 // -- leading zeros -------------------------------------------------
1989 // They all count towards the first group, but only one reaches `xtrc`:
1990 // strtod does not care how many were written, and keeping them would
1991 // just make the buffer longer.
1992 bool saw_digit = false;
1993 while (!at_end && !is_punct(c) && c == m_in.digits[0])
1994 {
1995 if (!saw_digit)
1996 {
1997 xtrc += '0';
1998 saw_digit = true;
1999 }
2000 ++group_len;
2001 advance();
2002 }
2003
2004 // -- mantissa, decimal point and exponent --------------------------
2005 bool saw_point = false;
2006 bool saw_exp = false;
2007 const char_type* const digit_table = m_in.digits.data();
2008
2009 while (!at_end)
2010 {
2011 // Punctuation is tested before the digit table, because a locale
2012 // may well have nominated a character that appears in both.
2013 if (!m_grouping.empty() && c == m_thousands_sep)
2014 {
2015 // Separators belong to the integer part alone; past a point or
2016 // an exponent one simply ends the field. A separator also has
2017 // to close a group that holds digits, which rules out a leading
2018 // separator and two in a row alike.
2019 if (saw_point || saw_exp)
2020 break;
2021 if (group_len == 0 || !close_group())
2022 {
2023 // Leaving `xtrc` empty is what tells convert_to_v to fail.
2024 xtrc.clear();
2025 break;
2026 }
2027 group_len = 0;
2028 }
2029 else if (c == m_decimal_point)
2030 {
2031 if (saw_point || saw_exp)
2032 break;
2033 // A grouping check only happens if the input used grouping at
2034 // all, so the run of digits before the point is worth recording
2035 // only when some separator already opened the tally.
2036 if (!groups_seen.empty() && !close_group())
2037 {
2038 xtrc.clear();
2039 break;
2040 }
2041 xtrc += '.';
2042 saw_point = true;
2043 }
2044 else
2045 {
2046 const char_type* q = std::find(digit_table, digit_table + 10, c);
2047 if (q != digit_table + 10)
2048 {
2049 xtrc += static_cast<char>('0' + (q - digit_table));
2050 saw_digit = true;
2051 ++group_len;
2052 }
2053 else if ((c == m_in.lower_e || c == m_in.upper_e)
2054 && !saw_exp && saw_digit)
2055 {
2056 // The exponent ends the integer part too, so an open tally
2057 // is closed here for the same reason as at a decimal point
2058 // -- unless a point already did it.
2059 if (!groups_seen.empty() && !saw_point && !close_group())
2060 {
2061 xtrc.clear();
2062 break;
2063 }
2064 xtrc += 'e';
2065 saw_exp = true;
2066
2067 if (++beg == end)
2068 {
2069 at_end = true;
2070 break;
2071 }
2072 c = *beg;
2073 if (!take_sign(c))
2074 // No exponent sign after all, so `c` is already the next
2075 // character to judge: go round again without advancing.
2076 continue;
2077 }
2078 else
2079 break;
2080 }
2081
2082 advance();
2083 }
2084
2085 bool success = true;
2086 // A grouping check only applies to input that actually used grouping.
2087 // When it did, the final group still has to be closed -- unless a point
2088 // or an exponent closed it already.
2089 if (!groups_seen.empty())
2090 {
2091 if (!saw_point && !saw_exp && !close_group())
2092 success = false;
2093 else
2094 success = FacetHelper::verify_grouping(m_grouping, groups_seen);
2095 }
2096
2097 return std::pair(success, beg);
2098 }

2139 template <typename TValue>
2140 bool convert_to_v(const char* s, TValue& v) const
2141 {
2142 // The parse happens on the carrier, and the narrowing back to TValue is deferred to
2143 // the end, where the range has been checked. See @ref float_carrier_t.
2144 using TCarrier = float_carrier_t<TValue>;
2145 static_assert(represents_v<TCarrier, TValue>,
2146 "no standard floating-point type represents TValue exactly");
2147
2148 TCarrier parsed{};
2149 char* parse_end = nullptr;
2150
2151 {
2152 clocale_wrapper inter_locale("C");
2153 clocale_user guard(inter_locale);
2154
2155 if constexpr (std::is_same_v<TCarrier, float>)
2156 parsed = strtof(s, &parse_end);
2157 else if constexpr (std::is_same_v<TCarrier, double>)
2158 parsed = strtod(s, &parse_end);
2159 else
2160 parsed = strtold(s, &parse_end);
2161 }
2162
2163 // Decide the semantic result before committing it to the caller. A malformed
2164 // field becomes zero, overflow becomes the matching finite extreme, and NaN
2165 // remains a successful conversion just as it was before this refactoring.
2166 if (parse_end == s || *parse_end != '\0')
2167 {
2168 v = TValue(0);
2169 return false;
2170 }
2171
2172 const TValue limit = std::numeric_limits<TValue>::max();
2173 const auto bound = static_cast<TCarrier>(limit);
2174 if (std::isinf(parsed) || parsed > bound || parsed < -bound)
2175 {
2176 v = std::signbit(parsed) ? -limit : limit;
2177 return false;
2178 }
2179
2180 v = static_cast<TValue>(parsed);
2181 return true;
2182 }

2219 static bool atoms_pairwise_distinct(const CharT* p, std::size_t n)
