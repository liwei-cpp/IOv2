458 template <typename TIter, std::sentinel_for<TIter> TSent>
459 TIter get(TIter beg, TSent end, ios_base<char_type>& io, bool& v) const
460 {
461 bool success = true;
462
463 if (!(io.flags() & ios_defs::boolalpha))
464 {
465 // Parse bool values as long.
466 long l = -1;
467 std::tie(success, beg) = extract_int(beg, end, io, l);
468
469 if (l == 0 || l == 1) v = bool(l);
470 else
471 {
472 // _GLIBCXX_RESOLVE_LIB_DEFECTS
473 // 23. Num_get overflow result.
474 v = true;
475 success = false;
476 }
477 }
478 else
479 {
480 // Parse bool values as alphanumeric.
481 bool testf = true;
482 bool testt = true;
483 bool donef = (m_false_name.empty());
484 bool donet = (m_true_name.empty());
485 std::size_t n = 0;
486 while (!donef || !donet)
487 {
488 if (beg == end)
489 break;
490
491 const CharT c = *beg;
492
493 if (!donef)
494 testf = c == m_false_name[n];
495
496 if (!testf && donet) break;
497
498 if (!donet)
499 testt = c == m_true_name[n];
500
501 if (!testt && donef) break;
502
503 if (!testt && !testf) break;
504
505 ++n;
506 ++beg;
507
508 donef = !testf || n >= m_false_name.size();
509 donet = !testt || n >= m_true_name.size();
510 }
511 if (testf && n == m_false_name.size() && n)
512 {
513 v = false;
514 if (testt && n == m_true_name.size())
515 success = false;
516 }
517 else if (testt && n == m_true_name.size() && n)
518 {
519 v = true;
520 }
521 else
522 {
523 // _GLIBCXX_RESOLVE_LIB_DEFECTS
524 // 23. Num_get overflow result.
525 v = false;
526 success = false;
527 }
528 }
529
530 if (!success) throw stream_error("numeric::get fail: parse boolean fail");
531 return beg;

741 template <typename TIter, typename TValue>
742 TIter insert_float(TIter s, ios_base<char_type>& io, TValue v, char mod) const
743 {
744 // precision() is now a bounded uint8_t (0-255); there is no negative /
745 // out-of-range sentinel to normalise here.
746 const std::streamsize prec = io.precision();
747 const int max_digits = std::numeric_limits<TValue>::digits10;
748
749 // Consume the field width up front, before any allocation or conversion
750 // that can throw: width() is one-shot and a stale value must not survive
751 // onto the stream if we leave by an exception. Used for padding below.
752 const std::size_t w = io.width();
753 io.width(0);
754
755 // [22.2.2.2.2] Stage 1, numeric conversion to character.
756 std::size_t len = 0;
757 std::array<char, 16> fbuf{};
758 format_float_(io.flags(), fbuf.data(), mod);
759
760 const ios_defs::fmtflags fltfield = io.flags() & ios_defs::floatfield;
761
762 // Initial buffer size estimate (GCC style).
763 // For non-fixed fields, max_digits * 3 is a safe heuristic.
764 std::size_t cs_size = static_cast<std::size_t>(max_digits) * 3 + 32;
765 if (fltfield == ios_defs::fixed)
766 cs_size = static_cast<std::size_t>(std::numeric_limits<TValue>::max_exponent10) + static_cast<std::size_t>(prec) + 32;
767
768 // Cap initial allocation to a reasonable size (e.g., 2048) to avoid huge initial pressure.
769 if (cs_size > 2048) cs_size = 2048;
770
771 std::vector<char> vec_cs(cs_size);
772
773 {
774 clocale_wrapper inter_locale("C");
775 clocale_user guard(inter_locale);
776
777 auto do_snprintf = [&](char* buf, std::size_t size) {
778 if (fltfield == (ios_defs::fixed | ios_defs::scientific))
779 return snprintf(buf, size, fbuf.data(), v);
780 else
781 return snprintf(buf, size, fbuf.data(), static_cast<int>(prec), v);
782 };
783
784 // snprintf returns int. Trap negative (encoding error) before
785 // promoting to size_t — otherwise -1 becomes SIZE_MAX and
786 // bypasses every downstream bound check.
787 int n = do_snprintf(vec_cs.data(), cs_size);
788 if (n < 0)
789 throw stream_error("numeric::put fail: floating-point conversion failed");
790 len = static_cast<std::size_t>(n);
791
792 // If buffer was too small, snprintf returns required length.
793 if (len >= cs_size)
794 {
795 cs_size = len + 1;
796 vec_cs.resize(cs_size);
797 n = do_snprintf(vec_cs.data(), cs_size);
798 if (n < 0)
799 throw stream_error("numeric::put fail: floating-point conversion failed");
800 len = static_cast<std::size_t>(n);
801 }
802 }
803
804 // len == 0 is also treated as failure: every well-formed numeric
805 // print (including 0, "inf", "nan") emits at least one character,
806 // and the downstream path assumes vec_ws(len) has at least one
807 // element before dereferencing vec_ws.data().
808 if (len == 0 || len >= cs_size)
809 throw stream_error("numeric::put fail: floating-point conversion failed");
810
811 const char* cs = vec_cs.data();
812
813 // [22.2.2.2.2] Stage 2, convert to char_type, using correct
814 // numeric_conf.decimal_point() values for '.' and adding grouping.
815 std::vector<CharT> vec_ws(len);
816 CharT* ws = vec_ws.data();
817 m_ctype->widen_seq(cs, cs + len, ws);
818
819 // Replace decimal point.
820 const char* p = std::find(cs, cs + len, '.');
821 CharT* wp = nullptr;
822 if (p != cs + len)
823 {
824 wp = ws + (p - cs);
825 *wp = m_decimal_point;
826 }
827
828 // Add grouping, if necessary.
829 if ((!m_grouping.empty())
830 && (wp || len < 3u || (cs[1] <= '9' && cs[2] <= '9' && cs[1] >= '0' && cs[2] >= '0')))
831 {
832 // Grouping can add (almost) as many separators as the
833 // number of digits, but no more.
834 std::vector<CharT> vec_ws2(len * 2);
835 CharT* ws2 = vec_ws2.data();
836
837 std::size_t off = 0;
838 if (cs[0] == '-' || cs[0] == '+')
839 {
840 off = 1;
841 ws2[0] = ws[0];
842 len -= 1;
843 }
844
845 group_float(m_grouping, m_thousands_sep, wp,
846 ws2 + off, ws + off, len);
847 len += off;
848 std::swap(vec_ws, vec_ws2);
849 ws = vec_ws.data();
850 }
851
852 // Pad.
853 if (std::cmp_greater(w, len))
854 {
855 if (w - len > ios_defs::max_pad_count)
856 throw stream_error("numeric put fail: fill count exceeds max_pad_count");
857
858 std::vector<CharT> vec_ws3(w);
859 CharT* ws3 = vec_ws3.data();
860 bool startSign = (ws[0] == m_out_atoms[s_ominus]) || (ws[0] == m_out_atoms[s_oplus]);
861 bool start0x = (ws[0] == m_out_atoms[s_odigits]) && (len > 1u) &&
862 ((ws[1] == m_out_atoms[s_ox]) || (ws[1] == m_out_atoms[s_oX]));
863 const ios_defs::fmtflags adjust = io.flags() & ios_defs::adjustfield;
864 pad(io.fill(), w, adjust, ws3, ws, len, startSign, start0x);
865
866 std::swap(vec_ws, vec_ws3);
867 ws = vec_ws.data();
868 }
869
870 // [22.2.2.2.2] Stage 4.
871 // Write resulting, fully-formatted string to output iterator.
872 return std::copy(ws, ws + len, s);
873 }

906 template <typename TIter, typename TValue>
907 TIter insert_int(TIter s, ios_base<char_type>& io, TValue v) const
908 {
909 using unsigned_type = std::make_unsigned_t<TValue>;
910
911 const ios_defs::fmtflags flags = io.flags();
912
913 // Consume the field width up front, before any allocation that can throw:
914 // width() is one-shot and a stale value must not survive onto the stream
915 // if we leave by an exception. Used for padding below.
916 const std::size_t w = io.width();
917 io.width(0);
918
919 // Long enough to hold hex, dec, and octal representations.
920 const auto ilen = 5 * sizeof(TValue);
921 std::vector<char_type> cs_vec(ilen);
922 char_type* cs = cs_vec.data();
923
924 // [22.2.2.2.2] Stage 1, numeric conversion to character.
925 // Result is returned right-justified in the buffer.
926 const ios_defs::fmtflags basefield = flags & ios_defs::basefield;
927 const bool dec = (basefield != ios_defs::oct && basefield != ios_defs::hex);
928 const unsigned_type u = ((v > 0 || !dec) ? unsigned_type(v)
929 : -unsigned_type(v));
930 auto len = static_cast<std::size_t>(int_to_char(cs + ilen, u, flags, dec));
931 cs += ilen - len;
932
933 // Add grouping, if necessary.
934 if (!m_grouping.empty())
935 {
936 // Grouping can add (almost) as many separators as the number
937 // of digits + space is reserved for numeric base or sign.
938 std::vector<char_type> cs_vec2((ilen + 1) * 2);
939 char_type* cs2 = cs_vec2.data() + 2;
940 len = static_cast<std::size_t>(FacetHelper::add_grouping(cs2, m_thousands_sep, m_grouping, cs, cs + len) - cs2);
941 std::swap(cs_vec, cs_vec2);
942 cs = cs_vec.data() + 2;
943 }
944
945 // Complete Stage 1, prepend numeric base or sign.
946 if (dec)
947 { // Decimal.
948 if (v >= 0)
949 {
950 if (bool(flags & ios_defs::showpos) && std::is_signed_v<TValue>)
951 *--cs = m_out_atoms[s_oplus], ++len;
952 }
953 else
954 *--cs = m_out_atoms[s_ominus], ++len;
955 }
956 else if (bool(flags & ios_defs::showbase) && v)
957 {
958 if (basefield == ios_defs::oct)
959 *--cs = m_out_atoms[s_odigits], ++len;
960 else
961 {
962 // 'x' or 'X'
963 const bool uppercase = flags & ios_defs::uppercase;
964 *--cs = m_out_atoms[s_ox + uppercase];
965 // '0'
966 *--cs = m_out_atoms[s_odigits];
967 len += 2;
968 }
969 }
970
971 // Pad.
972 if (std::cmp_greater(w, len))
973 {
974 if (w - len > ios_defs::max_pad_count)
975 throw stream_error("numeric put fail: fill count exceeds max_pad_count");
976
977 std::vector<char_type> cs_vec3(w);
978 char_type* cs3 = cs_vec3.data();
979 bool startSign = (cs[0] == m_out_atoms[s_ominus]) || (cs[0] == m_out_atoms[s_oplus]);
980 bool start0x = (cs[0] == m_out_atoms[s_odigits]) && (len > 1u) &&
981 ((cs[1] == m_out_atoms[s_ox]) || (cs[1] == m_out_atoms[s_oX]));
982 const ios_defs::fmtflags adjust = io.flags() & ios_defs::adjustfield;
983 pad(io.fill(), w, adjust, cs3, cs, len, startSign, start0x);
984 std::swap(cs_vec, cs_vec3);
985 cs = cs_vec.data();
986 }
987
988 // [22.2.2.2.2] Stage 4.
989 // Write resulting, fully-formatted string to output iterator.
990 return std::copy(cs, cs + len, s);
991 }

1020 template <typename TValue>
1021 int int_to_char(CharT* bufend, TValue v, ios_defs::fmtflags flags, bool dec) const
1022 {
1023 auto buf = bufend;
1024 if (dec)
1025 { // Decimal.
1026 do // NOLINT(cppcoreguidelines-avoid-do-while)
1027 {
1028 *--buf = m_out_atoms[(v % 10) + s_odigits];
1029 v /= 10;
1030 } while (v != 0);
1031 }
1032 else if ((flags & ios_defs::basefield) == ios_defs::oct)
1033 { // Octal.
1034 do // NOLINT(cppcoreguidelines-avoid-do-while)
1035 {
1036 *--buf = m_out_atoms[(v & 0x7) + s_odigits];
1037 v >>= 3;
1038 } while (v != 0);
1039 }
1040 else
1041 { // Hex.
1042 const bool uppercase = flags & ios_defs::uppercase;
1043 const int case_offset = uppercase ? s_oudigits : s_odigits;
1044 do // NOLINT(cppcoreguidelines-avoid-do-while)
1045 {
1046 *--buf = m_out_atoms[(v & 0xf) + case_offset];
1047 v >>= 4;
1048 } while (v != 0);
1049 }
1050 return bufend - buf;
1051 }

1143 [[nodiscard]] bool fill_alters_reading_(char_type fill, bool leads_digits,
1144 bool trails_value, bool negative) const
1145 {
1146 const char_type* const digits = m_out_atoms.data() + s_odigits;
1147 if (std::find(digits, digits + 10, fill) != digits + 10)
1148 return !(fill == digits[0] && leads_digits);
1149
1150 // Past the value nothing but a digit can still be read into it.
1151 if (trails_value)
1152 return false;
1153
1154 if (fill == m_decimal_point)
1155 return true;
1156 if (fill == m_out_atoms[s_ominus])
1157 return !negative;
1158 if (fill == m_out_atoms[s_oplus])
1159 return negative;
1160 return false;
1161 }

1193 void pad(char_type fill, std::size_t w, ios_defs::fmtflags adjust,
1194 char_type* new_buf, const char_type* cs, std::size_t& len,
1195 bool startSign, bool start0x) const
1196 {
1197 // Fill is about to be written, so this is where it gets vetted: a fill character
1198 // that would change the number this field reads as is rejected outright. Where
1199 // the run lands decides what is safe, and `pad_impl_` below fixes that: `left`
1200 // appends it after the value; `internal`, and `right` on content that starts with
1201 // a digit, put it directly in front of the digits; `right` on content that starts
1202 // with a sign or `0x` leaves that prefix between the fill and the digits.
1203 // The check belongs here rather than at the top of the insert functions because
1204 // `fill` is sticky stream state: a stream carrying setfill('1') must keep working
1205 // for every output whose width leaves nothing to pad.
1206 const bool trails_value = (adjust == ios_defs::left);
1207 const bool leads_digits = !trails_value
1208 && (adjust == ios_defs::internal || !(startSign || start0x));
1209 if (fill_alters_reading_(fill, leads_digits, trails_value,
1210 cs[0] == m_out_atoms[s_ominus]))
1211 throw stream_error("numeric put fail: fill would change the value the field reads as");
1212
1213 // [22.2.2.2.2] Stage 3.
1214 // If necessary, pad.
1215 pad_impl_(adjust, fill, new_buf, cs, w, len, startSign, start0x);
1216 len = w;
1217 }

1486 if (basefield == 0) base = 8;

1501 break;
1502
1503 if (++beg != end)
1504 {
1505 c = *beg;
1506 if (!found_zero) break;
1507 }
1508 else
1509 testeof = true;

1513 // base digits as valid input.

1516 // Extract.
1517 std::vector<uint8_t> found_grouping;
1518 if (!m_grouping.empty()) found_grouping.reserve(32);
1519 bool testfail = false;
1520 bool testoverflow = false;
1521 const unsigned_type max_val = (negative && std::is_signed_v<TValue>)
1522 ? -static_cast<unsigned_type>(std::numeric_limits<TValue>::min()) : std::numeric_limits<TValue>::max();
1523 const unsigned_type smax = max_val / base;
1524 unsigned_type result = 0;
1525 int digit = 0;
1526 const char_type* lit_zero = m_in_atoms.data() + s_izero;
1527
1528 while (!testeof)
1529 {
1530 // According to 22.2.2.1.2, p8-9, first look for thousands_sep
1531 // and decimal_point.
1532 if (!m_grouping.empty() && c == m_thousands_sep)
1533 {

1539 // uint8_t. A single group whose size cannot fit in that
1540 // byte is rejected rather than silently truncated.
1541 if (std::cmp_greater(sep_pos, std::numeric_limits<uint8_t>::max()))

1543 testfail = true;
1544 break;
1545 }
1546 found_grouping.push_back(static_cast<uint8_t>(sep_pos));
1547 sep_pos = 0;
1548 }
1549 else
1550 {
1551 testfail = true;
1552 break;
1553 }
1554 }
1555 else if (c == m_decimal_point) break;
1556 else

1558 const char_type* q = std::find(lit_zero, lit_zero + len, c);
1559 if (q == lit_zero + len) break;

1562 if (digit > 15) digit -= 6;
1563 if (result > smax) testoverflow = true;
1564 else

1567 testoverflow |= result > max_val - digit;
1568 result += digit;

1570 }
1571 }
1572
1573 if (++beg != end) c = *beg;

1575 }

1583 // exceeds the uint8_t storage in found_grouping.
1584 if (std::cmp_greater(sep_pos, std::numeric_limits<uint8_t>::max()))
1585 {
1586 testfail = true;
1587 }
1588 else
1589 {
1590 found_grouping.push_back(static_cast<uint8_t>(sep_pos));
1591 success = FacetHelper::verify_grouping(m_grouping, found_grouping);
1592 }
1593 }

1595 // LWG 23 (Num_get overflow result): when the field overflows,
1596 // v must be set to numeric_limits::max() / min() with failbit.
1597 //

1599 // digit-accumulation loop sets testoverflow, ++sep_pos is skipped

1601 // well-formed thousands_sep then fails the grouping check and sets
1602 // testfail — but that testfail is a side effect of the overflow

1607 // into uint32_t under grouping "\3") to v = 0 instead of v = max,
1608 // contradicting LWG 23. We diverge from libstdc++'s ordering here
1609 // (which has the same latent issue) to keep the overflow signal
1610 // dominant whenever it fires.
1611 if (testoverflow)
1612 {
1613 if (negative && std::is_signed_v<TValue>)
1614 v = std::numeric_limits<TValue>::min();
1615 else
1616 v = std::numeric_limits<TValue>::max();
1617 success = false;
1618 }
1619 else if ((!sep_pos && !found_zero && !found_grouping.size()) || testfail)
1620 {
1621 v = 0;
1622 success = false;
1623 }
1624 else
1625 v = negative ? -result : result;
1626
1627 return std::pair(success, beg);
