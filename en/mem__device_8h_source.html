411 CharT* put_buf(std::size_t len)
412 {
413 if (len > m_str.max_size() - m_next_pos)
414 throw device_error("mem_device::put_buf fail: size overflow");
415
416 m_put_buf_checkpoint = m_str.size();
417 const std::size_t needed = m_next_pos + len;
418 if (needed > m_str.size())
419 {
420 if (needed > m_str.capacity())
421 {
422 const std::size_t cap = m_str.capacity();
423 m_str.reserve(std::max(needed, cap + cap / 2));
424 }
425 m_str.resize(needed);
426 }
427 CharT* res = m_str.data() + m_next_pos;
428 m_next_pos = needed;
429 return res;
430 }

455 void put_rollback(std::size_t len)
456 {
457 if (!m_put_buf_checkpoint.has_value())
458 throw device_error("mem_device::put_rollback fail: no active put_buf");
459 if (len == 0)
460 throw device_error("mem_device::put_rollback fail: length cannot be zero");
461 if (m_next_pos < len)
462 throw device_error("mem_device::put_rollback fail: rollback length too large");
463
464 const std::size_t saved_size = m_put_buf_checkpoint.value();
465 m_put_buf_checkpoint.reset();
466
467 m_next_pos -= len;

469 }
