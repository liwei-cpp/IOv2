402 CharT* put_buf(std::size_t len)
403 {
404 if (len > m_str.max_size() - m_next_pos)
405 throw device_error("mem_device::put_buf fail: size overflow");
406
407 m_put_buf_checkpoint = m_str.size();
408 const std::size_t needed = m_next_pos + len;
409 if (needed > m_str.size())
410 {
411 if (needed > m_str.capacity())
412 {
413 const std::size_t cap = m_str.capacity();
414 m_str.reserve(std::max(needed, cap + cap / 2));
415 }
416 m_str.resize(needed);
417 }
418 CharT* res = m_str.data() + m_next_pos;
419 m_next_pos = needed;
420 return res;
421 }

446 void put_rollback(std::size_t len)
447 {
448 if (!m_put_buf_checkpoint.has_value())
449 throw device_error("mem_device::put_rollback fail: no active put_buf");
450 if (len == 0)
451 throw device_error("mem_device::put_rollback fail: length cannot be zero");
452 if (m_next_pos < len)
453 throw device_error("mem_device::put_rollback fail: rollback length too large");
454
455 const std::size_t saved_size = m_put_buf_checkpoint.value();
456 m_put_buf_checkpoint.reset();
457
458 m_next_pos -= len;
459 m_str.resize(std::max(m_next_pos, saved_size));
460 }
