48// Opens `p` with an fopen-style `mode`; nullptr with errno set on failure.
49inline FILE* file_open(const std::filesystem::path& p, const char* mode)
50{
51 const std::string_view m(mode);
52 const bool binary = m.find('b') != std::string_view::npos;
53 const bool update = m.find('+') != std::string_view::npos;
54
55 if (m.find('x') == std::string_view::npos)
56 {
57 const std::wstring wmode(m.begin(), m.end());
58 return _wfsopen(p.c_str(), wmode.c_str(), _SH_DENYNO);
59 }
60
61 const int oflag = _O_CREAT | _O_EXCL | (update ? _O_RDWR : _O_WRONLY)
62 | (binary ? _O_BINARY : _O_TEXT);
63 int fd = -1;
64 if (_wsopen_s(&fd, p.c_str(), oflag, _SH_DENYNO, _S_IREAD | _S_IWRITE) != 0)
65 return nullptr;
66
67 FILE* f = _fdopen(fd, update ? (binary ? "w+b" : "w+") : (binary ? "wb" : "w"));
68 if (f == nullptr)
69 {
70 // The file now exists, and stays: deleting it by name could hit another one.
71 const int saved = errno;
72 _close(fd);
73 errno = saved;
74 }
75 return f;
76}

88// The path in the active code page, as narrow text is read on Windows. Characters
89// the code page lacks become '?': path::string() would throw for them instead.
90inline std::string path_for_message(const std::filesystem::path& p)
91{
92 const std::wstring& w = p.native();
93 if (w.empty())
94 return {};
95 const int wlen = static_cast<int>(w.size());
96 const int len = WideCharToMultiByte(win_cp_acp, 0, w.data(), wlen, nullptr, 0, nullptr, nullptr);
97 if (len <= 0)
98 return "?";
99 std::string res(static_cast<std::size_t>(len), '\0');
100 WideCharToMultiByte(win_cp_acp, 0, w.data(), wlen, res.data(), len, nullptr, nullptr);
101 return res;
102}
