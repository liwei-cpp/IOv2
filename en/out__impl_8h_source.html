353 void reset()
354 {
355 std::lock_guard guard(this->io_mutex());
356 this->clear();
357 this->exceptions(ios_defs::goodbit);
358
359 auto detached = m_streambuf.detach();
360
361 try { detached.first.dflush(); }
362 catch (...)
363 {
364 if (!detached.second)
365 detached.second = std::current_exception();
366 }
367
368 try { m_streambuf.attach(); }
369 catch (...) { this->handle_exception(std::current_exception()); }
370
371 if (detached.second) this->handle_exception(detached.second);
372 }
