396 bool sync_with_stdio(bool sync = true)
397 {
398 if (!sync)
399 return m_sync_with_stdio.exchange(false);
400
401 // Flag and hand-over under one lock, so a caller that returns can rely on the
402 // bytes buffered before the call being in stdio's hands. The hand-over runs even
403 // when already synchronized: a failed write leaves bytes behind in that mode too.
404 std::lock_guard guard(this->io_mutex());
405 std::exception_ptr hand_over_error;
406 try
407 {
408 // The sentry's operation, not the stream-level flush(): no fflush of bytes
409 // that are not this stream's, and nothing at all on an empty buffer. No good()
410 // gate: a stream in a failed state may still hold bytes (root_cvt::flush keeps
411 // them for a retry). A retry that fails again re-reports the bit already set.
412 m_channel.flush();
413 }
414 catch (...)
415 {
416 hand_over_error = std::current_exception();
417 }
418
419 // Only now true: synced_with_stdio() reads without the lock, and a thread that sees
420 // true may printf at once. Reported after the flip, which an armed mask would skip.
421 const bool old_sync_state = m_sync_with_stdio.exchange(true);
422 if (hand_over_error)
423 this->handle_exception(hand_over_error);
424 return old_sync_state;
425 }

453 [[nodiscard]] bool synced_with_stdio() const noexcept

562 std::string switch_code(const std::string& new_code)
563 requires std::is_same_v<TChar, wchar_t>
564 {
565 // One lock over the query and the switch, so concurrent switches each return the
566 // code the one before them set.
567 std::lock_guard guard(this->io_mutex());
568 auto res = code();
569 if (res != new_code)
570 {
571 code_cvt_switch acc(new_code);
572 this->adjust(acc);
573 }
574 return res;

734 void reset()
735 {
736 std::lock_guard guard(this->io_mutex());
737 this->clear();
738 this->exceptions(ios_defs::goodbit);
739
740 auto detached = m_channel.detach();
741
742 try { detached.first.dflush(); }
743 catch (...)
744 {
745 if (!detached.second)
746 detached.second = std::current_exception();
747 }
748
749 try { m_channel.attach(); }
750 catch (...) { this->handle_exception(std::current_exception()); }
751
752 if (detached.second) this->handle_exception(detached.second);
753 }
