206inline std::vector<timeio_raw_era> timeio_read_eras()
207{
208 std::vector<timeio_raw_era> items;
209#if defined(__GLIBC__)
210 // NOLINTBEGIN(cppcoreguidelines-pro-type-reinterpret-cast)
211 struct era_wire_header
212 {
213 int32_t direction_marker;
214 int32_t offset;
215 int32_t from_year;
216 int32_t from_month;
217 int32_t from_day;
218 int32_t to_year;
219 int32_t to_month;
220 int32_t to_day;
221 };
222 static_assert(sizeof(era_wire_header) == 8 * sizeof(int32_t));
223
224 const auto record_count = static_cast<int32_t>(
225 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
226 if (record_count <= 0)
227 return items;
228
229 items.reserve(static_cast<std::size_t>(record_count));
230 const char* cursor = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
231 for (int32_t index = 0; index < record_count; ++index)
232 {
233 const char* const record = cursor;
234 era_wire_header header{};
235 std::memcpy(&header, cursor, sizeof header);
236 cursor += sizeof header;
237
238 timeio_raw_era value{};
239 constexpr int32_t epoch = 1900;
240 constexpr int32_t last_relative_year = std::numeric_limits<int32_t>::max() - epoch;
241 value.from_year = header.from_year > last_relative_year
242 ? std::numeric_limits<int32_t>::max()
243 : header.from_year + epoch;
244 value.from_month = static_cast<uint8_t>(header.from_month + 1);
245 value.from_day = static_cast<uint8_t>(header.from_day);
246
247 const bool open_ended = header.to_year > last_relative_year;
248 value.to_year = open_ended
249 ? std::numeric_limits<int32_t>::max()
250 : header.to_year + epoch;
251 value.to_month = open_ended ? uint8_t{12}
252 : static_cast<uint8_t>(header.to_month + 1);
253 value.to_day = open_ended ? uint8_t{31}
254 : static_cast<uint8_t>(header.to_day);
255 value.offset = header.offset;
256 value.direction = header.direction_marker == static_cast<int32_t>('+') ? '+' : '-';
257
258 value.name = cursor;
259 cursor += std::strlen(cursor) + 1;
260 value.format = cursor;
261 cursor += std::strlen(cursor) + 1;
262
263 constexpr std::size_t wire_alignment = sizeof(int32_t);
264 const auto narrow_size = static_cast<std::size_t>(cursor - record);
265 cursor += (wire_alignment - (narrow_size % wire_alignment)) % wire_alignment;
266
267 const auto* wide_name = reinterpret_cast<const wchar_t*>(cursor);
268 cursor += (std::wcslen(wide_name) + 1) * sizeof(wchar_t);
269 const auto* wide_format = reinterpret_cast<const wchar_t*>(cursor);
270 cursor += (std::wcslen(wide_format) + 1) * sizeof(wchar_t);
271
272 items.push_back(std::move(value));
273 }
274 // NOLINTEND(cppcoreguidelines-pro-type-reinterpret-cast)
275#endif
276 return items;
277}

279// The alternative digits of the calling thread's locale, from 0, at most 100. glibc stores them
280// as consecutive NUL-terminated strings ended by an empty one, trusted like the era table (the
281// walk is bounded only by that and the cap); elsewhere (macOS) there are none. Call it under a
282// clocale_user.
283inline std::vector<std::string> timeio_read_alt_digits()
284{
285 std::vector<std::string> res;
286#if defined(__GLIBC__)
287 const char* ptr = nl_langinfo(ALT_DIGITS);
288 for (std::size_t i = 0; ptr != nullptr && i < 100 && *ptr != '\0'; ++i)
289 {
290 res.emplace_back(ptr);
291 ptr += res.back().size() + 1;
292 }
293#endif
294 return res;
295}

297// The date and time data of the calling thread's current locale. Call it under a clocale_user.
298inline timeio_raw_data timeio_read_data()
299{
300 static constexpr std::array<nl_item, 7> day_items =
301 {DAY_1, DAY_2, DAY_3, DAY_4, DAY_5, DAY_6, DAY_7};
302 static constexpr std::array<nl_item, 7> abbr_day_items =
303 {ABDAY_1, ABDAY_2, ABDAY_3, ABDAY_4, ABDAY_5, ABDAY_6, ABDAY_7};
304 static constexpr std::array<nl_item, 12> month_items =
305 {MON_1, MON_2, MON_3, MON_4, MON_5, MON_6, MON_7, MON_8, MON_9, MON_10, MON_11, MON_12};
306 static constexpr std::array<nl_item, 12> abbr_month_items =
307 {ABMON_1, ABMON_2, ABMON_3, ABMON_4, ABMON_5, ABMON_6,
308 ABMON_7, ABMON_8, ABMON_9, ABMON_10, ABMON_11, ABMON_12};
309
310 timeio_raw_data res;
311 for (std::size_t i = 0; i < 7; ++i)
312 {
313 res.day[i] = timeio_item(day_items[i]);
314 res.abbr_day[i] = timeio_item(abbr_day_items[i]);
315 }
316 for (std::size_t i = 0; i < 12; ++i)
317 {
318 res.month[i] = timeio_item(month_items[i]);
319 res.abbr_month[i] = timeio_item(abbr_month_items[i]);
320 }
321 res.am = timeio_item(AM_STR);
322 res.pm = timeio_item(PM_STR);
323 res.date_format = timeio_item(D_FMT);
324 res.era_date_format = timeio_item(ERA_D_FMT);
325 res.time_format = timeio_item(T_FMT);
326 res.era_time_format = timeio_item(ERA_T_FMT);
327 res.date_time_format = timeio_item(D_T_FMT);
328 res.era_date_time_format = timeio_item(ERA_D_T_FMT);
329 res.am_pm_format = timeio_item(T_FMT_AMPM);
330 res.alt_digits = timeio_read_alt_digits();
331 res.eras = timeio_read_eras();
332 return res;
333}
