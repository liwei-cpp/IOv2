217 template <bool Saturate = false>
218 auto get_buf(std::size_t to_max)
219 {
220 if (to_max == 0)
221 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
222 if (to_max > m_buffer.size())
223 throw cvt_error("cvt_reader::get_buf fail, read size too large");
224
225 const auto rollback_size = m_end_pos - m_cur_pos;
226 if (to_max <= rollback_size)
227 {
228 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
229 m_cur_pos += to_max;
230 if constexpr (!Saturate)
231 return res;
232 else
233 return res.first;
234 }
235
236 if (to_max > m_buffer.size() - m_cur_pos)
237 {
238 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
239 m_cur_pos = 0;
240 m_end_pos = rollback_size;
241 }
242 const std::size_t aim_size = to_max - rollback_size;
243 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
244 m_end_pos += read_size;
245
246 if constexpr (Saturate)
247 {
248 while (read_size < aim_size)
249 {
250 const std::size_t new_aim_size = aim_size - read_size;
251 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
252 if (new_read_size == 0)
253 throw cvt_error("get_buf<Saturate> fail: meet eos");
254 m_end_pos += new_read_size;
255 read_size += new_read_size;
256 }
257 }
258
259 if constexpr (!Saturate)
260 {
261 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
262 m_cur_pos = m_end_pos;
263 return res;
264 }
265 else
266 {
267 const char_type* res = m_buffer.data() + m_cur_pos;
268 m_cur_pos = m_end_pos;
269 return res;
270 }
271 }

1544 template <typename Self>
1545 std::size_t get(this Self& self, internal_type* to, std::size_t to_max)
1546 requires requires(Self& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1547 t.get_main(r, data, len);
1548 }
1549 {
1550 // An error that get_main() hit after delivering characters was held back so that
1551 // the caller got those characters; it is reported now.
1552 if (self.m_get_error)
1553 std::rethrow_exception(std::exchange(self.m_get_error, nullptr));
1554
1555 // Read failures do not taint: they only advance the kernel cursor
1556 // and possibly leave a per-call mbstate undefined; the underlying
1557 // stream content is unchanged, and callers can reseek to recover.
1558 // We still honour any pre-existing taint so that a writer that
1559 // corrupted the stream cannot then read back a confused image.
1560 self.assert_not_tainted();
1561
1562 cvt_reader<KernelType> reader(self.m_kernel, self.m_tmp_io_buffer, self.m_rd_cur, self.m_rd_end);
1563
1564 if (!self.m_is_bos_done)
1565 {
1566 if (self.m_io_status == io_status::output)
1567 throw cvt_error("abs_cvt::get fail: cannot switch to input mode in the BOS phase");
1568 if (to_max == 0) return 0;
1569
1570 constexpr std::size_t ext_size = sizeof(external_type);
1571
1572 reader.reset(s_bos_chunk);
1573 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1574 char* dest_bytes = reinterpret_cast<char*>(to);
1575 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1576 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1577
1578 // BOS phase requires complete data: if the stream ends unexpectedly
1579 // before providing enough bytes, this is a format error and we throw.
1580 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1581
1582 // Read full external_type chunks
1583 while (dest_bytes + ext_size <= dest_bytes_end)
1584 {
1585 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1586
1587 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1588 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1589 dest_bytes += units_to_read_now * ext_size;
1590 }
1591
1592 // Read the final partial internal_type if needed
1593 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1594 if (final_remainder_bytes > 0)
1595 {
1596 const auto* src_buf = reader.template get_buf<true>(1);
1597 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1598 dest_bytes += final_remainder_bytes;
1599 }
1600
1601 return to_max;
1602 }
1603 else
1604 {
1605 if (self.m_io_status != io_status::input)
1606 {
1607 if constexpr (cvt_cpt::support_io_switch<Self>)
1608 self.switch_to_get();
1609 else
1610 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1611 }
1612 static_assert(std::is_same_v<decltype(self.get_main(reader, to, to_max)),
1613 std::pair<std::size_t, std::exception_ptr>>,
1614 "get_main() must return std::pair<std::size_t, std::exception_ptr>");
1615 static_assert(noexcept(self.get_main(reader, to, to_max)),
1616 "get_main() must be noexcept");
1617 auto [count, err] = self.get_main(reader, to, to_max);
1618 if (err)
1619 {
1620 if (count == 0)
1621 std::rethrow_exception(err);
1622 self.m_get_error = std::move(err);
1623 }
1624 return count;
1625 }
1626 }

1675 template <typename Self>
1676 void put(this Self& self, const internal_type* to, std::size_t to_size)
1677 requires requires(Self& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1678 { t.put_main(w, data, len) } -> std::same_as<void>;
1679 }
1680 {
1681 self.assert_not_tainted();
1682
1683 // IO-direction switching does not write bytes to the output stream:
1684 // a failed switch_to_put() leaves the stream unchanged, so the
1685 // converter is not corrupted and must not be tainted. Perform the
1686 // switch BEFORE entering the taint-on-exception region so that only
1687 // actual kernel writes can cause tainting.
1688 if (self.m_is_bos_done && self.m_io_status != io_status::output)
1689 {
1690 if constexpr (cvt_cpt::support_io_switch<Self>)
1691 self.switch_to_put();
1692 else
1693 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1694 }
1695 if (!self.m_is_bos_done && self.m_io_status == io_status::input)
1696 throw cvt_error("abs_cvt::put fail: cannot switch to output mode in the BOS phase");
1697
1698 try
1699 {
1700 cvt_writer<KernelType> writer(self.m_kernel, self.m_tmp_io_buffer);
1701 if (!self.m_is_bos_done)
1702 {
1703 if (to_size == 0) return;
1704
1705 constexpr std::size_t ext_size = sizeof(external_type);
1706
1707 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1708 auto to_bytes = reinterpret_cast<const char*>(to);
1709 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1710 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1711
1712 // BOS phase writes complete external_type units. If the input data
1713 // is not a multiple of ext_size, pad the final unit with zeros.
1714
1715 writer.reset(s_bos_chunk);
1716 while (to_bytes + ext_size <= to_bytes_end)
1717 {
1718 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1719 auto ptr = writer.put_buf(dest_count);
1720 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1721 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1722 to_bytes += dest_count * ext_size;
1723 }
1724
1725 if (to_bytes < to_bytes_end)
1726 {
1727 // The while loop above guarantees remaining < ext_size.
1728 // The modulo is logically redundant but helps the compiler's
1729 // static analyzer prove the bound for memset size calculation.
1730 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1731 auto ptr = writer.put_buf(1);
1732 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1733 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1734 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1735 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1736 }
1737
1738 writer.commit();
1739 }
1740 else
1741 {
1742 // switch_to_put() was already called above if needed;
1743 // m_io_status is guaranteed to be output at this point.
1744 self.put_main(writer, to, to_size);
1745 // commit() is the sole responsibility of abs_cvt::put: derived
1746 // put_main implementations must NOT call writer.commit() themselves.
1747 // If put_main throws, this commit is skipped and what the writer
1748 // still stages is dropped at writer destruction (output already
1749 // handed to the kernel stays); the catch below taints the converter.
1750 writer.commit();
1751 }
1752 }
1753 catch (...)
1754 {
1755 self.m_is_tainted = true;
1756 throw;
1757 }
1758 }
