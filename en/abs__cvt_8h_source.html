217 template <bool Saturate = false>
218 auto get_buf(std::size_t to_max)
219 {
220 if (to_max == 0)
221 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
222 if (to_max > m_buffer.size())
223 throw cvt_error("cvt_reader::get_buf fail, read size too large");
224
225 const auto rollback_size = m_end_pos - m_cur_pos;
226 if (to_max <= rollback_size)
227 {
228 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
229 m_cur_pos += to_max;
230 if constexpr (!Saturate)
231 return res;
232 else
233 return res.first;
234 }
235
236 if (to_max > m_buffer.size() - m_cur_pos)
237 {
238 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
239 m_cur_pos = 0;
240 m_end_pos = rollback_size;
241 }
242 const std::size_t aim_size = to_max - rollback_size;
243 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
244 m_end_pos += read_size;
245
246 if constexpr (Saturate)
247 {
248 while (read_size < aim_size)
249 {
250 const std::size_t new_aim_size = aim_size - read_size;
251 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
252 if (new_read_size == 0)
253 throw cvt_error("get_buf<Saturate> fail: meet eos");
254 m_end_pos += new_read_size;
255 read_size += new_read_size;
256 }
257 }
258
259 if constexpr (!Saturate)
260 {
261 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
262 m_cur_pos = m_end_pos;
263 return res;
264 }
265 else
266 {
267 const char_type* res = m_buffer.data() + m_cur_pos;
268 m_cur_pos = m_end_pos;
269 return res;
270 }
271 }

1528 template <typename Self>
1529 std::size_t get(this Self& self, internal_type* to, std::size_t to_max)
1530 requires requires(Self& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1531 t.get_main(r, data, len);
1532 }
1533 {
1534 // An error that get_main() hit after delivering characters was held back so that
1535 // the caller got those characters; it is reported now.
1536 if (self.m_get_error)
1537 std::rethrow_exception(std::exchange(self.m_get_error, nullptr));
1538
1539 // Read failures do not taint: they only advance the kernel cursor
1540 // and possibly leave a per-call mbstate undefined; the underlying
1541 // stream content is unchanged, and callers can reseek to recover.
1542 // We still honour any pre-existing taint so that a writer that
1543 // corrupted the stream cannot then read back a confused image.
1544 self.assert_not_tainted();
1545
1546 cvt_reader<KernelType> reader(self.m_kernel, self.m_tmp_io_buffer, self.m_rd_cur, self.m_rd_end);
1547
1548 if (!self.m_is_bos_done)
1549 {
1550 if (to_max == 0) return 0;
1551
1552 constexpr std::size_t ext_size = sizeof(external_type);
1553
1554 reader.reset(s_bos_chunk);
1555 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1556 char* dest_bytes = reinterpret_cast<char*>(to);
1557 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1558 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1559
1560 // BOS phase requires complete data: if the stream ends unexpectedly
1561 // before providing enough bytes, this is a format error and we throw.
1562 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1563
1564 // Read full external_type chunks
1565 while (dest_bytes + ext_size <= dest_bytes_end)
1566 {
1567 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1568
1569 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1570 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1571 dest_bytes += units_to_read_now * ext_size;
1572 }
1573
1574 // Read the final partial internal_type if needed
1575 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1576 if (final_remainder_bytes > 0)
1577 {
1578 const auto* src_buf = reader.template get_buf<true>(1);
1579 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1580 dest_bytes += final_remainder_bytes;
1581 }
1582
1583 return to_max;
1584 }
1585 else
1586 {
1587 if (self.m_io_status != io_status::input)
1588 {
1589 if constexpr (cvt_cpt::support_io_switch<Self>)
1590 self.switch_to_get();
1591 else
1592 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1593 }
1594 static_assert(std::is_same_v<decltype(self.get_main(reader, to, to_max)),
1595 std::pair<std::size_t, std::exception_ptr>>,
1596 "get_main() must return std::pair<std::size_t, std::exception_ptr>");
1597 static_assert(noexcept(self.get_main(reader, to, to_max)),
1598 "get_main() must be noexcept");
1599 auto [count, err] = self.get_main(reader, to, to_max);
1600 if (err)
1601 {
1602 if (count == 0)
1603 std::rethrow_exception(err);
1604 self.m_get_error = std::move(err);
1605 }
1606 return count;
1607 }

1655 template <typename Self>
1656 void put(this Self& self, const internal_type* to, std::size_t to_size)
1657 requires requires(Self& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1658 { t.put_main(w, data, len) } -> std::same_as<void>;
1659 }
1660 {
1661 self.assert_not_tainted();
1662
1663 // IO-direction switching does not write bytes to the output stream:
1664 // a failed switch_to_put() leaves the stream unchanged, so the
1665 // converter is not corrupted and must not be tainted. Perform the
1666 // switch BEFORE entering the taint-on-exception region so that only
1667 // actual kernel writes can cause tainting.
1668 if (self.m_is_bos_done && self.m_io_status != io_status::output)
1669 {
1670 if constexpr (cvt_cpt::support_io_switch<Self>)
1671 self.switch_to_put();
1672 else
1673 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1674 }
1675
1676 try
1677 {
1678 cvt_writer<KernelType> writer(self.m_kernel, self.m_tmp_io_buffer);
1679 if (!self.m_is_bos_done)
1680 {
1681 if (to_size == 0) return;
1682
1683 constexpr std::size_t ext_size = sizeof(external_type);
1684
1685 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1686 auto to_bytes = reinterpret_cast<const char*>(to);
1687 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1688 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1689
1690 // BOS phase writes complete external_type units. If the input data
1691 // is not a multiple of ext_size, pad the final unit with zeros.
1692
1693 writer.reset(s_bos_chunk);
1694 while (to_bytes + ext_size <= to_bytes_end)
1695 {
1696 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1697 auto ptr = writer.put_buf(dest_count);
1698 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1699 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1700 to_bytes += dest_count * ext_size;
1701 }
1702
1703 if (to_bytes < to_bytes_end)
1704 {
1705 // The while loop above guarantees remaining < ext_size.
1706 // The modulo is logically redundant but helps the compiler's
1707 // static analyzer prove the bound for memset size calculation.
1708 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1709 auto ptr = writer.put_buf(1);
1710 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1711 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1712 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1713 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1714 }
1715
1716 writer.commit();
1717 }
1718 else
1719 {
1720 // switch_to_put() was already called above if needed;
1721 // m_io_status is guaranteed to be output at this point.
1722 self.put_main(writer, to, to_size);
1723 // commit() is the sole responsibility of abs_cvt::put: derived
1724 // put_main implementations must NOT call writer.commit() themselves.
1725 // If put_main throws, this commit is skipped and any uncommitted
1726 // buffer contents are dropped at writer destruction; the catch
1727 // below taints the converter.
1728 writer.commit();
1729 }
1730 }
1731 catch (...)
1732 {
1733 self.m_is_tainted = true;
1734 throw;
1735 }
1736 }
