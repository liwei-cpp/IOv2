1441 requires requires(CurrentType& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1442 { t.get_main(r, data, len) } -> std::same_as<std::size_t>;
1443 }
1444 {
1445 // Read failures do not taint: they only advance the kernel cursor
1446 // and possibly leave a per-call mbstate undefined; the underlying
1447 // stream content is unchanged, and callers can reseek to recover.
1448 // We still honour any pre-existing taint so that a writer that
1449 // corrupted the stream cannot then read back a confused image.
1450 assert_not_tainted();

1452 cvt_reader<KernelType> reader(m_kernel, m_tmp_io_buffer);
1453
1454 if (!m_is_bos_done)
1455 {
1456 if (to_max == 0) return 0;
1457
1458 constexpr std::size_t ext_size = sizeof(external_type);
1459
1460 reader.reset(s_bos_chunk);
1461 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1462 char* dest_bytes = reinterpret_cast<char*>(to);
1463 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1464 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1465
1466 // BOS phase requires complete data: if the stream ends unexpectedly
1467 // before providing enough bytes, this is a format error and we throw.
1468 // Unlike get_main() which may return partial data, BOS is all-or-nothing.

1470 // Read full external_type chunks
1471 while (dest_bytes + ext_size <= dest_bytes_end)
1472 {
1473 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1474
1475 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1476 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1477 dest_bytes += units_to_read_now * ext_size;
1478 }
1479
1480 // Read the final partial internal_type if needed
1481 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1482 if (final_remainder_bytes > 0)
1483 {
1484 const auto* src_buf = reader.template get_buf<true>(1);
1485 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1486 dest_bytes += final_remainder_bytes;
1487 }

1489 return to_max;
1490 }
1491 else

1497 else

1549 void put(const internal_type* to, std::size_t to_size)
1550 requires requires(CurrentType& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1551 { t.put_main(w, data, len) } -> std::same_as<void>;
1552 }
1553 {
1554 assert_not_tainted();
1555
1556 // IO-direction switching does not write bytes to the output stream:
1557 // a failed switch_to_put() leaves the stream unchanged, so the
1558 // converter is not corrupted and must not be tainted. Perform the
1559 // switch BEFORE entering the taint-on-exception region so that only
1560 // actual kernel writes can cause tainting.
1561 if (m_is_bos_done && m_io_status != io_status::output)
1562 {
1563 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1564 static_cast<CurrentType*>(this)->switch_to_put();
1565 else
1566 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1567 }
1568
1569 try
1570 {
1571 cvt_writer<KernelType> writer(m_kernel, m_tmp_io_buffer);
1572 if (!m_is_bos_done)
1573 {
1574 if (to_size == 0) return;
1575
1576 constexpr std::size_t ext_size = sizeof(external_type);
1577
1578 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1579 auto to_bytes = reinterpret_cast<const char*>(to);
1580 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1581 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1582
1583 // BOS phase writes complete external_type units. If the input data
1584 // is not a multiple of ext_size, pad the final unit with zeros.
1585
1586 writer.reset(s_bos_chunk);
1587 while (to_bytes + ext_size <= to_bytes_end)
1588 {
1589 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1590 auto ptr = writer.put_buf(dest_count);
1591 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1592 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1593 to_bytes += dest_count * ext_size;
1594 }
1595
1596 if (to_bytes < to_bytes_end)
1597 {
1598 // The while loop above guarantees remaining < ext_size.
1599 // The modulo is logically redundant but helps the compiler's
1600 // static analyzer prove the bound for memset size calculation.
1601 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1602 auto ptr = writer.put_buf(1);
1603 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1604 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1605 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1606 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1607 }
1608
1609 writer.commit();
1610 }
1611 else

1613 // switch_to_put() was already called above if needed;
1614 // m_io_status is guaranteed to be output at this point.
1615 static_cast<CurrentType*>(this)->put_main(writer, to, to_size);
1616 // commit() is the sole responsibility of abs_cvt::put: derived
1617 // put_main implementations must NOT call writer.commit() themselves.
1618 // If put_main throws, this commit is skipped and any uncommitted
1619 // buffer contents are dropped at writer destruction; the catch
1620 // below taints the converter.
1621 writer.commit();
1622 }
1623 }
