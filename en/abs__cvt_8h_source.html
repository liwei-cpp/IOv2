204 template <bool Saturate = false>
205 auto get_buf(std::size_t to_max)
206 {
207 if (to_max == 0)
208 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
209 if (to_max > m_buffer.size())
210 throw cvt_error("cvt_reader::get_buf fail, read size too large");
211
212 const auto rollback_size = m_end_pos - m_cur_pos;
213 if (to_max <= rollback_size)
214 {
215 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
216 m_cur_pos += to_max;
217 if constexpr (!Saturate)
218 return res;
219 else
220 return res.first;
221 }
222
223 if (to_max > m_buffer.size() - m_cur_pos)
224 {
225 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
226 m_cur_pos = 0;
227 m_end_pos = rollback_size;
228 }
229 const std::size_t aim_size = to_max - rollback_size;
230 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
231 m_end_pos += read_size;
232
233 if constexpr (Saturate)
234 {
235 while (read_size < aim_size)
236 {
237 const std::size_t new_aim_size = aim_size - read_size;
238 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
239 if (new_read_size == 0)
240 throw cvt_error("get_buf<Saturate> fail: meet eos");
241 m_end_pos += new_read_size;
242 read_size += new_read_size;
243 }
244 }
245
246 if constexpr (!Saturate)
247 {
248 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
249 m_cur_pos = m_end_pos;
250 return res;
251 }
252 else
253 {
254 const char_type* res = m_buffer.data() + m_cur_pos;
255 m_cur_pos = m_end_pos;
256 return res;
257 }
258 }

1466 std::size_t get(internal_type* to, std::size_t to_max)
1467 requires requires(CurrentType& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1468 { t.get_main(r, data, len) } -> std::same_as<std::size_t>;
1469 }
1470 {
1471 // Read failures do not taint: they only advance the kernel cursor
1472 // and possibly leave a per-call mbstate undefined; the underlying
1473 // stream content is unchanged, and callers can reseek to recover.
1474 // We still honour any pre-existing taint so that a writer that
1475 // corrupted the stream cannot then read back a confused image.
1476 assert_not_tainted();
1477
1478 cvt_reader<KernelType> reader(m_kernel, m_tmp_io_buffer);
1479
1480 if (!m_is_bos_done)
1481 {
1482 if (to_max == 0) return 0;
1483
1484 constexpr std::size_t ext_size = sizeof(external_type);
1485
1486 reader.reset(s_bos_chunk);
1487 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1488 char* dest_bytes = reinterpret_cast<char*>(to);
1489 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1490 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1491
1492 // BOS phase requires complete data: if the stream ends unexpectedly
1493 // before providing enough bytes, this is a format error and we throw.
1494 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1495
1496 // Read full external_type chunks
1497 while (dest_bytes + ext_size <= dest_bytes_end)
1498 {
1499 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1500
1501 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1502 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1503 dest_bytes += units_to_read_now * ext_size;
1504 }
1505
1506 // Read the final partial internal_type if needed
1507 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1508 if (final_remainder_bytes > 0)
1509 {
1510 const auto* src_buf = reader.template get_buf<true>(1);
1511 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1512 dest_bytes += final_remainder_bytes;
1513 }
1514
1515 return to_max;
1516 }
1517 else
1518 {
1519 if (m_io_status != io_status::input)
1520 {
1521 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1522 static_cast<CurrentType*>(this)->switch_to_get();
1523 else
1524 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1525 }
1526 return static_cast<CurrentType*>(this)->get_main(reader, to, to_max);
1527 }
1528 }

1575 void put(const internal_type* to, std::size_t to_size)
1576 requires requires(CurrentType& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1577 { t.put_main(w, data, len) } -> std::same_as<void>;
1578 }
1579 {
1580 assert_not_tainted();
1581
1582 // IO-direction switching does not write bytes to the output stream:
1583 // a failed switch_to_put() leaves the stream unchanged, so the
1584 // converter is not corrupted and must not be tainted. Perform the
1585 // switch BEFORE entering the taint-on-exception region so that only
1586 // actual kernel writes can cause tainting.
1587 if (m_is_bos_done && m_io_status != io_status::output)
1588 {
1589 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1590 static_cast<CurrentType*>(this)->switch_to_put();
1591 else
1592 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1593 }
1594
1595 try
1596 {
1597 cvt_writer<KernelType> writer(m_kernel, m_tmp_io_buffer);
1598 if (!m_is_bos_done)
1599 {
1600 if (to_size == 0) return;
1601
1602 constexpr std::size_t ext_size = sizeof(external_type);
1603
1604 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1605 auto to_bytes = reinterpret_cast<const char*>(to);
1606 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1607 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1608
1609 // BOS phase writes complete external_type units. If the input data
1610 // is not a multiple of ext_size, pad the final unit with zeros.
1611
1612 writer.reset(s_bos_chunk);
1613 while (to_bytes + ext_size <= to_bytes_end)
1614 {
1615 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1616 auto ptr = writer.put_buf(dest_count);
1617 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1618 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1619 to_bytes += dest_count * ext_size;
1620 }
1621
1622 if (to_bytes < to_bytes_end)
1623 {
1624 // The while loop above guarantees remaining < ext_size.
1625 // The modulo is logically redundant but helps the compiler's
1626 // static analyzer prove the bound for memset size calculation.
1627 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1628 auto ptr = writer.put_buf(1);
1629 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1630 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1631 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1632 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1633 }
1634
1635 writer.commit();
1636 }
1637 else
1638 {
1639 // switch_to_put() was already called above if needed;
1640 // m_io_status is guaranteed to be output at this point.
1641 static_cast<CurrentType*>(this)->put_main(writer, to, to_size);
1642 // commit() is the sole responsibility of abs_cvt::put: derived
1643 // put_main implementations must NOT call writer.commit() themselves.
1644 // If put_main throws, this commit is skipped and any uncommitted
1645 // buffer contents are dropped at writer destruction; the catch
1646 // below taints the converter.
1647 writer.commit();
1648 }
1649 }
1650 catch (...)
1651 {
1652 m_is_tainted = true;
1653 throw;
1654 }
1655 }
