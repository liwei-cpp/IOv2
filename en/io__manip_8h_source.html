536template <istream_type T, typename TMoney>
537inline T& operator>>(T& is, _Get_money<TMoney> f)
538{
539 // Delegate to the generic formatted-extraction operator: it owns the sentry, the
540 // skipws handling, the stream iterator (i_iter is private and befriends only that
541 // template) and the eofbit bookkeeping. The explicit template argument list is
542 // required rather than stylistic -- `is >> f` would re-select *this* overload,
543 // because f is a named lvalue and the by-value candidate is the more specialized
544 // one, and recurse forever.
545 return IOv2::operator>><T, _Get_money<TMoney>>(is, f);
546}
