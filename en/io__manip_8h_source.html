421template <istream_type T, typename TMoney>
422inline T& operator>>(T& is, _Get_money<TMoney> f)
423{
424 // Delegate to the generic formatted-extraction operator: it owns the sentry, the
425 // skipws handling, the stream iterator (i_iter is private and befriends only that
426 // template) and the eofbit bookkeeping. The explicit template argument list is
427 // required rather than stylistic -- `is >> f` would re-select *this* overload,
428 // because f is a named lvalue and the by-value candidate is the more specialized
429 // one, and recurse forever.
430 return IOv2::operator>><T, _Get_money<TMoney>>(is, f);
431}

590 requires (std::is_same_v<TChar, typename TIter::value_type>)
591 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc, _Get_time<TChar>& f)
592 {
593 if (f.tmb == nullptr || f.fmt == nullptr)
594 throw stream_error("get_time fail: null tm or format pointer");
595
596 auto mp = loc.template get<timeio<TChar>>();
597 if (!mp)
598 throw stream_error("cannot get timeio facet");
599

601 // parse_context_type<TChar, std::tm>; that is also the combination whose
602 // explicit operator std::tm() is available.

604 auto res = mp->get(s, s_end, tmp, f.fmt);
605 *(f.tmb) = static_cast<std::tm>(tmp);

608 }
