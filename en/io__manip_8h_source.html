561template <istream_type T, typename TMoney>
562inline T& operator>>(T& is, _Get_money<TMoney> f)
563{
564 // Delegate to the generic formatted-extraction operator: it owns the sentry, the
565 // skipws handling, the stream iterator (i_iter is private and befriends only that
566 // template) and the eofbit bookkeeping. The explicit template argument list is
567 // required rather than stylistic -- `is >> f` would re-select *this* overload,
568 // because f is a named lvalue and the by-value candidate is the more specialized
569 // one, and recurse forever.
570 return IOv2::operator>><T, _Get_money<TMoney>>(is, f);
571}

864 template <typename TIter, std::sentinel_for<TIter> TSent>
865 requires (std::is_same_v<TChar, typename TIter::value_type>)
866 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc, _Get_time<TChar>& f)
867 {
868 if (f.tmb == nullptr || f.fmt == nullptr)
869 throw stream_error("get_time fail: null tm or format pointer");
870
871 auto mp = loc.template get<timeio<TChar>>();
872 if (!mp)
873 throw stream_error("cannot get timeio facet");
874
875 // The context is date+time without a time zone, matching
876 // parse_context_type<TChar, std::tm>; that is also the combination whose
877 // explicit operator std::tm() is available.
878 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
879 auto res = mp->get(s, s_end, tmp, f.fmt);
880 *(f.tmb) = static_cast<std::tm>(tmp);
881
882 return res;
883 }
