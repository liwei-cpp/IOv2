406template <istream_type T, typename TMoney>
407inline T& operator>>(T& is, _Get_money<TMoney> f)
408{
409 // Delegate to the generic formatted-extraction operator: it owns the sentry, the
410 // skipws handling, the stream iterator (i_iter is private and befriends only that
411 // template) and the eofbit bookkeeping. The explicit template argument list is
412 // required rather than stylistic -- `is >> f` would re-select *this* overload,
413 // because f is a named lvalue and the by-value candidate is the more specialized
414 // one, and recurse forever.
415 return operator>><T, _Get_money<TMoney>>(is, f);
416}
