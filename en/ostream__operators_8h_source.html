378 void try_flush() override
379 {
380 constexpr int attempts = 3;
381
382 T& obj = static_cast<T&>(*this);
383 for (int i = 0; i < attempts; ++i)
384 {
385 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
386 if (lk)
387 {
388 obj.flush();
389 return;
390 }
391 if (i + 1 < attempts)
392 std::this_thread::yield();
393 }
394 }

847template <ostream_type T, typename TValue>
848 requires detail::insertable<T, TValue>
849T& operator<<(T& obj, const TValue& value)
850{
851 using TChar = typename T::char_type;
852 using TDecay = std::decay_t<const TValue&>;
853
854 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
855 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
856 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay, TValue>;
857 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay, TValue>;
858
859 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
860 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
861 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
862 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
863 "See io/traits/traits_base.h.");
864
865 if constexpr (stream_v || stream_d)
866 {
867 try
868 {
869 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
870 }
871 catch (...)
872 {
873 obj.handle_exception(std::current_exception());
874 }
875 }
876 else if constexpr (iter_v || iter_d)
877 {
878 using sentry_type = typename T::out_sentry_type;
879 std::lock_guard guard(obj.io_mutex());
880 try
881 {
882 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
883 bool(obj.flags() & ios_defs::appmode));
884
885 auto iter = obj.o_iter();
886 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
887 iter, obj, obj.locale(), value);
888 }
889 catch (...)
890 {
891 obj.handle_exception(std::current_exception());
892 }
893 }
894 else
895 static_assert(dependent_false_v<TValue>,
896 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
897 "branch for it. The concept and the chain must stay in step -- see "
898 "io/utilities/ostream_operators.h.");
899
900 return obj;
901}
