432 void try_flush() noexcept override
433 {
434 constexpr int attempts = 3;
435
436 T& obj = static_cast<T&>(*this);
437 for (int i = 0; i < attempts; ++i)
438 {
439 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
440 if (lk)
441 {
442 try
443 {
444 obj.flush();
445 }
446 catch (...)
447 {
448 // Record on the tie target, never on the initiator, and never rethrow.
449 // <true> keeps the stored exception alive; see the note above.
450 obj.template handle_exception<true>(std::current_exception());
451 }
452 return;
453 }
454 if (i + 1 < attempts)
455 std::this_thread::yield();
456 }
457 }

910template <ostream_type T, typename TValue>
911 requires detail::insertable<T, TValue>
912T& operator<<(T& obj, const TValue& value)
913{
914 using TChar = typename T::char_type;
915 using TDecay = std::decay_t<const TValue&>;
916
917 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
918 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
919 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay, TValue>;
920 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay, TValue>;
921
922 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
923 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
924 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
925 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
926 "See io/traits/traits_base.h.");
927
928 if constexpr (stream_v || stream_d)
929 {
930 try
931 {
932 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
933 }
934 catch (...)
935 {
936 obj.handle_exception(std::current_exception());
937 }
938 }
939 else if constexpr (iter_v || iter_d)
940 {
941 using sentry_type = typename T::out_sentry_type;
942 std::lock_guard guard(obj.io_mutex());
943 try
944 {
945 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
946 bool(obj.flags() & ios_defs::appmode));
947
948 auto iter = obj.o_iter();
949 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
950 iter, obj, obj.locale(), value);
951 }
952 catch (...)
953 {
954 obj.handle_exception(std::current_exception());
955 }
956 }
957 else
958 static_assert(dependent_false_v<TValue>,
959 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
960 "branch for it. The concept and the chain must stay in step -- see "
961 "io/utilities/ostream_operators.h.");
962
963 return obj;
964}
