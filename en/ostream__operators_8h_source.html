396 void try_flush() override
397 {
398 constexpr int attempts = 3;
399
400 T& obj = static_cast<T&>(*this);
401 for (int i = 0; i < attempts; ++i)
402 {
403 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
404 if (lk)
405 {
406 obj.flush();
407 return;
408 }
