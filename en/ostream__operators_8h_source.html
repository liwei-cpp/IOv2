127 out_sentry(TStream& os, bool is_unit_buf, bool is_app_mode)
128 : m_os(os)
129 , m_is_unit_buf(is_unit_buf)
130 {
131 if (!static_cast<bool>(m_os))
132 throw stream_error("ostream_sentry create fail: Invalid ostream");
133
134 if (auto* tied = m_os.tie())
135 tied->try_flush();
136
137 if constexpr (is_std)
138 m_sync_with_stdio = os.m_sync_with_stdio.load();
139
140 if constexpr (involve_input)
141 os.m_channel.switch_to_put();
142
143 if constexpr (!is_std)

153 throw cvt_error(std::string("ostream_sentry create fail: appmode cannot "
154 "reposition to the end; appmode requires a "
155 "fixed-length, state-independent encoding: ")
156 + e.what());
157 }
158 }
159 }
160
161 if (!static_cast<bool>(m_os))
162 throw stream_error("ostream_sentry create fail: Invalid ostream");

199 ~out_sentry() noexcept(false)
200 {
201 try
202 {
203 if (m_os)
204 {
205 auto exited = [&]() noexcept {
206 if constexpr (is_std) return m_os.m_exited.load();
207 else return false;
208 };
209 if (m_is_unit_buf || m_sync_with_stdio || exited())
210 m_os.m_channel.flush();

212 // After the exit hook, stdio's own last flush may already be behind us:
213 // glibc leaves a stdout it never used fully buffered, so push the bytes out.
214 if (m_is_unit_buf || exited())
215 m_os.m_channel.device().dflush();
216 }
217 }
218 catch (...)
219 {
220 m_os.handle_exception(std::current_exception());
221 }
222 }

419template <typename T>
420struct out_tie_target : public tie_target
421{
481 void try_flush() noexcept final
482 {
483 // Downstream first, before taking our own lock; tie() keeps the chain acyclic.
484 if (auto ptr = tied_to(); ptr != nullptr)
485 ptr->try_flush();
486
487 constexpr int attempts = 3;
488
489 T& obj = static_cast<T&>(*this);
490 for (int i = 0; i < attempts; ++i)
491 {
492 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
493 if (lk)
494 {
495 try
496 {
497 obj.flush();
498 }
499 catch (...)
500 {
501 // Record on the tie target, never on the initiator, and never rethrow.
502 // <true> keeps the stored exception alive; see the note above.
503 obj.template handle_exception<true>(std::current_exception());
504 }
505 return;
506 }
507 if (i + 1 < attempts)
508 std::this_thread::yield();
509 }
510 }
511
517 [[nodiscard]] tie_target* tied_to() const noexcept final
518 {
519 return static_cast<const T&>(*this).tie();
520 }

963template <ostream_type T, typename TValue>
964 requires detail::insertable<T, TValue>
965T& operator<<(T& obj, const TValue& value)
966{
967 using TChar = typename T::char_type;
968 using TDecay = std::decay_t<const TValue&>;
969
970 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
971 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
972 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay, TValue>;
973 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay, TValue>;
974
975 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
976 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
977 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
978 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
979 "See io/traits/traits_base.h.");
980
981 if constexpr (stream_v || stream_d)
982 {
983 try
984 {
985 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
986 }
987 catch (...)
988 {
989 obj.handle_exception(std::current_exception());
990 }
991 }
992 else if constexpr (iter_v || iter_d)
993 {
994 using sentry_type = typename T::out_sentry_type;
995 std::lock_guard guard(obj.io_mutex());
996 try
997 {
998 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
999 bool(obj.flags() & ios_defs::appmode));
1000
1001 auto iter = obj.o_iter();
1002 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
1003 iter, obj, obj.locale(), value);
1004 }
1005 catch (...)
1006 {
1007 obj.handle_exception(std::current_exception());
1008 }
1009 }
1010 else
1011 static_assert(dependent_false_v<TValue>,
1012 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
1013 "branch for it. The concept and the chain must stay in step -- see "
1014 "IOv2/io/utilities/ostream_operators.h.");
1015
1016 return obj;
1017}
