397 void try_flush() override
398 {
399 constexpr int attempts = 3;
400
401 T& obj = static_cast<T&>(*this);
402 for (int i = 0; i < attempts; ++i)
403 {
404 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
405 if (lk)
406 {
407 obj.flush();
408 return;
409 }
410 if (i + 1 < attempts)
411 std::this_thread::yield();
412 }

844template <ostream_type T, typename TValue>
845 requires detail::insertable<T, TValue>
846T& operator<<(T& obj, const TValue& value)
847{
848 using TChar = typename T::char_type;
849 using TDecay = std::decay_t<TValue>;
850
851 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
852 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
853 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay>;
854 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay>;
855
856 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
857 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
858 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
859 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
860 "See io/traits/traits_base.h.");
861
862 if constexpr (stream_v || stream_d)
863 {
864 try
865 {
866 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
867 }
868 catch (...)
869 {
870 obj.handle_exception(std::current_exception());
871 }
872 }
873 else if constexpr (iter_v || iter_d)
874 {
875 using sentry_type = typename T::out_sentry_type;
876 std::lock_guard guard(obj.io_mutex());
877 try
878 {
879 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
880 bool(obj.flags() & ios_defs::appmode));
881
882 auto iter = obj.o_iter();
883 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
884 iter, obj, obj.locale(), value);
885 }
886 catch (...)
887 {
888 obj.handle_exception(std::current_exception());
889 }
890 }
891 else
892 static_assert(dependent_false_v<TValue>,
893 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
894 "branch for it. The concept and the chain must stay in step -- see "
895 "io/utilities/ostream_operators.h.");
896
897 return obj;
898}
