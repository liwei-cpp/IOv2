388 ostream& operator=(ostream&& other) noexcept
389 {
390 if (this == &other) return *this;
391
392 std::lock_guard guard(this->io_mutex());
393 // NOLINTBEGIN(bugprone-use-after-move): each `std::move(other)` below binds to a
394 // *different* base subobject, and every one of those base assignments touches only its
395 // own members; `m_streambuf` and `m_locale` belong to no base at all. Nothing is read
396 // after being moved from -- clang-tidy just cannot see that the operands are disjoint.
397 ios_state<TChar>::operator=(std::move(other));
398 out_flusher<ostream<TDevice, TChar>>::operator=(std::move(other));
399 ostream_operators<TChar>::operator=(std::move(other));
400 stream_common_operators::operator=(std::move(other));
401 m_streambuf = std::move(other.m_streambuf);
402 m_locale = std::move(other.m_locale);
403 // NOLINTEND(bugprone-use-after-move)
404 return *this;
405 }
406
429 // `override` because the `abs_flusher` base (reached through `out_flusher`) has a virtual
430 // destructor, so this one is virtual whether or not it says so.
