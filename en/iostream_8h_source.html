411 iostream& operator=(iostream&& other) noexcept
412 {
413 if (this == &other) return *this;
414
415 std::lock_guard guard(this->io_mutex());
416 // NOLINTBEGIN(bugprone-use-after-move): each `std::move(other)` below binds to a
417 // *different* base subobject, and every one of those base assignments touches only its
418 // own members; `m_channel` and `m_locale` belong to no base at all. Nothing is read
419 // after being moved from -- clang-tidy just cannot see that the operands are disjoint.
420 ios_state<TChar>::operator=(std::move(other));
421 istream_operators<TChar>::operator=(std::move(other));
422 out_tie_target<iostream<TDevice, TChar>>::operator=(std::move(other));
423 ostream_operators<TChar>::operator=(std::move(other));
424 stream_common_operators::operator=(std::move(other));
425 m_channel = std::move(other.m_channel);
426 m_locale = std::move(other.m_locale);
427 // NOLINTEND(bugprone-use-after-move)
428 return *this;
429 }
