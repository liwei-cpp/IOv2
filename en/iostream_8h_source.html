418 iostream& operator=(iostream&& other) noexcept
419 {
420 if (this == &other) return *this;
421
422 std::lock_guard guard(this->io_mutex());
423 // NOLINTBEGIN(bugprone-use-after-move): each `std::move(other)` below binds to a
424 // *different* base subobject, and every one of those base assignments touches only its
425 // own members; `m_streambuf` and `m_locale` belong to no base at all. Nothing is read
426 // after being moved from -- clang-tidy just cannot see that the operands are disjoint.
427 ios_state<TChar>::operator=(std::move(other));
428 istream_operators<TChar>::operator=(std::move(other));
