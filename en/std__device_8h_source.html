293 std::size_t dget(char* s, std::size_t n)
294 requires (ID == STDIN_FILENO)
295 {
296 if (n == 0 || m_eof_hit) return 0;
297 if (s == nullptr)
298 throw device_error("std_device::dget fail: null buffer");
299
300 if (m_c.has_value())
301 {
302 // Whether a read() would return at once: data is ready, or the peer hung
303 // up (read() then returns 0 without waiting). On a poll error the answer
304 // is "no"; the next blocking read() reports it.
305 const auto readable_now = []() noexcept
306 {
307 struct pollfd pfd{ .fd = ID, .events = POLLIN, .revents = 0 };
308 int r = poll(&pfd, 1, 0);
309 while (r == -1 && errno == EINTR)
310 r = poll(&pfd, 1, 0);
311 return r > 0 && (pfd.revents & (POLLIN | POLLHUP)) != 0;
312 };
313
314 *s = m_c.value();
315 if (n > 1 && readable_now())
316 {
317 // Top up with what is already there, so a probe followed by a bulk
318 // read still reads in bulk. Only when data is ready: waiting here
319 // would block on a pipe whose peer sent that one byte, or on a
320 // terminal where the user typed an empty line. If do_get throws,
321 // m_c must still hold the byte so the next dget() redelivers it.
322 const std::size_t got = do_get(s + 1, n - 1);
323 m_c.reset();
324 return 1 + got;
325 }
326 m_c.reset();
327 return 1;
328 }
329 return do_get(s, n);
330 }

394 std::size_t do_get(char* s, std::size_t n)
395 {
396 assert((s != nullptr) && (n != 0));
397
398 constexpr auto max_read = static_cast<std::size_t>(std::numeric_limits<ssize_t>::max());
399 ssize_t ret = 0;
400 while (true)
401 {
402 ret = read(ID, s, std::min(n, max_read));
403 if (ret != -1) break;
404
405 if (errno == EINTR) continue;
406 if (errno == EAGAIN || errno == EWOULDBLOCK)
407 {
408 struct pollfd pfd{ .fd = ID, .events = POLLIN, .revents = 0 };
409 if (poll(&pfd, 1, -1) == -1)
410 {
411 if (errno == EINTR) continue;
412 throw device_error("std_device::do_get fail: poll error");
413 }
414 else if (pfd.revents & (POLLERR | POLLNVAL))
415 throw device_error("std_device::do_get fail: poll revents error");
416 else if (pfd.revents & POLLIN) continue;
417 else if (pfd.revents & POLLHUP)
418 {
419 ret = 0;
420 break;
421 }
422 else
423 throw device_error("std_device::do_get fail: unexpected poll revents");
424 }
425 else
426 throw device_error("std_device::do_get fail: read error");
427 }
428
429 if (ret == 0)
430 m_eof_hit = true;
431 return static_cast<std::size_t>(ret);
432 }
433private:
434 [[no_unique_address]] std::conditional_t<ID == STDIN_FILENO, bool, std::monostate> m_eof_hit{};
435 [[no_unique_address]] std::conditional_t<ID == STDIN_FILENO, std::optional<char_type>, std::monostate> m_c;
436 bool m_active = true;
