269 std::size_t dget(char* s, std::size_t n)
270 requires (ID == STDIN_FILENO)
271 {
272 if (n == 0 || m_eof_hit) return 0;
273 if (s == nullptr)
274 throw device_error("std_device::dget fail: null buffer");
275
276 if (m_c.has_value())
277 {
278 // Whether a read() would return at once: data is ready, or the peer hung
279 // up (read() then returns 0 without waiting). On a poll error the answer
280 // is "no"; the next blocking read() reports it.
281 const auto readable_now = []() noexcept
282 {
283 struct pollfd pfd{ .fd = ID, .events = POLLIN, .revents = 0 };
284 int r = 0;
285 do { r = poll(&pfd, 1, 0); } while (r == -1 && errno == EINTR);
286 return r > 0 && (pfd.revents & (POLLIN | POLLHUP)) != 0;
287 };
288
289 *s = m_c.value();
290 if (n > 1 && readable_now())
291 {
292 // Top up with what is already there, so a probe followed by a bulk
293 // read still reads in bulk. Only when data is ready: waiting here
294 // would block on a pipe whose peer sent that one byte, or on a
295 // terminal where the user typed an empty line. If do_get throws,
296 // m_c must still hold the byte so the next dget() redelivers it.
297 const std::size_t got = do_get(s + 1, n - 1);
298 m_c.reset();
299 return 1 + got;
300 }
301 m_c.reset();
302 return 1;
303 }
304 return do_get(s, n);
305 }

322 void dput(const char* ch, std::size_t n)
323 requires ((ID == STDOUT_FILENO) || (ID == STDERR_FILENO))
324 {
325 if (n == 0) return;
326 if (ch == nullptr)
327 throw device_error("std_device::dput fail: null buffer");

329 bool put_res = false;

370 std::size_t do_get(char* s, std::size_t n)
371 {
372 assert((s != nullptr) && (n != 0));
373
374 constexpr auto max_read = static_cast<std::size_t>(std::numeric_limits<ssize_t>::max());
375 ssize_t ret = 0;
376 while (true)
377 {
378 ret = read(ID, s, std::min(n, max_read));
379 if (ret != -1) break;
380
381 if (errno == EINTR) continue;
382 if (errno == EAGAIN || errno == EWOULDBLOCK)
383 {
384 struct pollfd pfd{ .fd = ID, .events = POLLIN, .revents = 0 };
385 if (poll(&pfd, 1, -1) == -1)
386 {
387 if (errno == EINTR) continue;
388 throw device_error("std_device::do_get fail: poll error");
389 }
390 else if (pfd.revents & (POLLERR | POLLNVAL))
391 throw device_error("std_device::do_get fail: poll revents error");
392 else if (pfd.revents & POLLIN) continue;
393 else if (pfd.revents & POLLHUP)
394 {
395 ret = 0;
396 break;
397 }
398 else
399 throw device_error("std_device::do_get fail: unexpected poll revents");
400 }
401 else
402 throw device_error("std_device::do_get fail: read error");
403 }
404
405 if (ret == 0)
406 m_eof_hit = true;
407 return static_cast<std::size_t>(ret);
408 }

412 bool m_active = true;
