495 void dput(const char* ch, std::size_t n)
496 requires ((ID == detail::std_out_fd) || (ID == detail::std_err_fd))
497 {
498 if (n == 0) return;
499 if (ch == nullptr)
500 throw device_error("std_device::dput fail: null buffer");

502 std::size_t written = 0;
503 if constexpr (ID == detail::std_out_fd)
504 written = std::fwrite(ch, sizeof(char), n, stdout);

536private:
537 // fd mode keeps its own state: EOF in m_eof_hit, a probed byte in m_c.
538 std::size_t fd_get(char* s, std::size_t n)
539 requires (ID == detail::std_in_fd)
540 {
541 if (m_eof_hit) return 0;
542 if (m_c.has_value())
543 {
544 *s = m_c.value();
545 if (n > 1 && detail::std_fd_readable_now(ID))
546 {
547 // Top up with what is already there, so a probe followed by a bulk
548 // read still reads in bulk. Only when data is ready: waiting here
549 // would block on a pipe whose peer sent that one byte, or on a
550 // terminal where the user typed an empty line. If fd_read throws,
551 // m_c must still hold the byte so the next dget() redelivers it.
552 const std::size_t got = fd_read(s + 1, n - 1);
553 m_c.reset();
554 return 1 + got;
555 }
556 m_c.reset();
557 return 1;
558 }
559 return fd_read(s, n);
560 }
561
562 bool fd_eof()
563 requires (ID == detail::std_in_fd)
564 {
565 if (m_eof_hit) return true;
566 if (m_c.has_value()) return false;
567
568 char_type ch = 0;
569 if (fd_read(&ch, 1) == 0) return true; // fd_read has set m_eof_hit
570 m_c = ch;
571 return false;
572 }
573
574 std::size_t fd_read(char* s, std::size_t n)
575 requires (ID == detail::std_in_fd)
576 {
577 const std::size_t ret = detail::std_fd_read(ID, s, n);
578 if (ret == 0)
579 m_eof_hit = true;
580 return ret;
581 }
582
583 // stdio mode keeps its state in the FILE: a probed byte goes back with ungetc,
584 // EOF is feof(stdin). m_c here can only be a byte fd mode probed before the
585 // switch; it is still due first, and alone, as fd readiness says nothing now.
586 std::size_t stdio_get(char* s, std::size_t n)
587 requires (ID == detail::std_in_fd)
588 {
589 if (m_c.has_value())
590 {
591 *s = m_c.value();
592 m_c.reset();
593 return 1;
594 }
595 return detail::std_stdin_file_read(s, n);
596 }
597
598 bool stdio_eof()
599 requires (ID == detail::std_in_fd)
600 {
601 if (m_c.has_value()) return false;
602 return detail::std_stdin_file_eof();
603 }
604private:
605 IOV2_NO_UNIQUE_ADDRESS std::conditional_t<ID == detail::std_in_fd, bool, std::monostate> m_eof_hit{};
606 IOV2_NO_UNIQUE_ADDRESS std::conditional_t<ID == detail::std_in_fd, std::optional<char_type>, std::monostate> m_c;
607 IOV2_NO_UNIQUE_ADDRESS std::conditional_t<ID == detail::std_in_fd, bool, std::monostate> m_use_stdio{};
608 bool m_active = true;
609};
