46// The user's preferred UI languages, most preferred first, named as Windows names them
47// (BCP-47); empty if they cannot be had.
48inline std::vector<std::string> messages_system_languages()
49{
50 unsigned long count = 0;
51 unsigned long size = 0; // in wchar_t, the closing empty name included
52 if (!GetUserPreferredUILanguages(win_mui_language_name, &count, nullptr, &size) || size == 0)
53 return {};
54 std::wstring buf(size, L'\0');
55 if (!GetUserPreferredUILanguages(win_mui_language_name, &count, buf.data(), &size))
56 return {};
57 buf.resize(size);
58
59 // Null-terminated names, the last followed by an empty one. A name is ASCII; one
60 // that is not is no name a .mo directory could have, and is left out.
61 std::vector<std::string> res;
62 std::size_t pos = 0;
63 while (pos < buf.size() && buf[pos] != L'\0')
64 {
65 std::size_t end = buf.find(L'\0', pos);
66 if (end == std::wstring::npos)
67 end = buf.size();
68 std::string name;
69 for (std::size_t i = pos; i < end && buf[i] < 0x80; ++i)
70 name.push_back(static_cast<char>(buf[i]));
71 if (name.size() == end - pos)
72 res.push_back(std::move(name));
73 pos = end + 1;
74 }
75 return res;
76}
