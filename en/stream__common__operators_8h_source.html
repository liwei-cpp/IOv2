1025 bool check_tie(abs_flusher* str) const noexcept
1026 {
1027 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
1028 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
1029 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
1030 // tie target.
1031 auto next = [](stream_common_operators* p)
1032 { return dynamic_cast<stream_common_operators*>(p->tie()); };
1033
1034 stream_common_operators* slow = dynamic_cast<stream_common_operators*>(str);
1035 stream_common_operators* fast = slow;
1036
1037 while (slow != nullptr)
1038 {
1039 if (slow == this)
1040 return false;
1041
1042 slow = next(slow);
1043 fast = (fast != nullptr) ? next(fast) : nullptr;
1044 fast = (fast != nullptr) ? next(fast) : nullptr;
1045
1046 if (fast != nullptr && fast == slow)
1047 return false;
1048 }
1049
1050 return true;
1051 }
