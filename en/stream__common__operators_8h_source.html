876 template <typename TSelf>
877 abs_flusher* tie(this TSelf& self, abs_flusher* str)
878 {
879 abs_flusher* res = nullptr;
880 bool ok = true;
881
882 {
883 std::lock_guard graph_lock(tie_graph_mutex());
884 res = self.m_tie_stream.load();
885 if constexpr (std::derived_from<TSelf, abs_flusher>)
886 ok = self.check_tie(str);
887 if (ok)
888 self.m_tie_stream.store(str);
889 }
890
891 if (!ok)
892 self.handle_exception(std::make_exception_ptr(stream_error(
893 "stream tie fail: the requested tie would form a cycle in the tie chain, "
894 "or the chain already contains one")));
895
896 return res;

974 bool check_tie(abs_flusher* str) const noexcept
975 {
976 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
977 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
978 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
979 // tie target.
980 auto next = [](stream_common_operators* p)
981 { return dynamic_cast<stream_common_operators*>(p->tie()); };
982
983 auto slow = dynamic_cast<stream_common_operators*>(str);
984 auto fast = slow;
985
986 while (slow != nullptr)
987 {
988 if (slow == this)
989 return false;
990
991 slow = next(slow);
992 fast = (fast != nullptr) ? next(fast) : nullptr;
993 fast = (fast != nullptr) ? next(fast) : nullptr;
994
995 if (fast != nullptr && fast == slow)
996 return false;
997 }
998
999 return true;
1000 }
