865 abs_flusher* tie(this TSelf& self, abs_flusher* str)
866 {
867 abs_flusher* res = nullptr;
868 bool ok = true;
869
870 {
871 std::lock_guard graph_lock(tie_graph_mutex());
872 res = self.m_tie_stream.load();
873 if constexpr (std::derived_from<TSelf, abs_flusher>)
874 ok = self.check_tie(str);
875 if (ok)
876 self.m_tie_stream.store(str);
877 }
878
879 if (!ok)

881 "stream tie fail: the requested tie would form a cycle in the tie chain, "
882 "or the chain already contains one")));

884 return res;

962 bool check_tie(abs_flusher* str) const noexcept
963 {
964 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
965 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
966 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
967 // tie target.
968 auto next = [](stream_common_operators* p)
969 { return dynamic_cast<stream_common_operators*>(p->tie()); };
970
971 auto slow = dynamic_cast<stream_common_operators*>(str);
972 auto fast = slow;
973
974 while (slow != nullptr)
975 {
976 if (slow == this)
977 return false;
978
979 slow = next(slow);
980 fast = (fast != nullptr) ? next(fast) : nullptr;
981 fast = (fast != nullptr) ? next(fast) : nullptr;
982
983 if (fast != nullptr && fast == slow)
984 return false;
985 }
986
987 return true;
988 }
