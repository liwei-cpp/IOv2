744private:
791 void ensure_no_cycle(abs_flusher* str) const
792 {
793 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
794 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
795 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
796 // tie target.
797 auto next = [](stream_common_operators* p)
798 { return dynamic_cast<stream_common_operators*>(p->tie()); };
799
800 stream_common_operators* slow = dynamic_cast<stream_common_operators*>(str);
801 stream_common_operators* fast = slow;
802
803 while (slow != nullptr)
804 {
805 if (slow == this)
806 throw stream_error("stream tie fail: the requested tie would form a cycle in the tie chain");
807
808 slow = next(slow);
809 fast = (fast != nullptr) ? next(fast) : nullptr;
810 fast = (fast != nullptr) ? next(fast) : nullptr;
811
812 if (fast != nullptr && fast == slow)
813 throw stream_error("stream tie fail: the tie chain already contains a cycle; "
814 "something other than tie() has written a tie pointer");
815 }
816 }
