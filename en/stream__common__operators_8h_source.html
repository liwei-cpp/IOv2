825 void ensure_no_cycle(abs_flusher* str) const
826 {
827 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
828 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
829 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
830 // tie target.
831 auto next = [](stream_common_operators* p)
832 { return dynamic_cast<stream_common_operators*>(p->tie()); };
833
834 stream_common_operators* slow = dynamic_cast<stream_common_operators*>(str);
835 stream_common_operators* fast = slow;
836
837 while (slow != nullptr)
838 {
839 if (slow == this)
840 throw stream_error("stream tie fail: the requested tie would form a cycle in the tie chain");
841
842 slow = next(slow);
843 fast = (fast != nullptr) ? next(fast) : nullptr;
844 fast = (fast != nullptr) ? next(fast) : nullptr;
845
846 if (fast != nullptr && fast == slow)
847 throw stream_error("stream tie fail: the tie chain already contains a cycle; "
848 "something other than tie() has written a tie pointer");
849 }
850 }
