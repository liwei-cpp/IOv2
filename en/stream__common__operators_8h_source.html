827private:
874 void ensure_no_cycle(abs_flusher* str) const
875 {
876 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
877 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
878 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
879 // tie target.
880 auto next = [](stream_common_operators* p)
881 { return dynamic_cast<stream_common_operators*>(p->tie()); };
882
883 stream_common_operators* slow = dynamic_cast<stream_common_operators*>(str);
884 stream_common_operators* fast = slow;
885
886 while (slow != nullptr)
887 {
888 if (slow == this)
889 throw stream_error("stream tie fail: the requested tie would form a cycle in the tie chain");
890
891 slow = next(slow);
892 fast = (fast != nullptr) ? next(fast) : nullptr;
893 fast = (fast != nullptr) ? next(fast) : nullptr;
894
895 if (fast != nullptr && fast == slow)
896 throw stream_error("stream tie fail: the tie chain already contains a cycle; "
897 "something other than tie() has written a tie pointer");
898 }
899 }
