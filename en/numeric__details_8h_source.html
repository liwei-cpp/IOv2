85 numeric_conf(const std::string& name)
86 : ft_basic<numeric<char>>()
87 {
88 if (FacetHelper::is_c_locale_name(name))
89 { // "C" locale
90 m_decimal_point = '.';
91 m_thousands_sep = ',';
92 m_true_name = "true";
93 m_false_name = "false";
94 return;
95 }
96
97 // Snapshot all locale-dependent strings while the locale guard is
98 // active. The lconv* and nl_langinfo() pointers may be invalidated
99 // by any subsequent setlocale()/uselocale() call (e.g. inside
100 // FacetHelper::string_to_char_convert), so copy first
101 // and convert after.
102 std::string dp_raw, ts_raw, grp_raw, yes_raw, no_raw;
103 bool yes_set = false, no_set = false;
104 {
105 clocale_wrapper inter_locale(name.c_str());
106 clocale_user guard(inter_locale);
107 const lconv* lc = localeconv();
108 if (lc->decimal_point) dp_raw = lc->decimal_point;
109 if (lc->thousands_sep) ts_raw = lc->thousands_sep;
110 if (lc->grouping) grp_raw = lc->grouping;
111
112 // Treat empty YESSTR/NOSTR the same as a missing key: an empty
113 // m_true_name / m_false_name would make numeric::get(bool&) fail
114 // unconditionally (zero-length prefix never matches) and let
115 // numeric::put(bool) write nothing but padding.
116 if (auto* p = nl_langinfo(YESSTR); p && *p) { yes_raw = p; yes_set = true; }
117 if (auto* p = nl_langinfo(NOSTR); p && *p) { no_raw = p; no_set = true; }
118 }
119
120 m_decimal_point = FacetHelper::string_to_char_convert(dp_raw, name);
121 m_thousands_sep = FacetHelper::string_to_char_convert(ts_raw, name);
122
123 // string_to_char_convert hard-codes '\0' as its failure sentinel. A
124 // decimal point of '\0' would later be mistaken for an in-stream
125 // null byte during parsing; fall back to '.' so the contract that
126 // m_decimal_point is a real, distinguishable character is preserved.
127 if (m_decimal_point == '\0') m_decimal_point = '.';
128
129 if (m_thousands_sep != '\0' && !grp_raw.empty())
130 {
131 // Copy raw POSIX grouping bytes, then normalise into the
132 // internal convention here at the POSIX boundary.
133 // Downstream (numeric, user-derived numeric_conf) sees
134 // only the internal form.
135 m_grouping.resize(grp_raw.size());
136 for (std::size_t i = 0; i < grp_raw.size(); ++i)
137 m_grouping[i] = static_cast<uint8_t>(grp_raw[i]);
138 FacetHelper::adjust_grouping(m_grouping);
139 }
140
141 // If a locale (or a user-derived numeric_conf that overrides
142 // thousands_sep() / decimal_point()) ends up with the same
143 // character for both separators, downstream extract_int /
144 // extract_float in facet/numeric.h check thousands_sep BEFORE
145 // decimal_point against each input character. An equal pair
146 // makes the decimal-point branch unreachable and groups would
147 // be parsed where a decimal mark was intended. Drop grouping
148 // so parsing falls back to the no-grouping path. Mirrors the
149 // same guard in numeric_conf<char8_t>, which can hit this case
150 // after multi-byte UTF-8 separators collapse onto single-byte
151 // fallbacks.
152 if (m_thousands_sep == m_decimal_point)
153 m_grouping.clear();
154
155 if (yes_set) m_true_name = yes_raw;
156 else m_true_name = "true";
157 if (no_set) m_false_name = no_raw;
158 else m_false_name = "false";
159 }

548class numeric_conf<CharT> : public ft_basic<numeric<char8_t>>

550public:

562public:
585 numeric_conf(const std::string& name)
586 : ft_basic<numeric<char8_t>>()
587 {
588 if (FacetHelper::is_c_locale_name(name))
589 { // "C" locale
590 m_decimal_point = u8'.';
591 m_thousands_sep = u8',';
592 m_true_name = u8"true";
593 m_false_name = u8"false";
594 return;
595 }
596
597 init_from_u32<>(name);
598 }
599
600private:
601 // T is defaulted/constrained to char32_t; templating it makes the
602 // numeric_conf<T> reference below dependent, so the whole body is checked
603 // only upon instantiation of this helper (i.e. only on the non-"C"/"POSIX"
604 // construction path), never eagerly.
605 template <typename T = char32_t>
606 requires std::is_same_v<T, char32_t>
607 void init_from_u32(const std::string& name)
608 {
609 numeric_conf<T> numeric_temp(name);
610 {
611 const auto input = numeric_temp.decimal_point();
612 auto output = detail::to_u8string(input);
613 // Fall back to u8'.' when the UTF-8 encoding is not a single
614 // byte OR when it is a single u8'\0' byte. The latter would be
615 // indistinguishable from an in-stream null byte during parsing.
616 m_decimal_point = (output.size() != 1 || output[0] == u8'\0')
617 ? u8'.' : output[0];
618 }
619
620 bool ts_representable = false;
621 {
622 const auto input = numeric_temp.thousands_sep();
623 auto output = detail::to_u8string(input);
624 ts_representable = (output.size() == 1 && output[0] != u8'\0');
625 m_thousands_sep = ts_representable ? output[0] : u8',';
626 }
627
628 m_true_name = detail::to_u8string(numeric_temp.truename());
629 m_false_name = detail::to_u8string(numeric_temp.falsename());
630
631 m_grouping = numeric_temp.grouping();
632
633 if (!ts_representable)
634 m_grouping.clear();
635
636 // After the single-byte fallbacks above, m_thousands_sep may have
637 // collapsed onto u8',' while m_decimal_point also resolved to u8','
638 // (e.g. a locale where decimal_point is ',' and thousands_sep is a
639 // multi-byte UTF-8 codepoint such as U+202F NARROW NO-BREAK SPACE).
640 // When the two separators coincide, downstream extract_int /
641 // extract_float would compare each input byte against both fields
642 // with the same value, making grouping detection ambiguous. Drop
643 // grouping in that case so parsing falls back to the no-grouping
644 // path instead of misclassifying digits.
645 if (m_thousands_sep == m_decimal_point)
646 m_grouping.clear();
647 }
648
