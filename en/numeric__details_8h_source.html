42// The punctuation of the locale called name, copied out with the thread switched to it. The
43// conversions the configurations make afterwards switch the thread's locale themselves.
44inline numeric_raw_punct numeric_snapshot(const std::string& name)
45{
46 clocale_wrapper loc(name);
47 clocale_user guard(loc);
48 return numeric_read_punct();
49}

69class numeric_conf<char> : public ft_basic<numeric<char>>
70{
71public:
101 numeric_conf(const std::string& name)
102 : ft_basic<numeric<char>>()
103 {
104 if (FacetHelper::is_c_locale_name(name))
105 { // "C" locale
106 m_decimal_point = ".";
107 m_thousands_sep = ",";
108 m_true_name = "true";
109 m_false_name = "false";
110 return;
111 }
112
113 const detail::numeric_raw_punct raw = detail::numeric_snapshot(name);
114
115 // The separators are kept as the C library spells them, in the locale's narrow
116 // encoding, however many bytes that takes (U+202F is three in UTF-8).
117 m_decimal_point = raw.decimal_point.empty() ? "." : raw.decimal_point;
118 m_thousands_sep = raw.thousands_sep;
119 m_grouping = FacetHelper::lconv_grouping<char>(raw.grouping, m_thousands_sep, m_decimal_point);
120
121 // An empty name would make numeric::get(bool&) fail unconditionally (a
122 // zero-length prefix never matches) and numeric::put(bool) write nothing
123 // but padding, so empty counts as missing.
124 m_true_name = raw.truename.empty() ? "true" : raw.truename;
125 m_false_name = raw.falsename.empty() ? "false" : raw.falsename;
126 }

237template <typename CharT>
238 requires std::is_same_v<CharT, wchar_t> ||
239 (std::is_same_v<CharT, char32_t> &&
240 wchar_t_is_utf32)
241class numeric_conf<CharT> : public ft_basic<numeric<CharT>>
242{
243public:
253 using char_type = CharT;
254
255public:
284 numeric_conf(const std::string& name)
285 : ft_basic<numeric<CharT>>()
286 {
287 if (FacetHelper::is_c_locale_name(name))
288 { // "C" locale
289 if constexpr (std::is_same_v<CharT, wchar_t>)
290 {
291 m_decimal_point = L".";
292 m_thousands_sep = L",";
293 m_true_name = L"true";
294 m_false_name = L"false";
295 }
296 else
297 {
298 m_decimal_point = U".";
299 m_thousands_sep = U",";
300 m_true_name = U"true";
301 m_false_name = U"false";
302 }
303 return;
304 }
305
306 const detail::numeric_raw_punct raw = detail::numeric_snapshot(name);
307
308 // A separator the C library gives in wide form is taken as it is; otherwise the
309 // narrow one is converted whole.
310 const auto widen = [&name](const std::string& narrow,
311 const std::optional<std::wstring>& wide) {
312 if (wide)
313 return std::basic_string<CharT>(wide->begin(), wide->end());
314 if (narrow.empty())
315 return std::basic_string<CharT>();
316 if constexpr (std::is_same_v<CharT, wchar_t>)
317 return detail::to_wstring(narrow, name);
318 else
319 return detail::to_u32string<CharT>(narrow, name); // <CharT>: see facet_helper.h
320 };
321 m_decimal_point = widen(raw.decimal_point, raw.wdecimal_point);
322 if (m_decimal_point.empty())
323 m_decimal_point.assign(1, static_cast<CharT>('.'));
324 m_thousands_sep = widen(raw.thousands_sep, raw.wthousands_sep);
325 m_grouping = FacetHelper::lconv_grouping<CharT>(raw.grouping, m_thousands_sep, m_decimal_point);
326
327 // Empty counts as missing, as in numeric_conf<char>.
328 if (raw.truename.empty())
329 {
330 if constexpr (std::is_same_v<CharT, wchar_t>) m_true_name = L"true";
331 else m_true_name = U"true";
332 }
333 else
334 {
335 if constexpr(std::is_same_v<CharT, wchar_t>)
336 m_true_name = detail::to_wstring(raw.truename, name);
337 else
338 m_true_name = detail::to_u32string<CharT>(raw.truename, name); // <CharT>: see facet_helper.h
339 }
340
341 if (raw.falsename.empty())
342 {
343 if constexpr (std::is_same_v<CharT, wchar_t>) m_false_name = L"false";
344 else m_false_name = U"false";
345 }
346 else
347 {
348 if constexpr(std::is_same_v<CharT, wchar_t>)
349 m_false_name = detail::to_wstring(raw.falsename, name);
350 else
351 m_false_name = detail::to_u32string<CharT>(raw.falsename, name);
352 }
353 }
354
355public:
367 [[nodiscard]] virtual const std::basic_string<CharT>& decimal_point() const { return m_decimal_point; }
368
385 [[nodiscard]] virtual const std::basic_string<CharT>& thousands_sep() const { return m_thousands_sep; }
386

411 [[nodiscard]] virtual const std::basic_string<CharT>& falsename() const { return m_false_name; }
412
429 [[nodiscard]] virtual const std::vector<uint8_t>& grouping() const { return m_grouping; }
430
431private:
432 std::basic_string<CharT> m_decimal_point;

466template <typename CharT>
467 requires std::is_same_v<CharT, char8_t>
468class numeric_conf<CharT> : public ft_basic<numeric<char8_t>>
469{
470public:
480 using char_type = char8_t;
481
482public:
507 numeric_conf(const std::string& name)
508 : ft_basic<numeric<char8_t>>()
509 {
510 if (FacetHelper::is_c_locale_name(name))
511 { // "C" locale
512 m_decimal_point = u8".";
513 m_thousands_sep = u8",";
514 m_true_name = u8"true";
515 m_false_name = u8"false";
516 return;
517 }
518
519 if (clocale_wrapper(name).is_utf8())
520 {
521 // The narrow configuration's bytes are UTF-8 already.
522 const numeric_conf<char> narrow(name);
523 const auto bytes = [](const std::string& s) { return std::u8string(s.begin(), s.end()); };
524 m_decimal_point = bytes(narrow.decimal_point());
525 m_thousands_sep = bytes(narrow.thousands_sep());
526 m_grouping = narrow.grouping();
527 m_true_name = bytes(narrow.truename());
528 m_false_name = bytes(narrow.falsename());
529 }
530 else
531 init_from_wide<>(name);
532 }
533
534private:
535 // W only makes the body dependent, so where wchar_t is not Unicode the
536 // discarded branch, and the to_u8string it names, is never instantiated.
537 template <typename W = wchar_t>
538 void init_from_wide(const std::string& name)
539 {
540 if constexpr (wchar_t_is_utf32 || wchar_t_is_utf16)
541 {
542 const numeric_conf<W> wide(name);
543 const auto utf8 = [](const std::basic_string<W>& s) {
544 return detail::to_u8string(std::basic_string_view<W>(s));
545 };
546 m_decimal_point = utf8(wide.decimal_point());
547 m_thousands_sep = utf8(wide.thousands_sep());
548 m_grouping = wide.grouping();
549 m_true_name = utf8(wide.truename());
550 m_false_name = utf8(wide.falsename());
551 }
552 else
553 throw io_error("numeric_conf<char8_t>: " + name +
554 " is not a UTF-8 locale, and wchar_t is not Unicode on this platform");
555 }
