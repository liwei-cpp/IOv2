1583 template <typename OutIt>
1584 OutIt put(OutIt out, const std::tm& t, std::basic_string_view<CharT> fmt) const
1585 {
1586 using namespace std::chrono;
1587
1588 // Validate raw tm fields before constructing chrono types: year/month
1589 // truncate out-of-range values (short / unsigned char) and would silently
1590 // wrap, so range-check the integers first, then reject field combinations
1591 // that are individually in range but not a real calendar date (e.g. Feb 30).
1592 // tm_sec is intentionally limited to [0,59], NOT C's [0,60]: this facet
1593 // routes time-of-day through std::chrono::hh_mm_ss<seconds>, which cannot
1594 // represent a leap second (23:59:60 would normalise to 24:00:00 and be
1595 // mis-formatted), so a leap-second tm is rejected up front here rather
1596 // than silently corrupted on output.
1597 if (t.tm_mon < 0 || t.tm_mon > 11 ||
1598 t.tm_mday < 1 || t.tm_mday > 31 ||
1599 t.tm_hour < 0 || t.tm_hour > 23 ||
1600 t.tm_min < 0 || t.tm_min > 59 ||
1601 t.tm_sec < 0 || t.tm_sec > 59 ||
1602 t.tm_year > static_cast<int>(year::max()) - 1900 ||
1603 t.tm_year + 1900 < static_cast<int>(year::min()))
1604 throw stream_error("timeio put error: std::tm field out of range");
1605
1606 const int y = t.tm_year + 1900;
1607
1608 year_month_day ymd{ year{y}, month{static_cast<unsigned>(t.tm_mon) + 1}, day{static_cast<unsigned>(t.tm_mday)} };
1609 if (!ymd.ok())
1610 throw stream_error("timeio put error: std::tm is not a valid calendar date");
1611
1612 weekday wd{ymd};
1613
1614 seconds sec = hours{t.tm_hour} + minutes{t.tm_min} + seconds{t.tm_sec};
1615 std::chrono::hh_mm_ss hms{sec};
1616 return do_put(out, fmt, &ymd, &wd, &hms, nullptr);
1617 }

2645 const std::array<std::basic_string<CharT>, N>& abbr,

2647 {

2650 {
2651 if (full[i].empty() || abbr[i].empty())
2652 throw std::runtime_error(std::string("timeio: empty ") + what + " name in locale data");
2653 names[2 * i] = &full[i];

2659 if ((a / 2) != (b / 2) && *names[a] == *names[b])
2660 throw std::runtime_error(std::string("timeio: duplicate ") + what + " name in locale data");
2661 }

2770 template <typename OutIt>
2771 OutIt do_put(OutIt out, std::basic_string_view<CharT> format,
2772 const std::chrono::year_month_day* ymd,
2773 std::chrono::weekday* wd,
2774 const std::chrono::hh_mm_ss<std::chrono::seconds>* hms,
2775 const std::chrono::time_zone* tz) const
2776 {
2777 auto f = format.cbegin();
2778 while (f != format.cend())
2779 {
2780 if (*f != static_cast<CharT>('%'))
2781 {
2782 *out++ = *f++;
2783 continue;
2784 }
2785
2786 if (++f == format.cend()) break;
2787
2788 CharT modifier = 0;
2789 if (*f == static_cast<CharT>('E') || *f == static_cast<CharT>('O'))
2790 {
2791 modifier = *f++;
2792 if (f == format.cend()) break;
2793 }
2794
2795 CharT format_char = *f;
2796 switch (format_char)
2797 {
2798 case static_cast<CharT>('%'):
2799 if (modifier) goto bad_format;
2800 *out++ = *f;
2801 break;

2809 {

2825 }
2826 }

2829 case static_cast<CharT>('b'):
2830 case static_cast<CharT>('h'):

2832 else
2833 {

2835 if (m > 11) *out++ = static_cast<CharT>('?');
2836 else
2837 {
2838 const auto& mon = m_abbr_month[m];
