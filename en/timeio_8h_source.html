3785 void validate_format_recursion() const
3786 {
3787 // The node numbering. era_first is both the count of fixed nodes and the number of the
3788 // first era format: how many era formats a locale has varies, so they can only be
3789 // numbered from there on.
3790 enum class fmt_node : std::size_t
3791 {
3792 date_time, era_date_time,
3793 date, era_date,
3794 time, era_time,
3795 am_pm,
3796 era_first
3797 };
3798
3799 // The order here IS fmt_node; era formats take the numbers from era_first on.
3800 std::vector<const std::basic_string<CharT>*> nodes{
3801 &m_date_time_format, &m_era_date_time_format,
3802 &m_date_format, &m_era_date_format,
3803 &m_time_format, &m_era_time_format,
3804 &m_am_pm_format};
3805 assert(nodes.size() == std::to_underlying(fmt_node::era_first));
3806 for (const era_entry& entry : m_era_master) nodes.push_back(&entry.format);
3807
3808 // Modifier gating mirrors the three switches: %Oc / %Ox / %OX / %Er / %Or degrade to
3809 // literals there, so no edge here. %% needs no case -- the second % is consumed as the
3810 // specifier character. Brackets need none either: a %c inside a group still recurses.
3811 //
3812 // The graph deliberately over-approximates, since a shade too many edges costs nothing
3813 // while a missing one costs the guarantee: %EY edges to every era node (do_get does try
3814 // them all, do_put expands only the era the year lands in, and expand_and_filter leaves
3815 // %EY unexpanded), and an edge stands even where the value type could not supply the
3816 // specifier and the switch would drop it. A %EY inside an era format re-resolves to that
3817 // same era, so that self-loop is real rather than an approximation.
3818 auto scan_edges = [n_total = nodes.size()]
3819 (std::basic_string_view<CharT> fmt, auto on_edge)
3820 {
3821 for (auto p = fmt.cbegin(); p != fmt.cend(); )
3822 {
3823 if (*p != static_cast<CharT>('%')) { ++p; continue; }
3824 if (++p == fmt.cend()) break;
3825
3826 CharT modifier = 0;
3827 if (*p == static_cast<CharT>('E') || *p == static_cast<CharT>('O'))
3828 {
3829 modifier = *p;
3830 if (++p == fmt.cend()) break;
3831 }
3832
3833 const bool era = (modifier == static_cast<CharT>('E'));
3834 switch (*p)
3835 {
3836 case static_cast<CharT>('c'):
3837 if (modifier != static_cast<CharT>('O'))
3838 on_edge(era ? fmt_node::era_date_time : fmt_node::date_time);
3839 break;
3840
3841 case static_cast<CharT>('x'):
3842 if (modifier != static_cast<CharT>('O'))
3843 on_edge(era ? fmt_node::era_date : fmt_node::date);
3844 break;
3845
3846 case static_cast<CharT>('X'):
3847 if (modifier != static_cast<CharT>('O'))
3848 on_edge(era ? fmt_node::era_time : fmt_node::time);
3849 break;
3850
3851 case static_cast<CharT>('r'):
3852 if (!modifier) on_edge(fmt_node::am_pm);
3853 break;
3854
3855 case static_cast<CharT>('Y'):
3856 if (era)
3857 for (std::size_t i = std::to_underlying(fmt_node::era_first); i < n_total; ++i)
3858 on_edge(static_cast<fmt_node>(i));
3859 break;
3860
3861 default:
3862 break;
3863 }
3864 ++p;
3865 }
3866 };
3867
3868 // Three-colour DFS: 0 unvisited, 1 on the stack, 2 finished. A grey target is a back
3869 // edge, i.e. a cycle; a directly self-referential %c lands here with u == v.
3870 std::vector<char> color(nodes.size(), 0);
3871 auto visit = [&](auto&& self, std::size_t v) -> void
3872 {
3873 color[v] = 1;
3874 scan_edges(*nodes[v], [&](fmt_node node)
3875 {
3876 const std::size_t u = std::to_underlying(node);
3877 if (color[u] == 1)
3878 throw std::runtime_error("timeio: self-referential compound format in locale data");
3879 if (color[u] == 0) self(self, u);
3880 });
3881 color[v] = 2;
3882 };
3883
3884 for (std::size_t v = 0; v < nodes.size(); ++v)
3885 if (color[v] == 0) visit(visit, v);
3886 }

4564 anything precedes, forwards only at the very start. A closing bracket is never a
