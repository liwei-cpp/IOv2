3692 // degrades it to for a value with no zone to name.

3701 if (zone_res->is_abbrev) ctx.m_zone_abbrev = zone_res->text.c_str();

3708 if ((rp == rp_end) || (*rp != static_cast<CharT>('%')))
3709 {
3710 succ = false;
3711 return rp;
3712 }
3713
3714 ++rp;

3716 {
3717 if ((rp == rp_end) || (*rp != modifier))
3718 {
3719 succ = false;
3720 return rp;
3721 }
3722 ++rp;
3723 }
3724
3725 if ((rp == rp_end) || (*rp != *fmt))
3726 {
3727 succ = false;
3728 return rp;
3729 }
3730 ++rp;
3731 }
3732 }
3733 ++fmt;

3736 // A format tail consisting solely of whitespace / %n / %t matches
3737 // zero-or-more whitespace and is satisfied even at end of input,
3738 // matching POSIX strptime and std::get_time. Skip such a tail before
3739 // judging failure.

3741 {

3747 break;

3749 }
3750
3751 if ((fmt != _fmt.cend()) && (rp == rp_end))
3752 succ = false;
3753 return rp;
3754 }

3937 void validate_format_recursion() const
3938 {
3939 // The node numbering. era_first is both the count of fixed nodes and the number of the
3940 // first era format: how many era formats a locale has varies, so they can only be
3941 // numbered from there on.
3942 enum class fmt_node : std::size_t
3943 {
3944 date_time, era_date_time,
3945 date, era_date,
3946 time, era_time,
3947 am_pm,
3948 era_first
3949 };
3950
3951 // The order here IS fmt_node; era formats take the numbers from era_first on.
3952 std::vector<const std::basic_string<CharT>*> nodes{
3953 &m_date_time_format, &m_era_date_time_format,
3954 &m_date_format, &m_era_date_format,
3955 &m_time_format, &m_era_time_format,
3956 &m_am_pm_format};
3957 assert(nodes.size() == std::to_underlying(fmt_node::era_first));
3958 for (const era_entry& entry : m_era_master) nodes.push_back(&entry.format);
3959
3960 // Modifier gating mirrors the three switches: %Oc / %Ox / %OX / %Er / %Or degrade to
3961 // literals there, so no edge here. %% needs no case -- the second % is consumed as the
3962 // specifier character. Brackets need none either: a %c inside a group still recurses.
3963 //
3964 // The graph deliberately over-approximates, since a shade too many edges costs nothing
3965 // while a missing one costs the guarantee: %EY edges to every era node (do_get does try
3966 // them all, do_put expands only the era the year lands in, and expand_and_filter leaves
3967 // %EY unexpanded), and an edge stands even where the value type could not supply the
3968 // specifier and the switch would drop it. A %EY inside an era format re-resolves to that
3969 // same era, so that self-loop is real rather than an approximation.
3970 auto scan_edges = [n_total = nodes.size()]
3971 (std::basic_string_view<CharT> fmt, auto on_edge)
3972 {
3973 for (auto p = fmt.cbegin(); p != fmt.cend(); )
3974 {
3975 if (*p != static_cast<CharT>('%')) { ++p; continue; }
3976 if (++p == fmt.cend()) break;
3977
3978 CharT modifier = 0;
3979 if (*p == static_cast<CharT>('E') || *p == static_cast<CharT>('O'))
3980 {
3981 modifier = *p;
3982 if (++p == fmt.cend()) break;
3983 }
3984
3985 const bool era = (modifier == static_cast<CharT>('E'));
3986 switch (*p)
3987 {
3988 case static_cast<CharT>('c'):
3989 if (modifier != static_cast<CharT>('O'))
3990 on_edge(era ? fmt_node::era_date_time : fmt_node::date_time);
3991 break;
3992
3993 case static_cast<CharT>('x'):
3994 if (modifier != static_cast<CharT>('O'))
3995 on_edge(era ? fmt_node::era_date : fmt_node::date);
3996 break;
3997
3998 case static_cast<CharT>('X'):
3999 if (modifier != static_cast<CharT>('O'))
4000 on_edge(era ? fmt_node::era_time : fmt_node::time);
4001 break;
4002
4003 case static_cast<CharT>('r'):
4004 if (!modifier) on_edge(fmt_node::am_pm);
4005 break;
4006
4007 case static_cast<CharT>('Y'):
4008 if (era)
4009 for (std::size_t i = std::to_underlying(fmt_node::era_first); i < n_total; ++i)
4010 on_edge(static_cast<fmt_node>(i));
4011 break;
4012
4013 default:
4014 break;
4015 }
4016 ++p;
4017 }
4018 };
4019
4020 // Three-colour DFS: 0 unvisited, 1 on the stack, 2 finished. A grey target is a back
4021 // edge, i.e. a cycle; a directly self-referential %c lands here with u == v.
4022 std::vector<char> color(nodes.size(), 0);
4023 auto visit = [&](auto&& self, std::size_t v) -> void
4024 {
4025 color[v] = 1;
4026 scan_edges(*nodes[v], [&](fmt_node node)
4027 {
4028 const std::size_t u = std::to_underlying(node);
4029 if (color[u] == 1)
4030 throw std::runtime_error("timeio: self-referential compound format in locale data");
4031 if (color[u] == 0) self(self, u);
4032 });
4033 color[v] = 2;
4034 };
4035
4036 for (std::size_t v = 0; v < nodes.size(); ++v)
4037 if (color[v] == 0) visit(visit, v);
4038 }
