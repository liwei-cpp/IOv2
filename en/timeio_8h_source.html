231 [[nodiscard]] std::chrono::year_month_day compute_ymd() const
232 {
233 using namespace std::chrono;
234 if (m_have_year && m_have_mon && m_have_mday)
235 return year_month_day{ year{m_year}, month{static_cast<uint8_t>(m_month)}, day{m_mday} };
236 if (m_have_year && m_have_yday)
237 {
238 sys_days sd = sys_days{ year{m_year} / 1 / 1 } + days{ m_yday };
239 return year_month_day{sd};

242 auto deduced_year = m_year;
243 // Deduce year
244 if (!m_have_year)
245 {
246 if (m_have_century && m_have_year_in_century)
247 deduced_year = deduced_year % 100 + m_century * 100; // NOLINT(bugprone-branch-clone)
248 else if ((m_have_year_of_era) && (!m_era_items.empty()))
249 {
250 using namespace TimeioHelper;
251
252 // use month and day to decide
253 if (m_have_mon && m_have_mday)
254 {
255 auto it = m_era_items.begin();
256 for (; it != m_era_items.end(); ++it)
257 {
258 int64_t est_year_64 = static_cast<int64_t>(it->from_year)
259 + (static_cast<int64_t>(m_year_of_era) - static_cast<int64_t>(it->offset)) * it->direction;
260 int est_year = static_cast<int>(std::clamp<int64_t>(est_year_64,
261 std::numeric_limits<int>::min(), std::numeric_limits<int>::max()));
262 bool cmp1 = era_small_or_equal(it->from_year, it->from_month, it->from_day,
263 est_year, m_month, m_mday) &&
264 era_small_or_equal(est_year, m_month, m_mday,
265 it->to_year, it->to_month, it->to_day);
266 bool cmp2 = era_small_or_equal(est_year, m_month, m_mday,
267 it->from_year, it->from_month, it->from_day) &&
268 era_small_or_equal(it->to_year, it->to_month, it->to_day,
269 est_year, m_month, m_mday);
270 if (!cmp1 && !cmp2) continue;
271 deduced_year = est_year;
272 break;
273 }
274
275 // nothing matches, choose the first item.
276 if (it == m_era_items.end())
277 deduced_year = m_era_items.begin()->from_year;
278 }
279 else if (m_have_mon)
280 {
281 auto it = m_era_items.begin();
282 for (; it != m_era_items.end(); ++it)
283 {
284 int64_t est_year_64 = static_cast<int64_t>(it->from_year)
285 + (static_cast<int64_t>(m_year_of_era) - static_cast<int64_t>(it->offset)) * it->direction;
286 int est_year = static_cast<int>(std::clamp<int64_t>(est_year_64,
287 std::numeric_limits<int>::min(), std::numeric_limits<int>::max()));
288 bool cmp1 = era_small_or_equal(it->from_year, it->from_month, it->from_day,
289 est_year, m_month, m_mday) &&
290 era_small_or_equal(est_year, m_month, m_mday,
291 it->to_year, it->to_month, it->to_day);
292 bool cmp2 = era_small_or_equal(est_year, m_month, m_mday,
293 it->from_year, it->from_month, it->from_day) &&
294 era_small_or_equal(it->to_year, it->to_month, it->to_day,
295 est_year, m_month, m_mday);
296 if (!cmp1 && !cmp2) continue;
297 deduced_year = est_year;
298 break;
299 }
300
301 // nothing matches, choose the first item.
302 if (it == m_era_items.end())
303 deduced_year = m_era_items.begin()->from_year;
304 }
305 else
306 {
307 auto it = m_era_items.begin();
308 for (; it != m_era_items.end(); ++it)
309 {
310 int64_t est_year_64 = static_cast<int64_t>(it->from_year)
311 + (static_cast<int64_t>(m_year_of_era) - static_cast<int64_t>(it->offset)) * it->direction;
312 int est_year = static_cast<int>(std::clamp<int64_t>(est_year_64,
313 std::numeric_limits<int>::min(), std::numeric_limits<int>::max()));
314 const bool within_forward = it->from_year <= est_year && est_year <= it->to_year;
315 const bool within_backward = it->to_year <= est_year && est_year <= it->from_year;
316 if (!within_forward && !within_backward) continue;
317 deduced_year = est_year;
318 break;
319 }
320
321 // nothing matches, choose the first item.
322 if (it == m_era_items.end())
323 deduced_year = m_era_items.begin()->from_year;
324 }
325 }
326
327 // Fuzzy matching, we do not have enough information, but need to set year.
328 else if (m_have_year_in_century) { /* do nothing */ }
329 else if (m_have_century)
330 // %C with no year within the century: the year within the century is 0,
331 // as in POSIX strptime. Deriving it from m_year would make the result
332 // depend on whatever the fallback happens to be -- the wall-clock year
333 // for a default-constructed context, the caller's old value for a hinted
334 // one -- for a format string that says nothing about it.
335 deduced_year = m_century * 100;
336 else if (m_have_era && !m_era_items.empty())
337 deduced_year = m_era_items.begin()->from_year;
338 }
339
340 // ISO-8601 week date. Prefer the ISO year (%G); otherwise use the year deduced
341 // above -- an explicit %Y, a century plus %y, an era, or the fallback -- instead
342 // of dropping the week number. The two fully-specified branches above
343 // (year+mon+mday, year+yday) already returned, so reaching here means the date is
344 // not pinned down by an explicit month/day or day-of-year.
345 if (m_have_iso_8601_week && m_have_wday)
346 {
347 int iso_year = m_have_iso_8601_year ? m_iso_8601_year : deduced_year;
348 int iso_wd = (m_wday == 0 ? 7 : m_wday);
349 year_month_day jan4 = year{iso_year}/January/4;
350 weekday wd_jan4{sys_days{jan4}};
351 sys_days week1_monday = sys_days{jan4} - (wd_jan4 - Monday);
352 sys_days final = week1_monday + days{7 * (m_iso_8601_week - 1)} + days{iso_wd - 1};
353 return year_month_day{final};
354 }
355
356 auto deduced_month = m_month;
357 auto deduced_mday = m_mday;
358 // True when no deduction path below can touch the day, i.e. the day is whatever
359 // set_date_hint() or the default constructor left behind. Every assignment to
360 // deduced_mday sits inside one of those paths and is guarded by !m_have_mday.
361 // A lone weekday, with no week number and no month or day of its own, is resolved
362 // against the fallback date rather than against January 1: see the branch below.
363 const bool wday_snaps = m_have_wday && !m_have_mon && !m_have_mday
364 && !(m_have_yday || m_have_uweek || m_have_wweek);
365 const bool mday_is_fallback = !m_have_mday
366 && !(m_have_yday || m_have_uweek || m_have_wweek || wday_snaps);
367 int deduced_yday = static_cast<int>(m_yday);
368 bool have_yday = m_have_yday;
369 // Deduce month / mday. When neither is given, both are derived from the
370 // day-of-year. When the month IS given but the day is not, the day-of-month
371 // is computed relative to the *reported* month (deduced_month) rather than
372 // the yday-derived month, so the returned month and day stay mutually
373 // consistent (contradictory month-vs-week input is GIGO and may still yield
374 // an out-of-range day, since this conversion has no error channel).
375 if ((m_have_uweek || m_have_wweek) && m_have_wday && (!have_yday))
376 {
377 int w_offset = m_have_uweek ? 0 : 1;
378
379 // calculate the week of day for Jan 1
380 int wday = day_of_the_week(deduced_year, 1, 1);
381
382 deduced_yday = ((7 - (wday - w_offset)) % 7 + (m_week_no - 1) * 7 + (m_wday - w_offset + 7) % 7);
383 have_yday = true;
384 }
385
386 if (!m_have_mon || !m_have_mday)
387 {
388 if (have_yday)
389 {
390 if (deduced_yday < 0)
391 {
392 deduced_year -= 1;
393 deduced_yday += isleap(deduced_year) ? 366 : 365;
394 }
395 while (deduced_yday >= (isleap(deduced_year) ? 366 : 365))
396 {
397 deduced_yday -= isleap(deduced_year) ? 366 : 365;
398 deduced_year += 1;
399 }
400 int t_mon = 0;
401 while (t_mon < 12 && s_mon_yday[isleap(deduced_year)][t_mon] <= deduced_yday)
402 t_mon++;
403 if (!m_have_mon) deduced_month = t_mon;
404 if (!m_have_mday) deduced_mday = (deduced_yday - s_mon_yday[isleap(deduced_year)][deduced_month - 1] + 1);
405 }
406 else if (wday_snaps)
407 {
408 // A weekday on its own cannot locate a date; only the offset within some
409 // week is known. Resolve it as the first matching weekday on or after the
410 // fallback date, so the result stays within six days of the fallback
411 // instead of jumping to whichever week of the year happens to match.
412 auto last = static_cast<unsigned>(
413 year_month_day_last{year{deduced_year},
414 month_day_last{month{static_cast<unsigned>(deduced_month)}}}
415 .day());
416 sys_days base{year{deduced_year}
417 / month{static_cast<unsigned>(deduced_month)}
418 / day{std::min<unsigned>(deduced_mday, last)}};
419 auto cur = static_cast<int>(weekday{base}.c_encoding());
420 year_month_day snapped{base + days{(static_cast<int>(m_wday) - cur + 7) % 7}};
421 deduced_year = int(snapped.year());
422 deduced_month = static_cast<uint8_t>(static_cast<unsigned>(snapped.month()));
423 deduced_mday = static_cast<uint8_t>(static_cast<unsigned>(snapped.day()));
424 }
425 else if (m_have_uweek || m_have_wweek)
426 {
427 // assume wday is 1
428 int w_offset = m_have_uweek ? 0 : 1;
429 auto j1_wday = day_of_the_week(deduced_year, 1, 1);
430
431 if (!have_yday)
432 {
433 deduced_yday = ((7 - (j1_wday - w_offset)) % 7 + (m_week_no - 1) * 7 + (1 - w_offset + 7) % 7);
434 have_yday = true;
435 }
436
437 if (!m_have_mday || !m_have_mon)
438 {
439 if (deduced_yday < 0)
440 {
441 deduced_year -= 1;
442 deduced_yday += isleap(deduced_year) ? 366 : 365;
443 }
444 while (deduced_yday >= (isleap(deduced_year) ? 366 : 365))
445 {
446 deduced_yday -= isleap(deduced_year) ? 366 : 365;
447 deduced_year += 1;
448 }
449 int t_mon = 0;
450 while (t_mon < 12 && s_mon_yday[isleap(deduced_year)][t_mon] <= deduced_yday)
451 t_mon++;
452 if (!m_have_mon)
453 deduced_month = t_mon;
454 if (!m_have_mday)
455 deduced_mday = (deduced_yday - s_mon_yday[isleap(deduced_year)][deduced_month - 1] + 1);
456 }
457 }
458 }
459
460 // A day that came from the fallback rather than from the input must give way when
461 // it cannot exist in the deduced month: the fallback only supplies fields the
462 // format string did not parse, so it must never turn a well-formed parse into an
463 // invalid date. A day the input really did specify is left alone and reported as
464 // invalid.
465 if (mday_is_fallback && deduced_month >= 1 && deduced_month <= 12)
466 {
467 auto last = static_cast<unsigned>(
468 year_month_day_last{year{deduced_year},
469 month_day_last{month{static_cast<unsigned>(deduced_month)}}}
470 .day());
471 if (static_cast<unsigned>(deduced_mday) > last)
472 deduced_mday = static_cast<uint8_t>(last);
473 }
474
475 return year_month_day{ year{deduced_year}, month{static_cast<uint8_t>(deduced_month)}, day{static_cast<uint8_t>(deduced_mday)} };

511private:
512 constexpr static const std::array<std::array<unsigned short int, 13>, 2> s_mon_yday =
513 {{
514 // Row 0: days already gone by when each month opens in a common year.
515 {{ 0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334, 365 }},
516 // Row 1: the same for a leap year. The trailing entry of each row is the
517 // length of the year, so a day-of-year search can walk the row to its end
518 // without a special case.
519 {{ 0, 31, 60, 91, 121, 152, 182, 213, 244, 274, 305, 335, 366 }}
520 }};
521
522 // The Gregorian rule: every fourth year, except centuries, except every fourth
523 // century.
524 static bool isleap(int year)
525 {
526 return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
527 }
528
529 // Weekday of a date, 0 = Sunday. `month` is 1-based and expected to be a real
530 // month, 1 to 12.
531 //
532 // The count runs from 0001-01-01, which was a Monday in the proleptic Gregorian
533 // calendar, so the number of days elapsed since then is exactly one less than the
534 // weekday index.
535 //
536 // Years at or below zero are reachable here -- %Y accepts 0000 and %C a century
537 // of 0 -- and C++ division truncates towards zero, which would gain a day on every
538 // negative quotient in the leap-day terms below. Rather than special-case the
539 // sign, the year is first lifted above zero by a whole number of 400-year cycles.
540 // A cycle is 146097 days, which is exactly 20871 weeks, so shifting by any
541 // multiple of it moves the date without moving the weekday.
542 //
543 // (This is a deliberate correction, not a translation of what stood here before:
544 // the previous formulation was right for year >= 1 -- verified against this one
545 // over every int -- but reported 0000-01-01 as a Sunday when it was a Saturday.)
546 static int day_of_the_week(int year, int month, int mday)
547 {
548 constexpr int64_t cycles = 400LL * 5368710LL; // enough to lift any int year
549
550 const int64_t y = static_cast<int64_t>(year) - 1 + cycles;
551 const int64_t elapsed = 365 * y + y / 4 - y / 100 + y / 400
552 + s_mon_yday[isleap(year)][month - 1] + mday - 1;
553 return static_cast<int>((elapsed + 1) % 7);
554 }

3683 succ = false; return rp;

3690 ++n;
3691 if (rp == rp_end) break;

3694 ++rp;
3695 if (rp == rp_end || *rp < static_cast<CharT>('0')

3700 break;
3701 }
3702 }
3703 if (rp == rp_end) break;

3707 else if (n != 4)
3708 {
3709 // A digit run of any other length is not an offset.
3710 succ = false; return rp;
3711 }

3714 {
3715 succ = false; return rp;
3716 }

3720 }

3725 // degrades it to for a value with no zone to name.

3734 if (zone_res->is_abbrev) ctx.m_zone_abbrev = zone_res->text.c_str();

3742 {

3744 return rp;
3745 }
3746
3747 ++rp;
3748 if (modifier)
3749 {
3750 if ((rp == rp_end) || (*rp != modifier))
3751 {
3752 succ = false;
3753 return rp;
3754 }
3755 ++rp;
3756 }
3757
3758 if ((rp == rp_end) || (*rp != *fmt))
3759 {
3760 succ = false;
3761 return rp;
3762 }
3763 ++rp;
3764 }
3765 }

3770 // zero-or-more whitespace and is satisfied even at end of input,
3771 // matching POSIX strptime and std::get_time. So is a %p where the locale
3772 // has no designator, which likewise matches the empty string. Skip such a
3773 // tail before judging failure.
3774 while (fmt != _fmt.cend())
3775 {

3784 }
3785
3786 if ((fmt != _fmt.cend()) && (rp == rp_end))

3789 }

3973 void validate_format_recursion() const
3974 {
3975 // The node numbering. era_first is both the count of fixed nodes and the number of the
3976 // first era format: how many era formats a locale has varies, so they can only be
3977 // numbered from there on.
3978 // NOLINTNEXTLINE(performance-enum-size)
3979 enum class fmt_node : std::size_t
3980 {
3981 date_time, era_date_time,
3982 date, era_date,
3983 time, era_time,
3984 am_pm,
3985 era_first
3986 };
3987
3988 // The order here IS fmt_node; era formats take the numbers from era_first on.
3989 std::vector<const std::basic_string<CharT>*> nodes{
3990 &m_date_time_format, &m_era_date_time_format,
3991 &m_date_format, &m_era_date_format,
3992 &m_time_format, &m_era_time_format,
3993 &m_am_pm_format};
3994 assert(nodes.size() == std::to_underlying(fmt_node::era_first));
3995 for (const era_entry& entry : m_era_master) nodes.push_back(&entry.format);
3996
3997 // Modifier gating mirrors the three switches: %Oc / %Ox / %OX / %Er / %Or degrade to
3998 // literals there, so no edge here. %% needs no case -- the second % is consumed as the
3999 // specifier character. Brackets need none either: a %c inside a group still recurses.
4000 //
4001 // The graph deliberately over-approximates, since a shade too many edges costs nothing
4002 // while a missing one costs the guarantee: %EY edges to every era node (do_get does try
4003 // them all, do_put expands only the era the year lands in, and expand_and_filter leaves
4004 // %EY unexpanded), and an edge stands even where the value type could not supply the
4005 // specifier and the switch would drop it. A %EY inside an era format re-resolves to that
4006 // same era, so that self-loop is real rather than an approximation.
4007 auto scan_edges = [n_total = nodes.size()]
4008 (std::basic_string_view<CharT> fmt, auto on_edge)
4009 {
4010 for (auto p = fmt.cbegin(); p != fmt.cend(); )
4011 {
4012 if (*p != static_cast<CharT>('%')) { ++p; continue; }
4013 if (++p == fmt.cend()) break;
4014
4015 CharT modifier = 0;
4016 if (*p == static_cast<CharT>('E') || *p == static_cast<CharT>('O'))
4017 {
4018 modifier = *p;
4019 if (++p == fmt.cend()) break;
4020 }
4021
4022 const bool era = (modifier == static_cast<CharT>('E'));
4023 switch (*p)
4024 {
4025 case static_cast<CharT>('c'):
4026 if (modifier != static_cast<CharT>('O'))
4027 on_edge(era ? fmt_node::era_date_time : fmt_node::date_time);
4028 break;
4029
4030 case static_cast<CharT>('x'):
4031 if (modifier != static_cast<CharT>('O'))
4032 on_edge(era ? fmt_node::era_date : fmt_node::date);
4033 break;
4034
4035 case static_cast<CharT>('X'):
4036 if (modifier != static_cast<CharT>('O'))
4037 on_edge(era ? fmt_node::era_time : fmt_node::time);
4038 break;
4039
4040 case static_cast<CharT>('r'):
4041 if (!modifier) on_edge(fmt_node::am_pm);
4042 break;
4043
4044 case static_cast<CharT>('Y'):
4045 if (era)
4046 for (std::size_t i = std::to_underlying(fmt_node::era_first); i < n_total; ++i)
4047 on_edge(static_cast<fmt_node>(i));
4048 break;
4049
4050 default:
4051 break;
4052 }
4053 ++p;
4054 }
4055 };
4056
4057 // Three-colour DFS: 0 unvisited, 1 on the stack, 2 finished. A grey target is a back
4058 // edge, i.e. a cycle; a directly self-referential %c lands here with u == v.
4059 std::vector<char> color(nodes.size(), 0);
4060 auto visit = [&](auto&& self, std::size_t v) -> void
4061 {
4062 color[v] = 1;
4063 scan_edges(*nodes[v], [&](fmt_node node)
4064 {
4065 const std::size_t u = std::to_underlying(node);
4066 if (color[u] == 1)
4067 throw std::runtime_error("timeio: self-referential compound format in locale data");
4068 if (color[u] == 0) self(self, u);
4069 });
4070 color[v] = 2;
4071 };
4072
4073 for (std::size_t v = 0; v < nodes.size(); ++v)
4074 if (color[v] == 0) visit(visit, v);
4075 }

4749 void expand_and_filter(std::basic_string<CharT>& result, std::basic_string_view<CharT> fmt) const

4751 auto closer_for = [](CharT c)
4752 {
4753 if (c == static_cast<CharT>('(')) return static_cast<CharT>(')');
4754 if (c == static_cast<CharT>('[')) return static_cast<CharT>(']');
4755 if (c == static_cast<CharT>('{')) return static_cast<CharT>('}');
4756 return CharT();
4757 };
4758
4759 auto f = fmt.cbegin();
4760
4761 /* Drop one separator adjacent to whatever was just removed, or an orphaned piece of
4762 punctuation is left behind: "%T %Z" has to become "%T" rather than "%T ", and
4763 "%m/%d/%Y %T" on a time-only value "%T" rather than "// %T". Trim backwards when
4764 anything precedes, forwards only at the very start. A closing bracket is never a
4765 separator -- its opener still stands, and eating it would orphan that one -- and
4766 symmetrically an opening bracket is never one going forwards. */
4767 auto trim_separator = [&]
4768 {
4769 if (!result.empty())
4770 {
4771 auto n = result.size();

4773
4774 if (n > 0 && is_punct(result[n - 1]) &&
4775 result[n - 1] != static_cast<CharT>(')') &&
4776 result[n - 1] != static_cast<CharT>(']') &&

5207 template <typename TIter, std::sentinel_for<TIter> TSent>
5208 static TIter scan_int_field(TIter beg, TSent end, int& out, int min_val, int max_val, std::size_t width, bool& succ) // NOLINT(bugprone-easily-swappable-parameters)
5209 {
5210 constexpr auto zero = static_cast<CharT>('0');
5211 constexpr auto nine = static_cast<CharT>('9');
5212
5213 int value = 0;
5214 std::size_t digits = 0;
5215 bool fits = true;
5216
5217 // A digit is committed -- iterator advanced, counter bumped -- only once the
5218 // value it produces has been checked against `max_val`. So the digit that
5219 // pushes the field past its ceiling is left unread for whatever parses next,
5220 // while `value` keeps it: that is what stops the range test below from
5221 // mistaking the truncated prefix for a field that fits. `max_val` is at most
5222 // 9999 over `width` <= 4 digits, so the running value cannot overflow `int`.
5223 while (digits < width && beg != end)
5224 {
5225 const CharT c = *beg;
5226 if (c < zero || c > nine)
5227 break;
5228
5229 value = value * 10 + static_cast<int>(c - zero);
5230 if (value > max_val)
5231 {
5232 fits = false;
5233 break;
5234 }
5235
5236 ++beg;
5237 ++digits;
5238 }
5239
5240 // `fits` already implies `value <= max_val`, so only the lower bound is
5241 // left to check. Failure never clears a `succ` the caller had set.
5242 if (fits && digits != 0 && value >= min_val)
5243 out = value;
5244 else
5245 succ = false;
5246
5247 return beg;
