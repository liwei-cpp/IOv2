189 explicit basic_file_device(const std::filesystem::path& file_name, file_open_flag flags = file_open_flag::none)
190 {
191 const char* mode = fopen_mode(flags);
192 m_file.reset(detail::file_open(file_name, mode));
193 if (!m_file)
194 {
195 const int err = errno;
196 throw device_error("cannot open file \"" + detail::path_for_message(file_name)
197 + "\" (errno=" + std::to_string(err) + ")");
198 }
199
200 if (detail::file_seek(m_file.get(), 0, SEEK_END) != 0)
201 throw device_error("cannot get file length for \"" + detail::path_for_message(file_name) + "\"");
202
203 int64_t len = detail::file_tell(m_file.get());
204 if (len < 0)
205 throw device_error("cannot get file length for \"" + detail::path_for_message(file_name) + "\"");
206 else if (!std::in_range<std::size_t>(len))
207 throw device_error("file too large for this platform");
208
209 if (detail::file_seek(m_file.get(), 0, SEEK_SET) != 0)
210 throw device_error("cannot reset the file for \"" + detail::path_for_message(file_name) + "\"");
211
212 m_file_len = static_cast<std::size_t>(len);
213 }

358 void close()
359 {
360 if (is_open())
361 {
362 std::exception_ptr ptr = nullptr;
363 if constexpr (IsOut)
364 {
365 try {
366 dflush();
367 } catch (...) {
368 ptr = std::current_exception();
369 }
370 }
371 // Not m_file.reset(): the deleter has to discard fclose()'s result, and a close
372 // can fail on its own even after a successful flush (EIO on close(2), say).
373 FILE* fp = m_file.release();
374 m_file_len = 0;
375 m_last_dir = {};
376 if (std::fclose(fp) != 0 && !ptr) // NOLINT(cppcoreguidelines-owning-memory)
377 {
378 const int err = errno;
379 ptr = std::make_exception_ptr(
380 device_error("cannot close file (errno=" + std::to_string(err) + ")"));

499 void dseek(std::size_t v)
500 {
501 if (!is_open())
502 throw device_error("file_device::dseek fail: file is closed");
503
504 // In input-enabled modes, seeking is restricted to the current file length.
505 // This ensures that any subsequent read operation (dget) starts from a valid
506 // position within the existing data bounds.
507 if constexpr (IsIn)
508 {
509 if (v > m_file_len)
510 throw device_error("file_device::dseek fail: invalid parameter");
511 }
512
513 if (!std::in_range<int64_t>(v))
514 throw device_error("file_device::dseek fail: position exceeds maximum seekable range");
515
516 if (detail::file_seek(m_file.get(), static_cast<int64_t>(v), SEEK_SET) != 0)
517 throw device_error("file_device::dseek fail: fseek invalid response");
518 m_last_dir = {};
519 }

570 void dput(const CharType* ch, std::size_t n)
571 requires (IsOut)
572 {
573 if (n == 0) return;
574 if (ch == nullptr)
575 throw device_error("file_device::dput fail: null buffer");
576 if (!is_open())
577 throw device_error("file_device::dput fail: file is closed");
578 if constexpr (IsIn)
579 {
580 if (!turn_to(io_dir::write))
581 throw device_error("file_device::dput fail: cannot switch from reading to writing");
582 }
583 std::size_t written = std::fwrite(ch, sizeof(CharType), n, m_file.get());
584
585 auto res = detail::file_tell(m_file.get());
586 if (res >= 0)
587 {
588 if (!std::in_range<std::size_t>(res))
589 throw dput_error("file_device::dput fail: file size overflow", written);
590 if (static_cast<std::size_t>(res) > m_file_len)
591 m_file_len = static_cast<std::size_t>(res);
592 }
593 else
594 throw dput_error("file_device::dput fail: cannot determine file position", written);
595
596 if (written != n)
597 throw dput_error("file_device::dput fail: partial write", written);
598 }

619 if (std::fflush(m_file.get()) == EOF)
620 throw device_error("file_device::dflush fail: fflush error");

624private:
625 static const char* fopen_mode(file_open_flag flags)
626 {
627 using enum file_open_flag;
628
629 // Always binary: a text-mode FILE translates newlines below the converters, which
630 // breaks their position arithmetic. The deprecated `binary` bit is masked off here
631 // (not named, which would warn in every user's build).
632 file_open_flag checkres = flags & (trunc | noreplace);
633
634 if constexpr (IsIn && IsOut)
635 {
636 if (checkres == none)
637 return "r+b";
638 else if (checkres == trunc)
639 return "w+b";
640 // noreplace (O_EXCL) implies creating a new file, so it's always empty.
641 // Therefore, trunc is implicit in this mode.
642 else if (checkres == noreplace || checkres == (trunc | noreplace))
643 return "w+bx";
644 else
645 throw device_error("Invalid open mode combination for read-write file");
646 }
647 else if constexpr (IsIn)
648 {
649 if (checkres == none)
650 return "rb";
651 else
652 throw device_error("Invalid open mode combination for read-only file");
653 }
654 else if constexpr (IsOut)
655 {
656 // In write-only mode, "w" already truncates by default if noreplace is not present.
657 if (checkres == none || checkres == trunc)
658 return "wb";
659 // noreplace implies creation of a new file.
660 else if (checkres == noreplace || checkres == (trunc | noreplace))
661 return "wbx";
662 else
663 throw device_error("Invalid open mode combination for write-only file");
664 }
665 else
666 {
667 static_assert(dependent_false_nttp_v<IsIn, IsOut>, "file device cannot open with neither in nor out mode");
668 }
669 }

671private:
672 std::unique_ptr<FILE, file_deleter> m_file;
673
674 // Cached file length, in bytes.
675 //
676 // Type is `size_t` (not `uint64_t`) because the public API — `dsize()`,
677 // `dseek(size_t)`, `drseek(size_t)` — is `size_t`-typed; using `size_t`
678 // here keeps the value lossless across the boundary on every platform
679 // (including 32-bit, where a `uint64_t` field could hold values that
680 // `dsize()` cannot return without truncation).
681 //
682 // Invariant: `m_file_len <= INT64_MAX`. All writes to this field source
683 // from a non-negative `int64_t` (the result of `detail::file_tell()`), narrowed
684 // through `std::in_range<size_t>` in the constructor and `dput()`.
685 // This invariant is what makes `-static_cast<int64_t>(offset)` in
686 // `drseek()` safe: any `offset <= m_file_len` fits in `int64_t`'s
687 // positive range, so the negation cannot overflow. Do not introduce
688 // setters or alternate sources that bypass the `int64_t`-derived path.
689 std::size_t m_file_len = 0;
690
691 // Which way the FILE was last used. C requires a seek or flush between
692 // reading and writing on one stream (C17 7.21.5.3p7); glibc lets it go,
693 // the MSVC CRT fails the write. Only a read-write device can switch.
694 enum class io_dir : std::uint8_t { none, read, write };
695
696 IOV2_NO_UNIQUE_ADDRESS
697 std::conditional_t<IsIn && IsOut, io_dir, std::monostate> m_last_dir{};
698
699 // Records that the next operation goes `to`, seeking to the current
700 // position first if the last one went the other way. False if that seek failed.
701 bool turn_to(io_dir to)
702 requires (IsIn && IsOut)
703 {
704 if (m_last_dir != io_dir::none && m_last_dir != to
705 && detail::file_seek(m_file.get(), 0, SEEK_CUR) != 0)
706 return false;
707 m_last_dir = to;
708 return true;
709 }
