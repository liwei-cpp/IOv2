334 void adjust(const cvt_behavior& acc)
335 {
336 if (this->is_tainted()) recover();
337 if (const auto* ptr = dynamic_cast<const code_cvt_switch*>(&acc); ptr)
338 {
339 if (!this->m_cvt_kernel.is_init_state() || this->m_cvt_kernel.is_mid_seq())
340 throw cvt_error("code_cvt_stdio::adjust fail: invalid state");
341 // Build the new kernel before committing: the commit move below MUST be
342 // noexcept, otherwise a partial throw would leave the converter without a
343 // usable kernel.
344 static_assert(
345 std::is_nothrow_move_assignable_v<codecvt_kernel<char, wchar_t>>,
346 "codecvt_kernel<char, wchar_t> move-assign must be noexcept; "
347 "otherwise code_cvt_stdio::adjust loses its strong exception guarantee");
348 codecvt_kernel<char, wchar_t> new_kernel(ptr->code);
349 if (new_kernel.is_state_dep())
350 throw cvt_error("code_cvt_stdio::adjust fail: state-dependent encoding");
351 this->m_cvt_kernel = std::move(new_kernel);
352 }
353 else if (const auto* st = dynamic_cast<const code_cvt_stdio_state*>(&acc); st)
354 {
355 if (!st->kernel)

361 "codecvt_kernel<char, wchar_t> destructor must be noexcept; "
362 "otherwise putting a code_cvt_stdio_state back can fail after the move");
363 this->m_cvt_kernel = std::move(*st->kernel);
364 st->kernel.reset();
365 }
