290 // The kernel takes a contiguous run; epc() bytes per character leave room for all.
291 const std::wstring chars(first, last);

293 const wchar_t* from = chars.data();

296 throw cvt_error("code_cvt_stdio::unget fail: cannot encode");

343 void adjust(const cvt_behavior& acc)
