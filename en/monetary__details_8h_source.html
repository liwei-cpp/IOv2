201 static pattern s_construct_pattern(pattern_spec spec)
202 {
203 using enum part;
204 const auto [precedes, sp, posn] = spec;
205
206 // sign_posn is the only field with a range to fall outside of; POSIX fills it
207 // with CHAR_MAX when the locale has nothing to say, and there is nothing to
208 // build a layout from.
209 if (posn < 0 || posn > 4)
210 return s_default_pattern;
211
212 // Two things get laid out: the amount, and the currency symbol -- which for
213 // sign_posn 3 and 4 has the sign glued to its front or its back, and so
214 // travels with it. Whichever of the two goes first is what `precedes` says.
215 std::array<part, 2> group{};
216 std::size_t group_len = 0;
217 if (posn == 3)
218 group[group_len++] = sign;
219 group[group_len++] = symbol;
220 if (posn == 4)
221 group[group_len++] = sign;
222
223 pattern ret{};
224 std::size_t at = 0;
225
226 // A sign that was not glued to the symbol brackets the whole field instead:
227 // in front for sign_posn 0 and 1, behind for 2.
228 if (posn <= 1)
229 ret[at++] = sign;
230
231 // The separator lives on the seam between the symbol and the amount, which is
232 // the one place a space is allowed to be: it can open no field and close none.
233 if (precedes)
234 {
235 for (std::size_t k = 0; k < group_len; ++k)
236 ret[at++] = group[k];
237 if (sp)
238 ret[at++] = space;
239 ret[at++] = value;
240 }
241 else
242 {
243 ret[at++] = value;
244 if (sp)
245 ret[at++] = space;
246 for (std::size_t k = 0; k < group_len; ++k)
247 ret[at++] = group[k];
248 }
249
250 if (posn == 2)
251 ret[at++] = sign;
252
253 // Without a space the field is one slot short. `none` takes up the slack, and
254 // being last is the one position it is allowed to hold.
255 if (at < ret.size())
256 ret[at++] = none;
257
258 return ret;
259 }

456 // No lc-> access beyond this point: the conversions may
457 // invalidate the lconv pointers.

461 // the fallbacks below regress to '.' / ','.

494 }

497 else if (!grouping_raw.empty())
498 {
499 // Normalise raw POSIX grouping bytes into the internal
500 // convention here at the POSIX boundary.
501 m_grouping = std::move(grouping_raw);

503 }
504
505 if (m_thousands_sep == m_decimal_point)
506 m_grouping.clear();
507 }
