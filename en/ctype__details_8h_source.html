75// The ctype lookup tables in this header and in ctype.h are sized as
76// std::numeric_limits<unsigned char>::max() + 1, on the assumption that a
77// byte is 8 bits and that the table covers all values an unsigned char can
78// hold. Exotic platforms with CHAR_BIT != 8 (some DSPs, historical mainframes)
79// would silently grow every per-byte table by 2^(CHAR_BIT-8)x. Reject them at
80// compile time rather than ship a non-obviously-broken binary.
81static_assert(CHAR_BIT == 8,
82 "IOv2/facet/ctype tables assume an 8-bit byte; see ctype.h::s_len");

216class ctype_conf<char> : public ft_basic<ctype<char>>
217{
218public:
234 ctype_conf(const std::string& name)
235 : ft_basic<ctype<char>>()
236 , m_inter_locale(name.c_str())
237 {
238 // Build lookup tables using standard POSIX functions
239 for (unsigned c = 0; c <= std::numeric_limits<unsigned char>::max(); ++c)
240 {
241 m_toupper_table[c] = toupper_l(static_cast<int>(c), m_inter_locale.c_locale);
242 m_tolower_table[c] = tolower_l(static_cast<int>(c), m_inter_locale.c_locale);
243
244 mask m = 0;
245 if (isupper_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::upper;
246 if (islower_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::lower;
247 if (isalpha_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::alpha;
248 if (isdigit_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::digit;
249 if (isxdigit_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::xdigit;
250 if (isspace_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::space;
251 if (isprint_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::print;
252 if (iscntrl_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::cntrl;
253 if (ispunct_l(static_cast<int>(c), m_inter_locale.c_locale)) m |= base_ft<ctype>::punct;
254 m_table[c] = m;
255 }
256 }
257
258public:
277 [[nodiscard]] virtual mask is(char c) const
278 {
279 return m_table[static_cast<unsigned char>(c)];
280 }
281
300 [[nodiscard]] virtual char toupper(char c) const
301 {
302 return static_cast<char>(m_toupper_table[static_cast<unsigned char>(c)]);
303 }
304
323 [[nodiscard]] virtual char tolower(char c) const
324 {
325 return static_cast<char>(m_tolower_table[static_cast<unsigned char>(c)]);
326 }
327
345 [[nodiscard]] virtual char widen(char c) const { return c; }
346
364 [[nodiscard]] virtual std::optional<char> narrow(char c) const { return c; }
365
366private:
369 clocale_wrapper m_inter_locale;
370
374 std::array<int, std::numeric_limits<unsigned char>::max() + 1> m_toupper_table {};
375
379 std::array<int, std::numeric_limits<unsigned char>::max() + 1> m_tolower_table {};
380
384 std::array<mask, std::numeric_limits<unsigned char>::max() + 1> m_table {};

509 [[nodiscard]] virtual base_ft<ctype>::mask is(CharT _c) const
510 {
511 if (out_of_wchar_range(_c)) return 0;
512
513 base_ft<ctype>::mask res = 0;
514 const auto c = static_cast<wchar_t>(_c);
515 if (iswctype_l(c, m_wmask_upper, m_inter_locale.c_locale)) res |= base_ft<ctype>::upper;
516 if (iswctype_l(c, m_wmask_lower, m_inter_locale.c_locale)) res |= base_ft<ctype>::lower;
517 if (iswctype_l(c, m_wmask_alpha, m_inter_locale.c_locale)) res |= base_ft<ctype>::alpha;
518 if (iswctype_l(c, m_wmask_digit, m_inter_locale.c_locale)) res |= base_ft<ctype>::digit;
519 if (iswctype_l(c, m_wmask_xdigit, m_inter_locale.c_locale)) res |= base_ft<ctype>::xdigit;
520 if (iswctype_l(c, m_wmask_space, m_inter_locale.c_locale)) res |= base_ft<ctype>::space;
521 if (iswctype_l(c, m_wmask_print, m_inter_locale.c_locale)) res |= base_ft<ctype>::print;
522 if (iswctype_l(c, m_wmask_cntrl, m_inter_locale.c_locale)) res |= base_ft<ctype>::cntrl;
