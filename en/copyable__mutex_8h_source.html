187class copyable_mutex

189public:
190 copyable_mutex() noexcept

196 // Copy/move construct a fresh, unlocked mutex; no lock state is carried over.
197 copyable_mutex(const copyable_mutex&) noexcept : copyable_mutex() {}
198 copyable_mutex& operator=(const copyable_mutex&) noexcept { return *this; }

203
204 void lock() { refresh_after_fork(); m_mutex.lock(); }
205 void unlock() { m_mutex.unlock(); }

223private:
224 // In the child of a fork() the mutex may be held by a thread that did not come along;
225 // the first lock there rebuilds it unlocked. Never taken in a process that has not forked.
226 void refresh_after_fork()
227 {
228 const unsigned gen = detail::copyable_mutex_fork_state().generation.load(std::memory_order_relaxed);
229 if (m_generation.load(std::memory_order_acquire) != gen) [[unlikely]]
230 rebuild(gen);
231 }
232
233 void rebuild(unsigned gen)
234 {
235 std::lock_guard guard(detail::copyable_mutex_fork_state().refresh);
236 if (m_generation.load(std::memory_order_relaxed) != gen)
237 {
238 std::construct_at(&m_mutex);
239 m_generation.store(gen, std::memory_order_release);
240 }
241 }
242
243 TMutex m_mutex;
244 std::atomic<unsigned> m_generation;
245};
