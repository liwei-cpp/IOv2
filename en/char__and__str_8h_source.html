61template <typename TIter, typename TChar>
62 requires (char_sink_for<TIter, TChar>)
63TIter ostream_insert(TIter iter, ios_base<TChar>& io, const TChar* s, std::size_t n)
64{
65 const std::size_t w = io.width();
66 // Consumed up front, so it cannot leak into the next insertion if a step below throws.
67 io.width(0);
68 if (w > n)
69 {
70 const std::size_t pad = w - n;
71 if (pad > ios_defs::max_pad_count)
72 throw stream_error("ostream insert fail: fill count exceeds max_pad_count");
73
74 const bool left = ((io.flags() & ios_defs::adjustfield) == ios_defs::left);
75 const TChar f = io.fill();
76 if (!left)
77 iter = std::fill_n(iter, pad, f);
78 iter = std::copy(s, s + n, iter);
79 if (left)
80 iter = std::fill_n(iter, pad, f);
81 }
82 else
83 iter = std::copy(s, s + n, iter);
84 return iter;
85}

137template <typename TIter, std::sentinel_for<TIter> TSent, typename TChar>
138 requires (std::is_same_v<TChar, typename TIter::value_type>)
139TIter istream_extract(TIter iter, TSent iter_end, ios_base<TChar>& io, const locale<TChar>& loc, TChar* s, std::size_t num)
140{
141 const std::size_t width = io.width();
142 // Consumed up front, so it cannot leak into the next extraction if a step below throws.
143 io.width(0);
144 if (0 < width && width < num)
145 num = width;
146
147 if (num <= 1)
148 {
149 if (num == 1) *s = TChar{};
150 throw stream_error("Do not have enough buffer to save character");
151 }
152
153 auto ct = loc.template get<ctype<TChar>>();
154 if (!ct)
155 {
156 *s = TChar{};
157 throw stream_error("cannot get ctype facet");
158 }
159
160 std::size_t extracted = 0;
161
162 while (extracted < num - 1
163 && (iter != iter_end)
164 && !(ct->is_any(base_ft<ctype>::space, *iter)))
165 {
166 *s++ = *iter;
167 ++extracted;
168 ++iter;
169 }
170
171 *s = TChar{};
172
173 if (extracted == 0)
174 throw stream_error("istream extraction fail: no characters extracted");
175
176 return iter;
177}

610 template <typename TIter, std::sentinel_for<TIter> TSent>
611 requires (std::is_same_v<TChar, typename TIter::value_type>)
612 static TIter sread(TIter iter, TSent iter_end, ios_base<TChar>& io, const locale<TChar>& loc, std::basic_string<TChar, TTraits, TAlloc>& str)
613 {
614 str.erase();
615 TChar buf[128];
616 std::size_t len = 0;
617 const std::size_t w = io.width();
618 io.width(0);
619 const std::size_t n = w > 0 ? w : str.max_size();
620 std::size_t extracted = 0;
621
622 auto ct = loc.template get<ctype<TChar>>();
623 if (!ct)
624 throw stream_error("cannot get ctype facet");
625 while (extracted < n
626 && (iter != iter_end)
627 && !(ct->is_any(base_ft<ctype>::space, *iter)))
628 {
629 if (len == 128)
630 {
631 str.append(buf, 128);
632 len = 0;
633 }
634 buf[len++] = *iter;
635 ++extracted;
636 ++iter;
637 }
638 str.append(buf, len);
639
640 if (extracted == 0)
641 throw stream_error("istream extraction fail: no characters extracted");
642
643 return iter;
644 }
