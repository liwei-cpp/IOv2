13template <typename TIter, typename TChar>
14 requires (std::is_same_v<TChar, typename TIter::value_type>)
15TIter ostream_insert(TIter iter, ios_base<TChar>& io, const TChar* s, std::streamsize n)
16{
17 const std::streamsize w = io.width();
18 if (w > n)
19 {
20 const bool left = ((io.flags() & ios_defs::adjustfield) == ios_defs::left);
21 const TChar f = io.fill();
22 if (!left)
23 iter = std::fill_n(iter, w - n, f);
24 iter = std::copy(s, s + n, iter);
25 if (left)
26 iter = std::fill_n(iter, w - n, f);
27 }
28 else
29 iter = std::copy(s, s + n, iter);
30 io.width(0);
31 return iter;
32}

34template <typename TIter, std::sentinel_for<TIter> TSent, typename TChar>
35 requires (std::is_same_v<TChar, typename TIter::value_type>)
36TIter istream_extract(TIter iter, TSent iter_end, ios_base<TChar>& io, const locale<TChar>& loc, TChar* s, std::streamsize num)
37{
38 std::streamsize width = io.width();
39 if (0 < width && width < num)
40 num = width;
41
42 if (num <= 1)
43 {
44 if (num == 1) *s = TChar{};
45 io.width(0);
46 throw stream_error("DO not have enough buffer to save character");
47 }
48
49 auto ct = loc.template get<ctype<TChar>>();
50 if (!ct)
51 throw stream_error("cannot get ctype facet");
52
53 if (iter == iter_end) throw stream_error("cannot extract character from empty stream");
54
55 std::streamsize extracted = 0;
56
57 while (extracted < num - 1
58 && (iter != iter_end)
59 && !(ct->is_any(base_ft<ctype>::space, *iter)))
60 {
61 *s++ = *iter;
62 ++extracted;
63 ++iter;
64 }
65
66 *s = TChar{};
67
68 io.width(0);
69
70 if (extracted == 0)
71 throw stream_error("istream extraction fail: no characters extracted");
72
73 return iter;
74}
