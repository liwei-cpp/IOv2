351 // NOLINTBEGIN(bugprone-use-after-move): each `std::move(other)` below binds to a
352 // *different* base subobject, and every one of those base assignments touches only its

354 // after being moved from -- clang-tidy just cannot see that the operands are disjoint.
355 ios_state<TChar>::operator=(std::move(other));

357 stream_common_operators::operator=(std::move(other));
358 m_streambuf = std::move(other.m_streambuf);
