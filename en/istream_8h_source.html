354 istream& operator=(istream&& other) noexcept
355 {

359 // NOLINTBEGIN(bugprone-use-after-move): each `std::move(other)` below binds to a
360 // *different* base subobject, and every one of those base assignments touches only its

362 // after being moved from -- clang-tidy just cannot see that the operands are disjoint.
363 ios_state<TChar>::operator=(std::move(other));
364 istream_operators<TChar>::operator=(std::move(other));
365 stream_common_operators::operator=(std::move(other));
366 m_streambuf = std::move(other.m_streambuf);
