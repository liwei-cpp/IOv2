456 static std::unordered_map<std::u8string, std::u8string> get_translate_dictionary(const std::string& filename)
457 {
458 std::FILE* fp = fopen(filename.c_str(), "rb"); // NOLINT(cppcoreguidelines-owning-memory)
459 if (!fp)
460 throw stream_error("get_translate_dictionary fail: cannot open file " + filename);
461
462 file_closer guard(fp);
463
464 if (std::fseek(fp, 0, SEEK_END) != 0)
465 throw stream_error("get_translate_dictionary fail: invalid format");
466 const long file_size = std::ftell(fp);
467 if (file_size < 0)
468 throw stream_error("get_translate_dictionary fail: file inconsistent");
469 if (std::fseek(fp, 0, SEEK_SET) != 0)
470 throw stream_error("get_translate_dictionary fail: invalid format");
471
472 auto read_num = [fp](unsigned char* buf, bool need_swap)
473 {
474 if (std::fread(buf, 1, 4, fp) != 4)
475 throw stream_error("get_translate_dictionary fail: invalid format");
476
477 std::uint32_t res = 0;
478 std::memcpy(&res, buf, 4);
479 if (need_swap)
480 res = std::byteswap(res);
481 return res;
482 };
483
484 std::array<unsigned char, 8> buff{};
485 if (std::fread(buff.data(), 1, 8, fp) != 8)
486 throw stream_error("get_translate_dictionary fail: invalid format");
487
488 bool need_swap = false;
489 if ((buff[0] == 0x95) && (buff[1] == 0x04) && (buff[2] == 0x12) && (buff[3] == 0xde))
490 need_swap = (std::endian::native == std::endian::little);
491 else if ((buff[0] == 0xde) && (buff[1] == 0x12) && (buff[2] == 0x04) && (buff[3] == 0x95))
492 need_swap = (std::endian::native == std::endian::big);
493 else throw stream_error("get_translate_dictionary fail: invalid format");
494
495 // Accept any .mo whose major revision (the high 16 bits) is 0, regardless of
496 // minor revision. Only minor 0 (basic) and minor 1 (system-dependent strings)
497 // are defined today, but minor revisions are backward-compatible additions:
498 // they only add separate tables (sysdep strings, hash table) and never change
499 // the regular orig/trans tables this parser reads. Sysdep strings live in their
500 // own orig/trans_sysdep tables we never touch, so a sysdep string is simply
501 // absent (lookup misses and the caller keeps the original) rather than returned
502 // with its segment markers unexpanded. A non-zero major revision uses a layout
503 // this basic parser does not commit to, so it is rejected.
504 std::uint32_t revision = 0;
505 std::memcpy(&revision, buff.data() + 4, 4);
506 if (need_swap)
507 revision = std::byteswap(revision);
508 if ((revision >> 16) != 0)
509 throw stream_error("get_translate_dictionary fail: unsupported .mo major revision");
510
511 auto str_num = read_num(buff.data(), need_swap);
512 auto ori_offset = read_num(buff.data(), need_swap);
513 auto aim_offset = read_num(buff.data(), need_swap);
514
515 // Each entry needs an 8-byte (length, offset) descriptor in both the
516 // original and translation tables, so a valid str_num cannot exceed
517 // file_size / 16. Rejecting larger counts caps the O(str_num) growth of
518 // oris and the result map, which the per-string length cap does not.
519 if (str_num > static_cast<std::uint64_t>(file_size) / 16u)
520 throw stream_error("get_translate_dictionary fail: implausible string count");
521
522 // The .mo file offsets (and string lengths) are unsigned 32-bit, but
523 // std::fseek takes a signed long. Where long is 64-bit (e.g. LP64
524 // Linux) every uint32_t is representable, so all seeks below are exact.
525 // On a 32-bit-long platform (LLP64/ILP32) an offset >= 2 GiB would
526 // convert to a negative long and seek wrongly — but that can only arise
527 // from a > 2 GiB or corrupt .mo, and every fseek/fread return value is
528 // checked, so the worst case is a clean "invalid format" throw, never a
529 // misread or out-of-bounds access.
530 if (std::fseek(fp, ori_offset, SEEK_SET) != 0)
531 throw stream_error("get_translate_dictionary fail: invalid format");
532
533 // The .mo format stores each string length as an unbounded 32-bit
534 // integer, so the buffer is grown to fit each entry rather than
535 // assuming a fixed maximum. A sanity cap guards against a corrupt or
536 // malicious file requesting a huge allocation.
537 constexpr std::uint32_t k_max_str_len = 64u * 1024u * 1024u; // 64 MiB
538 std::uint64_t total_str_bytes = 0;
539 std::vector<char8_t> str_buf;
540 std::vector<std::u8string> oris;
541 for (std::size_t i = 0; i < str_num; ++i)
542 {
543 auto length = read_num(buff.data(), need_swap);
544 auto offset = read_num(buff.data(), need_swap);
545 auto cur_pos = std::ftell(fp);
546 if (cur_pos == -1L) throw stream_error("get_translate_dictionary fail: file inconsistent");
547
548 if (length > k_max_str_len)
549 throw stream_error("get_translate_dictionary fail: string too long");
550 total_str_bytes += length;
551 if (total_str_bytes > static_cast<std::uint64_t>(file_size))
552 throw stream_error("get_translate_dictionary fail: total string bytes exceed file size");
553 if (str_buf.size() < static_cast<std::size_t>(length) + 1)
554 str_buf.resize(static_cast<std::size_t>(length) + 1);
555
556 if (std::fseek(fp, offset, SEEK_SET) != 0)
557 throw stream_error("get_translate_dictionary fail: invalid format");
558
559 if (std::fread(str_buf.data(), 1, length, fp) != length)
560 throw stream_error("get_translate_dictionary fail: invalid format");
561
562 str_buf[length] = u8'\0';
563 oris.emplace_back(str_buf.data());
564
565 if (std::fseek(fp, cur_pos, SEEK_SET) != 0)
566 throw stream_error("get_translate_dictionary fail: invalid format");
567 }
568
569 if (std::fseek(fp, aim_offset, SEEK_SET) != 0)
570 throw stream_error("get_translate_dictionary fail: invalid format");
571 std::unordered_map<std::u8string, std::u8string> res;
572
573 for (std::size_t i = 0; i < str_num; ++i)
574 {
575 auto length = read_num(buff.data(), need_swap);
576 auto offset = read_num(buff.data(), need_swap);
577 auto cur_pos = std::ftell(fp);
578 if (cur_pos == -1L) throw stream_error("get_translate_dictionary fail: file inconsistent");
579
580 if (length > k_max_str_len)
581 throw stream_error("get_translate_dictionary fail: string too long");
582 total_str_bytes += length;
583 if (total_str_bytes > static_cast<std::uint64_t>(file_size))
584 throw stream_error("get_translate_dictionary fail: total string bytes exceed file size");
585 if (str_buf.size() < static_cast<std::size_t>(length) + 1)
586 str_buf.resize(static_cast<std::size_t>(length) + 1);
587
588 if (std::fseek(fp, offset, SEEK_SET) != 0)
589 throw stream_error("get_translate_dictionary fail: invalid format");
590
591 if (std::fread(str_buf.data(), 1, length, fp) != length)
592 throw stream_error("get_translate_dictionary fail: invalid format");
593
594 str_buf[length] = u8'\0';
595 res.insert({oris[i], str_buf.data()});
596
597 if (std::fseek(fp, cur_pos, SEEK_SET) != 0)
598 throw stream_error("get_translate_dictionary fail: invalid format");
599 }
600
601 return res;
602 }

686 static std::string filter_lang(const text_domain& td, const std::string& p_lang)
687 {
688 auto match_lang = [&td](const std::string& str_lang)
689 {
690 std::size_t start = 0;
691 std::size_t end = 0;
692 while ((end = str_lang.find(':', start)) != std::string::npos)
693 {
694 auto cur_lang = str_lang.substr(start, end - start);
695 if ((!cur_lang.empty()) && (base_ft<messages>::available(td, cur_lang)))
696 return cur_lang;
697 start = end + 1;
698 }
699
700 auto last_str = str_lang.substr(start);
701 if ((!last_str.empty()) && (base_ft<messages>::available(td, last_str)))
702 return last_str;
703
704 return std::string{};
705 };
706
707 if (p_lang.empty())
708 {
709 // Environment fallback (mirrors gettext): LANGUAGE, then
710 // LC_ALL / LC_MESSAGES / LANG. std::getenv is NOT synchronised
711 // against a concurrent setenv()/putenv() on another thread — the
712 // returned pointer and the bytes it addresses may be invalidated
713 // mid-read — and neither C++ nor POSIX offers a portable
714 // thread-safe environment read, so a lock here could not close the
715 // race against mutators elsewhere. This code therefore assumes the
716 // process environment is not modified concurrently with messages
717 // facet construction (true for the usual "configure locale once at
718 // startup" usage). Each value is copied into a std::string at once
719 // to keep the read window minimal.
720 std::string res;
721 if (const char* p = std::getenv("LANGUAGE"))
722 {
723 res = match_lang(p);
724 if (!res.empty()) return res;
725 }
726
727 for (const char* var : {"LC_ALL", "LC_MESSAGES", "LANG"})
728 {
729 if (const char* p = std::getenv(var); p && p[0] != '\0')
730 {
731 res = p;
732 // Unlike LANGUAGE, these are a single locale name, not a
733 // ':'-separated list (gettext never splits them). A value
734 // containing ':' is therefore not a valid locale name:
735 // skip it and try the next variable.
736 if (res.find(':') == std::string::npos && base_ft<messages>::available(td, res))
737 return res;
738 }
739 }
740
741 return "";
742 }
743 else
744 return match_lang(p_lang);
745 }
