95inline std::vector<std::string> messages_language_candidates(const std::vector<std::string>& names)
96{
97 const auto is_alpha = [](char c) { return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'); };
98 const auto is_digit = [](char c) { return c >= '0' && c <= '9'; };
99 const auto all_of = [](const std::string& s, auto pred) { return std::all_of(s.begin(), s.end(), pred); };
100
101 std::vector<std::string> res;
102 const auto add = [&res](std::string c)
103 {
104 if (std::ranges::find(res, c) == res.end())
105 res.push_back(std::move(c));
106 };
107
108 for (const std::string& name : names)
109 {
110 std::vector<std::string> subtags(1);
111 for (const char c : name)
112 {
113 if (c == '-' || c == '_') subtags.emplace_back();
114 else subtags.back().push_back(c);
115 }
116
117 std::string lang = subtags[0];
118 if (lang.size() < 2 || lang.size() > 8 || !all_of(lang, is_alpha))
119 continue;
120 for (char& c : lang)
121 if (c >= 'A' && c <= 'Z') c = static_cast<char>(c - 'A' + 'a');
122
123 // The region comes before any extension, which a single-character subtag starts.
124 std::string region;
125 for (std::size_t i = 1; i < subtags.size() && subtags[i].size() > 1; ++i)
126 {
127 const std::string& t = subtags[i];
128 if ((t.size() == 2 && all_of(t, is_alpha)) || (t.size() == 3 && all_of(t, is_digit)))
129 {
130 region = t;
131 for (char& c : region)
132 if (c >= 'a' && c <= 'z') c = static_cast<char>(c - 'a' + 'A');
133 break;
134 }
135 }
136
137 if (!region.empty())
138 add((lang + '_').append(region));
139 add(lang);
140 }
141 return res;
142}

560 static std::unordered_map<std::u8string, std::u8string> get_translate_dictionary(const std::filesystem::path& filename)
561 {
562 std::FILE* fp = detail::file_open(filename, "rb"); // NOLINT(cppcoreguidelines-owning-memory)
563 if (!fp)
564 throw io_error("get_translate_dictionary fail: cannot open file " + detail::path_for_message(filename));
565
566 file_closer guard(fp);
567
568 if (detail::file_seek(fp, 0, SEEK_END) != 0)
569 throw io_error("get_translate_dictionary fail: invalid format");
570 const std::int64_t file_size = detail::file_tell(fp);
571 if (file_size < 0)
572 throw io_error("get_translate_dictionary fail: file inconsistent");
573 if (detail::file_seek(fp, 0, SEEK_SET) != 0)
574 throw io_error("get_translate_dictionary fail: invalid format");
575
576 auto read_num = [fp](unsigned char* buf, bool need_swap)
577 {
578 if (std::fread(buf, 1, 4, fp) != 4)
579 throw io_error("get_translate_dictionary fail: invalid format");
580
581 std::uint32_t res = 0;
582 std::memcpy(&res, buf, 4);
583 if (need_swap)
584 res = std::byteswap(res);
585 return res;
586 };
587
588 std::array<unsigned char, 8> buff{};
589 if (std::fread(buff.data(), 1, 8, fp) != 8)
590 throw io_error("get_translate_dictionary fail: invalid format");
591
592 bool need_swap = false;
593 if ((buff[0] == 0x95) && (buff[1] == 0x04) && (buff[2] == 0x12) && (buff[3] == 0xde))
594 need_swap = (std::endian::native == std::endian::little);
595 else if ((buff[0] == 0xde) && (buff[1] == 0x12) && (buff[2] == 0x04) && (buff[3] == 0x95))
596 need_swap = (std::endian::native == std::endian::big);
597 else throw io_error("get_translate_dictionary fail: invalid format");
598
599 // Accept any .mo whose major revision (the high 16 bits) is 0, regardless of
600 // minor revision. Only minor 0 (basic) and minor 1 (system-dependent strings)
601 // are defined today, but minor revisions are backward-compatible additions:
602 // they only add separate tables (sysdep strings, hash table) and never change
603 // the regular orig/trans tables this parser reads. Sysdep strings live in their
604 // own orig/trans_sysdep tables we never touch, so a sysdep string is simply
605 // absent (lookup misses and the caller keeps the original) rather than returned
606 // with its segment markers unexpanded. A non-zero major revision uses a layout
607 // this basic parser does not commit to, so it is rejected.
608 std::uint32_t revision = 0;
609 std::memcpy(&revision, buff.data() + 4, 4);
610 if (need_swap)
611 revision = std::byteswap(revision);
612 if ((revision >> 16) != 0)
613 throw io_error("get_translate_dictionary fail: unsupported .mo major revision");
614
615 auto str_num = read_num(buff.data(), need_swap);
616 auto ori_offset = read_num(buff.data(), need_swap);
617 auto aim_offset = read_num(buff.data(), need_swap);
618
619 // Each entry needs an 8-byte (length, offset) descriptor in both the
620 // original and translation tables, so a valid str_num cannot exceed
621 // file_size / 16. Rejecting larger counts caps the O(str_num) growth of
622 // oris and the result map, which the per-string length cap does not.
623 if (str_num > static_cast<std::uint64_t>(file_size) / 16u)
624 throw io_error("get_translate_dictionary fail: implausible string count");
625
626 // The .mo offsets are unsigned 32-bit. file_seek takes a 64-bit offset, so
627 // every one is exact on every platform; std::fseek's long is 32-bit on Windows.
628 if (detail::file_seek(fp, ori_offset, SEEK_SET) != 0)
629 throw io_error("get_translate_dictionary fail: invalid format");
630
631 // The .mo format stores each string length as an unbounded 32-bit
632 // integer, so the buffer is grown to fit each entry rather than
633 // assuming a fixed maximum. A sanity cap guards against a corrupt or
634 // malicious file requesting a huge allocation.
635 constexpr std::uint32_t k_max_str_len = 64u * 1024u * 1024u; // 64 MiB
636 std::uint64_t total_str_bytes = 0;
637 std::vector<char8_t> str_buf;
638 std::vector<std::u8string> oris;
639 for (std::size_t i = 0; i < str_num; ++i)
640 {
641 auto length = read_num(buff.data(), need_swap);
642 auto offset = read_num(buff.data(), need_swap);
643 const std::int64_t cur_pos = detail::file_tell(fp);
644 if (cur_pos < 0) throw io_error("get_translate_dictionary fail: file inconsistent");
645
646 if (length > k_max_str_len)
647 throw io_error("get_translate_dictionary fail: string too long");
648 total_str_bytes += length;
649 if (std::cmp_greater(total_str_bytes, file_size))
650 throw io_error("get_translate_dictionary fail: total string bytes exceed file size");
651 if (str_buf.size() < static_cast<std::size_t>(length) + 1)
652 str_buf.resize(static_cast<std::size_t>(length) + 1);
653
654 if (detail::file_seek(fp, offset, SEEK_SET) != 0)
655 throw io_error("get_translate_dictionary fail: invalid format");
656
657 if (std::fread(str_buf.data(), 1, length, fp) != length)
658 throw io_error("get_translate_dictionary fail: invalid format");
659
660 str_buf[length] = u8'\0';
661 oris.emplace_back(str_buf.data());
662
663 if (detail::file_seek(fp, cur_pos, SEEK_SET) != 0)
664 throw io_error("get_translate_dictionary fail: invalid format");
665 }
666
667 if (detail::file_seek(fp, aim_offset, SEEK_SET) != 0)
668 throw io_error("get_translate_dictionary fail: invalid format");
669 std::unordered_map<std::u8string, std::u8string> res;
670
671 for (std::size_t i = 0; i < str_num; ++i)
672 {
673 auto length = read_num(buff.data(), need_swap);
674 auto offset = read_num(buff.data(), need_swap);
675 const std::int64_t cur_pos = detail::file_tell(fp);
676 if (cur_pos < 0) throw io_error("get_translate_dictionary fail: file inconsistent");
677
678 if (length > k_max_str_len)
679 throw io_error("get_translate_dictionary fail: string too long");
680 total_str_bytes += length;
681 if (std::cmp_greater(total_str_bytes, file_size))
682 throw io_error("get_translate_dictionary fail: total string bytes exceed file size");
683 if (str_buf.size() < static_cast<std::size_t>(length) + 1)
684 str_buf.resize(static_cast<std::size_t>(length) + 1);
685
686 if (detail::file_seek(fp, offset, SEEK_SET) != 0)
687 throw io_error("get_translate_dictionary fail: invalid format");
688
689 if (std::fread(str_buf.data(), 1, length, fp) != length)
690 throw io_error("get_translate_dictionary fail: invalid format");
691
692 str_buf[length] = u8'\0';
693 res.insert({oris[i], str_buf.data()});
694
695 if (detail::file_seek(fp, cur_pos, SEEK_SET) != 0)
696 throw io_error("get_translate_dictionary fail: invalid format");
697 }
698
699 return res;
700 }

807 static std::string filter_lang(const text_domain& td, const std::string& p_lang)
808 {
809 auto match_lang = [&td](const std::string& str_lang)
810 {
811 std::size_t start = 0;
812 std::size_t end = 0;
813 while ((end = str_lang.find(':', start)) != std::string::npos)
814 {
815 auto cur_lang = str_lang.substr(start, end - start);
816 if ((!cur_lang.empty()) && (base_ft<messages>::available(td, cur_lang)))
817 return cur_lang;
818 start = end + 1;
819 }
820
821 auto last_str = str_lang.substr(start);
822 if ((!last_str.empty()) && (base_ft<messages>::available(td, last_str)))
823 return last_str;
824
825 return std::string{};
826 };
827
828 if (p_lang.empty())
829 {
830 // Environment fallback in gettext's order, names used verbatim; see the doc
831 // comment above for how this differs from gettext and for concurrent setenv.
832 std::string res = match_lang(detail::get_env("LANGUAGE"));
833 if (!res.empty()) return res;
834
835 bool locale_set = false;
836 for (const char* var : {"LC_ALL", "LC_MESSAGES", "LANG"})
837 {
838 res = detail::get_env(var);
839 if (res.empty())
840 continue;
841 locale_set = true;
842 // Unlike LANGUAGE, these are a single locale name, not a ':'-separated
843 // list (gettext never splits them), so a value with ':' is skipped.
844 if (res.find(':') == std::string::npos && base_ft<messages>::available(td, res))
845 return res;
846 }
847
848 // gettext's rule: only a locale nobody set falls back on the system's list. One
849 // set explicitly asked for its language, catalogue or not.
850 if (!locale_set)
851 {
852 for (const std::string& c : detail::messages_language_candidates(detail::messages_system_languages()))
853 if (base_ft<messages>::available(td, c))
854 return c;
855 }
856
857 return "";
858 }
859 else
860 return match_lang(p_lang);
861 }

863 // Global mapping from text domain name to its directory path.
864 // Protected by s_domain_mutex; read/write only while holding the lock.
865 // A reference to a map that is never deleted: the standard stream objects are
866 // not destroyed at exit, so the data their messages facet reaches must not be
867 // either (see common/sing_temp.h). The mutex is trivially destructible. Only
868 // bad_alloc can escape the new, at startup, where nothing could handle it anyway.
869 // NOLINTNEXTLINE(bugprone-throwing-static-initialization)
870 inline static std::unordered_map<std::string, std::filesystem::path>& s_domain_dirs =
871 *new std::unordered_map<std::string, std::filesystem::path>;
872 inline static copyable_mutex<std::mutex> s_domain_mutex;
873
874private:
875 // Declared first: initialised before m_filtered_lang and m_domain_info (both
876 // of which depend on this snapshot) because members are initialised in
877 // declaration order. See the constructor for the full rationale.
878 const std::filesystem::path m_dirname;
879 const std::string m_filtered_lang;
880 const std::string m_domain_info;

1379 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
1380 static std::unordered_map<TString, TString> init(const std::filesystem::path& dirname, const std::string& domain, const std::string& lang, const std::string& cvt_ft, bool throw_if_fail)
1381 {
1382 try
1383 {
1384 if (lang.empty())
1385 throw io_error("messages_conf init fail: no language available");
1386
1387 const bool utf8 = clocale_wrapper(cvt_ft).is_utf8();
1388 if constexpr (!std::is_same_v<CharT, char>)
1389 {
1390 if (!utf8)
1391 throw io_error("messages_conf<wchar_t> init fail: " + cvt_ft +
1392 " is not a UTF-8 locale, and wchar_t is not Unicode on this platform");
1393 }
1394
1395 const auto dict = base_ft<messages>::get_translate_dictionary(
1396 base_ft<messages>::get_domain_file({.dirname = dirname, .domain = domain}, lang));
1397 if constexpr (std::is_same_v<CharT, char>)
1398 {
1399 if (!utf8)
1400 return s_through_wide(dict, cvt_ft);
1401
1402 std::unordered_map<TString, TString> res;
1403 for (const auto& [k, v] : dict)
1404 res.insert({TString(k.begin(), k.end()), TString(v.begin(), v.end())});
1405 return res;
1406 }
1407 else
1408 {
1409 // macOS's wchar_t: the locale's C library decodes, through one converter
1410 // that attach() resets for each string.
1411 auto cvt = IOv2::code_cvt_creator<char, CharT>(cvt_ft).create(rb_root_cvt{mem_device{""}});
1412 const auto decode = [&cvt](const std::u8string& s)
1413 {
1414 cvt.attach(mem_device{std::string(s.begin(), s.end())});
1415 cvt.bos(); cvt.main_cont_beg();
1416 TString res;
1417 std::array<CharT, 8> buf{};
1418 while (true)
1419 {
1420 const auto c = cvt.get(buf.data(), buf.size());
1421 if (c == 0) break;
1422 res.append(buf.data(), c);
1423 }
1424 return res;
1425 };
1426
1427 std::unordered_map<TString, TString> res;
1428 for (const auto& [k, v] : dict)
1429 res.insert({decode(k), decode(v)});
1430 return res;
1431 }
1432 }
1433 catch(...)
1434 {
1435 if (throw_if_fail)
1436 throw;
1437 }
1438 return std::unordered_map<TString, TString>{};
1439 }
1440
1441 // char in a locale that is not UTF-8: UTF-8 to a Unicode wchar_t, then to cvt_ft's
1442 // encoding. W only makes the body dependent, so where wchar_t is not Unicode (macOS)
1443 // the discarded branch, and the to_wstring it names, is never instantiated.
1444 template <typename W = wchar_t>
1445 static std::unordered_map<TString, TString> s_through_wide(const std::unordered_map<std::u8string, std::u8string>& dict,
1446 const std::string& cvt_ft)
1447 {
1448 if constexpr (wchar_t_is_utf32 || wchar_t_is_utf16)
1449 {
1450 auto cvt = IOv2::code_cvt_creator<char, W>(cvt_ft).create(rb_root_cvt{mem_device{""}});
1451 const auto encode = [&cvt](const std::u8string& s)
1452 {
1453 const std::basic_string<W> wide = detail::to_wstring<W>(s);
1454 cvt.bos(); cvt.main_cont_beg();
1455 cvt.put(wide.data(), wide.size());
1456 auto [dev, err] = cvt.detach();
1457 cvt.attach(mem_device{""});
1458 if (err) std::rethrow_exception(err);
1459 return dev.str();
1460 };
1461
1462 std::unordered_map<TString, TString> res;
1463 for (const auto& [k, v] : dict)
1464 res.insert({encode(k), encode(v)});
1465 return res;
1466 }
1467 else
1468 throw io_error("messages_conf<char> init fail: " + cvt_ft +
1469 " is not a UTF-8 locale, and wchar_t is not Unicode on this platform");
1470 }

1472private:
1473 const std::unordered_map<TString, TString> m_dict; // NOLINT(cppcoreguidelines-avoid-const-or-ref-data-members)
1474};
