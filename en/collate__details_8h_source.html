296 n);
297 else
298 static_assert(dependent_false_v<CharT>, "collate_conf::transform is not implemented.");
299
300 if (res == xfrm_failed)
301 throw io_error("collate_conf::transform: strxfrm/wcsxfrm failed");

303 if (res >= n)
