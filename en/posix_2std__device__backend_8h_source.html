36// Whether a read() would return at once: data is ready, or the peer hung
37// up (read() then returns 0 without waiting). On a poll error the answer
38// is "no"; the next blocking read() reports it.
39inline bool std_fd_readable_now(int fd) noexcept
40{
41 struct pollfd pfd{ .fd = fd, .events = POLLIN, .revents = 0 };
42 int r = poll(&pfd, 1, 0);
43 while (r == -1 && errno == EINTR)
44 r = poll(&pfd, 1, 0);
45 return r > 0 && (pfd.revents & (POLLIN | POLLHUP)) != 0;
46}

48// After a read on fd failed with EAGAIN: waits in poll() until reading is worth
49// another try. Returns false on a bare hang-up, where read() returns 0.
50inline bool std_fd_wait_input(int fd)
51{
52 struct pollfd pfd{ .fd = fd, .events = POLLIN, .revents = 0 };
53 if (poll(&pfd, 1, -1) == -1)
54 {
55 if (errno == EINTR) return true;
56 throw device_error("std_device stdin read fail: poll error");
57 }
58 if (pfd.revents & (POLLERR | POLLNVAL))
59 throw device_error("std_device stdin read fail: poll revents error");
60 if (pfd.revents & POLLIN) return true;
61 if (pfd.revents & POLLHUP) return false;
62 throw device_error("std_device stdin read fail: unexpected poll revents");
63}

65inline std::size_t std_fd_read(int fd, char* s, std::size_t n)
66{
67 assert((s != nullptr) && (n != 0));
68
69 constexpr auto max_read = static_cast<std::size_t>(std::numeric_limits<ssize_t>::max());
70 ssize_t ret = 0;
71 while (true)
72 {
73 ret = read(fd, s, std::min(n, max_read));
74 if (ret != -1) break;
75
76 if (errno == EINTR) continue;
77 if (errno == EAGAIN || errno == EWOULDBLOCK)
78 {
79 if (std_fd_wait_input(fd)) continue;
80 ret = 0;
81 break;
82 }
83 throw device_error("std_device::fd_read fail: read error");
84 }
85 return static_cast<std::size_t>(ret);
86}

101// Reads n bytes through stdin's FILE and, like fread(), waits until all have
102// come or the stream ends. 0 means end of file, FILE's sticky flag: until a
103// clearerr(stdin) every call returns 0 at once. EINTR and EAGAIN are retried;
104// on any other error the bytes already read are returned first, and the next
105// call meets the error again and throws.
106inline std::size_t std_stdin_file_read(char* s, std::size_t n)
107{
108 assert((s != nullptr) && (n != 0));
109
110 stdin_file_lock guard;
111 if (std::feof(stdin)) return 0;
112 // A stale error flag must not pass for one raised by this read.
113 std::clearerr(stdin);
114
115 std::size_t got = 0;
116 while (true)
117 {
118 got += std::fread(s + got, 1, n - got, stdin);
119 if (got == n || !std::ferror(stdin)) return got;
120
121 const int err = errno;
122 if (err == EINTR || err == EAGAIN || err == EWOULDBLOCK)
123 {
124 std::clearerr(stdin);
125 // glibc's getc and fread also hold the lock across a blocking read().
126 if (err != EINTR) (void)std_fd_wait_input(std_in_fd);
127 continue;
128 }
129 if (got != 0) return got;
130 throw device_error("std_device stdin FILE read fail: read error");
131 }
132}

134// Whether stdin is at end of file, waiting for a byte if need be. A byte read
135// to find out goes back with ungetc(), which guarantees one byte of pushback;
136// the lock keeps another thread from reading in between.
137inline bool std_stdin_file_eof()
138{
139 stdin_file_lock guard;
140 char c = 0;
141 if (std_stdin_file_read(&c, 1) == 0) return true;
142 if (std::ungetc(static_cast<unsigned char>(c), stdin) == EOF)
143 throw device_error("std_device::deof fail: ungetc error");
144 return false;
145}
146
147// Pushes [first, last) back into stdin's FILE, last byte first, so the next read
148// gets them in order. Beyond one byte ungetc() may fail (glibc and BSD only when
149// out of memory, musl past a small reserve); then the bytes already pushed are
150// read off again and false comes back. ungetc() clears the EOF flag and there is
151// no call to set it again, so a failed attempt may leave that flag cleared.
152template <std::bidirectional_iterator It>
153bool std_stdin_file_unget(It first, It last)
154{
155 assert(first != last);
156
157 stdin_file_lock guard;
158 std::size_t pushed = 0;
159 while (last != first)
160 {
161 --last;
162 if (std::ungetc(static_cast<unsigned char>(*last), stdin) == EOF)
163 {
164 while (pushed-- != 0)
165 (void)getc_unlocked(stdin);
166 return false;
167 }
168 ++pushed;
169 }
170 return true;
171}
