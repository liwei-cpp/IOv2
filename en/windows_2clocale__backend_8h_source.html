76// Puts this thread in per-thread locale mode, where setlocale changes this thread alone,
77// and returns what clocale_restore needs to undo it.
78inline clocale_saved clocale_enter_thread_mode()
79{
80 clocale_saved res;
81 res.mode = _configthreadlocale(_ENABLE_PER_THREAD_LOCALE);
82 try
83 {
84 res.name = std::setlocale(LC_ALL, nullptr);
85 }
86 catch (...)
87 {
88 _configthreadlocale(res.mode);
89 throw;
90 }
91 return res;
92}
93
94// Puts back the thread locale and mode clocale_use (or clocale_enter_thread_mode) replaced.
95inline void clocale_restore(const clocale_saved& old) noexcept
96{
97 std::setlocale(LC_ALL, old.name.c_str());
98 _configthreadlocale(old.mode);
99}

101// A locale with all categories from name; null if the CRT rejects it. Throws only
102// std::bad_alloc.
103inline clocale_handle clocale_create(const char* name)
104{
105 auto res = std::make_unique<windows_clocale>();
106 const clocale_saved old = clocale_enter_thread_mode();
107 const char* all = nullptr;
108 try
109 {
110 all = std::setlocale(LC_ALL, name);
111 if (all)
112 {
113 res->all_name = all; // copied before the next setlocale reuses the buffer
114 res->ctype_name = std::setlocale(LC_CTYPE, nullptr);
115 if (const wchar_t* bcp47 = ___lc_locale_name_func()[LC_CTYPE])
116 res->ctype_bcp47 = bcp47;
117 if (const wchar_t* bcp47 = ___lc_locale_name_func()[LC_COLLATE])
118 res->collate_bcp47 = bcp47;
119 res->collate_cp = ___lc_collate_cp_func();
120 }
121 }
122 catch (...)
123 {
124 clocale_restore(old);
125 throw;
126 }
127 clocale_restore(old);
128 if (!all)
129 return nullptr;
130
131 res->loc = _create_locale(LC_ALL, res->all_name.c_str());
132 return res->loc ? res.release() : nullptr;
133}

135// An independent copy of loc; null on failure. Throws only std::bad_alloc.
136inline clocale_handle clocale_duplicate(clocale_handle loc)
137{
138 auto res = std::make_unique<windows_clocale>(
139 windows_clocale{.all_name = loc->all_name, .ctype_name = loc->ctype_name,
140 .ctype_bcp47 = loc->ctype_bcp47, .collate_bcp47 = loc->collate_bcp47,
141 .collate_cp = loc->collate_cp});
142 res->loc = _create_locale(LC_ALL, res->all_name.c_str());
143 return res->loc ? res.release() : nullptr;
144}

146inline void clocale_release(clocale_handle loc) noexcept
147{
148 // Owned since clocale_create / clocale_duplicate released it from a unique_ptr.
149 const std::unique_ptr<windows_clocale> owned(loc);
150 _free_locale(owned->loc);
151}

153// The name of loc's LC_CTYPE category, as setlocale reports it. "POSIX" is not a name the
154// CRT knows, so no locale has it.
155inline const char* clocale_ctype_name(clocale_handle loc) noexcept
156{
157 return loc->ctype_name.c_str();
158}

160// Makes loc the calling thread's locale. Throws io_error, with nothing changed, if
161// setlocale refuses the switch.
162inline clocale_saved clocale_use(clocale_handle loc)
163{
164 clocale_saved old = clocale_enter_thread_mode();
165 if (!std::setlocale(LC_ALL, loc->all_name.c_str()))
166 {
167 clocale_restore(old);
168 throw io_error("clocale_user: setlocale rejected " + loc->all_name);
169 }
170 return old;
171}
