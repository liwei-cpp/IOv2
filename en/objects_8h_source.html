218 // Each reason is recovered by rethrowing the pointer and catching it -- which leaves the
219 // pointer intact, so `failed()` still hands out every exception afterwards.
220 static std::string describe(const failures& f)
221 {
222 std::string msg = "IOv2::sync_with_stdio failed for";
223 bool first = true;
224
225 const auto add = [&](const char* name, const std::exception_ptr& err)
226 {
227 if (!err) return;
228 msg += first ? " " : "; ";
229 first = false;
230 msg += name;
231 msg += ": ";
232 try { std::rethrow_exception(err); }
233 catch (const std::exception& e) { msg += e.what(); }
234 catch (...) { msg += "unknown exception"; }
235 };
236
237 add("cout", f.cout_err); add("cerr", f.cerr_err); add("clog", f.clog_err);
238 add("wcout", f.wcout_err); add("wcerr", f.wcerr_err); add("wclog", f.wclog_err);
239 add("cin", f.cin_err); add("wcin", f.wcin_err);
240 return msg;
241 }

243 failures m_failed;

301inline void sync_with_stdio(bool sync = true)
302{
303 sync_error::failures failed;
304
305 // Every stream is attempted, so one that throws cannot keep the others from
306 // switching. Collecting costs nothing that can fail: assigning an exception_ptr
307 // is noexcept, and the table is a local.
308 const auto one = [sync](auto& stream, std::exception_ptr& slot) noexcept
309 {
310 try { stream.sync_with_stdio(sync); }
311 catch (...) { slot = std::current_exception(); }
312 };
313
314 one(cout, failed.cout_err);
315 one(cerr, failed.cerr_err);
316 one(clog, failed.clog_err);
317
318 one(wcout, failed.wcout_err);
319 one(wcerr, failed.wcerr_err);
320 one(wclog, failed.wclog_err);
321
322 one(cin, failed.cin_err);
323 one(wcin, failed.wcin_err);
324
325 if (failed.any()) throw sync_error(std::move(failed));
326}
327}
