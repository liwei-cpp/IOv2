153 // Each reason is recovered by rethrowing the pointer and catching it -- which leaves the
154 // pointer intact, so `failed()` still hands out every exception afterwards.
155 static std::string describe(const failures& f)
156 {
157 std::string msg = "IOv2::sync_with_stdio failed for";
158 bool first = true;
159
160 const auto add = [&](const char* name, const std::exception_ptr& err)
161 {
162 if (!err) return;
163 msg += first ? " " : "; ";
164 first = false;
165 msg += name;
166 msg += ": ";
167 try { std::rethrow_exception(err); }
168 catch (const std::exception& e) { msg += e.what(); }
169 catch (...) { msg += "unknown exception"; }
170 };
171
172 add("cout", f.cout_err); add("cerr", f.cerr_err); add("clog", f.clog_err);
173 add("wcout", f.wcout_err); add("wcerr", f.wcerr_err); add("wclog", f.wclog_err);
174 add("cin", f.cin_err); add("wcin", f.wcin_err);
175 return msg;
176 }
177
178 failures m_failed;

226inline void sync_with_stdio(bool sync = true)
227{
228 sync_error::failures failed;
229
230 // Every stream is attempted, so one that throws cannot keep the others from
231 // switching. Collecting costs nothing that can fail: assigning an exception_ptr
232 // is noexcept, and the table is a local.
233 const auto one = [sync](auto& stream, std::exception_ptr& slot) noexcept
234 {
235 try { stream.sync_with_stdio(sync); }
236 catch (...) { slot = std::current_exception(); }
237 };
238
239 one(cout, failed.cout_err);
240 one(cerr, failed.cerr_err);
241 one(clog, failed.clog_err);
242
243 one(wcout, failed.wcout_err);
244 one(wcerr, failed.wcerr_err);
245 one(wclog, failed.wclog_err);
246
247 one(cin, failed.cin_err);
248 one(wcin, failed.wcin_err);
249
250 if (failed.any()) throw sync_error(std::move(failed));
251}
252}
