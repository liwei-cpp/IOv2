397 template <typename TIter, std::integral TVal>
398 requires (!std::same_as<TVal, bool>)
399 TIter put(TIter s, bool intl, ios_base<char_type>& io, TVal v) const
400 {
401 constexpr std::size_t buf_size = std::numeric_limits<TVal>::digits10 + 3;
402 std::array<char_type, buf_size> vec;
403
404 char_type* p = vec.data() + buf_size;
405 *--p = '\0';
406
407 using TU = std::make_unsigned_t<TVal>;
408 bool negative = false;
409 TU uv;
410
411 if constexpr(std::is_unsigned_v<TVal>)
412 uv = v;
413 else
414 {
415 negative = (v < 0);
416 uv = negative ? static_cast<TU>(-(v + 1)) + 1
417 : static_cast<TU>(v);
418 }
419
420 do { // NOLINT(cppcoreguidelines-avoid-do-while)
421 *--p = static_cast<char_type>('0' + (uv % 10));
422 uv /= 10;
423 } while (uv != 0);
424
425 if (negative)
426 *--p = '-';
427
428 return intl ? insert<true>(s, io, p)
429 : insert<false>(s, io, p);
430 }

565 template <typename TIter, std::sentinel_for<TIter> TSent>
566 TIter get(TIter beg, TSent end, bool intl, ios_base<char_type>& io, std::basic_string<char_type>& digits) const
567 {
568 bool succ = true;
569
570 std::string str;
571 std::tie(succ, beg) = intl ? extract<true>(beg, end, io, str)
572 : extract<false>(beg, end, io, str);
573 const auto len = str.size();
574 std::basic_string<char_type> tmp;
575 if (len)
576 {
577 tmp.reserve(len);
578 for (auto ch : str)
579 {
580 if (ch == '-')
581 tmp.push_back(s_atoms[0]);
582 else if ((ch >= '0') && (ch <= '9'))
583 tmp.push_back(s_atoms[ch - '0' + 1]);
584 }
585 }
586 if (!succ)
587 throw stream_error("monetary parse fail");
588 if (len)
589 digits.swap(tmp);
590 return beg;
591 }

686 [[nodiscard]] bool fill_alters_reading(char_type fill, const split_info& info,
687 bool leads_digits, bool trails_value,
688 bool negative) const
689 {
690 const char_type* const digits = s_atoms.data() + s_zero;
691 if (std::find(digits, digits + 10, fill) != digits + 10)
692 return !(fill == digits[0] && leads_digits);
693
694 // Past the amount nothing but a digit can still be read into it.
695 if (trails_value)
696 return false;
697
698 if (fill == m_decimal_point)
699 return true;
700 if (!info.m_negative_sign.empty() && fill == info.m_negative_sign[0])
701 return !negative;
702 if (!info.m_positive_sign.empty() && fill == info.m_positive_sign[0])
703 return negative;
704 return false;
705 }

761 template <bool isIntl, typename TIter>
762 TIter insert(TIter s, ios_base<char_type>& io, const std::basic_string<char_type>& digits) const
763 {
764 const split_info& info = isIntl ? m_int : m_nat;
765 using part = base_ft<monetary>::part;
766
767 // Capture and consume the field width up front. width() is one-shot, so
768 // it must be cleared exactly once per put; resetting it here — before any
769 // allocation or copy that could throw — guarantees no leftover width
770 // leaks into the next output operation on whichever path we leave by,
771 // including an exception thrown while formatting. The captured value is
772 // used for padding below.
773 const std::size_t width = io.width();
774 io.width(0);
775
776 // A leading minus is how the caller says the amount is negative. It selects
777 // this locale's negative pattern and negative sign string and is then dropped:
778 // the sign the field ends up carrying comes from that string, which may spell
779 // it quite differently (or not at all).
780 const char_type* first = digits.data();
781 const char_type* const stop = first + digits.size();
782
783 const bool negative = (*first == s_atoms[s_minus]);
784 // The pattern is four bytes and is read once per slot while the strings below
785 // are being built; taken by value it stays in a register, where a reference
786 // into the facet would have to be reloaded around every write.
787 const base_ft<monetary>::pattern order = negative ? info.m_neg_format : info.m_pos_format;
788 const std::basic_string<char_type>& sign = negative ? info.m_negative_sign : info.m_positive_sign;
789 if (negative)
790 ++first;
791
792 // The amount runs up to the first character that is not one of this facet's
793 // digits; whatever the caller put after that is not ours to format.
794 const char_type* const digit_table = s_atoms.data() + s_zero;
795 std::size_t len = 0;
796 for (const char_type* q = first; q != stop; ++q)
797 {
798 if (std::find(digit_table, digit_table + 10, *q) == digit_table + 10)
799 break;
800 ++len;
801 }
802 if (!len)
803 return s;
804
805 // The digits are the smallest units of the currency, so the amount they spell
806 // is read off their right-hand end: the run splits into an integer head and a
807 // fractional tail with the cut m_frac_digits places from that end. A run too
808 // short to reach the cut has a head of nothing, and the fraction picks up the
809 // shortfall as leading zeros — two fractional places turn 7 into .07, not 7.0.
810 // A negative frac_digits asks for no fraction at all and keeps the whole run.
811 const int frac = info.m_frac_digits;
812 // Widened once, and only from a value the `frac < 0` arm has already ruled
813 // out as negative, so every length comparison below stays unsigned-to-unsigned.
814 const std::size_t fracw = (frac > 0) ? static_cast<std::size_t>(frac) : 0;
815 const std::size_t head = (frac < 0) ? len
816 : (len > fracw) ? len - fracw
817 : 0;
818
819 std::basic_string<char_type> value;
820 value.reserve(2 * len);
821
822 if (head)
823 {
824 if (m_grouping.empty())
825 value.assign(first, head);
826 else
827 {
828 // add_grouping writes at most one separator per digit, so twice the
829 // head is always room enough; the slack is trimmed off afterwards.
830 value.assign(2 * head, char_type());
831 const char_type* const wrote =
832 FacetHelper::add_grouping(value.data(), m_thousands_sep, m_grouping, first, first + head);
833 value.erase(static_cast<std::size_t>(wrote - value.data()));
834 }
835 }
836
837 if (frac > 0)
838 {
839 value += m_decimal_point;
840 if (fracw > len)
841 value.append(fracw - len, s_atoms[s_zero]);
842 value.append(first + head, len - head);
843 }
844
845 const ios_defs::fmtflags adjust = io.flags() & ios_defs::adjustfield;
846 const bool show_symbol = (io.flags() & ios_defs::showbase) != 0;
847
848 // What the field is worth before any padding: the amount, the sign string in
849 // full (even the part that trails), and the currency symbol if it is shown.
850 len = value.size() + sign.size() + (show_symbol ? info.m_curr_symbol.size() : 0);
851
852 if (width > len && width - len > ios_defs::max_pad_count)
853 throw stream_error("monetary put fail: fill count exceeds max_pad_count");
854
855 std::basic_string<char_type> res;
856 res.reserve(2 * len);
857
858 // Non-zero only under `internal`, where the shortfall is not tacked onto an
859 // end but poured into whichever pattern slot is spare.
860 const std::size_t spread = (adjust == ios_defs::internal && len < width) ? width - len : 0;
861
862 // Where each run of fill lands is recorded as it is written, so that the
863 // readability check below can be made against the finished field instead of
864 // against the pattern: whether a run abuts the amount depends on which
865 // parts between them turn out to be empty, which the pattern alone does not
866 // say. A pattern writes at most one run per part, plus one final pad.
867 struct fill_run { std::size_t pos; std::size_t len; };
868 std::array<fill_run, 5> runs{};
869 std::size_t run_count = 0;
870 std::size_t value_pos = std::basic_string<char_type>::npos;
871
872 // Lay the parts down in the order this locale asks for.
873 for (const part which : order)
874 {
875 switch (which)
876 {
877 case part::symbol:
878 if (show_symbol)
879 res += info.m_curr_symbol;
880 break;
881 case part::sign:
882 // Only the sign's first character belongs to this slot; a longer sign
883 // string has its remainder appended once the pattern is fully down.
884 if (!sign.empty())
885 res += sign[0];
886 break;
887 case part::value:
888 value_pos = res.size();
889 res += value;
890 break;
891 case part::space:
892 // This slot owes at least one fill character, and takes the whole
893 // internal spread when there is one to take.
894 runs[run_count++] = {res.size(), spread ? spread : 1};
895 res.append(spread ? spread : 1, io.fill());
896 break;
897 case part::none:
898 // A slot that writes nothing of its own is still somewhere the
899 // internal spread can go.
900 if (spread)
901 {
902 runs[run_count++] = {res.size(), spread};
903 res.append(spread, io.fill());
904 }
905 break;
906 }
907 }
908
909 // A sign spelled with more than one character wraps the field: its first
910 // character sat in the sign slot, the rest trails everything.
911 if (sign.size() > 1)
912 res.append(sign, 1);
913
914 // Whatever width is still unaccounted for is fill.
915 len = res.size();
916 if (width > len)
917 {
918 const std::size_t pad = width - len;
919 if (adjust == ios_defs::left)
920 {
921 runs[run_count++] = {len, pad};
922 res.append(pad, io.fill());
923 }
924 else
925 {
926 // Padding in front shifts everything already recorded — the amount
927 // included — right by that much.
928 res.insert(0, pad, io.fill());
929 for (std::size_t r = 0; r < run_count; ++r)
930 runs[r].pos += pad;
931 if (value_pos != std::basic_string<char_type>::npos)
932 value_pos += pad;
933 runs[run_count++] = {0, pad};
934 }
935 len = width;
936 }
937
938 // Fill has now been written, so this is where it gets vetted: reject a fill
939 // character that would change the amount this field reads as. The check sits
940 // here rather than at the top of the function because `fill` is sticky stream
941 // state — a stream carrying setfill('1') must keep working for every output
942 // whose width leaves nothing to pad, and those take no run at all.
943 for (std::size_t r = 0; r < run_count; ++r)
944 {
945 const bool leads_digits = (value_pos != std::basic_string<char_type>::npos)
946 && (runs[r].pos + runs[r].len == value_pos);
947 const bool trails_value = (value_pos != std::basic_string<char_type>::npos)
948 && (runs[r].pos >= value_pos + value.size());
949 if (fill_alters_reading(io.fill(), info, leads_digits, trails_value, negative))
950 throw stream_error("monetary put fail: fill would change the value the field reads as");
951 }
952
953 return std::copy(res.data(), res.data() + len, s);
954 }

1024 template <bool isIntl, typename TIter, std::sentinel_for<TIter> TSent>
1025 std::pair<bool, TIter> extract(TIter beg, TSent end, ios_base<char_type>& io, std::string& units) const
1026 {
1027 const split_info& info = isIntl ? m_int : m_nat;
1028 using part = base_ft<monetary>::part;
1029
1030 // The negative pattern is the one walked, whichever sign the input turns out
1031 // to carry: it is the wider of the two, since a locale may spell the positive
1032 // sign as nothing at all, and the polarity is settled by the sign slot itself.
1033 const base_ft<monetary>::pattern& order = info.m_neg_format;
1034
1035 const char_type* const digit_table = s_atoms.data() + s_zero;
1036 const bool show_symbol = (io.flags() & ios_defs::showbase) != 0;
1037
1038 // A locale that spells both signs cannot have the field leave one out.
1039 const bool sign_required = !info.m_positive_sign.empty() && !info.m_negative_sign.empty();
1040
1041 bool negative = false; // polarity the sign slot deduced
1042 bool valid = true; // still looking at a well-formed field
1043 bool saw_point = false; // the decimal point has gone by
1044 int sign_len = 0; // length of the sign string the input turned out to carry
1045
1046 // Digits seen since the last structural character. Before the decimal point
1047 // that is the current group; when the point arrives the count is handed to
1048 // `int_run` and restarts, so afterwards it counts fractional places.
1049 int run = 0;
1050 int int_run = 0;
1051
1052 // Group widths in the order met, checked against m_grouping once the whole
1053 // amount is in.
1054 std::vector<uint8_t> grouping_tmp;
1055 if (!m_grouping.empty())
1056 grouping_tmp.reserve(32);
1057
1058 // Digits accumulate here as ASCII, and are handed to the caller only if the
1059 // field turns out to be well-formed all the way to its end.
1060 std::string out;
1061 out.reserve(32);
1062
1063 // Which part carries the amount: a run of fill consumed before that index sits
1064 // in front of the amount, one consumed after it sits behind.
1065 int value_idx = 3;
1066 for (int i = 0; i < 4; ++i)
1067 if (order[i] == part::value)
1068 {
1069 value_idx = i;
1070 break;
1071 }
1072
1073 for (int i = 0; i < 4 && valid; ++i)
1074 {
1075 // Set when this part consumes at least one fill character, which is the
1076 // only case in which fill decides how much of the input counts as the
1077 // amount — and so the only case worth vetting.
1078 bool ate_fill = false;
1079 switch (order[i])
1080 {
1081 case part::symbol:
1082 {
1083 // [locale.money.get.virtuals]: showbase makes the currency symbol
1084 // mandatory. Otherwise it is optional, and is read only where letting
1085 // it go would strand a part that still has to be reached — which
1086 // depends on what this slot is followed by, hence the case analysis.
1087 bool required = show_symbol || sign_len > 1;
1088 if (!required)
1089 switch (i)
1090 {
1091 case 0:
1092 // Nothing has been read yet; parsing has to start somewhere.
1093 required = true;
1094 break;
1095 case 1:
1096 required = sign_required || order[0] == part::sign
1097 || order[2] == part::space;
1098 break;
1099 case 2:
1100 required = order[3] == part::value
1101 || (sign_required && order[3] == part::sign);
1102 break;
1103 default:
1104 break;
1105 }
1106 if (!required)
1107 break;
1108
1109 const std::basic_string<char_type>& symbol = info.m_curr_symbol;
1110 std::size_t taken = 0;
1111 while (taken != symbol.size() && beg != end && *beg == symbol[taken])
1112 {
1113 ++beg;
1114 ++taken;
1115 }
1116 // Stopping part-way through the symbol always makes the field
1117 // malformed; stopping before its first character only does so when
1118 // showbase promised the symbol would be there.
1119 if (taken != symbol.size() && (taken != 0 || show_symbol))
1120 valid = false;
1121 break;
1122 }
1123 case part::sign:
1124 {
1125 // Whichever sign string the input opens with is the one it wears.
1126 const std::basic_string<char_type>& pos = info.m_positive_sign;
1127 const std::basic_string<char_type>& neg = info.m_negative_sign;
1128 const std::basic_string<char_type>* worn = nullptr;
1129 if (beg != end)
1130 {
1131 if (!pos.empty() && *beg == pos[0])
1132 worn = &pos;
1133 else if (!neg.empty() && *beg == neg[0])
1134 worn = &neg;
1135 }
1136
1137 if (worn)
1138 {
1139 negative = (worn == &neg);
1140 sign_len = static_cast<int>(worn->size());
1141 ++beg;
1142 }
1143 else if (pos.empty() != neg.empty())
1144 // [locale.money.get.virtuals]: where one of the two sign strings
1145 // is empty, a field with no sign in it takes the polarity of that
1146 // empty string. Only an empty negative string makes that negative.
1147 negative = neg.empty();
1148 else if (sign_required)
1149 valid = false;
1150 break;
1151 }
1152 case part::value:
1153 // Digits go straight into the result; the separators between them are
1154 // not kept, only the widths of the groups they mark off.
1155 for (; beg != end; ++beg)
1156 {
1157 const char_type c = *beg;
1158 const char_type* d = std::find(digit_table, digit_table + 10, c);
1159 if (d != digit_table + 10)
1160 {
1161 out += static_cast<char>('0' + (d - digit_table));
1162 ++run;
1163 continue;
1164 }
1165
1166 // Punctuation structures the integer part only. Past the decimal
1167 // point the fraction is a plain run of digits, so a separator or a
1168 // second point there ends the amount instead of shaping it.
1169 if (saw_point)
1170 break;
1171
1172 if (c == m_decimal_point)
1173 {
1174 // A locale with no fractional places has nothing for a decimal
1175 // point to introduce; the character is not ours to consume.
1176 if (info.m_frac_digits <= 0)
1177 break;
1178
1179 int_run = run;
1180 run = 0;
1181 saw_point = true;
1182 }
1183 else if (!m_grouping.empty() && c == m_thousands_sep)
1184 {
1185 // A separator with no preceding digits, or a group
1186 // longer than the largest representable group size,
1187 // can never satisfy any grouping spec: reject outright
1188 // rather than truncating the count.
1189 if (run == 0 || std::cmp_greater(run, std::numeric_limits<uint8_t>::max()))
1190 {
1191 valid = false;
1192 break;
1193 }
1194 grouping_tmp.push_back(static_cast<uint8_t>(run));
1195 run = 0;
1196 }
1197 else
1198 break;
1199 }
1200 // An amount with no digits at all is not an amount.
1201 if (out.empty())
1202 valid = false;
1203 break;
1204 case part::space:
1205 // This slot owes at least one fill character.
1206 if (beg != end && (*beg == io.fill()))
1207 {
1208 ++beg;
1209 ate_fill = true;
1210 }
1211 else
1212 valid = false;
1213 [[fallthrough]];
1214 case part::none:
1215 // Either slot then soaks up whatever further fill follows — except in
1216 // the last one, where trailing fill belongs to the stream, not to us.
1217 if (i != 3)
1218 for (; beg != end && (*beg == io.fill()); ++beg)
1219 ate_fill = true;
1220 // Reject fill that a reader would have taken for part of the amount:
1221 // `is >> setfill('1') >> get_money(v)` on "112" must fail loudly rather
1222 // than silently hand back 2. Whether the run leads the digits is read
1223 // off the input itself — the character it stopped on — because the
1224 // pattern does not say which parts in between came out empty.
1225 if (ate_fill
1226 && fill_alters_reading(io.fill(), info,
1227 i < value_idx && beg != end
1228 && (*beg == m_decimal_point
1229 || std::find(digit_table, digit_table + 10, *beg)
1230 != digit_table + 10),
1231 i > value_idx, negative))
1232 throw stream_error("monetary get fail: fill would change the value the field reads as");
1233 break;
1234 }

1293 }

1296 if (!valid)

1342

1346 constexpr TVal min_value = std::numeric_limits<TVal>::min();
1347

1349 negative = (s[i] == '-');
1350 ++i;
1351 }
1352
1353 if (i == s.size()) return false;
1354

1357 char c = s[i];
1358 if (c < '0' || c > '9') return false;
1359
1360 int digit = c - '0';
1361

1363 {

1365 return false;
1366 }
1367 else if (value < (min_value + digit) / 10)

1372 }

1388 if (value > (max_value - digit) / 10)
