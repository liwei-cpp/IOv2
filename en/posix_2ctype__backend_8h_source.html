78// Byte case mapping in loc. macOS maps a byte as a code point too and can answer with
79// one above 0xFF (0xFF -> U+0178), which is no byte; c is then kept.
80inline int ctype_toupper(int c, clocale_handle loc) noexcept
81{
82 if (!ctype_byte_is_char(c, loc)) return c;
83 const int res = toupper_l(c, loc);
84 return (res >= 0 && res <= 0xFF) ? res : c;
85}
86
87inline int ctype_tolower(int c, clocale_handle loc) noexcept
88{
89 if (!ctype_byte_is_char(c, loc)) return c;
90 const int res = tolower_l(c, loc);
91 return (res >= 0 && res <= 0xFF) ? res : c;
92}
