667 template <std::same_as<bool> B = bool>
668 locale involve_msg(const std::string& domain, const std::string& lang = "",
669 B throw_if_fail = false) const requires (!detail::messages_locale_encoded<TChar>)
670 {
671 // messages_conf<TChar> exists for char8_t, char32_t and a Unicode wchar_t here
672 // (char, and macOS's wchar_t, take the cvt overload below); for char16_t it stays
673 // incomplete. Reject that with a clear message instead of an opaque "incomplete
674 // type" error from make_shared below.
675 static_assert(requires { sizeof(messages_conf<TChar>); },
676 "involve_msg: messages translation is unsupported for this character "
677 "type -- no messages_conf<TChar> specialization exists (e.g. char16_t).");
678 const std::string filtered_lang = base_ft<messages>::filter_lang(domain, lang);
679 const std::filesystem::path dirname = base_ft<messages>::get_dirname(domain);
680 auto ft = s_ori_facet_buf.try_get_msg<TChar>(domain, filtered_lang, dirname);
681 if (!ft)
682 {
683 // Build strictly so a load failure surfaces as an exception, and intern only
684 // on success: a degraded (load-failed) facet is never cached, so a later
685 // throw_if_fail=true call re-attempts the load and can still throw.
686 try
687 {
688 ft = std::make_shared<messages_conf<TChar>>(domain, filtered_lang, dirname, true);
689 // Caching is separate from the load: a cache-insert failure (e.g. an
690 // allocation failure while interning) must never degrade or drop a
691 // successfully-loaded facet. On such a failure keep it, just uncached.
692 try { ft = s_ori_facet_buf.put_msg<TChar>(ft, domain, filtered_lang, dirname); }
693 catch (...) {} // NOLINT(bugprone-empty-catch)
694 }
695 catch (...)
696 {
697 if (throw_if_fail) throw;
698 ft = std::make_shared<messages_conf<TChar>>(domain, filtered_lang, dirname, false);
699 }
700 }
701
702 // At most one messages facet per locale (keyed by the single messages_conf
703 // type id). If *this already carries one, this replaces it in the returned
704 // copy -- matching involve()'s same-id replace semantics; *this is unchanged.
705 locale res(*this);
706 res.m_facet_confs[ft->id()] = std::move(ft);
707 res.m_facets.clear();
708 return res;
709 }
