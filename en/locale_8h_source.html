611 locale involve_msg(const std::string& domain, const std::string& lang = "",
612 bool throw_if_fail = false) const requires (!std::is_same_v<TChar, char>)
613 {
614 // messages_conf<TChar> is only specialized for char8_t, char32_t and
615 // UTF-32 wchar_t (plus the char overload above); for char16_t or a
616 // non-UTF-32 wchar_t it stays incomplete. Reject those here with a clear
617 // message instead of an opaque "incomplete type" error from make_shared
618 // below. Safe as a completeness check because messages_conf<TChar>'s
619 // completeness is fixed program-wide (a type either has a specialization
620 // everywhere or nowhere).
621 static_assert(requires { sizeof(messages_conf<TChar>); },
622 "involve_msg: messages translation is unsupported for this character "
623 "type -- no messages_conf<TChar> specialization exists (e.g. char16_t, "
624 "or wchar_t on a non-UTF-32 platform).");
625 const std::string filtered_lang = base_ft<messages>::filter_lang(domain, lang);
626 const std::string dirname = base_ft<messages>::get_dirname(domain);
627 auto ft = s_ori_facet_buf.try_get_msg<TChar>(domain, filtered_lang, dirname);
628 if (!ft)
629 {
630 // Build strictly so a load failure surfaces as an exception, and intern only
631 // on success: a degraded (load-failed) facet is never cached, so a later
632 // throw_if_fail=true call re-attempts the load and can still throw.
633 try
634 {
635 ft = std::make_shared<messages_conf<TChar>>(domain, filtered_lang, dirname, true);
636 // Caching is separate from the load: a cache-insert failure (e.g. an
637 // allocation failure while interning) must never degrade or drop a
638 // successfully-loaded facet. On such a failure keep it, just uncached.
639 try { ft = s_ori_facet_buf.put_msg<TChar>(ft, domain, filtered_lang, dirname); }
640 catch (...) {} // NOLINT(bugprone-empty-catch)
641 }
642 catch (...)
643 {
644 if (throw_if_fail) throw;
645 ft = std::make_shared<messages_conf<TChar>>(domain, filtered_lang, dirname, false);
646 }
647 }
648
649 // At most one messages facet per locale (keyed by the single messages_conf
650 // type id). If *this already carries one, this replaces it in the returned
651 // copy -- matching involve()'s same-id replace semantics; *this is unchanged.
652 locale res(*this);
653 res.m_facet_confs[ft->id()] = std::move(ft);
654 res.m_facets.clear();
655 return res;
656 }

980 && !std::derived_from<TF, abs_ft>)
981 std::shared_ptr<TF> get() const
982 {
983 {
984 std::shared_lock g(m_facet_mutex);
985 if (auto it = m_facets.find(type_id_v<TF>()); it != m_facets.end())
986 return std::static_pointer_cast<TF>(it->second);
987 }
988

990 auto res = obj.template get<TF>();
991 if (res)
992 {

994 auto [it, inserted] = m_facets.insert({type_id_v<TF>(), res});
995 if (!inserted)
