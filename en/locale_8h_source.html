656 locale involve_msg(const std::string& domain, const std::string& lang = "",
657 bool throw_if_fail = false) const requires (!std::is_same_v<TChar, char>)
658 {
659 // messages_conf<TChar> is only specialized for char8_t, char32_t and
660 // UTF-32 wchar_t (plus the char overload above); for char16_t or a
661 // non-UTF-32 wchar_t it stays incomplete. Reject those here with a clear
662 // message instead of an opaque "incomplete type" error from make_shared
663 // below. Safe as a completeness check because messages_conf<TChar>'s
664 // completeness is fixed program-wide (a type either has a specialization
665 // everywhere or nowhere).
666 static_assert(requires { sizeof(messages_conf<TChar>); },
667 "involve_msg: messages translation is unsupported for this character "
668 "type -- no messages_conf<TChar> specialization exists (e.g. char16_t, "
669 "or wchar_t on a non-UTF-32 platform).");
670 const std::string filtered_lang = base_ft<messages>::filter_lang(domain, lang);
671 const std::string dirname = base_ft<messages>::get_dirname(domain);
672 auto ft = s_ori_facet_buf.try_get_msg<TChar>(domain, filtered_lang, dirname);
673 if (!ft)
674 {
675 // Build strictly so a load failure surfaces as an exception, and intern only
676 // on success: a degraded (load-failed) facet is never cached, so a later
677 // throw_if_fail=true call re-attempts the load and can still throw.
678 try
679 {
680 ft = std::make_shared<messages_conf<TChar>>(domain, filtered_lang, dirname, true);
681 // Caching is separate from the load: a cache-insert failure (e.g. an
682 // allocation failure while interning) must never degrade or drop a
683 // successfully-loaded facet. On such a failure keep it, just uncached.
684 try { ft = s_ori_facet_buf.put_msg<TChar>(ft, domain, filtered_lang, dirname); }
685 catch (...) {} // NOLINT(bugprone-empty-catch)
686 }
687 catch (...)
688 {
689 if (throw_if_fail) throw;
690 ft = std::make_shared<messages_conf<TChar>>(domain, filtered_lang, dirname, false);
691 }
692 }
693
694 // At most one messages facet per locale (keyed by the single messages_conf
695 // type id). If *this already carries one, this replaces it in the returned
696 // copy -- matching involve()'s same-id replace semantics; *this is unchanged.
697 locale res(*this);
698 res.m_facet_confs[ft->id()] = std::move(ft);
699 res.m_facets.clear();
700 return res;
701 }

767 const std::string& cvt = "", bool throw_if_fail = false) const requires (std::is_same_v<TChar, char>)
768 {
769 const std::string filtered_lang = base_ft<messages>::filter_lang(domain, lang);
770 // Resolve the encoding up front and use it consistently as both the cache
771 // key and the construction argument. An empty cvt maps to the (stable,
772 // env-derived) CTYPE name; keying by this effective value -- rather than by
773 // the raw cvt -- ensures the cached facet is always looked up under the
774 // exact encoding it was built with, and lets an explicit cvt equal to the
775 // CTYPE name share the same cache entry as the empty-cvt case.
776 const std::string effective_cvt = cvt.empty() ? s_ori_facet_buf.locale_name(LC_CTYPE) : cvt;
