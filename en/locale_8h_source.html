683 template <std::same_as<bool> B = bool>
684 locale involve_msg(const std::string& domain, const std::string& lang = "",
685 B throw_if_fail = false) const requires (!detail::messages_locale_encoded<TChar>)
686 {
687 // messages_conf<TChar> exists for char8_t, char32_t and a Unicode wchar_t here
688 // (char, and macOS's wchar_t, take the cvt overload below); for char16_t it stays
689 // incomplete. Reject that with a clear message instead of an opaque "incomplete
690 // type" error from make_shared below.
691 static_assert(requires { sizeof(messages_conf<TChar>); },
692 "involve_msg: messages translation is unsupported for this character "
693 "type -- no messages_conf<TChar> specialization exists (e.g. char16_t).");
694 const std::string filtered_lang = base_ft<messages>::filter_lang(domain, lang);
695 const std::filesystem::path dirname = base_ft<messages>::get_dirname(domain);
696 auto ft = s_ori_facet_buf.try_get_msg<TChar>(domain, filtered_lang, dirname);
697 if (!ft)
698 {
699 // Build strictly so a load failure surfaces as an exception, and intern only
700 // on success: a degraded (load-failed) facet is never cached, so a later
701 // throw_if_fail=true call re-attempts the load and can still throw.
702 try
703 {
704 ft = std::make_shared<messages_conf<TChar>>(domain, filtered_lang, dirname, true);
705 // Caching is separate from the load: a cache-insert failure (e.g. an
706 // allocation failure while interning) must never degrade or drop a
707 // successfully-loaded facet. On such a failure keep it, just uncached.
708 try { ft = s_ori_facet_buf.put_msg<TChar>(ft, domain, filtered_lang, dirname); }
709 catch (...) {} // NOLINT(bugprone-empty-catch)
710 }
711 catch (...)
712 {
713 if (throw_if_fail) throw;
714 ft = std::make_shared<messages_conf<TChar>>(domain, filtered_lang, dirname, false);
715 }
716 }
717
718 // At most one messages facet per locale (keyed by the single messages_conf
719 // type id). If *this already carries one, this replaces it in the returned
720 // copy -- matching involve()'s same-id replace semantics; *this is unchanged.
721 locale res(*this);
722 res.m_facet_confs[ft->id()] = std::move(ft);
723 res.m_facets.clear();
724 return res;
725 }

803 locale involve_msg(const std::string& domain, const std::string& lang = "", // NOLINT(bugprone-easily-swappable-parameters)
804 const std::string& cvt = "", B throw_if_fail = false) const requires detail::messages_locale_encoded<TChar>
805 {
806 const std::string filtered_lang = base_ft<messages>::filter_lang(domain, lang);
807 // Resolve the encoding up front and use it consistently as both the cache
808 // key and the construction argument. An empty cvt maps to the (stable,
809 // env-derived) CTYPE name; keying by this effective value -- rather than by

813 const std::string effective_cvt = cvt.empty() ? s_ori_facet_buf.locale_name(LC_CTYPE) : cvt;
