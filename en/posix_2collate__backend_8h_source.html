50// Runs call with errno cleared; nullopt if it set errno. The caller's errno is kept.
51template <typename Call>
52std::optional<decltype(std::declval<Call&>()())> collate_errno_call(Call call) noexcept
53{
54 const int saved = errno;
55 errno = 0;
56 const auto res = call();
57 const bool failed = (errno != 0);
58 errno = saved;
59 if (failed)
60 return std::nullopt;
61 return res;
62}

64inline std::optional<int> collate_strcoll(const char* s1, const char* s2, clocale_handle loc) noexcept
65{
66 const auto res = collate_errno_call([&] { return strcoll_l(s1, s2, loc); });
67 if (!res) return std::nullopt;
68 return (*res > 0) - (*res < 0);
69}
