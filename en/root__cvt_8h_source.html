408 std::pair<device_type, std::exception_ptr> detach() noexcept
409 {

534 void main_cont_beg()
535 {
536 // bos() must be called before main_cont_beg() to establish io_status
537 assert(m_io_status != io_status::neutral);
538
539 if constexpr (dev_cpt::support_put<device_type>)
540 {
541 if (m_io_status == io_status::output)
542 flush();
543 }
544
545 if constexpr (dev_cpt::support_positioning<device_type>)
546 {
547 // Invariant: dtell() >= buffered data. The buffer is filled via dget(),
548 // which advances the device position by the amount read. Consuming data
549 // from the buffer only moves m_buf_cur forward, never increasing the
550 // buffered amount beyond what was read from the device.
551 const auto buffered = static_cast<std::size_t>(m_buf_end - m_buf_cur);
552 assert(m_device.dtell() >= buffered);
553 m_bos_len = m_device.dtell() - buffered;
554 }
555 }

602 io_status bos()
603 {
604 if constexpr (dev_cpt::support_get<device_type> &&
605 dev_cpt::support_put<device_type>)
606 {
607 if (m_device.deof())

609 else
610 m_io_status = io_status::input;

613 m_io_status = io_status::input;

615 m_io_status = io_status::output;
616 else
617 static_assert(dependent_false_v<device_type>, "device does not support get nor put");

696 std::size_t get(internal_type* to, std::size_t to_max)
697 requires (dev_cpt::support_get<device_type>)
698 {
699 if constexpr (dev_cpt::support_put<device_type>)
700 {
701 if (m_io_status != io_status::input)
702 switch_to_get();
703 }
704 if (m_io_status != io_status::input)
705 throw cvt_error("root_cvt::get fails: invalid io status");
706
707 if constexpr (!HasInBuffer)
708 return m_device.dget(to, to_max);
709 else
710 {
711 if (m_buf_cur == m_buf_end)
712 {
713 const std::size_t act_len = m_device.dget(m_buffer.data(), s_buffer_length);
714 m_buf_cur = m_buffer.data();
715 m_buf_end = m_buf_cur + act_len;
716 }
717
718 const std::size_t res_len = std::min<std::size_t>(m_buf_end - m_buf_cur, to_max);
719 std::copy(m_buf_cur, m_buf_cur + res_len, to);
720 m_buf_cur += res_len;
721 return res_len;
722 }

756 void put(const internal_type* to, std::size_t to_size)
757 requires (dev_cpt::support_put<device_type>)
758 {
759 if constexpr (dev_cpt::support_get<device_type>)
760 {
761 if (m_io_status != io_status::output)
762 switch_to_put();
763 }
764 if (m_io_status != io_status::output)
765 throw cvt_error("root_cvt::put fails: invalid io status");
766
767 const std::size_t buf_used = m_buf_cur - m_buffer.data();
768 const std::size_t remain = s_buffer_length - buf_used;
769 if (to_size < remain)
770 {
771 m_buf_cur = std::copy(to, to + to_size, m_buf_cur);
772 return;
773 }
774
775 if (to_size == remain)
776 {
777 std::copy(to, to + to_size, m_buf_cur);
778 m_device.dput(m_buffer.data(), s_buffer_length);
779 m_buf_cur = m_buffer.data();
780 return;
781 }
782
783 if (buf_used > 0)
784 m_device.dput(m_buffer.data(), buf_used);
785 if (to_size < s_buffer_length / 2)
786 m_buf_cur = std::copy(to, to + to_size, m_buffer.data());
787 else
788 {
789 m_buf_cur = m_buffer.data();
790 m_device.dput(to, to_size);
791 }
792 }

806 void flush()
807 requires (dev_cpt::support_put<device_type>)
808 {
809 if (m_io_status != io_status::output) return;
810
811 if ((m_buf_cur != nullptr) && (m_buf_cur != m_buffer.data()))
812 {
813 m_device.dput(m_buffer.data(), m_buf_cur - m_buffer.data());
814 m_buf_cur = m_buffer.data();
815 }

840 [[nodiscard]] std::size_t tell() const
841 requires (dev_cpt::support_positioning<device_type>)
842 {
843 const std::size_t device_tell = m_device.dtell();
844
845 switch(m_io_status)
846 {
847 case io_status::output:
848 {
849 // Invariant: device_tell >= m_bos_len. The device position only advances
850 // via flush operations, never moving backward from the stream start.
851 assert(device_tell >= m_bos_len);
852 const auto buf_used = static_cast<std::size_t>(m_buf_cur - m_buffer.data());
853 return device_tell - m_bos_len + buf_used;
854 }
855 case io_status::input:
856 {
857 // Invariant: device_tell - m_bos_len >= buffered. Same invariant as
858 // main_cont_beg(): buffer data comes from dget(), which advances device
859 // position by the amount read.
860 const auto buffered = static_cast<std::size_t>(m_buf_end - m_buf_cur);
861 assert(device_tell >= m_bos_len);
862 assert(device_tell - m_bos_len >= buffered);
863 return device_tell - m_bos_len - buffered;
864 }
865 default:
866 throw cvt_error("root_cvt::tell fails: invalid io status");
867 }
868 }

888 void seek(std::size_t pos)
889 requires (dev_cpt::support_positioning<device_type>)
890 {
891 if constexpr (dev_cpt::support_put<device_type>)
892 {
893 if (m_io_status == io_status::output)
894 flush();
895 }
896
897 if (pos > std::numeric_limits<std::size_t>::max() - m_bos_len)
898 throw cvt_error("root_cvt::seek fails: position overflow");

901 if (m_io_status == io_status::input)
902 m_buf_cur = m_buf_end;
903 }

925 void rseek(std::size_t pos)
926 requires (dev_cpt::support_positioning<device_type>)
927 {
928 if constexpr (dev_cpt::support_put<device_type>)
929 {
930 if (m_io_status == io_status::output)
931 flush();
932 }
933

959 void switch_to_get()
960 requires (dev_cpt::support_get<device_type> && dev_cpt::support_put<device_type>)
961 {
962 switch(m_io_status)
963 {
964 case io_status::input:
965 return;
966 case io_status::output:
967 flush();
968 [[fallthrough]];
969 default:
970 m_io_status = io_status::input;
971 m_buf_cur = m_buf_end = m_buffer.data();
972 }
973 }
974
997 void switch_to_put()
998 requires (dev_cpt::support_get<device_type> && dev_cpt::support_put<device_type>)
999 {
1000 switch(m_io_status)
1001 {
1002 case io_status::output:
1003 return;
1004 case io_status::input:
1005 if (m_buf_cur != m_buf_end)
1006 {
1007 if constexpr (!dev_cpt::support_positioning<device_type>)
1008 throw cvt_error("root_cvt::switch_to_put fails: device does not support positioning");
1009 else
1010 seek(tell());
1011 }
1012 [[fallthrough]];
1013 default:
1014 m_io_status = io_status::output;
1015 m_buf_cur = m_buf_end = m_buffer.data();
1016 }
1017 }

1750 template <bool Saturate = false>
1751 auto get_buf(std::size_t to_max)
1752 {
1753 if (to_max == 0)
1754 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
1755 if (to_max > KernelType::s_buffer_length)
1756 throw cvt_error("cvt_reader::get_buf fail, read size too large");
1757
1758 if constexpr (dev_cpt::support_put<device_type>)
1759 {
1760 if (m_kernel.m_io_status != io_status::input)
1761 m_kernel.switch_to_get();
1762 }
1763 if (m_kernel.m_io_status != io_status::input)
1764 throw cvt_error("cvt_reader::get_buf fail, invalid io status");
1765
1766 const std::size_t remain = m_kernel.m_buf_end - m_kernel.m_buf_cur;
1767 // need to read, then read at most as it can
1768 if (remain < to_max)
1769 {
1770 if (remain != 0)
1771 m_kernel.m_buf_end = std::copy(m_kernel.m_buf_cur, m_kernel.m_buf_end, m_kernel.m_buffer.data());
1772 else
1773 m_kernel.m_buf_end = m_kernel.m_buffer.data();
1774
1775 m_kernel.m_buf_cur = m_kernel.m_buffer.data();
1776 m_kernel.m_buf_end += m_kernel.m_device.dget(m_kernel.m_buf_end, KernelType::s_buffer_length - remain);
1777 }
1778
1779 if constexpr (Saturate)
1780 {
1781 const auto aim_ptr = to_max + m_kernel.m_buf_cur;
1782 while (m_kernel.m_buf_end < aim_ptr)
1783 {
1784 auto new_size = m_kernel.m_device.dget(m_kernel.m_buf_end, aim_ptr - m_kernel.m_buf_end);
1785 if (new_size == 0)
1786 throw cvt_error("get_buf<Saturate> fail: meet eos");
1787 m_kernel.m_buf_end += new_size;
1788 }
1789 }
