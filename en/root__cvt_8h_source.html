528 void main_cont_beg()
529 {
530 // bos() must be called before main_cont_beg() to establish io_status
531 assert(m_io_status != io_status::neutral);
532
533 if constexpr (dev_cpt::support_put<device_type>)
534 {
535 if (m_io_status == io_status::output)
536 flush();
537 }
538
539 if constexpr (dev_cpt::support_positioning<device_type>)
540 {
541 // Invariant: dtell() >= buffered data. The buffer is filled via dget(),
542 // which advances the device position by the amount read. Consuming data
543 // from the buffer only moves m_buf_cur forward, never increasing the
544 // buffered amount beyond what was read from the device.
545 const auto buffered = static_cast<std::size_t>(m_buf_end - m_buf_cur);
546 assert(m_device.dtell() >= buffered);
547 m_bos_len = m_device.dtell() - buffered;
548 }
549 }

596 io_status bos()
597 {
598 if constexpr (dev_cpt::support_get<device_type> &&
599 dev_cpt::support_put<device_type>)
600 {
601 if (m_device.deof())
602 m_io_status = io_status::output;
603 else
604 m_io_status = io_status::input;
605 }
606 else if constexpr (dev_cpt::support_get<device_type>)
607 m_io_status = io_status::input;
608 else if constexpr (dev_cpt::support_put<device_type>)
609 m_io_status = io_status::output;
610 else
611 static_assert(dependent_false_v<device_type>, "device does not support get nor put");
612
613 return m_io_status;
614 }

690 std::size_t get(internal_type* to, std::size_t to_max)
691 requires (dev_cpt::support_get<device_type>)
692 {
693 if constexpr (dev_cpt::support_put<device_type>)
694 {
695 if (m_io_status != io_status::input)
696 switch_to_get();
697 }
698 if (m_io_status != io_status::input)
699 throw cvt_error("root_cvt::get fails: invalid io status");
700
701 if constexpr (!HasInBuffer)
702 return m_device.dget(to, to_max);
703 else
704 {
705 if (m_buf_cur == m_buf_end)
706 {
707 const std::size_t act_len = m_device.dget(m_buffer.data(), s_buffer_length);
708 m_buf_cur = m_buffer.data();
709 m_buf_end = m_buf_cur + act_len;
710 }
711
712 const std::size_t res_len = std::min<std::size_t>(m_buf_end - m_buf_cur, to_max);
713 std::copy(m_buf_cur, m_buf_cur + res_len, to);
714 m_buf_cur += res_len;
715 return res_len;
716 }

750 void put(const internal_type* to, std::size_t to_size)
751 requires (dev_cpt::support_put<device_type>)
752 {
753 if constexpr (dev_cpt::support_get<device_type>)
754 {
755 if (m_io_status != io_status::output)
756 switch_to_put();
757 }
758 if (m_io_status != io_status::output)
759 throw cvt_error("root_cvt::put fails: invalid io status");
760
761 const std::size_t buf_used = m_buf_cur - m_buffer.data();
762 const std::size_t remain = s_buffer_length - buf_used;
763 if (to_size < remain)
764 {
765 m_buf_cur = std::copy(to, to + to_size, m_buf_cur);
766 return;
767 }
768
769 if (to_size == remain)
770 {
771 std::copy(to, to + to_size, m_buf_cur);
772 m_device.dput(m_buffer.data(), s_buffer_length);
773 m_buf_cur = m_buffer.data();
774 return;
775 }
776
777 if (buf_used > 0)
778 m_device.dput(m_buffer.data(), buf_used);
779 if (to_size < s_buffer_length / 2)
780 m_buf_cur = std::copy(to, to + to_size, m_buffer.data());
781 else
782 {
783 m_buf_cur = m_buffer.data();
784 m_device.dput(to, to_size);
785 }
786 }

800 void flush()
801 requires (dev_cpt::support_put<device_type>)
802 {
803 if (m_io_status != io_status::output) return;
804
805 if ((m_buf_cur != nullptr) && (m_buf_cur != m_buffer.data()))
806 {
807 m_device.dput(m_buffer.data(), m_buf_cur - m_buffer.data());
808 m_buf_cur = m_buffer.data();
809 }

834 [[nodiscard]] std::size_t tell() const
835 requires (dev_cpt::support_positioning<device_type>)
836 {
837 const std::size_t device_tell = m_device.dtell();
838
839 switch(m_io_status)
840 {
841 case io_status::output:
842 {
843 // Invariant: device_tell >= m_bos_len. The device position only advances
844 // via flush operations, never moving backward from the stream start.
845 assert(device_tell >= m_bos_len);
846 const auto buf_used = static_cast<std::size_t>(m_buf_cur - m_buffer.data());
847 return device_tell - m_bos_len + buf_used;
848 }
849 case io_status::input:
850 {
851 // Invariant: device_tell - m_bos_len >= buffered. Same invariant as
852 // main_cont_beg(): buffer data comes from dget(), which advances device
853 // position by the amount read.
854 const auto buffered = static_cast<std::size_t>(m_buf_end - m_buf_cur);
855 assert(device_tell >= m_bos_len);
856 assert(device_tell - m_bos_len >= buffered);
857 return device_tell - m_bos_len - buffered;
858 }
859 default:
860 throw cvt_error("root_cvt::tell fails: invalid io status");
861 }
862 }

882 void seek(std::size_t pos)
883 requires (dev_cpt::support_positioning<device_type>)
884 {
885 if constexpr (dev_cpt::support_put<device_type>)
886 {
887 if (m_io_status == io_status::output)
888 flush();

920 requires (dev_cpt::support_positioning<device_type>)

922 if constexpr (dev_cpt::support_put<device_type>)

924 if (m_io_status == io_status::output)

953 void switch_to_get()
954 requires (dev_cpt::support_get<device_type> && dev_cpt::support_put<device_type>)
955 {
956 switch(m_io_status)
957 {
958 case io_status::input:
959 return;
960 case io_status::output:
961 flush();
962 [[fallthrough]];
963 default:
964 m_io_status = io_status::input;

991 void switch_to_put()
992 requires (dev_cpt::support_get<device_type> && dev_cpt::support_put<device_type>)
993 {
994 switch(m_io_status)
995 {
996 case io_status::output:
997 return;
998 case io_status::input:
999 if (m_buf_cur != m_buf_end)
1000 {
1001 if constexpr (!dev_cpt::support_positioning<device_type>)
1002 throw cvt_error("root_cvt::switch_to_put fails: device does not support positioning");
1003 else
1004 seek(tell());
1005 }
1006 [[fallthrough]];
1007 default:
1008 m_io_status = io_status::output;
1009 m_buf_cur = m_buf_end = m_buffer.data();
1010 }
1011 }

1744 template <bool Saturate = false>
1745 auto get_buf(std::size_t to_max)
1746 {
1747 if (to_max == 0)
1748 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
1749 if (to_max > KernelType::s_buffer_length)
1750 throw cvt_error("cvt_reader::get_buf fail, read size too large");
1751
1752 if constexpr (dev_cpt::support_put<device_type>)
1753 {
1754 if (m_kernel.m_io_status != io_status::input)
1755 m_kernel.switch_to_get();
1756 }
1757 if (m_kernel.m_io_status != io_status::input)
1758 throw cvt_error("cvt_reader::get_buf fail, invalid io status");
1759
1760 const std::size_t remain = m_kernel.m_buf_end - m_kernel.m_buf_cur;
1761 // need to read, then read at most as it can
1762 if (remain < to_max)
1763 {
1764 if (remain != 0)
1765 m_kernel.m_buf_end = std::copy(m_kernel.m_buf_cur, m_kernel.m_buf_end, m_kernel.m_buffer.data());
1766 else
1767 m_kernel.m_buf_end = m_kernel.m_buffer.data();
1768
1769 m_kernel.m_buf_cur = m_kernel.m_buffer.data();
1770 m_kernel.m_buf_end += m_kernel.m_device.dget(m_kernel.m_buf_end, KernelType::s_buffer_length - remain);
1771 }
1772
1773 if constexpr (Saturate)
1774 {
1775 const auto aim_ptr = to_max + m_kernel.m_buf_cur;
1776 while (m_kernel.m_buf_end < aim_ptr)
1777 {
1778 auto new_size = m_kernel.m_device.dget(m_kernel.m_buf_end, aim_ptr - m_kernel.m_buf_end);
1779 if (new_size == 0)
1780 throw cvt_error("get_buf<Saturate> fail: meet eos");
1781 m_kernel.m_buf_end += new_size;
1782 }
1783 }
1784
