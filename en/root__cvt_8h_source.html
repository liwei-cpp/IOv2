487 void attach(device_type&& dev = device_type{})
488 {
489 const bool was_output = (m_io_status == io_status::output);
490 auto detach_res = detach();
491
492 if constexpr (dev_cpt::support_put<device_type>)
493 {
494 if (was_output)
495 {
496 try
497 {
498 detach_res.first.dflush();

500 catch (...)
501 {
502 // detach() performs the converter-level flush first, so its
503 // error has priority over a later device-level flush error.
504 if (!detach_res.second)

567 void main_cont_beg()
568 {
569 // bos() must be called before main_cont_beg() to establish io_status
570 assert(m_io_status != io_status::neutral);
571
572 if constexpr (dev_cpt::support_put<device_type>)
573 {
574 if (m_io_status == io_status::output)
575 flush();
576 }
577
578 if constexpr (dev_cpt::support_positioning<device_type>)
579 {
580 // Invariant: dtell() >= buffered data. The buffer is filled via dget(),
581 // which advances the device position by the amount read. Consuming data
582 // from the buffer only moves m_buf_cur forward, never increasing the
583 // buffered amount beyond what was read from the device.
584 const auto buffered = static_cast<std::size_t>(m_buf_end - m_buf_cur);
585 assert(m_device.dtell() >= buffered);
586 m_bos_len = m_device.dtell() - buffered;
587 }

729 std::size_t get(this auto& self, internal_type* to, std::size_t to_max)
730 requires (dev_cpt::support_get<device_type>)
731 {
732 if constexpr (dev_cpt::support_put<device_type>)
733 {
734 if (self.m_io_status != io_status::input)
735 self.switch_to_get();
736 }
737 if (self.m_io_status != io_status::input)
738 throw cvt_error("root_cvt::get fails: invalid io status");
739
740 if constexpr (!HasInBuffer)
741 return self.m_device.dget(to, to_max);
742 else
743 {
744 if (self.m_buf_cur == self.m_buf_end)
745 {
746 // A derived root may read less than a full buffer (see read_ahead).
747 std::size_t want = s_buffer_length;
748 if constexpr (requires { self.read_ahead(to_max, want); })
749 want = self.read_ahead(std::min(to_max, want), want);
750 const std::size_t act_len = self.m_device.dget(self.m_buffer.data(), want);
751 self.m_buf_cur = self.m_buffer.data();
752 self.m_buf_end = self.m_buf_cur + act_len;
753 }
754
755 const std::size_t res_len = std::min<std::size_t>(self.m_buf_end - self.m_buf_cur, to_max);
756 std::copy(self.m_buf_cur, self.m_buf_cur + res_len, to);
757 self.m_buf_cur += res_len;
758 return res_len;
759 }

793 void put(const internal_type* to, std::size_t to_size)
794 requires (dev_cpt::support_put<device_type>)
795 {
796 if constexpr (dev_cpt::support_get<device_type>)
797 {
798 if (m_io_status != io_status::output)
799 switch_to_put();
800 }
801 if (m_io_status != io_status::output)
802 throw cvt_error("root_cvt::put fails: invalid io status");
803
804 const std::size_t buf_used = m_buf_cur - m_buffer.data();
805 const std::size_t remain = s_buffer_length - buf_used;
806 if (to_size < remain)
807 {
808 m_buf_cur = std::copy(to, to + to_size, m_buf_cur);
809 return;
810 }
811
812 if (to_size == remain)
813 {
814 std::copy(to, to + to_size, m_buf_cur);
815 dput_buffer(s_buffer_length, buf_used);
816 return;
817 }
818
819 if (buf_used > 0)
820 dput_buffer(buf_used, buf_used);
821 if (to_size < s_buffer_length / 2)
822 m_buf_cur = std::copy(to, to + to_size, m_buffer.data());
823 else
824 {
825 m_buf_cur = m_buffer.data();
826 m_device.dput(to, to_size);
827 }
828 }

878 [[nodiscard]] std::size_t tell() const
879 requires (dev_cpt::support_positioning<device_type>)
880 {
881 const std::size_t device_tell = m_device.dtell();
882
883 switch(m_io_status)
884 {
885 case io_status::output:
886 {
887 // Invariant: device_tell >= m_bos_len. The device position only advances
888 // via flush operations, never moving backward from the stream start.
889 assert(device_tell >= m_bos_len);
890 const auto buf_used = static_cast<std::size_t>(m_buf_cur - m_buffer.data());
891 return device_tell - m_bos_len + buf_used;
892 }
893 case io_status::input:
894 {
895 // Invariant: device_tell - m_bos_len >= buffered. Same invariant as
896 // main_cont_beg(): buffer data comes from dget(), which advances device
897 // position by the amount read.
898 const auto buffered = static_cast<std::size_t>(m_buf_end - m_buf_cur);
899 assert(device_tell >= m_bos_len);
900 assert(device_tell - m_bos_len >= buffered);
901 return device_tell - m_bos_len - buffered;
902 }
903 default:
904 throw cvt_error("root_cvt::tell fails: invalid io status");
905 }

1132 void dput_buffer(std::size_t len, std::size_t old)
1133 requires (dev_cpt::support_put<device_type>)
1134 {
1135 try
1136 {
1137 m_device.dput(m_buffer.data(), len);
1138 }
1139 catch (const dput_error& e)
1140 {
1141 const std::size_t done = std::min(e.written(), len);
1142 const std::size_t keep_end = (done > old) ? len : old;
1143 if (done == 0)
1144 m_buf_cur = m_buffer.data() + keep_end;
1145 else
1146 m_buf_cur = std::copy(m_buffer.data() + done, m_buffer.data() + keep_end, m_buffer.data());
1147 throw;
1148 }
1149 catch (...)
1150 {
1151 m_buf_cur = m_buffer.data() + old;
1152 throw;
1153 }
1154 m_buf_cur = m_buffer.data();
1155 }

1166 std::vector<external_type> m_buffer;
1169 external_type* m_buf_cur;
1172 external_type* m_buf_end;

1904 // need to read, then read at most as it can

1908 m_kernel.m_buf_end = std::copy(m_kernel.m_buf_cur, m_kernel.m_buf_end, m_kernel.m_buffer.data());
1909 else
1910 m_kernel.m_buf_end = m_kernel.m_buffer.data();

1912 m_kernel.m_buf_cur = m_kernel.m_buffer.data();
1913 // A derived root may read less than fills the buffer (see read_ahead).
1914 std::size_t want = KernelType::s_buffer_length - remain;
1915 if constexpr (requires { m_kernel.read_ahead(to_max - remain, want); })
1916 want = m_kernel.read_ahead(to_max - remain, want);
1917 m_kernel.m_buf_end += m_kernel.m_device.dget(m_kernel.m_buf_end, want);
1918 }
1919

1923 while (m_kernel.m_buf_end < aim_ptr)
1924 {

1926 if (new_size == 0)
1927 throw cvt_error("get_buf<Saturate> fail: meet eos");
1928 m_kernel.m_buf_end += new_size;
1929 }
1930 }
1931
1932 if constexpr (!Saturate)
1933 {
1934 auto res_len = std::min<std::size_t>(m_kernel.m_buf_end - m_kernel.m_buf_cur, to_max);
1935 std::pair<const char_type*, std::size_t> res{m_kernel.m_buf_cur, res_len};
1936 m_kernel.m_buf_cur += res_len;
1937 return res;
1938 }
1939 else
1940 {
1941 const char_type* res = m_kernel.m_buf_cur;
1942 m_kernel.m_buf_cur += to_max;
1943 return res;
1944 }
1945 }

1968 {
