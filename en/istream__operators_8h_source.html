467 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
468 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
469 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
470 TOut get(this TSelf& self, TOut s, size_t n, TChar delim)
471 {
472 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
473
474 size_t gcount = 0;
475 bool at_eof = false;
476 std::lock_guard guard(self.io_mutex());
477 try
478 {
479 using sentry_type = typename TSelf::in_sentry_type;
480 sentry_type cerb(self, true);
481 if constexpr (std::is_pointer_v<TOut>)
482 {
483 if (s == nullptr)
484 throw stream_error("istream get fail: null character sequence");
485 }
486 if (n == 0)
487 throw stream_error("istream get fail: zero buffer size");
488 auto c = self.m_streambuf.sgetc();
489 while ((gcount + is_cstr < n) &&
490 (c.has_value()) &&
491 (c.value() != delim))
492 {
493 *s++ = c.value();
494 ++gcount;
495 c = self.m_streambuf.snextc();
496 }
497
498 at_eof = (gcount + is_cstr < n) && (!c.has_value());
499
500 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
501 {
502 if (c.has_value())
503 {
504 if (c.value() == delim)
505 {
506 self.m_streambuf.sbumpc();
507 ++gcount;
508 }
509 else
510 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
511 }
512 }
513
514 if (gcount == 0)
515 throw stream_error{"istream get fail: no character extracted"};
516
517 if (at_eof)
518 self.setstate(ios_defs::eofbit);
519 }
520 catch(...)
521 {
522 if constexpr (is_cstr)
523 {
524 if constexpr (std::is_pointer_v<TOut>)
525 {
526 if (s != nullptr && n != 0)
527 *s++ = TChar{};
528 }
529 else if (n != 0)
530 *s++ = TChar{};
531 }
532 self.handle_exception(std::current_exception(), at_eof);
533 return s;
534 }
535
536 if constexpr (is_cstr)
537 *s++ = TChar{};
538 return s;

956template <istream_type T, typename TValue>
957T& operator>>(T& obj, TValue&& value)
958{
959 using TChar = typename T::char_type;
960 using TV = std::remove_cvref_t<TValue>;
961 using TCtx = typename parse_context_type<TChar, TV>::type;
962 using TTarget = std::conditional_t<std::is_lvalue_reference_v<TValue>,
963 std::remove_reference_t<TValue>, const TV>;
964
965 constexpr bool stream_r = requires
966 { io_traits<TChar, TV>::sread(obj, value); };
967 constexpr bool iter_v = std::is_same_v<TCtx, TV> && requires (TTarget& v)
968 { io_traits<TChar, TV>::sread(obj.i_iter(), std::default_sentinel,
969 obj, obj.locale(), v); };
970 constexpr bool iter_c = !std::is_same_v<TCtx, TV> && requires (TTarget& v, TCtx& c)
971 {
972 io_traits<TChar, TCtx>::sread(obj.i_iter(), std::default_sentinel,
973 obj, obj.locale(), c);
974 v = static_cast<TV>(c);
975 };
976
977 if constexpr (stream_r)
978 {
979 try
980 {
981 io_traits<TChar, TV>::sread(obj, value);
982 }
983 catch (...)
984 {
985 obj.handle_exception(std::current_exception());
986 }
987 }
988 else if constexpr (iter_v || iter_c)
989 {
990 using sentry_type = typename T::in_sentry_type;
991
992 bool saw_eof = false;
993 std::lock_guard guard(obj.io_mutex());
994 try
995 {
996 auto iter = obj.i_iter(&saw_eof);
997 bool skip = bool(obj.flags() & ios_defs::skipws);
998 sentry_type cerb(obj, !skip);
999
1000 if (obj.eof())
1001 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1002
1003 if constexpr (iter_v)
1004 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(), value);
1005 else
1006 {
1007 TCtx tmp = [&value]() -> TCtx {
1008 if constexpr (requires (const TV& v)
1009 { parse_context_type<TChar, TV>::make_parse_context(v); })
1010 return parse_context_type<TChar, TV>::make_parse_context(value);
1011 else
1012 return TCtx{};
1013 }();
1014 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1015 value = static_cast<TV>(tmp);
1016 }
1017
1018 if (saw_eof) obj.setstate(ios_defs::eofbit);
1019 }
1020 catch(...)
1021 {
1022 obj.handle_exception(std::current_exception(), saw_eof);
1023 }
1024 }
1025 else if constexpr (has_io_traits<TChar, TV> || has_io_traits<TChar, TCtx>)
1026 static_assert(dependent_false_v<TValue>,
1027 "IOv2: cannot extract into this type. An io_traits<char_type, TValue> exists but "
1028 "offers no sread() usable with this stream. The usual cause is a direction mismatch "
1029 "-- the type provides swrite() only and is meant for insertion (write `os << x`, not "
1030 "`is >> x`). Otherwise the target is not writable: extraction needs a non-const "
1031 "lvalue, so `is >> 5` and `is >> const_obj` are rejected here. A fill character whose "
1032 "type differs from the stream's char_type lands here too. "
1033 "See io/traits/traits_base.h.");
1034 else
1035 static_assert(dependent_false_v<TValue>,
1036 "IOv2: cannot extract into this type. No io_traits<char_type, TValue> is defined for "
1037 "it. Define one -- see io/traits/traits_base.h -- or extract into a supported type "
1038 "such as an arithmetic type, a character type, CharT* or std::basic_string.");
1039
1040 return obj;
1041}
