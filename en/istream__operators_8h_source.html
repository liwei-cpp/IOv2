712 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
713 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
714 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
715 TOut get(this TSelf& self, TOut s, std::ptrdiff_t n, TChar delim)
716 {
717 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
718
719 std::size_t gcount = 0;
720 bool at_eof = false;
721 std::lock_guard guard(self.io_mutex());
722 try
723 {
724 using sentry_type = typename TSelf::in_sentry_type;
725 sentry_type cerb(self, true);
726 if constexpr (std::is_pointer_v<TOut>)
727 {
728 if (s == nullptr)
729 throw stream_error("istream get fail: null character sequence");
730 }
731 if (n < 0)
732 throw stream_error("istream get fail: negative buffer size");
733 if (n == 0)
734 throw stream_error("istream get fail: zero buffer size");
735 const auto cap = static_cast<std::size_t>(n);
736 // Room for no character: fail before asking the device for one. getline still
737 // looks, since a delimiter there is an empty line it consumes.
738 if constexpr (std::is_same_v<DelimPolicy, keep_sep>)
739 {
740 if (cap - is_cstr == 0)
741 throw stream_error{"istream get fail: no character extracted"};
742 }
743 auto c = self.m_channel.getc();
744 while ((gcount + is_cstr < cap) &&
745 (c.has_value()) &&
746 (c.value() != delim))
747 {
748 *s++ = c.value();
749 ++gcount;
750 // Full: consume this one and stop. A peek at the next would read a byte this
751 // call does not need. getline must peek, to tell a delimiter from overflow.
752 if constexpr (std::is_same_v<DelimPolicy, keep_sep>)
753 {
754 if (gcount + is_cstr == cap)
755 {
756 self.m_channel.bumpc();
757 break;
758 }
759 }
760 c = self.m_channel.nextc();
761 }
762
763 at_eof = (gcount + is_cstr < cap) && (!c.has_value());
764
765 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
766 {
767 if (c.has_value())
768 {
769 if (c.value() == delim)
770 {
771 self.m_channel.bumpc();
772 ++gcount;
773 }
774 else
775 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
776 }
777 }
778
779 if (gcount == 0)
780 throw stream_error{"istream get fail: no character extracted"};
781
782 if (at_eof)
783 self.setstate(ios_defs::eofbit);
784 }
785 catch(...)
786 {
787 if constexpr (is_cstr)
788 {
789 if constexpr (std::is_pointer_v<TOut>)
790 {
791 if (s != nullptr && n > 0)
792 *s++ = TChar{};
793 }
794 else if (n > 0)
795 *s++ = TChar{};
796 }
797 self.handle_exception(std::current_exception(), at_eof);
798 return s;
799 }
800
801 if constexpr (is_cstr)
802 *s++ = TChar{};
803 return s;
804 }

1065 template <typename TSelf>
1066 TSelf& ignore(this TSelf& self, std::ptrdiff_t n, TChar delim)
1067 {
1068 std::ptrdiff_t gcount = 0;
1069 bool at_eof = false;
1070 std::lock_guard guard(self.io_mutex());
1071 try
1072 {
1073 using sentry_type = typename TSelf::in_sentry_type;
1074 sentry_type cerb(self, true);
1075 if (n < 0)
1076 throw stream_error{"istream ignore fail: negative character count"};
1077 if (n == 0) return self;
1078
1079 auto c = self.m_channel.getc();
1080 while (gcount < n
1081 && c.has_value()
1082 && (c.value() != delim))
1083 {
1084 // The n-th one: consume it and stop, without peeking at one more.
1085 if (++gcount == n)
1086 {
1087 self.m_channel.bumpc();
1088 break;
1089 }
1090 c = self.m_channel.nextc();
1091 }
1092
1093 at_eof = (gcount < n) && (!c.has_value());
1094
1095 if (gcount < n)
1096 {
1097 if (c.has_value())
1098 {
1099 ++gcount;
1100 self.m_channel.bumpc();
1101 }
1102 }
1103
1104 if (at_eof)
1105 self.setstate(ios_defs::eofbit);
1106 }
1107 catch(...)
1108 {
1109 self.handle_exception(std::current_exception(), at_eof);

1269template <istream_type T, typename TValue>
1270 requires detail::extractable<T, TValue>
1271// The forwarding reference below is a binder plus a value-category carrier, never a forwarder;
1272// see the @note above for why `std::forward` here would be wrong rather than merely redundant.
1273// NOLINTNEXTLINE(cppcoreguidelines-missing-std-forward)
1274T& operator>>(T& obj, TValue&& value)
1275{
1276 using TChar = typename T::char_type;
1277 using TV = std::remove_cvref_t<TValue>;
1278 using TCtx = detail::in_ctx_t<TChar, TValue>;
1279 using TTarget = detail::in_target_t<TValue>;
1280
1281 constexpr bool stream_r = detail::extractable_with_stream<T, TValue>;
1282 constexpr bool iter_v = detail::extractable_with_iter<T, TValue>;
1283 constexpr bool iter_c = detail::extractable_with_ctx<T, TValue>;
1284
1285 static_assert(!(stream_r && (iter_v || iter_c)),
1286 "IOv2: this io_traits specialization provides both forms of sread() -- the stream form "
1287 "sread(stream, value) and the iterator form sread(iter, end, io, loc, value). Provide "
1288 "exactly one: the iterator form for formatted extraction, the stream form for a "
1289 "manipulator. See io/traits/traits_base.h.");
1290
1291 if constexpr (stream_r)
1292 {
1293 try
1294 {
1295 io_traits<TChar, TV>::sread(obj, static_cast<TTarget&>(value));
1296 }
1297 catch (...)
1298 {
1299 obj.handle_exception(std::current_exception());
1300 }
1301 }
1302 else if constexpr (iter_v || iter_c)
1303 {
1304 using sentry_type = typename T::in_sentry_type;
1305
1306 bool saw_eof = false;
1307 std::lock_guard guard(obj.io_mutex());
1308 try
1309 {
1310 auto iter = obj.i_iter(&saw_eof);
1311 bool skip = bool(obj.flags() & ios_defs::skipws);
1312 sentry_type cerb(obj, !skip);
1313
1314 if (obj.eof())
1315 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1316
1317 if constexpr (iter_v)
1318 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(),
1319 static_cast<TTarget&>(value));
1320 else
1321 {
1322 using TPct = parse_context_type<TChar, TV>;
1323 TCtx tmp = [&value]() -> TCtx {
1324 // Detect the member by name only, so a mis-written one reaches a static_assert
1325 // instead of being read as absent and silently default constructed.
1326 if constexpr (detail::declares_maker<TPct, TTarget>)
1327 {
1328 if constexpr (requires { &TPct::make_parse_context; })
1329 {
1330 static_assert(
1331 detail::maker_signature_v<TCtx, TV,
1332 decltype(&TPct::make_parse_context)>,
1333 "IOv2: this parse_context_type's make_parse_context() must be "
1334 "declared exactly `static type make_parse_context(const T&)` -- it "
1335 "seeds the context from the target, so it takes the target by const "
1336 "reference and returns parse_context_type<TChar, T>::type by value. "
1337 "See io/traits/traits_base.h.");
1338 return TPct::make_parse_context(static_cast<const TV&>(value));
1339 }
1340 else
1341 static_assert(dependent_false_v<TCtx>,
1342 "IOv2: this parse_context_type's make_parse_context() must be a "
1343 "single non-template, non-overloaded static member function, and "
1344 "must be public and not deleted. See io/traits/traits_base.h.");
1345 }
1346 else if constexpr (std::default_initializable<TCtx>)
1347 return TCtx{};
1348 else
1349 static_assert(dependent_false_v<TCtx>,
1350 "IOv2: this parse_context_type's context has no make_parse_context() "
1351 "and is not default constructible, so there is no way to obtain one. "
1352 "Give the context a default constructor, or give parse_context_type a "
1353 "static make_parse_context(const T&). See io/traits/traits_base.h.");
1354 }();
1355 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1356 tmp.convert_to(static_cast<TTarget&>(value));
1357 }
1358
1359 if (saw_eof) obj.setstate(ios_defs::eofbit);
1360 }
1361 catch(...)
1362 {
1363 obj.handle_exception(std::current_exception(), saw_eof);
1364 }
1365 }
1366 else
1367 static_assert(dependent_false_v<TValue>,
1368 "IOv2 internal: detail::extractable admitted this type but the dispatch chain has no "
1369 "branch for it. The concept and the chain must stay in step -- see "
1370 "IOv2/io/utilities/istream_operators.h.");
1371
1372 return obj;
1373}
