130 in_sentry(TStream& is, bool noskip)
131 : m_is(is)
132 {
133 // good(), not operator bool: a sticky eofbit stops the work here, as the standard's sentry
134 // does. The check at the end stays !m_is -- a skip that reaches EOF sets eofbit only.
135 if (!m_is.good())
136 throw stream_error("istream_sentry create fail: Invalid istream");
137
138 if (auto* tied = is.tie())
139 tied->try_flush();
140
141 if constexpr (involve_output)
142 is.m_channel.switch_to_get();
143
144 if (!noskip)
145 {
146 try
147 {
148 auto ct = is.m_locale.template get<IOv2::ctype<typename TStream::char_type>>();
149 if (!ct)
150 throw stream_error{"istream ignore_ws fail: no ctype facet"};
151 auto c = is.m_channel.getc();
152 while (c.has_value() &&
153 ct->is_any(base_ft<ctype>::space, c.value()))
154 {
155 c = is.m_channel.nextc();
156 }
157
158 if (!c.has_value())
159 throw eof_error{};
160 }
161 catch(...)
162 {
163 is.handle_exception(std::current_exception());
164 }
165 }
166
167 if (!m_is)
168 throw stream_error("istream_sentry create fail: Invalid istream");
169 }

743 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
744 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
745 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
746 TOut get(this TSelf& self, TOut s, std::ptrdiff_t n, TChar delim)
747 {
748 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
749
750 std::size_t gcount = 0;
751 bool at_eof = false;
752 std::lock_guard guard(self.io_mutex());
753 try
754 {
755 using sentry_type = typename TSelf::in_sentry_type;
756 sentry_type cerb(self, true);
757 if constexpr (std::is_pointer_v<TOut>)
758 {
759 if (s == nullptr)
760 throw stream_error("istream get fail: null character sequence");
761 }
762 if (n < 0)
763 throw stream_error("istream get fail: negative buffer size");
764 if (n == 0)
765 throw stream_error("istream get fail: zero buffer size");
766 const auto cap = static_cast<std::size_t>(n);
767 // Room for no character: fail before asking the device for one. getline still
768 // looks, since a delimiter there is an empty line it consumes.
769 if constexpr (std::is_same_v<DelimPolicy, keep_sep>)
770 {
771 if (cap - is_cstr == 0)
772 throw stream_error{"istream get fail: no character extracted"};
773 }
774 auto c = self.m_channel.getc();
775 while ((gcount + is_cstr < cap) &&
776 (c.has_value()) &&
777 (c.value() != delim))
778 {
779 *s++ = c.value();
780 ++gcount;
781 // Full: consume this one and stop. A peek at the next would read a byte this
782 // call does not need. getline must peek, to tell a delimiter from overflow.
783 if constexpr (std::is_same_v<DelimPolicy, keep_sep>)
784 {
785 if (gcount + is_cstr == cap)
786 {
787 self.m_channel.bumpc();
788 break;
789 }
790 }
791 c = self.m_channel.nextc();
792 }
793
794 at_eof = (gcount + is_cstr < cap) && (!c.has_value());
795
796 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
797 {
798 if (c.has_value())
799 {
800 if (c.value() == delim)
801 {
802 self.m_channel.bumpc();
803 ++gcount;
804 }
805 else
806 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
807 }
808 }
809
810 if (gcount == 0)
811 throw stream_error{"istream get fail: no character extracted"};
812
813 if (at_eof)
814 self.setstate(ios_defs::eofbit);
815 }
816 catch(...)
817 {
818 if constexpr (is_cstr)
819 {
820 if constexpr (std::is_pointer_v<TOut>)
821 {
822 if (s != nullptr && n > 0)
823 *s++ = TChar{};
824 }
825 else if (n > 0)
826 *s++ = TChar{};
827 }
828 self.handle_exception(std::current_exception(), at_eof);
829 return s;
830 }
831
832 if constexpr (is_cstr)
833 *s++ = TChar{};
834 return s;
835 }

1097 TSelf& ignore(this TSelf& self, std::ptrdiff_t n, TChar delim)
1098 {
1099 std::ptrdiff_t gcount = 0;
1100 bool at_eof = false;
1101 std::lock_guard guard(self.io_mutex());
1102 try
1103 {
1104 using sentry_type = typename TSelf::in_sentry_type;
1105 sentry_type cerb(self, true);
1106 if (n < 0)
1107 throw stream_error{"istream ignore fail: negative character count"};
1108 if (n == 0) return self;
1109
1110 auto c = self.m_channel.getc();
1111 while (gcount < n
1112 && c.has_value()
1113 && (c.value() != delim))
1114 {
1115 // The n-th one: consume it and stop, without peeking at one more.
1116 if (++gcount == n)
1117 {
1118 self.m_channel.bumpc();
1119 break;
1120 }
1121 c = self.m_channel.nextc();
1122 }
1123
1124 at_eof = (gcount < n) && (!c.has_value());
1125
1126 if (gcount < n)
1127 {
1128 if (c.has_value())
1129 {
1130 ++gcount;
1131 self.m_channel.bumpc();
1132 }
1133 }
1134
1135 if (at_eof)
1136 self.setstate(ios_defs::eofbit);
1137 }
1138 catch(...)
1139 {
1140 self.handle_exception(std::current_exception(), at_eof);
1141 }
1142

1311template <istream_type T, typename TValue>
1312 requires detail::extractable<T, TValue>
1313// The forwarding reference below is a binder plus a value-category carrier, never a forwarder;
1314// see the @note above for why `std::forward` here would be wrong rather than merely redundant.
1315// NOLINTNEXTLINE(cppcoreguidelines-missing-std-forward)
1316T& operator>>(T& obj, TValue&& value)
1317{
1318 using TChar = typename T::char_type;
1319 using TV = std::remove_cvref_t<TValue>;
1320 using TCtx = detail::in_ctx_t<TChar, TValue>;
1321 using TTarget = detail::in_target_t<TValue>;
1322
1323 constexpr bool stream_r = detail::extractable_with_stream<T, TValue>;
1324 constexpr bool iter_v = detail::extractable_with_iter<T, TValue>;
1325 constexpr bool iter_c = detail::extractable_with_ctx<T, TValue>;
1326
1327 static_assert(!(stream_r && (iter_v || iter_c)),
1328 "IOv2: this io_traits specialization provides both forms of sread() -- the stream form "
1329 "sread(stream, value) and the iterator form sread(iter, end, io, loc, value). Provide "
1330 "exactly one: the iterator form for formatted extraction, the stream form for a "
1331 "manipulator. See io/traits/traits_base.h.");
1332
1333 if constexpr (stream_r)
1334 {
1335 try
1336 {
1337 io_traits<TChar, TV>::sread(obj, static_cast<TTarget&>(value));
1338 }
1339 catch (...)
1340 {
1341 obj.handle_exception(std::current_exception());
1342 }
1343 }
1344 else if constexpr (iter_v || iter_c)
1345 {
1346 using sentry_type = typename T::in_sentry_type;
1347
1348 bool saw_eof = false;
1349 std::lock_guard guard(obj.io_mutex());
1350 try
1351 {
1352 auto iter = obj.i_iter(&saw_eof);
1353 bool skip = bool(obj.flags() & ios_defs::skipws);
1354 sentry_type cerb(obj, !skip);
1355
1356 // ws is the one extraction that may read nothing: like std::ws it reports the end
1357 // with eofbit alone, so a skip that ran off the end is not a failure for it.
1358 if constexpr (!std::is_same_v<TV, ws_t>)
1359 {
1360 if (obj.eof())
1361 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1362 }
1363
1364 if constexpr (iter_v)
1365 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(),
1366 static_cast<TTarget&>(value));
1367 else
1368 {
1369 using TPct = parse_context_type<TChar, TV>;
1370 TCtx tmp = [&value]() -> TCtx {
1371 // Detect the member by name only, so a mis-written one reaches a static_assert
1372 // instead of being read as absent and silently default constructed.
1373 if constexpr (detail::declares_maker<TPct, TTarget>)
1374 {
1375 if constexpr (requires { &TPct::make_parse_context; })
1376 {
1377 static_assert(
1378 detail::maker_signature_v<TCtx, TV,
1379 decltype(&TPct::make_parse_context)>,
1380 "IOv2: this parse_context_type's make_parse_context() must be "
1381 "declared exactly `static type make_parse_context(const T&)` -- it "
1382 "seeds the context from the target, so it takes the target by const "
1383 "reference and returns parse_context_type<TChar, T>::type by value. "
1384 "See io/traits/traits_base.h.");
1385 return TPct::make_parse_context(static_cast<const TV&>(value));
1386 }
1387 else
1388 static_assert(dependent_false_v<TCtx>,
1389 "IOv2: this parse_context_type's make_parse_context() must be a "
1390 "single non-template, non-overloaded static member function, and "
1391 "must be public and not deleted. See io/traits/traits_base.h.");
1392 }
1393 else if constexpr (std::default_initializable<TCtx>)
1394 return TCtx{};
1395 else
1396 static_assert(dependent_false_v<TCtx>,
1397 "IOv2: this parse_context_type's context has no make_parse_context() "
1398 "and is not default constructible, so there is no way to obtain one. "
1399 "Give the context a default constructor, or give parse_context_type a "
1400 "static make_parse_context(const T&). See io/traits/traits_base.h.");
1401 }();
1402 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1403 tmp.convert_to(static_cast<TTarget&>(value));
1404 }
1405
1406 if (saw_eof) obj.setstate(ios_defs::eofbit);
1407 }
1408 catch(...)
1409 {
1410 obj.handle_exception(std::current_exception(), saw_eof);
1411 }
1412 }
1413 else
1414 static_assert(dependent_false_v<TValue>,
1415 "IOv2 internal: detail::extractable admitted this type but the dispatch chain has no "
1416 "branch for it. The concept and the chain must stay in step -- see "
1417 "IOv2/io/utilities/istream_operators.h.");
1418
1419 return obj;
1420}
