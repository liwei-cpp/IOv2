712 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
713 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
714 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
715 TOut get(this TSelf& self, TOut s, std::ptrdiff_t n, TChar delim)
716 {
717 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
718
719 std::size_t gcount = 0;
720 bool at_eof = false;
721 std::lock_guard guard(self.io_mutex());
722 try
723 {
724 using sentry_type = typename TSelf::in_sentry_type;
725 sentry_type cerb(self, true);
726 if constexpr (std::is_pointer_v<TOut>)
727 {
728 if (s == nullptr)
729 throw stream_error("istream get fail: null character sequence");
730 }
731 if (n < 0)
732 throw stream_error("istream get fail: negative buffer size");
733 if (n == 0)
734 throw stream_error("istream get fail: zero buffer size");
735 const auto cap = static_cast<std::size_t>(n);
736 auto c = self.m_channel.getc();
737 while ((gcount + is_cstr < cap) &&
738 (c.has_value()) &&
739 (c.value() != delim))
740 {
741 *s++ = c.value();
742 ++gcount;
743 c = self.m_channel.nextc();
744 }
745
746 at_eof = (gcount + is_cstr < cap) && (!c.has_value());
747
748 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
749 {
750 if (c.has_value())
751 {
752 if (c.value() == delim)
753 {
754 self.m_channel.bumpc();
755 ++gcount;
756 }
757 else
758 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
759 }
760 }
761
762 if (gcount == 0)
763 throw stream_error{"istream get fail: no character extracted"};
764
765 if (at_eof)
766 self.setstate(ios_defs::eofbit);
767 }
768 catch(...)
769 {
770 if constexpr (is_cstr)
771 {
772 if constexpr (std::is_pointer_v<TOut>)
773 {
774 if (s != nullptr && n > 0)
775 *s++ = TChar{};
776 }
777 else if (n > 0)
778 *s++ = TChar{};
779 }
780 self.handle_exception(std::current_exception(), at_eof);
781 return s;
782 }
783
784 if constexpr (is_cstr)
785 *s++ = TChar{};
786 return s;
787 }

1247template <istream_type T, typename TValue>
1248 requires detail::extractable<T, TValue>
1249// The forwarding reference below is a binder plus a value-category carrier, never a forwarder;
1250// see the @note above for why `std::forward` here would be wrong rather than merely redundant.
1251// NOLINTNEXTLINE(cppcoreguidelines-missing-std-forward)
1252T& operator>>(T& obj, TValue&& value)
1253{
1254 using TChar = typename T::char_type;
1255 using TV = std::remove_cvref_t<TValue>;
1256 using TCtx = detail::in_ctx_t<TChar, TValue>;
1257 using TTarget = detail::in_target_t<TValue>;
1258
1259 constexpr bool stream_r = detail::extractable_with_stream<T, TValue>;
1260 constexpr bool iter_v = detail::extractable_with_iter<T, TValue>;
1261 constexpr bool iter_c = detail::extractable_with_ctx<T, TValue>;
1262
1263 static_assert(!(stream_r && (iter_v || iter_c)),
1264 "IOv2: this io_traits specialization provides both forms of sread() -- the stream form "
1265 "sread(stream, value) and the iterator form sread(iter, end, io, loc, value). Provide "
1266 "exactly one: the iterator form for formatted extraction, the stream form for a "
1267 "manipulator. See io/traits/traits_base.h.");
1268
1269 if constexpr (stream_r)
1270 {
1271 try
1272 {
1273 io_traits<TChar, TV>::sread(obj, static_cast<TTarget&>(value));
1274 }
1275 catch (...)
1276 {
1277 obj.handle_exception(std::current_exception());
1278 }
1279 }
1280 else if constexpr (iter_v || iter_c)
1281 {
1282 using sentry_type = typename T::in_sentry_type;
1283
1284 bool saw_eof = false;
1285 std::lock_guard guard(obj.io_mutex());
1286 try
1287 {
1288 auto iter = obj.i_iter(&saw_eof);
1289 bool skip = bool(obj.flags() & ios_defs::skipws);
1290 sentry_type cerb(obj, !skip);
1291
1292 if (obj.eof())
1293 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1294
1295 if constexpr (iter_v)
1296 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(),
1297 static_cast<TTarget&>(value));
1298 else
1299 {
1300 using TPct = parse_context_type<TChar, TV>;
1301 TCtx tmp = [&value]() -> TCtx {
1302 // Detect the member by name only, so a mis-written one reaches a static_assert
1303 // instead of being read as absent and silently default constructed.
1304 if constexpr (detail::declares_maker<TPct, TTarget>)
1305 {
1306 if constexpr (requires { &TPct::make_parse_context; })
1307 {
1308 static_assert(
1309 detail::maker_signature_v<TCtx, TV,
1310 decltype(&TPct::make_parse_context)>,
1311 "IOv2: this parse_context_type's make_parse_context() must be "
1312 "declared exactly `static type make_parse_context(const T&)` -- it "
1313 "seeds the context from the target, so it takes the target by const "
1314 "reference and returns parse_context_type<TChar, T>::type by value. "
1315 "See io/traits/traits_base.h.");
1316 return TPct::make_parse_context(static_cast<const TV&>(value));
1317 }
1318 else
1319 static_assert(dependent_false_v<TCtx>,
1320 "IOv2: this parse_context_type's make_parse_context() must be a "
1321 "single non-template, non-overloaded static member function, and "
1322 "must be public and not deleted. See io/traits/traits_base.h.");
1323 }
1324 else if constexpr (std::default_initializable<TCtx>)
1325 return TCtx{};
1326 else
1327 static_assert(dependent_false_v<TCtx>,
1328 "IOv2: this parse_context_type's context has no make_parse_context() "
1329 "and is not default constructible, so there is no way to obtain one. "
1330 "Give the context a default constructor, or give parse_context_type a "
1331 "static make_parse_context(const T&). See io/traits/traits_base.h.");
1332 }();
1333 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1334 tmp.convert_to(static_cast<TTarget&>(value));
1335 }
1336
1337 if (saw_eof) obj.setstate(ios_defs::eofbit);
1338 }
1339 catch(...)
1340 {
1341 obj.handle_exception(std::current_exception(), saw_eof);
1342 }
1343 }
1344 else
1345 static_assert(dependent_false_v<TValue>,
1346 "IOv2 internal: detail::extractable admitted this type but the dispatch chain has no "
1347 "branch for it. The concept and the chain must stay in step -- see "
1348 "IOv2/io/utilities/istream_operators.h.");
1349
1350 return obj;
1351}
