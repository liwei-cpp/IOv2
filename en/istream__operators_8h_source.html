455 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>

458 TOut get(this TSelf& self, TOut s, size_t n, TChar delim)
459 {

461
462 size_t gcount = 0;
463 bool at_eof = false;
464 std::lock_guard guard(self.io_mutex());
465 try
466 {
467 using sentry_type = typename TSelf::in_sentry_type;
468 sentry_type cerb(self, true);
469 if constexpr (std::is_pointer_v<TOut>)
470 {

472 throw stream_error("istream get fail: null character sequence");
473 }
474 if (n == 0)
475 throw stream_error("istream get fail: zero buffer size");

478 (c.has_value()) &&
479 (c.value() != delim))
480 {
481 *s++ = c.value();

483 c = self.m_streambuf.snextc();

489 {

491 {
492 if (c.value() == delim)
493 {
494 self.m_streambuf.sbumpc();
495 ++gcount;
496 }
497 else
498 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
499 }
500 }
501
502 if (gcount == 0)
503 throw stream_error{"istream get fail: no character extracted"};
504
505 if (at_eof)
506 self.setstate(ios_defs::eofbit);
507 }
508 catch(...)
509 {
510 if constexpr (is_cstr)
511 {
512 if constexpr (std::is_pointer_v<TOut>)
513 {
514 if (s != nullptr && n != 0)
515 *s++ = TChar{};
516 }
517 else if (n != 0)
518 *s++ = TChar{};
519 }
520 self.handle_exception(std::current_exception(), at_eof);
521 return s;
522 }

525 *s++ = TChar{};
