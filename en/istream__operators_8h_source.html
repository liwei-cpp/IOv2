635 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
636 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
637 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
638 TOut get(this TSelf& self, TOut s, std::ptrdiff_t n, TChar delim)
639 {
640 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
641
642 std::size_t gcount = 0;
643 bool at_eof = false;
644 std::lock_guard guard(self.io_mutex());
645 try
646 {
647 using sentry_type = typename TSelf::in_sentry_type;
648 sentry_type cerb(self, true);
649 if constexpr (std::is_pointer_v<TOut>)
650 {
651 if (s == nullptr)
652 throw stream_error("istream get fail: null character sequence");
653 }
654 if (n < 0)
655 throw stream_error("istream get fail: negative buffer size");
656 if (n == 0)
657 throw stream_error("istream get fail: zero buffer size");
658 const auto cap = static_cast<std::size_t>(n);
659 auto c = self.m_streambuf.sgetc();
660 while ((gcount + is_cstr < cap) &&
661 (c.has_value()) &&
662 (c.value() != delim))
663 {
664 *s++ = c.value();
665 ++gcount;
666 c = self.m_streambuf.snextc();
667 }
668
669 at_eof = (gcount + is_cstr < cap) && (!c.has_value());
670
671 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
672 {
673 if (c.has_value())
674 {
675 if (c.value() == delim)
676 {
677 self.m_streambuf.sbumpc();
678 ++gcount;
679 }
680 else
681 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
682 }
683 }
684
685 if (gcount == 0)
686 throw stream_error{"istream get fail: no character extracted"};
687
688 if (at_eof)
689 self.setstate(ios_defs::eofbit);
690 }
691 catch(...)
692 {
693 if constexpr (is_cstr)
694 {
695 if constexpr (std::is_pointer_v<TOut>)
696 {
697 if (s != nullptr && n > 0)
698 *s++ = TChar{};
699 }
700 else if (n > 0)
701 *s++ = TChar{};
702 }
703 self.handle_exception(std::current_exception(), at_eof);
704 return s;
705 }
706
707 if constexpr (is_cstr)
708 *s++ = TChar{};
709 return s;
710 }

1177template <istream_type T, typename TValue>
1178 requires detail::extractable<T, TValue>
1179// The forwarding reference below is a binder plus a value-category carrier, never a forwarder;
1180// see the @note above for why `std::forward` here would be wrong rather than merely redundant.
1181// NOLINTNEXTLINE(cppcoreguidelines-missing-std-forward)
1182T& operator>>(T& obj, TValue&& value)
1183{
1184 using TChar = typename T::char_type;
1185 using TV = std::remove_cvref_t<TValue>;
1186 using TCtx = detail::in_ctx_t<TChar, TValue>;
1187 using TTarget = detail::in_target_t<TValue>;
1188
1189 constexpr bool stream_r = detail::extractable_with_stream<T, TValue>;
1190 constexpr bool iter_v = detail::extractable_with_iter<T, TValue>;
1191 constexpr bool iter_c = detail::extractable_with_ctx<T, TValue>;
1192
1193 static_assert(!(stream_r && (iter_v || iter_c)),
1194 "IOv2: this io_traits specialization provides both forms of sread() -- the stream form "
1195 "sread(stream, value) and the iterator form sread(iter, end, io, loc, value). Provide "
1196 "exactly one: the iterator form for formatted extraction, the stream form for a "
1197 "manipulator. See io/traits/traits_base.h.");
1198
1199 if constexpr (stream_r)
1200 {
1201 try
1202 {
1203 io_traits<TChar, TV>::sread(obj, static_cast<TTarget&>(value));
1204 }
1205 catch (...)
1206 {
1207 obj.handle_exception(std::current_exception());
1208 }
1209 }
1210 else if constexpr (iter_v || iter_c)
1211 {
1212 using sentry_type = typename T::in_sentry_type;
1213
1214 bool saw_eof = false;
1215 std::lock_guard guard(obj.io_mutex());
1216 try
1217 {
1218 auto iter = obj.i_iter(&saw_eof);
1219 bool skip = bool(obj.flags() & ios_defs::skipws);
1220 sentry_type cerb(obj, !skip);
1221
1222 if (obj.eof())
1223 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1224
1225 if constexpr (iter_v)
1226 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(),
1227 static_cast<TTarget&>(value));
1228 else
1229 {
1230 TCtx tmp = [&value]() -> TCtx {
1231 if constexpr (requires (const TV& v)
1232 { parse_context_type<TChar, TV>::make_parse_context(v); })
1233 return parse_context_type<TChar, TV>::make_parse_context(value);
1234 else if constexpr (std::default_initializable<TCtx>)
1235 return TCtx{};
1236 else
1237 static_assert(dependent_false_v<TCtx>,
1238 "IOv2: this parse_context_type's context has no make_parse_context() "
1239 "and is not default constructible, so there is no way to obtain one. "
1240 "Give the context a default constructor, or give parse_context_type a "
1241 "static make_parse_context(const T&). See io/traits/traits_base.h.");
1242 }();
1243 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1244 tmp.convert_to(static_cast<TTarget&>(value));
1245 }
1246
1247 if (saw_eof) obj.setstate(ios_defs::eofbit);
1248 }
1249 catch(...)
1250 {
1251 obj.handle_exception(std::current_exception(), saw_eof);
1252 }
1253 }
1254 else
1255 static_assert(dependent_false_v<TValue>,
1256 "IOv2 internal: detail::extractable admitted this type but the dispatch chain has no "
1257 "branch for it. The concept and the chain must stay in step -- see "
1258 "IOv2/io/utilities/istream_operators.h.");
1259
1260 return obj;
1261}
