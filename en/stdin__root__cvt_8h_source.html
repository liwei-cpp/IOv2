199 template <bool Saturate = false>
200 auto get_buf(std::size_t to_max)
201 {
202 if (to_max == 0)
203 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
204 if (to_max > s_buffer_length)
205 throw cvt_error("cvt_reader::get_buf fail, read size too large");
206
207 if (m_cur == m_end) restore_size();
208 char* const buf = m_buffer.data();
209 const std::size_t remain = m_end - m_cur;
210 if (remain < to_max)
211 {
212 std::copy(buf + m_cur, buf + m_end, buf);
213 m_cur = 0;
214 m_end = remain;
215 const std::size_t room = s_buffer_length - remain;
216 const std::size_t want = synced() ? std::min(to_max - remain, room) : room;
217 m_end += m_device.dget(buf + m_end, want);
218 }
219
220 if constexpr (Saturate)
221 {
222 while (m_end < m_cur + to_max)
223 {
224 const std::size_t got = m_device.dget(buf + m_end, m_cur + to_max - m_end);
225 if (got == 0)
226 throw cvt_error("get_buf<Saturate> fail: meet eos");
227 m_end += got;
228 }
229 const char* res = buf + m_cur;
230 m_cur += to_max;
231 return res;
232 }
233 else
234 {
235 const std::size_t len = std::min(m_end - m_cur, to_max);
236 std::pair<const char*, std::size_t> res{buf + m_cur, len};
237 m_cur += len;
238 return res;
239 }
240 }

259 void rollback(std::size_t len)
260 {
261 if (len == 0)
262 throw cvt_error("cvt_reader::rollback fail, length cannot be zero");
263 if (len > m_cur)
264 throw cvt_error("cvt_reader::rollback fail, rollback length too large");
265
266 m_cur -= len;
267 if (m_device.dunget(m_buffer.data() + m_cur, m_end - m_cur))
268 m_cur = m_end = 0;
269 }

305 template <std::bidirectional_iterator It>
306 requires std::same_as<std::iter_value_t<It>, char>
307 void unget(It first, It last)
308 {
309 if (first == last) return;
310 if (m_cur == m_end && m_device.dunget(first, last)) return;
311
312 const auto n = static_cast<std::size_t>(std::distance(first, last));
313 if (n > m_cur) make_room(n);
314 m_cur -= n;
315 std::copy(first, last, m_buffer.data() + m_cur);
316 }
317
318private:
319 // Moves what is left to the very end, so that at least n bytes fit in front of it and
320 // later calls find room too. A larger buffer is built before anything changes, so a
321 // bad_alloc leaves the state as it was.
322 void make_room(std::size_t n)
323 {
324 const std::size_t remain = m_end - m_cur;
325 char* const buf = m_buffer.data();
326 if (n + remain > m_buffer.size())
327 {
328 std::vector<char> larger(std::max(n + remain, m_buffer.size() * 2));
329 std::copy(buf + m_cur, buf + m_end, larger.data() + larger.size() - remain);
330 m_buffer.swap(larger);
331 }
332 else
333 std::copy_backward(buf + m_cur, buf + m_end, buf + m_buffer.size());
334 m_end = m_buffer.size();
335 m_cur = m_end - remain;
336 }
337
338 // Gives back what unget() grew the buffer by; only for a buffer with nothing left in it.
339 void restore_size()
340 {
341 if (m_buffer.size() == s_buffer_length) return;
342 m_cur = m_end = 0;
343 m_buffer.resize(s_buffer_length);
344 m_buffer.shrink_to_fit();
345 }
346
347private:
348 std_input_device m_device;
349 std::vector<char> m_buffer;
350 std::size_t m_cur = 0;
351 std::size_t m_end = 0;
352 std::atomic<bool> m_synced;
353};
