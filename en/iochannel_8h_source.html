392 std::size_t getn(char_type* s, std::size_t n, std::size_t* got = nullptr) requires (IsIn)
393 {
394 if (got) *got = 0;
395 if ((s == nullptr) || (n == 0)) return 0;
396
397 if constexpr (IsOut)
398 switch_to_get();
399
400 std::size_t res = 0;
401 while (!m_read_buf.empty())
402 {
403 *s++ = m_read_buf.back();
404 m_read_buf.pop_back();
405 if (++res == n) break;
406 }
407 if (got) *got = res;
408
409 while (res < n)
410 {
411 std::size_t c = m_cvt.get(s, n - res);
412 assert(c <= n - res);
413 if (c == 0) break;
414 res += c;
415 s += c;
416 if (got) *got = res;
417 }
418 return res;
419 }

777 void switch_to_put() requires (IsIn && IsOut)
778 {
779 if (!m_read_buf.empty())
780 {
781 try
782 {
783 const std::size_t pos = tell();
784 m_cvt.seek(pos);
785 }
786 catch (const cvt_error& e)
787 {
788 throw cvt_error("base_channel::switch_to_put fails: cannot reposition "
789 + std::to_string(m_read_buf.size())
790 + " buffered/put-back character(s) before switching to output mode: "
791 + e.what());
792 }
793 // Must immediately follow a *successful* seek: the seek consumed the buffer, and
794 // tell() subtracts m_read_buf.size(), so keeping it would subtract twice. A failed
795 // seek skips this and leaves the buffer intact; a throw from the switch below
796 // therefore finds it already cleared. Both are by design -- see iostream's @warning.
797 m_read_buf.clear();
798 }
799 m_cvt.switch_to_put();
800 }

869 std::pair<device_type, std::exception_ptr> detach() noexcept
870 {
871 if constexpr (IsIn)
872 {
873 if (!m_read_buf.empty())
874 {
875 try
876 {
877 // On over-putback (more put-back than actually read), tell()
878 // saturates at 0, so the device is handed back positioned at the
879 // stream origin. That is consistent with the logical read cursor
880 // being at 0; see tell()/putbackc() and switch_to_put().
881 const std::size_t pos = tell();
882 m_cvt.seek(pos);
883 }
884 catch (...) // NOLINT(bugprone-empty-catch)
885 {
886 // Swallowed on purpose: a device that doesn't support positioning
887 // (e.g. a pipe/tty-backed stdin) can never honor this reposition,
888 // and that is an inherent property of the device, not an
889 // actionable failure - the caller could not have repositioned it
890 // either after getting it back. Reporting this error would make
891 // routine detach()/attach() cycles on such devices start throwing
892 // (e.g. reset() right after a formatted read has left
893 // one buffered lookahead character), even though nothing is
894 // actually broken; losing that lookahead (and, below, the root
895 // converter's read buffer) is the
896 // accepted, unavoidable cost for non-positionable devices.
897 }
898 m_read_buf.clear();
899 }
900 }
901 return m_cvt.detach();
902 }
