126 static type make_parse_context(const std::tm& tmb)
127 {
128 using namespace std::chrono;
129
130 std::int64_t total_mon = (static_cast<std::int64_t>(tmb.tm_year) + 1900) * 12 + tmb.tm_mon;
131 std::int64_t norm_year = total_mon / 12;
132 std::int64_t norm_mon = total_mon % 12;
133 if (norm_mon < 0)
134 {
135 norm_mon += 12;
136 --norm_year;
137 }
138
139 norm_year = std::clamp<std::int64_t>(norm_year,
140 static_cast<int>(year::min()),
141 static_cast<int>(year::max()));
142
143 constexpr sys_days cal_min{year::min() / January / 1};
144 constexpr sys_days cal_max{year::max() / December / 31};
145 constexpr std::int64_t secs_per_day = 24 * 60 * 60;
146
147 // The time group is folded into one second-of-day count and normalized with the same
148 // carry the date group uses, so an out-of-range hour/minute/second moves the date
149 // instead of silently wrapping within it. A leap second is the one exception: it
150 // becomes 59 because hh_mm_ss cannot represent it.
151 std::int64_t sec = (tmb.tm_sec == 60) ? 59 : tmb.tm_sec;
152 std::int64_t tod = static_cast<std::int64_t>(tmb.tm_hour) * 3600
153 + static_cast<std::int64_t>(tmb.tm_min) * 60
154 + sec;
155 std::int64_t day_carry = (tod >= 0) ? tod / secs_per_day
156 : -((secs_per_day - 1 - tod) / secs_per_day);
157
158 auto first = sys_days{year{static_cast<int>(norm_year)}
159 / month{static_cast<unsigned>(norm_mon) + 1} / day{1}};
160 auto offset = std::clamp<std::int64_t>(
161 static_cast<std::int64_t>(tmb.tm_mday) - 1 + day_carry, -4'000'000, 4'000'000);
162
163 type ctx;
164 ctx.set_hint(
165 year_month_day{std::clamp(first + days{static_cast<int>(offset)}, cal_min, cal_max)});
166 ctx.set_hint(hh_mm_ss<seconds>{seconds{tod - day_carry * secs_per_day}});
167 return ctx;
168 }
