54// The name of loc's LC_CTYPE category. glibc reports "POSIX" as "C"; macOS keeps it.
55inline const char* clocale_ctype_name(clocale_handle loc) noexcept
56{
57#if defined(__GLIBC__)

59#else
60 return querylocale(LC_CTYPE_MASK, loc);
