206 if ((s & ios_defs::cvtfailbit) == ios_defs::goodbit) m_exp_cvt_fail = std::exception_ptr{};
207 if ((s & ios_defs::strfailbit) == ios_defs::goodbit) m_exp_str_fail = std::exception_ptr{};

444 template <bool ignore_exception_mask = false>
445 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
446 {
447 if (!ex) return;
448 std::lock_guard guard(m_state_mutex);
449 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
450 try
451 {
452 std::rethrow_exception(ex);
453 }
454 catch (device_error&)
455 {
456 if (!m_exp_dev_fail)
457 m_exp_dev_fail = std::current_exception();
458 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
459 }
460 catch (cvt_error&)
461 {
462 if (!m_exp_cvt_fail)
463 m_exp_cvt_fail = std::current_exception();
464 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
465 }
466 catch (stream_error&)
467 {
468 if (!m_exp_str_fail)
469 m_exp_str_fail = std::current_exception();
470 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
471 }
472 catch (eof_error&)
473 {
474 setstate<ignore_exception_mask>(ios_defs::eofbit);
475 }
476 catch(...)
477 {
478 if (!m_exp_other_fail)
479 m_exp_other_fail = std::current_exception();
480 setstate<ignore_exception_mask>(ios_defs::otherfailbit | eof);
481 }
