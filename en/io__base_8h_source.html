831 void access_callbacks(const locale<TChar>& new_loc)
832 {
833 std::lock_guard guard(io_mutex());
834 std::exception_ptr throw_exception = nullptr;
835
836 for (const auto& [cb, id] : m_callbacks)
837 {
838 try
839 {
840 auto pre = m_pwords.find(id);
841 std::shared_ptr<void> old_data =
842 (pre != m_pwords.end()) ? pre->second : nullptr;
843
844 std::shared_ptr<void> new_data = cb(new_loc, old_data);
845
846 // Re-locate after the callback: cb may reentrantly mutate
847 // m_pwords (via set_pword), which can rehash/erase and would
848 // invalidate an iterator held across the call.
849 auto it = m_pwords.find(id);
850 if (new_data)
851 {
852 if (it != m_pwords.end()) it->second = std::move(new_data);
853 else m_pwords.insert({id, std::move(new_data)});
854 }
855 else if (it != m_pwords.end())
856 {
857 m_pwords.erase(it);
858 }
859 }
860 catch(...)
861 {
862 if (!throw_exception)
863 throw_exception = std::current_exception();
864 }
865 }
866
867 if (throw_exception)
868 std::rethrow_exception(throw_exception);
869 }

1049 void clear(ios_defs::iostate s = ios_defs::goodbit) noexcept(ignore_exception_mask)
1050 {
1051 std::lock_guard guard(this->io_mutex());
1052 m_stream_state.store(s);
1053 if ((s & ios_defs::devfailbit) == ios_defs::goodbit) m_exp_dev_fail = std::exception_ptr{};
1054 if ((s & ios_defs::cvtfailbit) == ios_defs::goodbit) m_exp_cvt_fail = std::exception_ptr{};
1055 if ((s & ios_defs::strfailbit) == ios_defs::goodbit) m_exp_str_fail = std::exception_ptr{};
1056 if ((s & ios_defs::otherfailbit) == ios_defs::goodbit) m_exp_other_fail = std::exception_ptr{};

1059 {
1060 ios_defs::iostate state_in_exp = m_exception & s;
1061 if (state_in_exp & ios_defs::devfailbit)
1062 {
1063 if (m_exp_dev_fail)
1064 std::rethrow_exception(std::exchange(m_exp_dev_fail, nullptr));
1065 else
1066 throw device_error("device failure bit has been set");
1067 }
1068 else if (state_in_exp & ios_defs::cvtfailbit)
1069 {
1070 if (m_exp_cvt_fail)
1071 std::rethrow_exception(std::exchange(m_exp_cvt_fail, nullptr));
1072 else
1073 throw cvt_error("converter failure bit has been set");
1074 }
1075 else if (state_in_exp & ios_defs::strfailbit)
1076 {
1077 if (m_exp_str_fail)
1078 std::rethrow_exception(std::exchange(m_exp_str_fail, nullptr));
1079 else

1332 template <bool ignore_exception_mask = false>
1333 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
1334 noexcept(ignore_exception_mask)
1335 {
1336 if (!ex) return;
1337 if constexpr (!ignore_exception_mask)
1338 {
1339 if (std::uncaught_exceptions() != 0)
1340 {
1341 handle_exception<true>(ex, at_eof);
1342 return;
1343 }
1344 }
1345 std::lock_guard guard(this->io_mutex());
1346 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
1347 try
1348 {
1349 std::rethrow_exception(ex);
1350 }
1351 catch (device_error&)
1352 {
1353 if (!m_exp_dev_fail)
1354 m_exp_dev_fail = std::current_exception();
1355 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
1356 }
1357 catch (cvt_error&)
1358 {
1359 if (!m_exp_cvt_fail)
1360 m_exp_cvt_fail = std::current_exception();
1361 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
1362 }
1363 catch (stream_error&)
1364 {
1365 if (!m_exp_str_fail)
1366 m_exp_str_fail = std::current_exception();
1367 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
1368 }
1369 catch (eof_error&)
1370 {
1371 setstate<ignore_exception_mask>(ios_defs::eofbit);
1372 }
1373 catch(...)
1374 {
1375 if (!m_exp_other_fail)
1376 m_exp_other_fail = std::current_exception();
1377 setstate<ignore_exception_mask>(ios_defs::otherfailbit | eof);
1378 }
1379 }
