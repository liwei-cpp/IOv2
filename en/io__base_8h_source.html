623 void access_callbacks(const locale<TChar>& new_loc)
624 {
625 std::lock_guard guard(m_io_mutex);
626 std::exception_ptr throw_exception = nullptr;
627
628 for (const auto& [cb, id] : m_callbacks)
629 {
630 try
631 {
632 auto pre = m_pwords.find(id);
633 std::shared_ptr<void> old_data =
634 (pre != m_pwords.end()) ? pre->second : nullptr;
635
636 std::shared_ptr<void> new_data = cb(new_loc, old_data);
637
638 // Re-locate after the callback: cb may reentrantly mutate
639 // m_pwords (via set_pword), which can rehash/erase and would
640 // invalidate an iterator held across the call.
641 auto it = m_pwords.find(id);
642 if (new_data)
643 {
644 if (it != m_pwords.end()) it->second = std::move(new_data);
645 else m_pwords.insert({id, std::move(new_data)});
646 }
647 else if (it != m_pwords.end())
648 {
649 m_pwords.erase(it);
650 }
651 }
652 catch(...)
653 {
654 if (!throw_exception)
655 throw_exception = std::current_exception();
656 }
657 }
658
659 if (throw_exception)
660 std::rethrow_exception(throw_exception);
661 }

1073 template <bool ignore_exception_mask = false>
1074 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
1075 {
1076 if (!ex) return;
1077 std::lock_guard guard(this->io_mutex());
1078 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
1079 try
1080 {
1081 std::rethrow_exception(ex);
1082 }
1083 catch (device_error&)
1084 {
1085 if (!m_exp_dev_fail)
1086 m_exp_dev_fail = std::current_exception();
1087 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
1088 }
1089 catch (cvt_error&)
1090 {
1091 if (!m_exp_cvt_fail)
1092 m_exp_cvt_fail = std::current_exception();
1093 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
1094 }
1095 catch (stream_error&)
1096 {
1097 if (!m_exp_str_fail)
1098 m_exp_str_fail = std::current_exception();
1099 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
1100 }
1101 catch (eof_error&)
1102 {
1103 setstate<ignore_exception_mask>(ios_defs::eofbit);
1104 }
1105 catch(...)
