625 void access_callbacks(const locale<TChar>& new_loc)
626 {
627 std::lock_guard guard(m_io_mutex);
628 std::exception_ptr throw_exception = nullptr;
629
630 for (const auto& [cb, id] : m_callbacks)
631 {
632 try
633 {
634 auto pre = m_pwords.find(id);
635 std::shared_ptr<void> old_data =
636 (pre != m_pwords.end()) ? pre->second : nullptr;
637
638 std::shared_ptr<void> new_data = cb(new_loc, old_data);
639
640 // Re-locate after the callback: cb may reentrantly mutate
641 // m_pwords (via set_pword), which can rehash/erase and would
642 // invalidate an iterator held across the call.
643 auto it = m_pwords.find(id);
644 if (new_data)
645 {
646 if (it != m_pwords.end()) it->second = std::move(new_data);
647 else m_pwords.insert({id, std::move(new_data)});
648 }
649 else if (it != m_pwords.end())
650 {
651 m_pwords.erase(it);
652 }
653 }
654 catch(...)
655 {
656 if (!throw_exception)
657 throw_exception = std::current_exception();
658 }
659 }
660
661 if (throw_exception)
662 std::rethrow_exception(throw_exception);
663 }

1097 template <bool ignore_exception_mask = false>
1098 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
1099 {
1100 if (!ex) return;
1101 if constexpr (!ignore_exception_mask)
1102 {
1103 if (std::uncaught_exceptions() != 0)
1104 {
1105 try { handle_exception<true>(ex, at_eof); }
1106 catch (...) {} // NOLINT(bugprone-empty-catch)
1107 return;
1108 }
1109 }
1110 std::lock_guard guard(this->io_mutex());
1111 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
1112 try
1113 {
1114 std::rethrow_exception(ex);
1115 }
1116 catch (device_error&)
1117 {
1118 if (!m_exp_dev_fail)
1119 m_exp_dev_fail = std::current_exception();
1120 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
1121 }
1122 catch (cvt_error&)
1123 {
1124 if (!m_exp_cvt_fail)
1125 m_exp_cvt_fail = std::current_exception();
1126 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
1127 }
1128 catch (stream_error&)
1129 {
1130 if (!m_exp_str_fail)
1131 m_exp_str_fail = std::current_exception();
1132 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
1133 }
1134 catch (eof_error&)
1135 {
1136 setstate<ignore_exception_mask>(ios_defs::eofbit);
1137 }
1138 catch(...)
1139 {
1140 if (!m_exp_other_fail)
1141 m_exp_other_fail = std::current_exception();
1142 setstate<ignore_exception_mask>(ios_defs::otherfailbit | eof);
1143 }
1144 }
