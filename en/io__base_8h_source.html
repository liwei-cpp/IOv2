685 // Re-locate after the callback: cb may reentrantly mutate
686 // m_pwords (via set_pword), which can rehash/erase and would
687 // invalidate an iterator held across the call.
688 auto it = m_pwords.find(id);

690 {

692 else m_pwords.insert({id, std::move(new_data)});
693 }
694 else if (it != m_pwords.end())
695 {
696 m_pwords.erase(it);
697 }
698 }

703 }
704 }
705

708 }

1143 template <bool ignore_exception_mask = false>
1144 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
1145 {
1146 if (!ex) return;
1147 if constexpr (!ignore_exception_mask)
1148 {
1149 if (std::uncaught_exceptions() != 0)
1150 {
1151 try { handle_exception<true>(ex, at_eof); }
1152 catch (...) {} // NOLINT(bugprone-empty-catch)
1153 return;
1154 }
1155 }
1156 std::lock_guard guard(this->io_mutex());
1157 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
1158 try
1159 {
1160 std::rethrow_exception(ex);
1161 }
1162 catch (device_error&)
1163 {
1164 if (!m_exp_dev_fail)
1165 m_exp_dev_fail = std::current_exception();
1166 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
1167 }
1168 catch (cvt_error&)
1169 {
1170 if (!m_exp_cvt_fail)
1171 m_exp_cvt_fail = std::current_exception();
1172 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
1173 }
1174 catch (stream_error&)
1175 {
1176 if (!m_exp_str_fail)
1177 m_exp_str_fail = std::current_exception();
1178 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
1179 }
1180 catch (eof_error&)
1181 {
1182 setstate<ignore_exception_mask>(ios_defs::eofbit);
1183 }
1184 catch(...)
1185 {
1186 if (!m_exp_other_fail)
1187 m_exp_other_fail = std::current_exception();
1188 setstate<ignore_exception_mask>(ios_defs::otherfailbit | eof);
1189 }
1190 }
