429 template <bool ignore_exception_mask = false>
430 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
431 {
432 if (!ex) return;
433 std::lock_guard guard(m_state_mutex);
434 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
435 try
436 {
437 std::rethrow_exception(ex);
438 }
439 catch (device_error&)
440 {
441 if (!m_exp_dev_fail)
442 m_exp_dev_fail = std::current_exception();
443 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
444 }
445 catch (cvt_error&)

447 if (!m_exp_cvt_fail)
448 m_exp_cvt_fail = std::current_exception();
449 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
450 }
451 catch (stream_error&)
452 {
453 if (!m_exp_str_fail)
454 m_exp_str_fail = std::current_exception();
455 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
456 }
457 catch (eof_error&)

459 setstate<ignore_exception_mask>(ios_defs::eofbit);

980 void access_callbacks(const locale<TChar>& new_loc)
981 {
982 std::lock_guard guard(m_io_mutex);
983 std::exception_ptr throw_exception = nullptr;
984
985 for (const auto& [cb, id] : m_callbacks)
986 {
987 try
988 {
989 auto pre = m_pwords.find(id);
990 std::shared_ptr<void> old_data =
991 (pre != m_pwords.end()) ? pre->second : nullptr;
992
993 std::shared_ptr<void> new_data = cb(new_loc, old_data);
994
995 // Re-locate after the callback: cb may reentrantly mutate
996 // m_pwords (via set_pword), which can rehash/erase and would
997 // invalidate an iterator held across the call.
998 auto it = m_pwords.find(id);
999 if (new_data)
1000 {
1001 if (it != m_pwords.end()) it->second = std::move(new_data);
1002 else m_pwords.insert({id, std::move(new_data)});
1003 }
1004 else if (it != m_pwords.end())
1005 {
1006 m_pwords.erase(it);
1007 }
1008 }
