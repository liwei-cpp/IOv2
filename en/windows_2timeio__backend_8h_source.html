143inline std::optional<std::wstring> timeio_from_picture(std::wstring_view pic)
144{
145 std::wstring res;
146 const auto literal = [&res](wchar_t c) {
147 if (c == L'%') res += L'%';
148 res += c;
149 };
150
151 for (std::size_t i = 0; i < pic.size();)
152 {
153 const wchar_t c = pic[i];
154 if (c == L'\'')
155 {
156 // A quoted run up to the next lone quote; '' is a quote, inside a run or out of one.
157 ++i;
158 if (i < pic.size() && pic[i] == L'\'') { res += L'\''; ++i; continue; }
159 for (; i < pic.size(); ++i)
160 {
161 if (pic[i] != L'\'') { literal(pic[i]); continue; }
162 if (i + 1 < pic.size() && pic[i + 1] == L'\'') { res += L'\''; ++i; continue; }
163 break;
164 }
165 ++i;
166 continue;
167 }
168
169 const bool letter = (c >= L'a' && c <= L'z') || (c >= L'A' && c <= L'Z');
170 if (!letter) { literal(c); ++i; continue; }
171
172 std::size_t n = 1;
173 while (i + n < pic.size() && pic[i + n] == c) ++n;
174
175 std::wstring_view spec;
176 switch (c)
177 {
178 case L'd': spec = n <= 2 ? L"%d" : n == 3 ? L"%a" : n == 4 ? L"%A" : L""; break;
179 case L'M': spec = n <= 2 ? L"%m" : n == 3 ? L"%b" : n == 4 ? L"%B" : L""; break;
180 case L'y': spec = n == 2 ? L"%y" : n == 4 ? L"%Y" : L""; break;
181 case L'h': spec = n <= 2 ? L"%I" : L""; break;
182 case L'H': spec = n <= 2 ? L"%H" : L""; break;
183 case L'm': spec = n <= 2 ? L"%M" : L""; break;
184 case L's': spec = n <= 2 ? L"%S" : L""; break;
185 case L't': spec = n == 2 ? L"%p" : L""; break;
186 default: break;
187 }
188 if (spec.empty())
189 return std::nullopt;
190 res += spec;
191 i += n;
192 }
193 return res;
194}

196// Whether a strftime format holds the specifier spec (a %% is a literal, not a specifier).
197inline bool timeio_has_specifier(std::wstring_view fmt, wchar_t spec)
198{
199 for (std::size_t i = 0; i + 1 < fmt.size(); ++i)
200 {
201 if (fmt[i] != L'%') continue;
202 if (fmt[i + 1] == spec) return true;
203 ++i;
204 }
205 return false;
206}

208// One GetLocaleInfoEx item of the locale named bcp47 ("" being the invariant locale); empty if
209// NLS has none.
210inline std::wstring timeio_nls(const std::wstring& bcp47, unsigned long item)
211{
212 const int len = GetLocaleInfoEx(bcp47.c_str(), item, nullptr, 0);
213 if (len <= 1)
214 return {};
215 std::wstring res(static_cast<std::size_t>(len), L'\0');
216 if (GetLocaleInfoEx(bcp47.c_str(), item, res.data(), len) != len)
217 return {};
218 res.resize(static_cast<std::size_t>(len) - 1); // the terminating NUL
219 return res;
220}

222// src in the code page cp, as the UCRT converts its own date and time strings (no flags: a
223// character the code page lacks becomes its best fit or the default character).
224inline std::string timeio_narrow(const std::wstring& src, unsigned int cp)
225{
226 if (src.empty())
227 return {};
228 const int src_len = static_cast<int>(src.size());
229 const int len = WideCharToMultiByte(cp, 0, src.data(), src_len, nullptr, 0, nullptr, nullptr);
230 if (len <= 0)
231 return {};
232 std::string res(static_cast<std::size_t>(len), '\0');
233 if (WideCharToMultiByte(cp, 0, src.data(), src_len, res.data(), len, nullptr, nullptr) != len)
234 return {};
235 return res;
236}

250// The date and time data of loc's LC_TIME, through NLS by its BCP-47 name.
251inline timeio_raw_data timeio_read_data(clocale_handle loc)
252{
253 const std::wstring& bcp47 = loc->time_bcp47;
254 timeio_raw_wide w;
255 for (std::size_t i = 0; i < w.day.size(); ++i)
256 {
257 // NLS counts weekdays from Monday, timeio from Sunday.
258 const auto nls_index = static_cast<unsigned long>((i + 6) % 7);
259 w.day[i] = timeio_nls(bcp47, win_locale_sdayname1 + nls_index);
260 w.abbr_day[i] = timeio_nls(bcp47, win_locale_sabbrevdayname1 + nls_index);
261 }
262 for (std::size_t i = 0; i < w.month.size(); ++i)
263 {
264 w.month[i] = timeio_nls(bcp47, win_locale_smonthname1 + static_cast<unsigned long>(i));
265 w.abbr_month[i] = timeio_nls(bcp47, win_locale_sabbrevmonthname1 + static_cast<unsigned long>(i));
266 }
267 w.am = timeio_nls(bcp47, win_locale_sam);
268 w.pm = timeio_nls(bcp47, win_locale_spm);
269
270 w.date_format = timeio_from_picture(timeio_nls(bcp47, win_locale_sshortdate))
271 .value_or(std::wstring(timeio_c_date_picture));
272 w.time_format = timeio_from_picture(timeio_nls(bcp47, win_locale_stimeformat))
273 .value_or(std::wstring(timeio_c_time_picture));
274 w.date_time_format = w.date_format + L' ' + w.time_format;
275 if (timeio_has_specifier(w.time_format, L'I') && timeio_has_specifier(w.time_format, L'p'))
276 w.am_pm_format = w.time_format;
277
278 timeio_raw_data res;
279 const auto narrow = [cp = loc->time_cp](const std::wstring& src) { return timeio_narrow(src, cp); };
280 for (std::size_t i = 0; i < w.day.size(); ++i)
281 {
282 res.day[i] = narrow(w.day[i]);
283 res.abbr_day[i] = narrow(w.abbr_day[i]);
284 }
285 for (std::size_t i = 0; i < w.month.size(); ++i)
286 {
287 res.month[i] = narrow(w.month[i]);
288 res.abbr_month[i] = narrow(w.abbr_month[i]);
289 }
290 res.am = narrow(w.am);
291 res.pm = narrow(w.pm);
292 res.date_format = narrow(w.date_format);
293 res.time_format = narrow(w.time_format);
294 res.date_time_format = narrow(w.date_time_format);
295 res.am_pm_format = narrow(w.am_pm_format);
296 res.wide = std::move(w);
297 return res;
298}
