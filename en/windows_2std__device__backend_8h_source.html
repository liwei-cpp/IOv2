45// Whether a read would return at once. A disk file never blocks; a pipe does
46// unless it holds data or its write end is gone (the read then returns 0).
47// A console cannot be asked without consuming input events, so the answer
48// there is "no", which costs only a short read.
49inline bool std_fd_readable_now(int fd) noexcept
50{
51 // -1 is INVALID_HANDLE_VALUE; -2 means fd 0 has no stream (a GUI process).
52 const std::intptr_t raw = _get_osfhandle(fd);
53 if (raw == -1 || raw == -2)
54 return false;
55 void* const h = reinterpret_cast<void*>(raw);
56
57 switch (GetFileType(h))
58 {
59 case win_file_type_disk:
60 return true;
61 case win_file_type_pipe:
62 {
63 unsigned long avail = 0;
64 if (PeekNamedPipe(h, nullptr, 0, nullptr, &avail, nullptr))
65 return avail != 0;
66 return GetLastError() == win_error_broken_pipe;
67 }
68 default:
69 return false;
70 }
71}

73inline std::size_t std_fd_read(int fd, char* s, std::size_t n)
74{
75 assert((s != nullptr) && (n != 0));
76
77 constexpr auto max_read = static_cast<std::size_t>(INT_MAX);
78 const int ret = _read(fd, s, static_cast<unsigned int>((std::min)(n, max_read)));
79 if (ret == -1)
80 throw device_error("std_device::do_get fail: read error");
81 return static_cast<std::size_t>(ret);
82}
