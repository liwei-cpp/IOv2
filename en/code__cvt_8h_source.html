162 explicit codecvt_kernel(const std::string& name)
163 : m_inter_locale(name)
164 {
165 clocale_user guard(m_inter_locale);
166
167 // there are no known constant-length encodings
168 // m_epc == 1 means fixed length
169 // Defensive check: the C standard requires MB_CUR_MAX to be a positive
170 // integer, but a misconfigured CRT / exotic libc could report 0, which
171 // would lead to division-by-zero UB in seek / rseek. Reject it here so
172 // m_epc >= 1 holds as a class invariant for any constructed instance.
173 const auto cur_max = MB_CUR_MAX;
174 if (cur_max == 0) [[unlikely]]
175 throw cvt_error("codecvt_kernel: locale reports MB_CUR_MAX == 0");
176 m_epc = static_cast<unsigned>(cur_max);
177
178 // Note: mbtowc uses internal static state, so concurrent construction of
179 // codecvt_kernel instances is NOT thread-safe. However, this is acceptable
180 // because this class does not support multi-threaded construction.
181 // Once constructed, instance-level operations use explicit mbstate_t (m_state)
182 // and are safe for single-instance usage.
183 // NOLINTNEXTLINE(cppcoreguidelines-prefer-member-initializer)
184 m_is_state_dep = (std::mbtowc(nullptr, nullptr, MB_CUR_MAX) != 0);
185 init_state();
186 }

437 std::pair<bool, std::size_t> in_helper(const char*& from, const char* from_end,
438 TInt*& to, TInt* to_end)
439 {
440 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
441 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
442 clocale_user guard(m_inter_locale);
443 // No wchar_t value glibc decodes to: it tells a shift-only run from a character.
444 constexpr auto no_char = static_cast<wchar_t>(-1);
445 std::array<char, MB_LEN_MAX> joined{};
446 std::size_t i_count = 0;
447
448 const std::size_t to_max = to_end - to;
449 while (i_count < to_max && (from < from_end || m_pending_len != 0))
450 {
451 // Held bytes go first, then as many new ones as fit behind them.
452 const auto remaining = static_cast<std::size_t>(from_end - from);
453 const char* src = from;
454 std::size_t taken = remaining;
455 if (m_pending_len != 0)
456 {
457 taken = std::min(remaining, joined.size() - m_pending_len);
458 std::copy_n(m_pending.data(), m_pending_len, joined.data());
459 std::copy_n(from, taken, joined.data() + m_pending_len);
460 src = joined.data();
461 }
462 const std::size_t avail = m_pending_len + taken;
463
464 auto tmp_state = m_state;
465 wchar_t wch = no_char;
466 std::size_t conv = mbrtowc(&wch, src, avail, &tmp_state);
467 if (conv == static_cast<std::size_t>(-1)) // NOLINT(modernize-use-integer-sign-comparison)
468 {
469 // Only tmp_state is unspecified after EILSEQ (C11 7.29.6.3.2); m_state
470 // still holds the shift state before the bad byte. Drop the held bytes.
471 m_pending_len = 0;
472 return std::pair{false, i_count};
473 }
474 else if (conv == static_cast<std::size_t>(-2)) // NOLINT(modernize-use-integer-sign-comparison)
475 {
476 // Hold every byte of the prefix; no character is longer than MB_LEN_MAX.
477 if (taken != remaining || avail > m_pending.size())
478 {
479 init_state();
480 return std::pair{false, i_count};
481 }
482 std::copy_n(from, taken, m_pending.data() + m_pending_len);
483 m_pending_len = avail;
484 from = from_end;
485 break;
486 }
487 else if (conv == 0)
488 {
489 // A null character: find how many bytes encode it.
490 std::size_t n = 1;
491 for (; n <= avail; ++n)
492 {
493 auto tmp_state2(m_state);
494 if (mbrtowc(nullptr, src, n, &tmp_state2) == 0)
495 {
496 tmp_state = tmp_state2;
497 break;
498 }
499 }
500 if (n > avail)
501 return std::pair{false, i_count};
502 conv = n;
503 wch = L'\0';
504 }
505
506 // conv bytes of src made up one complete sequence; the held bytes come first.
507 m_state = tmp_state;
508 if (conv <= m_pending_len)
509 {
510 std::copy(m_pending.begin() + conv, m_pending.begin() + m_pending_len, m_pending.begin());
511 m_pending_len -= conv;
512 }
513 else
514 {
515 from += conv - m_pending_len;
516 m_pending_len = 0;
517 }
518 if (wch == no_char)
519 continue; // a shift sequence alone
520 *to++ = static_cast<TInt>(wch);
521 ++i_count;
522 }
523
524 return std::pair{true, i_count};
525 }

725 bool out_helper(TInt ch, char8_t*& to, char8_t* to_end)
726 {
727 if (std::greater<>{}(to, to_end)) [[unlikely]]
728 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
729 // Check for maximum UTF-8 length (4 bytes) upfront
730 if (to_end - to < 4)
731 return false;
732 const auto c = static_cast<uint32_t>(ch);
733 if (0xD800U <= c && c <= 0xDFFFU) [[unlikely]]
734 return false;
735
736 if (c < 0x80U)
737 *to++ = static_cast<char8_t>(c);
738 else if (c <= 0x7ffU)
739 {
740 *to++ = static_cast<char8_t>((c >> 6) + 0xC0U);
741 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
742 }
743 else if (c <= 0xFFFFU)
744 {
745 *to++ = static_cast<char8_t>((c >> 12) + 0xE0U);
746 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);
747 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
748 }
749 else if (c <= 0x10FFFFU)
750 {
751 *to++ = static_cast<char8_t>((c >> 18) + 0xF0U);
752 *to++ = static_cast<char8_t>(((c >> 12) & 0x3FU) + 0x80U);
753 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);
754 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
755 }
756 else [[unlikely]]
757 return false;
758 return true;
759 }

802 std::pair<bool, std::size_t> in_helper(const char8_t*& from, const char8_t* from_end,
803 TInt*& to, TInt* to_end)
804 {
805 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
806 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
807 const TInt* const ori_to = to;
808 const auto written = [&] { return static_cast<std::size_t>(to - ori_to); };
809
810 // Held bytes first: complete that character from the front of the new input.
811 if (m_pending_len != 0 && from != from_end && to != to_end)
812 {
813 std::array<char8_t, 4> joined = m_pending;
814 const std::size_t held = m_pending_len;
815 const auto taken = std::min(static_cast<std::size_t>(from_end - from), joined.size() - held);
816 for (std::size_t i = 0; i < taken; ++i)
817 joined[held + i] = from[i];
818
819 const char8_t* p = joined.data();
820 if (decode_run(p, joined.data() + held + taken, to, to + 1) == run_status::bad)
821 {
822 init_state();
823 return std::pair{false, written()};
824 }
825 const auto used = static_cast<std::size_t>(p - joined.data());
826 if (used == 0)
827 {
828 // Still short: the new input ran out before the character did.
829 m_pending = joined;
830 m_pending_len = held + taken;
831 from += taken;
832 return std::pair{true, written()};
833 }
834 from += used - held;
835 m_pending_len = 0;
836 }
837
838 switch (decode_run(from, from_end, to, to_end))
839 {
840 case run_status::bad:
841 return std::pair{false, written()};
842 case run_status::short_input:
843 // An incomplete character at the end (at most three bytes): hold it.
844 m_pending_len = static_cast<std::size_t>(from_end - from);
845 for (std::size_t i = 0; i < m_pending_len; ++i)
846 m_pending[i] = from[i];
847 from = from_end;
848 break;
849 case run_status::done:
850 break;
851 }
852 return std::pair{true, written()};
853 }
854
855private:
856 enum class run_status { done, short_input, bad };
857
858 // Decodes characters from [from, from_end) into [to, to_end) until either runs
859 // out: done, short_input when the bytes left at from are the start of a
860 // character but too few, or bad at an invalid sequence.
861 static run_status decode_run(const char8_t*& from, const char8_t* from_end, TInt*& to, TInt* to_end)
862 {
863 while ((from != from_end) && (to != to_end))
864 {
865 auto c1 = static_cast<uint32_t>(*from);
866 if (c1 < 0x80U) [[likely]]
867 {
868 ++from;
869 *to++ = static_cast<TInt>(c1);
870 }
871 else if (c1 < 0xE0U)
872 {
873 if (c1 < 0xC0U) [[unlikely]]
874 return run_status::bad;
875 if (from_end - from < 2) return run_status::short_input;
876 auto c2 = static_cast<uint32_t>(from[1]);
877 if ((c2 & 0xC0U) != 0x80U) [[unlikely]]
878 return run_status::bad;
879 const auto encoded = c2 + (c1 << 6);
880 constexpr auto prefix_bias = (0xC0U << 6) + 0x80U;
881 const auto c = encoded - prefix_bias;
882 if (c < 0x80U) return run_status::bad;
883 *to++ = c;
884 from += 2;
885 }
886 else if (c1 < 0xF0U)
887 {
888 if (from_end - from < 3) return run_status::short_input;
889 auto c2 = static_cast<uint32_t>(from[1]);
890 auto c3 = static_cast<uint32_t>(from[2]);
891 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U)) [[unlikely]]
892 return run_status::bad;
893 const auto encoded = c3 + (c2 << 6) + (c1 << 12);
894 constexpr auto prefix_bias = (0xE0U << 12) + (0x80U << 6) + 0x80U;
895 const auto c = encoded - prefix_bias;
896 if (c < 0x800U) return run_status::bad;
897 if (c >= 0xD800U && c <= 0xDFFFU) [[unlikely]]
898 return run_status::bad;
899 *to++ = c;
900 from += 3;
901 }
902 else if (c1 < 0xF8U)
903 {
904 if (from_end - from < 4) return run_status::short_input;
905 auto c2 = static_cast<uint32_t>(from[1]);
906 auto c3 = static_cast<uint32_t>(from[2]);
907 auto c4 = static_cast<uint32_t>(from[3]);
908 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U) || ((c4 & 0xC0U) != 0x80U)) [[unlikely]]
909 return run_status::bad;
910 const auto encoded = c4 + (c3 << 6) + (c2 << 12) + (c1 << 18);
911 constexpr auto prefix_bias =
912 (0xF0U << 18) + (0x80U << 12) + (0x80U << 6) + 0x80U;
913 const auto c = encoded - prefix_bias;
914 if (c < 0x10000U || c > 0x10FFFFU) return run_status::bad;
915 *to++ = c;
916 from += 4;
917 }
918 else
919 return run_status::bad;
920 }
921
922 return run_status::done;
923 }

1257 std::size_t get_main(cvt_reader<KernelType>& reader, internal_type* to, std::size_t to_max)
1258 requires (cvt_cpt::support_get<KernelType>)
1259 {
1260 if (to_max == 0) return 0;
1261 reader.reset(s_max_buf_size);
1262 std::size_t total_size = 0;
1263
1264 std::size_t prev_rollback = 0;
1265 while (total_size < to_max)
1266 {
1267 std::size_t dest_size = std::min<std::size_t>(to_max - total_size, s_max_buf_size);
1268 dest_size = std::max(dest_size, prev_rollback + 1);
1269
1270 if (dest_size > s_max_buf_size) [[unlikely]]
1271 throw cvt_error("code_cvt::get fail: input sequence too long");
1272
1273 auto [ptr, cur_size] = reader.get_buf(dest_size);
1274 if (cur_size == prev_rollback)
1275 {
1276 if (cur_size == 0 && (total_size != 0 || !m_cvt_kernel.is_mid_seq()))
1277 return total_size;
1278 throw cvt_error("code_cvt::get fail: partial input sequence");
1279 }
1280
1281 auto ext_cur = ptr;
1282 auto [succ, int_len] = m_cvt_kernel.in_helper(ext_cur, ptr + cur_size, to, to + to_max - total_size);
1283
1284 // Update accumulated length BEFORE the failure check: in_helper may
1285 // have written `int_len` chars into the caller's `to` buffer before
1286 // hitting an invalid byte. Throwing without this update would leave
1287 // m_accu_len inconsistent with the chars already produced.
1288 m_accu_len += int_len;
1289 total_size += int_len;
1290
1291 if (!succ)
1292 throw cvt_error("code_cvt::get fail: invalid external sequence");
1293
1294 if (ext_cur == ptr + cur_size)
1295 prev_rollback = 0;
1296 else
1297 {
1298 prev_rollback = ptr + cur_size - ext_cur;
1299 reader.rollback(prev_rollback);
1300 }
1301 }
1302 return total_size;
1303 }
