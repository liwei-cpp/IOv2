163 explicit codecvt_kernel(const std::string& name)
164 : m_inter_locale(name)
165 {
166 clocale_user guard(m_inter_locale);
167
168 // there are no known constant-length encodings
169 // m_epc == 1 means fixed length
170 // Defensive check: the C standard requires MB_CUR_MAX to be a positive
171 // integer, but a misconfigured CRT / exotic libc could report 0, which
172 // would lead to division-by-zero UB in seek / rseek. Reject it here so
173 // m_epc >= 1 holds as a class invariant for any constructed instance.
174 const auto cur_max = MB_CUR_MAX;
175 if (cur_max == 0) [[unlikely]]
176 throw cvt_error("codecvt_kernel: locale reports MB_CUR_MAX == 0");
177 m_epc = static_cast<unsigned>(cur_max);
178
179 // Note: mbtowc uses internal static state, so concurrent construction of
180 // codecvt_kernel instances is NOT thread-safe. However, this is acceptable
181 // because this class does not support multi-threaded construction.
182 // Once constructed, instance-level operations use explicit mbstate_t (m_state)
183 // and are safe for single-instance usage.
184 // NOLINTNEXTLINE(cppcoreguidelines-prefer-member-initializer)
185 m_is_state_dep = (std::mbtowc(nullptr, nullptr, MB_CUR_MAX) != 0);
186 init_state();
187 }

366 {
367 if (std::greater<>{}(to, to_end)) [[unlikely]]
368 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
369 clocale_user guard(m_inter_locale);
370 if (static_cast<std::size_t>(to_end - to) < m_epc) // NOLINT(modernize-use-integer-sign-comparison)
371 return false;

373 const std::mbstate_t before = m_state;

375 const std::size_t conv = std::wcrtomb(local.data(), ch, &m_state);

445 std::pair<bool, std::size_t> in_helper(const char*& from, const char* from_end,
446 TInt*& to, TInt* to_end)
447 {
448 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
449 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
450 clocale_user guard(m_inter_locale);
451 // No wchar_t value glibc decodes to: it tells a shift-only run from a character.
452 constexpr auto no_char = static_cast<wchar_t>(-1);
453 std::array<char, MB_LEN_MAX> joined{};
454 std::size_t i_count = 0;
455
456 const auto find_nul = [&]
457 {
458 if (!m_is_state_dep || from >= from_end)
459 return from_end;
460 const void* p = std::memchr(from, 0, static_cast<std::size_t>(from_end - from));
461 return p != nullptr ? static_cast<const char*>(p) : from_end;
462 };
463 const char* nul = find_nul();
464
465 const std::size_t to_max = to_end - to;
466 while (i_count < to_max && (from < from_end || m_pending_len != 0))
467 {
468 if (nul < from)
469 nul = find_nul();
470 if (m_pending_len == 0 && from < from_end && *from == '\0')
471 {
472 ++from;
473 m_state = std::mbstate_t{};
474 *to++ = TInt{};
475 ++i_count;
476 continue;
477 }
478
479 // Held bytes go first, then as many new ones as fit behind them.
480 const auto remaining = static_cast<std::size_t>(nul - from);
481 const char* src = from;
482 std::size_t taken = remaining;
483 if (m_pending_len != 0)
484 {
485 taken = std::min(remaining, joined.size() - m_pending_len);
486 std::copy_n(m_pending.data(), m_pending_len, joined.data());
487 std::copy_n(from, taken, joined.data() + m_pending_len);
488 src = joined.data();
489 }
490 const std::size_t avail = m_pending_len + taken;
491
492 auto tmp_state = m_state;
493 wchar_t wch = no_char;
494 std::size_t conv = mbrtowc(&wch, src, avail, &tmp_state);
495 if (m_is_state_dep && (conv == static_cast<std::size_t>(-1) || conv == static_cast<std::size_t>(-2))) // NOLINT(modernize-use-integer-sign-comparison)
496 {
497 // glibc judges a run that opens with a complete shift sequence as a
498 // whole; take that sequence off first, so the error or the held bytes
499 // start behind it.
500 for (std::size_t k = 1; k < avail; ++k)
501 {
502 auto st = m_state;
503 wchar_t w = no_char;
504 const std::size_t r = mbrtowc(&w, src, k, &st);
505 if (r == static_cast<std::size_t>(-2)) continue; // NOLINT(modernize-use-integer-sign-comparison)
506 if (r != static_cast<std::size_t>(-1) && r != 0 && w == no_char) // NOLINT(modernize-use-integer-sign-comparison)
507 {
508 conv = r;
509 tmp_state = st;
510 }
511 break;
512 }
513 }
514 if (conv == static_cast<std::size_t>(-1)) // NOLINT(modernize-use-integer-sign-comparison)
515 {
516 // Only tmp_state is unspecified after EILSEQ (C11 7.29.6.3.2); m_state
517 // still holds the shift state before the bad byte. Drop the held bytes.
518 m_pending_len = 0;
519 return std::pair{false, i_count};
520 }
521 else if (conv == static_cast<std::size_t>(-2)) // NOLINT(modernize-use-integer-sign-comparison)
522 {
523 if (nul != from_end)
524 {
525 m_pending_len = 0;
526 return std::pair{false, i_count};
527 }
528 // Hold every byte of the prefix; no character is longer than MB_LEN_MAX.
529 if (taken != remaining || avail > m_pending.size())
530 {
531 m_pending_len = 0;
532 return std::pair{false, i_count};
533 }
534 std::copy_n(from, taken, m_pending.data() + m_pending_len);
535 m_pending_len = avail;
536 from = from_end;
537 break;
538 }
539 else if (conv == 0)
540 return std::pair{false, i_count};
541
542 // conv bytes of src made up one complete sequence; the held bytes come first.
543 m_state = tmp_state;
544 if (conv <= m_pending_len)
545 {
546 std::copy(m_pending.begin() + conv, m_pending.begin() + m_pending_len, m_pending.begin());
547 m_pending_len -= conv;
548 }
549 else
550 {
551 from += conv - m_pending_len;
552 m_pending_len = 0;
553 }
554 if (wch == no_char)
555 continue; // a shift sequence alone
556 *to++ = static_cast<TInt>(wch);
557 ++i_count;
558 }
559
560 return std::pair{true, i_count};
561 }

565 unsigned m_epc = 0;
566 std::mbstate_t m_state{};
567 std::array<char, MB_LEN_MAX> m_pending{};
568 std::size_t m_pending_len = 0;
569 bool m_is_state_dep = false;

761 bool out_helper(TInt ch, char8_t*& to, char8_t* to_end)
762 {
763 if (std::greater<>{}(to, to_end)) [[unlikely]]
764 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
765 // Check for maximum UTF-8 length (4 bytes) upfront
766 if (to_end - to < 4)
767 return false;
768 const auto c = static_cast<uint32_t>(ch);
769 if (0xD800U <= c && c <= 0xDFFFU) [[unlikely]]
770 return false;
771
772 if (c < 0x80U)
773 *to++ = static_cast<char8_t>(c);
774 else if (c <= 0x7ffU)
775 {
776 *to++ = static_cast<char8_t>((c >> 6) + 0xC0U);
777 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
778 }
779 else if (c <= 0xFFFFU)
780 {
781 *to++ = static_cast<char8_t>((c >> 12) + 0xE0U);
782 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);
783 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
784 }
785 else if (c <= 0x10FFFFU)
786 {
787 *to++ = static_cast<char8_t>((c >> 18) + 0xF0U);
788 *to++ = static_cast<char8_t>(((c >> 12) & 0x3FU) + 0x80U);
789 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);
790 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
791 }
792 else [[unlikely]]
793 return false;
794 return true;

838 std::pair<bool, std::size_t> in_helper(const char8_t*& from, const char8_t* from_end,
839 TInt*& to, TInt* to_end)
840 {
841 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
842 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
843 const TInt* const ori_to = to;
844 const auto written = [&] { return static_cast<std::size_t>(to - ori_to); };
845
846 // Held bytes first: complete that character from the front of the new input.
847 if (m_pending_len != 0 && from != from_end && to != to_end)
848 {
849 std::array<char8_t, 4> joined = m_pending;
850 const std::size_t held = m_pending_len;
851 const auto taken = std::min(static_cast<std::size_t>(from_end - from), joined.size() - held);
852 for (std::size_t i = 0; i < taken; ++i)
853 joined[held + i] = from[i];
854
855 const char8_t* p = joined.data();
856 if (decode_run(p, joined.data() + held + taken, to, to + 1) == run_status::bad)
857 {
858 init_state();
859 return std::pair{false, written()};
860 }
861 const auto used = static_cast<std::size_t>(p - joined.data());
862 if (used == 0)
863 {
864 // Still short: the new input ran out before the character did.
865 m_pending = joined;
866 m_pending_len = held + taken;
867 from += taken;
868 return std::pair{true, written()};
869 }
870 from += used - held;
871 m_pending_len = 0;
872 }
873
874 switch (decode_run(from, from_end, to, to_end))
875 {
876 case run_status::bad:
877 return std::pair{false, written()};
878 case run_status::short_input:
879 // An incomplete character at the end (at most three bytes): hold it.
880 m_pending_len = static_cast<std::size_t>(from_end - from);
881 for (std::size_t i = 0; i < m_pending_len; ++i)
882 m_pending[i] = from[i];
883 from = from_end;
884 break;
885 case run_status::done:
886 break;
887 }
888 return std::pair{true, written()};
889 }

891private:
892 enum class run_status : std::uint8_t { done, short_input, bad };
893
894 // Decodes characters from [from, from_end) into [to, to_end) until either runs
895 // out: done, short_input when the bytes left at from are the start of a
896 // character but too few, or bad at an invalid sequence.
897 static run_status decode_run(const char8_t*& from, const char8_t* from_end, TInt*& to, TInt* to_end)
898 {
899 while ((from != from_end) && (to != to_end))
900 {
901 auto c1 = static_cast<uint32_t>(*from);
902 if (c1 < 0x80U) [[likely]]
903 {
904 ++from;
905 *to++ = static_cast<TInt>(c1);
906 }
907 else if (c1 < 0xE0U)
908 {
909 if (c1 < 0xC0U) [[unlikely]]
910 return run_status::bad;
911 if (from_end - from < 2) return run_status::short_input;
912 auto c2 = static_cast<uint32_t>(from[1]);
913 if ((c2 & 0xC0U) != 0x80U) [[unlikely]]
914 return run_status::bad;
915 const auto encoded = c2 + (c1 << 6);
916 constexpr auto prefix_bias = (0xC0U << 6) + 0x80U;
917 const auto c = encoded - prefix_bias;
918 if (c < 0x80U) return run_status::bad;
919 *to++ = c;
920 from += 2;
921 }
922 else if (c1 < 0xF0U)
923 {
924 if (from_end - from < 3) return run_status::short_input;
925 auto c2 = static_cast<uint32_t>(from[1]);
926 auto c3 = static_cast<uint32_t>(from[2]);
927 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U)) [[unlikely]]
928 return run_status::bad;
929 const auto encoded = c3 + (c2 << 6) + (c1 << 12);
930 constexpr auto prefix_bias = (0xE0U << 12) + (0x80U << 6) + 0x80U;
931 const auto c = encoded - prefix_bias;
932 if (c < 0x800U) return run_status::bad;
933 if (c >= 0xD800U && c <= 0xDFFFU) [[unlikely]]
934 return run_status::bad;
935 *to++ = c;
936 from += 3;
937 }
938 else if (c1 < 0xF8U)
939 {
940 if (from_end - from < 4) return run_status::short_input;
941 auto c2 = static_cast<uint32_t>(from[1]);
942 auto c3 = static_cast<uint32_t>(from[2]);
943 auto c4 = static_cast<uint32_t>(from[3]);
944 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U) || ((c4 & 0xC0U) != 0x80U)) [[unlikely]]
945 return run_status::bad;
946 const auto encoded = c4 + (c3 << 6) + (c2 << 12) + (c1 << 18);
947 constexpr auto prefix_bias =
948 (0xF0U << 18) + (0x80U << 12) + (0x80U << 6) + 0x80U;
949 const auto c = encoded - prefix_bias;
950 if (c < 0x10000U || c > 0x10FFFFU) return run_status::bad;
951 *to++ = c;
952 from += 4;
953 }
954 else
955 return run_status::bad;
956 }
957
958 return run_status::done;
959 }

1314 std::size_t get_main(cvt_reader<KernelType>& reader, internal_type* to, std::size_t to_max)
1315 requires (cvt_cpt::support_get<KernelType>)
1316 {
1317 if (to_max == 0) return 0;
1318 if (m_get_failed)
1319 {
1320 m_get_failed = false;
1321 throw cvt_error("code_cvt::get fail: invalid external sequence");
1322 }
1323 reader.reset(s_max_buf_size);
1324 std::size_t total_size = 0;
1325
1326 std::size_t prev_rollback = 0;
1327 while (total_size < to_max)
1328 {
1329 std::size_t dest_size = std::min<std::size_t>(to_max - total_size, s_max_buf_size);
1330 dest_size = std::max(dest_size, prev_rollback + 1);
1331
1332 if (dest_size > s_max_buf_size) [[unlikely]]
1333 throw cvt_error("code_cvt::get fail: input sequence too long");
1334
1335 auto [ptr, cur_size] = reader.get_buf(dest_size);
1336 if (cur_size == prev_rollback)
1337 {
1338 if (cur_size == 0 && (total_size != 0 || !m_cvt_kernel.is_mid_seq()))
1339 return total_size;
1340 throw cvt_error("code_cvt::get fail: partial input sequence");
1341 }
1342
1343 auto ext_cur = ptr;
1344 const bool held = m_cvt_kernel.is_mid_seq();
1345 auto [succ, int_len] = m_cvt_kernel.in_helper(ext_cur, ptr + cur_size, to, to + to_max - total_size);
1346
1347 // Update accumulated length BEFORE the failure check: in_helper may
1348 // have written `int_len` chars into the caller's `to` buffer before
1349 // hitting an invalid byte. Throwing without this update would leave
1350 // m_accu_len inconsistent with the chars already produced.
1351 m_accu_len += int_len;
1352 total_size += int_len;
1353
1354 if (!succ)
1355 {
1356 const auto* resume = (held && ext_cur == ptr && int_len == 0)
1357 ? ext_cur : std::min(ext_cur + 1, ptr + cur_size);
1358 if (resume != ptr + cur_size)
1359 reader.rollback(ptr + cur_size - resume);
1360 if (total_size == 0)
1361 throw cvt_error("code_cvt::get fail: invalid external sequence");
1362 m_get_failed = true;
1363 return total_size;
1364 }
1365
1366 if (ext_cur == ptr + cur_size)
1367 prev_rollback = 0;
1368 else
1369 {
1370 prev_rollback = ptr + cur_size - ext_cur;
1371 reader.rollback(prev_rollback);
1372 }
1373 }
1374 return total_size;
1375 }

1524 void seek_impl(std::size_t pos)
1525 requires (cvt_cpt::support_positioning<KernelType>)
1526 {
1527 const bool needs_state_reset =
1528 m_cvt_kernel.is_var_length() || m_cvt_kernel.is_state_dep();
1529
1530 if (needs_state_reset && (pos != 0 || BT::m_io_status == io_status::output))
1531 throw cvt_error("code_cvt::seek fail: cannot seek with dependent converter");
1532
1533 const unsigned epc = m_cvt_kernel.epc();
1534 if (pos > std::numeric_limits<std::size_t>::max() / epc)
1535 throw cvt_error("code_cvt::seek fail: position overflow");
1536
1537 // Commit lower layer first. If it throws, this object is unchanged
1538 // (strong exception guarantee). After this point the remaining
1539 // mutations are noexcept (mbstate_t reset / scalar assignment), so
1540 // we cannot end up half-updated.
1541 BT::kernel_seek(pos * epc);
1542
1543 if (needs_state_reset)
1544 m_cvt_kernel.init_state();
1545 m_accu_len = pos;
1546 m_get_failed = false;
1547 }

1573 void rseek_impl(std::size_t pos)
1574 requires (cvt_cpt::support_positioning<KernelType>)
1575 {
1576 if (m_cvt_kernel.is_var_length() || m_cvt_kernel.is_state_dep())
1577 throw cvt_error("code_cvt::rseek fail: cannot seek with dependent converter");
1578
1579 const unsigned epc = m_cvt_kernel.epc();
1580 if (pos > std::numeric_limits<std::size_t>::max() / epc)
1581 throw cvt_error("code_cvt::rseek fail: position overflow");
1582
1583 const std::size_t saved_dev_pos = BT::kernel_tell();
1584 BT::kernel_rseek(pos * epc);
1585 const std::size_t new_dev_pos = BT::kernel_tell();
1586
1587 if (new_dev_pos % epc != 0)
1588 {
1589 // Restore device so this call appears atomic. If the restore
1590 // itself throws, suppress it — the caller already gets a
1591 // cvt_error and is expected to re-seek before further use;
1592 // surfacing a different exception here would only obscure the
1593 // failure mode.
1594 try { BT::kernel_seek(saved_dev_pos); } catch (...) {} // NOLINT(bugprone-empty-catch)
1595 throw cvt_error("code_cvt::rseek fail: partial sequence");
1596 }
1597
1598 m_accu_len = new_dev_pos / epc;
1599 m_cvt_kernel.init_state();
1600 m_get_failed = false;
1601 }
