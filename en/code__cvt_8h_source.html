356 std::pair<bool, std::size_t> in_helper(const char*& from, const char* from_end,
357 TInt*& to, TInt* to_end)
358 {
359 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
360 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
361 clocale_user guard(m_inter_locale);
362 wchar_t wch = 0;
363 std::size_t i_count = 0;
364
365 const std::size_t to_max = to_end - to;
366 while (from < from_end && (i_count < to_max))
367 {
368 auto tmp_state = m_state;
369 std::size_t conv = mbrtowc(&wch, from, from_end - from, &tmp_state);
370 if (conv == static_cast<std::size_t>(-1)) // NOLINT(modernize-use-integer-sign-comparison)
371 return std::pair{false, i_count};
372 else if (conv == static_cast<std::size_t>(-2)) // NOLINT(modernize-use-integer-sign-comparison)
373 {
374 from = from_end;
375 m_state = tmp_state;
376 break;
377 }
378 else if (conv == 0)
379 {
380 // Find the actual byte length of the null character encoding
381 // before writing to output buffer to ensure consistency
382 std::size_t n = 1;
383 const auto max_n = static_cast<std::size_t>(from_end - from);
384 for (; n <= max_n; ++n)
385 {
386 auto tmp_state2(m_state);
387 if (mbrtowc(nullptr, from, n, &tmp_state2) == 0)
388 {
389 m_state = tmp_state2;
390 break;
391 }
392 }
393 if (n > max_n)
394 return std::pair{false, i_count};
395
396 // Only write to output after validation succeeds
397 *to++ = static_cast<TInt>(0);
398 from += n;
399 ++i_count;
400 }
401 else
402 {
403 // mbrtowc returning > 0 guarantees a complete character was converted;
404 // no partial character can exist in m_state at this point
405 m_state = tmp_state;
406 *to++ = wch;
407 from += conv;
408 ++i_count;
409 }
410 }
411
412 return std::pair{true, i_count};
413 }

966 std::size_t get_main(cvt_reader<KernelType>& reader, internal_type* to, std::size_t to_max)
967 requires (cvt_cpt::support_get<KernelType>)
968 {
969 if (to_max == 0) return 0;
970 reader.reset(s_max_buf_size);
971 std::size_t total_size = 0;
972
973 std::size_t prev_rollback = 0;
974 while (total_size < to_max)
975 {
976 std::size_t dest_size = std::min<std::size_t>(to_max - total_size, s_max_buf_size);
977 dest_size = std::max(dest_size, prev_rollback + 1);
978
979 if (dest_size > s_max_buf_size) [[unlikely]]
980 throw cvt_error("code_cvt::get fail: input sequence too long");
981
982 auto [ptr, cur_size] = reader.get_buf(dest_size);
983 if (cur_size == prev_rollback)
984 {
985 if (cur_size == 0) return total_size;
986 throw cvt_error("code_cvt::get fail: partial input sequence");
987 }
988
989 auto ext_cur = ptr;
990 auto [succ, int_len] = m_cvt_kernel.in_helper(ext_cur, ptr + cur_size, to, to + to_max - total_size);
991
992 // Update accumulated length BEFORE the failure check: in_helper may
993 // have written `int_len` chars into the caller's `to` buffer before
994 // hitting an invalid byte. Throwing without this update would leave
995 // m_accu_len inconsistent with the chars already produced.
996 m_accu_len += int_len;
997 total_size += int_len;
998
999 if (!succ)
1000 throw cvt_error("code_cvt::get fail: invalid external sequence");
1001
1002 if (ext_cur == ptr + cur_size)
1003 prev_rollback = 0;
1004 else
1005 {
1006 prev_rollback = ptr + cur_size - ext_cur;
1007 reader.rollback(prev_rollback);
1008 }
1009 }
1010 return total_size;
