161 explicit codecvt_kernel(const std::string& name)

167 // m_epc == 1 means fixed length
168 // Defensive check: the C standard requires MB_CUR_MAX to be a positive
169 // integer, but a misconfigured CRT / exotic libc could report 0, which
170 // would lead to division-by-zero UB in seek / rseek. Reject it here so
171 // m_epc >= 1 holds as a class invariant for any constructed instance.

302 bool out_helper(TInt ch, char*& to, char* to_end)
303 {
304 if (std::greater<>{}(to, to_end)) [[unlikely]]
305 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
306 clocale_user guard(m_inter_locale);
307 if (static_cast<std::size_t>(to_end - to) < m_epc) // NOLINT(modernize-use-integer-sign-comparison)
308 return false;
309
310 const std::size_t conv = std::wcrtomb(to, ch, &m_state);
311 if (conv == static_cast<std::size_t>(-1)) // NOLINT(modernize-use-integer-sign-comparison)
312 {
313 init_state(); // Reset to known state per C standard
314 return false;
315 }
316 to += conv;
317
318 return true;

359 std::pair<bool, std::size_t> in_helper(const char*& from, const char* from_end,
360 TInt*& to, TInt* to_end)
361 {
362 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
363 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
364 clocale_user guard(m_inter_locale);
365 wchar_t wch = 0;
366 std::size_t i_count = 0;
367
368 const std::size_t to_max = to_end - to;
369 while (from < from_end && (i_count < to_max))
370 {
371 auto tmp_state = m_state;
372 std::size_t conv = mbrtowc(&wch, from, from_end - from, &tmp_state);
373 if (conv == static_cast<std::size_t>(-1)) // NOLINT(modernize-use-integer-sign-comparison)
374 return std::pair{false, i_count};
375 else if (conv == static_cast<std::size_t>(-2)) // NOLINT(modernize-use-integer-sign-comparison)
376 {
377 from = from_end;
378 m_state = tmp_state;
379 break;
380 }
381 else if (conv == 0)
382 {
383 // Find the actual byte length of the null character encoding
384 // before writing to output buffer to ensure consistency
385 std::size_t n = 1;
386 const auto max_n = static_cast<std::size_t>(from_end - from);
387 for (; n <= max_n; ++n)
388 {
389 auto tmp_state2(m_state);
390 if (mbrtowc(nullptr, from, n, &tmp_state2) == 0)
391 {
392 m_state = tmp_state2;
393 break;
394 }
395 }
396 if (n > max_n)
397 return std::pair{false, i_count};
398
399 // Only write to output after validation succeeds
400 *to++ = static_cast<TInt>(0);
401 from += n;
402 ++i_count;
403 }
404 else
405 {
406 // mbrtowc returning > 0 guarantees a complete character was converted;
407 // no partial character can exist in m_state at this point
408 m_state = tmp_state;
409 *to++ = wch;
410 from += conv;
411 ++i_count;
412 }
413 }
414
415 return std::pair{true, i_count};
416 }
417
418private:
419 clocale_wrapper m_inter_locale;
420 unsigned m_epc = 0;
421 std::mbstate_t m_state{};
422 bool m_is_state_dep = false;

570 {
571 if (std::greater<>{}(to, to_end)) [[unlikely]]
572 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
573 // Check for maximum UTF-8 length (4 bytes) upfront
574 if (to_end - to < 4)
575 return false;
576 const auto c = static_cast<uint32_t>(ch);
577 if (0xD800U <= c && c <= 0xDFFFU) [[unlikely]]
578 return false;
579
580 if (c < 0x80U)
581 *to++ = static_cast<char8_t>(c);
582 else if (c <= 0x7ffU)
583 {
584 *to++ = static_cast<char8_t>((c >> 6) + 0xC0U);
585 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
586 }
587 else if (c <= 0xFFFFU)
588 {

590 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);

646 std::pair<bool, std::size_t> in_helper(const char8_t*& from, const char8_t* from_end,
647 TInt*& to, TInt* to_end)
648 {
649 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
650 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
651 const TInt* const ori_to = to;
652
653 while ((from != from_end) && (to != to_end))
654 {
655 auto c1 = static_cast<uint32_t>(*from);
656 if (c1 < 0x80U) [[likely]]
657 {
658 ++from;
659 *to++ = static_cast<TInt>(c1);
660 }
661 else if (c1 < 0xE0U)
662 {
663 if (c1 < 0xC0U) [[unlikely]]
664 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
665 if (from_end - from < 2) break;
666 auto c2 = static_cast<uint32_t>(from[1]);
667 if ((c2 & 0xC0U) != 0x80U) [[unlikely]]
668 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
669 const auto encoded = c2 + (c1 << 6);
670 constexpr auto prefix_bias = (0xC0U << 6) + 0x80U;
671 const auto c = encoded - prefix_bias;
672 if (c < 0x80U) return std::pair{false, static_cast<std::size_t>(to - ori_to)};
673 *to++ = c;
674 from += 2;
675 }
676 else if (c1 < 0xF0U)
677 {
678 if (from_end - from < 3) break;
679 auto c2 = static_cast<uint32_t>(from[1]);
680 auto c3 = static_cast<uint32_t>(from[2]);
681 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U)) [[unlikely]]
682 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
683 const auto encoded = c3 + (c2 << 6) + (c1 << 12);
684 constexpr auto prefix_bias = (0xE0U << 12) + (0x80U << 6) + 0x80U;
685 const auto c = encoded - prefix_bias;
686 if (c < 0x800U) return std::pair{false, static_cast<std::size_t>(to - ori_to)};
687 if (c >= 0xD800U && c <= 0xDFFFU) [[unlikely]]
688 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
689 *to++ = c;
690 from += 3;
691 }
692 else if (c1 < 0xF8U)
693 {
694 if (from_end - from < 4) break;
695 auto c2 = static_cast<uint32_t>(from[1]);
696 auto c3 = static_cast<uint32_t>(from[2]);
697 auto c4 = static_cast<uint32_t>(from[3]);
698 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U) || ((c4 & 0xC0U) != 0x80U)) [[unlikely]]
699 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
700 const auto encoded = c4 + (c3 << 6) + (c2 << 12) + (c1 << 18);
701 constexpr auto prefix_bias =
702 (0xF0U << 18) + (0x80U << 12) + (0x80U << 6) + 0x80U;
703 const auto c = encoded - prefix_bias;
704 if (c < 0x10000U || c > 0x10FFFFU) return std::pair{false, static_cast<std::size_t>(to - ori_to)};
705 *to++ = c;
706 from += 4;
707 }
708 else
709 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
710 }
711
712 return std::pair{true, static_cast<std::size_t>(to - ori_to)};
713 }

937 void put_main(cvt_writer<KernelType>& writer, const internal_type* to, std::size_t to_size)
938 requires (cvt_cpt::support_put<KernelType>)
939 {
940 writer.reset(s_max_buf_size);

942 for (std::size_t i = 0; i < to_size; ++i)
943 {
944 external_type* out_beg = writer.put_buf(buf_len);
945 external_type* out_next = out_beg;
946 put_buf_guard guard{writer, buf_len};

951 throw cvt_error("code_cvt::put fail: input character cannot be encoded");
952
953 guard.used(static_cast<std::size_t>(out_next - out_beg));
954 }

956 // updated only on success so that a thrown encoding error leaves

976 std::size_t get_main(cvt_reader<KernelType>& reader, internal_type* to, std::size_t to_max)
977 requires (cvt_cpt::support_get<KernelType>)
978 {
979 if (to_max == 0) return 0;
980 reader.reset(s_max_buf_size);
981 std::size_t total_size = 0;
982
983 std::size_t prev_rollback = 0;
984 while (total_size < to_max)
985 {
986 std::size_t dest_size = std::min<std::size_t>(to_max - total_size, s_max_buf_size);
987 dest_size = std::max(dest_size, prev_rollback + 1);
988
989 if (dest_size > s_max_buf_size) [[unlikely]]
990 throw cvt_error("code_cvt::get fail: input sequence too long");
991
992 auto [ptr, cur_size] = reader.get_buf(dest_size);
993 if (cur_size == prev_rollback)
994 {
995 if (cur_size == 0) return total_size;
996 throw cvt_error("code_cvt::get fail: partial input sequence");
997 }
998
999 auto ext_cur = ptr;
1000 auto [succ, int_len] = m_cvt_kernel.in_helper(ext_cur, ptr + cur_size, to, to + to_max - total_size);
1001
1002 // Update accumulated length BEFORE the failure check: in_helper may
1003 // have written `int_len` chars into the caller's `to` buffer before
1004 // hitting an invalid byte. Throwing without this update would leave
1005 // m_accu_len inconsistent with the chars already produced.
1006 m_accu_len += int_len;
1007 total_size += int_len;
1008
1009 if (!succ)
1010 throw cvt_error("code_cvt::get fail: invalid external sequence");
1011
1012 if (ext_cur == ptr + cur_size)
1013 prev_rollback = 0;
1014 else
1015 {
1016 prev_rollback = ptr + cur_size - ext_cur;
1017 reader.rollback(prev_rollback);
1018 }
1019 }
1020 return total_size;
1021 }

1126 void seek_impl(std::size_t pos)
1127 requires (cvt_cpt::support_positioning<KernelType>)
1128 {
1129 const bool needs_state_reset =
1130 m_cvt_kernel.is_var_length() || m_cvt_kernel.is_state_dep();
1131
1132 if (needs_state_reset && (pos != 0 || BT::m_io_status == io_status::output))
1133 throw cvt_error("code_cvt::seek fail: cannot seek with dependent converter");
1134
1135 const unsigned epc = m_cvt_kernel.epc();
1136 if (pos > std::numeric_limits<std::size_t>::max() / epc)
1137 throw cvt_error("code_cvt::seek fail: position overflow");

1142 // we cannot end up half-updated.

1174 void rseek_impl(std::size_t pos)
1175 requires (cvt_cpt::support_positioning<KernelType>)
1176 {
1177 if (m_cvt_kernel.is_var_length() || m_cvt_kernel.is_state_dep())
1178 throw cvt_error("code_cvt::rseek fail: cannot seek with dependent converter");
1179
1180 const unsigned epc = m_cvt_kernel.epc();
1181 if (pos > std::numeric_limits<std::size_t>::max() / epc)
1182 throw cvt_error("code_cvt::rseek fail: position overflow");
1183
1184 const std::size_t saved_dev_pos = BT::m_kernel.tell();
1185 BT::m_kernel.rseek(pos * epc);
1186 const std::size_t new_dev_pos = BT::m_kernel.tell();

1188 if (new_dev_pos % epc != 0)
1189 {

1191 // itself throws, suppress it — the caller already gets a
1192 // cvt_error and is expected to re-seek before further use;
1193 // surfacing a different exception here would only obscure the
1194 // failure mode.
1195 try { BT::m_kernel.seek(saved_dev_pos); } catch (...) {} // NOLINT(bugprone-empty-catch)
1196 throw cvt_error("code_cvt::rseek fail: partial sequence");
1197 }
1198
1199 m_accu_len = new_dev_pos / epc;
1200 m_cvt_kernel.init_state();
1201 }
