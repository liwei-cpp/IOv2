162 explicit codecvt_kernel(const std::string& name)
163 : m_inter_locale(name)
164 {
165 clocale_user guard(m_inter_locale);
166
167 // there are no known constant-length encodings
168 // m_epc == 1 means fixed length
169 // Defensive check: the C standard requires MB_CUR_MAX to be a positive
170 // integer, but a misconfigured CRT / exotic libc could report 0, which
171 // would lead to division-by-zero UB in seek / rseek. Reject it here so
172 // m_epc >= 1 holds as a class invariant for any constructed instance.
173 const auto cur_max = MB_CUR_MAX;
174 if (cur_max == 0) [[unlikely]]
175 throw cvt_error("codecvt_kernel: locale reports MB_CUR_MAX == 0");
176 m_epc = static_cast<unsigned>(cur_max);
177
178 // Note: mbtowc uses internal static state, so concurrent construction of
179 // codecvt_kernel instances is NOT thread-safe. However, this is acceptable
180 // because this class does not support multi-threaded construction.
181 // Once constructed, instance-level operations use explicit mbstate_t (m_state)
182 // and are safe for single-instance usage.
183 // NOLINTNEXTLINE(cppcoreguidelines-prefer-member-initializer)
184 m_is_state_dep = (std::mbtowc(nullptr, nullptr, MB_CUR_MAX) != 0);
185 init_state();
186 }

204 std::size_t unshift(char* to, std::size_t capacity)
205 {
206 if (to == nullptr || capacity < m_epc)
207 throw cvt_error("codecvt_kernel::unshift fail: insufficient output buffer");

338 bool out_helper(TInt ch, char*& to, char* to_end)
339 {
340 if (std::greater<>{}(to, to_end)) [[unlikely]]
341 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
342 clocale_user guard(m_inter_locale);
343 if (static_cast<std::size_t>(to_end - to) < m_epc) // NOLINT(modernize-use-integer-sign-comparison)
344 return false;
345
346 const std::size_t conv = std::wcrtomb(to, ch, &m_state);
347 if (conv == static_cast<std::size_t>(-1)) // NOLINT(modernize-use-integer-sign-comparison)
348 {
349 init_state(); // Reset to known state per C standard
350 return false;
351 }
352 to += conv;
353
354 return true;
355 }

395 std::pair<bool, std::size_t> in_helper(const char*& from, const char* from_end,
396 TInt*& to, TInt* to_end)
397 {
398 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
399 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
400 clocale_user guard(m_inter_locale);
401 wchar_t wch = 0;
402 std::size_t i_count = 0;
403
404 const std::size_t to_max = to_end - to;
405 while (from < from_end && (i_count < to_max))
406 {
407 auto tmp_state = m_state;
408 std::size_t conv = mbrtowc(&wch, from, from_end - from, &tmp_state);
409 if (conv == static_cast<std::size_t>(-1)) // NOLINT(modernize-use-integer-sign-comparison)
410 {
411 // EILSEQ leaves the state unspecified (C11 7.29.6.3.2); a partial
412 // sequence kept from an earlier -2 must not pair with what follows.
413 init_state();
414 return std::pair{false, i_count};
415 }
416 else if (conv == static_cast<std::size_t>(-2)) // NOLINT(modernize-use-integer-sign-comparison)
417 {
418 from = from_end;
419 m_state = tmp_state;
420 break;
421 }
422 else if (conv == 0)
423 {
424 // Find the actual byte length of the null character encoding
425 // before writing to output buffer to ensure consistency
426 std::size_t n = 1;
427 const auto max_n = static_cast<std::size_t>(from_end - from);
428 for (; n <= max_n; ++n)
429 {
430 auto tmp_state2(m_state);
431 if (mbrtowc(nullptr, from, n, &tmp_state2) == 0)
432 {
433 m_state = tmp_state2;
434 break;
435 }
436 }
437 if (n > max_n)
438 return std::pair{false, i_count};
439
440 // Only write to output after validation succeeds
441 *to++ = static_cast<TInt>(0);
442 from += n;
443 ++i_count;
444 }
445 else
446 {
447 // mbrtowc returning > 0 guarantees a complete character was converted;
448 // no partial character can exist in m_state at this point
449 m_state = tmp_state;
450 *to++ = wch;
451 from += conv;
452 ++i_count;
453 }
454 }
455
456 return std::pair{true, i_count};
457 }

459private:
460 clocale_wrapper m_inter_locale;
461 unsigned m_epc = 0;
462 std::mbstate_t m_state{};
463 bool m_is_state_dep = false;

634 bool out_helper(TInt ch, char8_t*& to, char8_t* to_end)
635 {
636 if (std::greater<>{}(to, to_end)) [[unlikely]]
637 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
638 // Check for maximum UTF-8 length (4 bytes) upfront
639 if (to_end - to < 4)
640 return false;
641 const auto c = static_cast<uint32_t>(ch);
642 if (0xD800U <= c && c <= 0xDFFFU) [[unlikely]]
643 return false;
644
645 if (c < 0x80U)
646 *to++ = static_cast<char8_t>(c);
647 else if (c <= 0x7ffU)
648 {
649 *to++ = static_cast<char8_t>((c >> 6) + 0xC0U);
650 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
651 }
652 else if (c <= 0xFFFFU)
653 {
654 *to++ = static_cast<char8_t>((c >> 12) + 0xE0U);
655 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);
656 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
657 }
658 else if (c <= 0x10FFFFU)
659 {
660 *to++ = static_cast<char8_t>((c >> 18) + 0xF0U);
661 *to++ = static_cast<char8_t>(((c >> 12) & 0x3FU) + 0x80U);
662 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);
663 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);

711 std::pair<bool, std::size_t> in_helper(const char8_t*& from, const char8_t* from_end,
712 TInt*& to, TInt* to_end)
713 {
714 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
715 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
716 const TInt* const ori_to = to;
717
718 while ((from != from_end) && (to != to_end))
719 {
720 auto c1 = static_cast<uint32_t>(*from);
721 if (c1 < 0x80U) [[likely]]
722 {
723 ++from;
724 *to++ = static_cast<TInt>(c1);
725 }
726 else if (c1 < 0xE0U)
727 {
728 if (c1 < 0xC0U) [[unlikely]]
729 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
730 if (from_end - from < 2) break;
731 auto c2 = static_cast<uint32_t>(from[1]);
732 if ((c2 & 0xC0U) != 0x80U) [[unlikely]]
733 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
734 const auto encoded = c2 + (c1 << 6);
735 constexpr auto prefix_bias = (0xC0U << 6) + 0x80U;
736 const auto c = encoded - prefix_bias;
737 if (c < 0x80U) return std::pair{false, static_cast<std::size_t>(to - ori_to)};
738 *to++ = c;
739 from += 2;
740 }
741 else if (c1 < 0xF0U)
742 {
743 if (from_end - from < 3) break;
744 auto c2 = static_cast<uint32_t>(from[1]);
745 auto c3 = static_cast<uint32_t>(from[2]);
746 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U)) [[unlikely]]
747 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
748 const auto encoded = c3 + (c2 << 6) + (c1 << 12);
749 constexpr auto prefix_bias = (0xE0U << 12) + (0x80U << 6) + 0x80U;
750 const auto c = encoded - prefix_bias;
751 if (c < 0x800U) return std::pair{false, static_cast<std::size_t>(to - ori_to)};
752 if (c >= 0xD800U && c <= 0xDFFFU) [[unlikely]]
753 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
754 *to++ = c;
755 from += 3;
756 }
757 else if (c1 < 0xF8U)
758 {
759 if (from_end - from < 4) break;
760 auto c2 = static_cast<uint32_t>(from[1]);
761 auto c3 = static_cast<uint32_t>(from[2]);
762 auto c4 = static_cast<uint32_t>(from[3]);
763 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U) || ((c4 & 0xC0U) != 0x80U)) [[unlikely]]
764 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
765 const auto encoded = c4 + (c3 << 6) + (c2 << 12) + (c1 << 18);
766 constexpr auto prefix_bias =
767 (0xF0U << 18) + (0x80U << 12) + (0x80U << 6) + 0x80U;
768 const auto c = encoded - prefix_bias;
769 if (c < 0x10000U || c > 0x10FFFFU) return std::pair{false, static_cast<std::size_t>(to - ori_to)};
770 *to++ = c;
771 from += 4;
772 }
773 else
774 return std::pair{false, static_cast<std::size_t>(to - ori_to)};
775 }
776
777 return std::pair{true, static_cast<std::size_t>(to - ori_to)};
778 }

1070 void put_main(cvt_writer<KernelType>& writer, const internal_type* to, std::size_t to_size)
1071 requires (cvt_cpt::support_put<KernelType>)
1072 {
1073 writer.reset(s_max_buf_size);
1074 const std::size_t buf_len = m_cvt_kernel.epc();
1075 for (std::size_t i = 0; i < to_size; ++i)
1076 {
1077 external_type* out_beg = writer.put_buf(buf_len);
1078 external_type* out_next = out_beg;
1079 put_buf_guard guard{writer, buf_len};
1080
1081 internal_type ch = *to++;
1082
1083 if (!m_cvt_kernel.out_helper(ch, out_next, out_beg + buf_len))
1084 throw cvt_error("code_cvt::put fail: input character cannot be encoded");
1085
1086 guard.used(static_cast<std::size_t>(out_next - out_beg));
1087 }
1088 // commit() is the responsibility of abs_cvt::put; m_accu_len is
1089 // updated only on success so that a thrown encoding error leaves
1090 // it untouched (the converter will be tainted by abs_cvt::put,
1091 // so further reads of m_accu_len are blocked anyway).
1092 m_accu_len += to_size;
1093 }

1109 std::size_t get_main(cvt_reader<KernelType>& reader, internal_type* to, std::size_t to_max)
1110 requires (cvt_cpt::support_get<KernelType>)
1111 {
1112 if (to_max == 0) return 0;
1113 reader.reset(s_max_buf_size);
1114 std::size_t total_size = 0;
1115
1116 std::size_t prev_rollback = 0;
1117 while (total_size < to_max)
1118 {
1119 std::size_t dest_size = std::min<std::size_t>(to_max - total_size, s_max_buf_size);
1120 dest_size = std::max(dest_size, prev_rollback + 1);
1121
1122 if (dest_size > s_max_buf_size) [[unlikely]]
1123 throw cvt_error("code_cvt::get fail: input sequence too long");
1124
1125 auto [ptr, cur_size] = reader.get_buf(dest_size);
1126 if (cur_size == prev_rollback)
1127 {
1128 if (cur_size == 0) return total_size;
1129 throw cvt_error("code_cvt::get fail: partial input sequence");
1130 }
1131
1132 auto ext_cur = ptr;
1133 auto [succ, int_len] = m_cvt_kernel.in_helper(ext_cur, ptr + cur_size, to, to + to_max - total_size);
1134
1135 // Update accumulated length BEFORE the failure check: in_helper may
1136 // have written `int_len` chars into the caller's `to` buffer before
1137 // hitting an invalid byte. Throwing without this update would leave
1138 // m_accu_len inconsistent with the chars already produced.
1139 m_accu_len += int_len;
1140 total_size += int_len;
1141
1142 if (!succ)
1143 throw cvt_error("code_cvt::get fail: invalid external sequence");
1144
1145 if (ext_cur == ptr + cur_size)
1146 prev_rollback = 0;
1147 else
1148 {
1149 prev_rollback = ptr + cur_size - ext_cur;
1150 reader.rollback(prev_rollback);
1151 }
1152 }
1153 return total_size;
1154 }

1259 void seek_impl(std::size_t pos)
1260 requires (cvt_cpt::support_positioning<KernelType>)
1261 {
1262 const bool needs_state_reset =
1263 m_cvt_kernel.is_var_length() || m_cvt_kernel.is_state_dep();
1264
1265 if (needs_state_reset && (pos != 0 || BT::m_io_status == io_status::output))
1266 throw cvt_error("code_cvt::seek fail: cannot seek with dependent converter");
1267
1268 const unsigned epc = m_cvt_kernel.epc();
1269 if (pos > std::numeric_limits<std::size_t>::max() / epc)
1270 throw cvt_error("code_cvt::seek fail: position overflow");
1271
1272 // Commit lower layer first. If it throws, this object is unchanged
1273 // (strong exception guarantee). After this point the remaining
1274 // mutations are noexcept (mbstate_t reset / scalar assignment), so
1275 // we cannot end up half-updated.
1276 BT::m_kernel.seek(pos * epc);
1277
1278 if (needs_state_reset)
1279 m_cvt_kernel.init_state();
1280 m_accu_len = pos;
1281 }

1307 void rseek_impl(std::size_t pos)
1308 requires (cvt_cpt::support_positioning<KernelType>)
1309 {
1310 if (m_cvt_kernel.is_var_length() || m_cvt_kernel.is_state_dep())
1311 throw cvt_error("code_cvt::rseek fail: cannot seek with dependent converter");
1312
1313 const unsigned epc = m_cvt_kernel.epc();
1314 if (pos > std::numeric_limits<std::size_t>::max() / epc)
1315 throw cvt_error("code_cvt::rseek fail: position overflow");
1316
1317 const std::size_t saved_dev_pos = BT::m_kernel.tell();
1318 BT::m_kernel.rseek(pos * epc);
1319 const std::size_t new_dev_pos = BT::m_kernel.tell();
1320
1321 if (new_dev_pos % epc != 0)
1322 {
1323 // Restore device so this call appears atomic. If the restore
1324 // itself throws, suppress it — the caller already gets a
1325 // cvt_error and is expected to re-seek before further use;
1326 // surfacing a different exception here would only obscure the
1327 // failure mode.
1328 try { BT::m_kernel.seek(saved_dev_pos); } catch (...) {} // NOLINT(bugprone-empty-catch)
1329 throw cvt_error("code_cvt::rseek fail: partial sequence");
1330 }
1331
1332 m_accu_len = new_dev_pos / epc;
1333 m_cvt_kernel.init_state();
