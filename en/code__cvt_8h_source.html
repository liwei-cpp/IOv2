93 // The byte's value, 0 to 255 whether or not char is signed. The UTF-8 code below takes
94 // char8_t (the UTF-8 kernels) or char (the locale kernel under a UTF-8 locale on Windows):
95 // one cannot be read through a pointer to the other, char8_t lacking char's aliasing
96 // privilege, so it is instantiated for each.
97 template <typename TByte>
98 requires (std::is_same_v<TByte, char8_t> || std::is_same_v<TByte, char>)
99 constexpr uint32_t utf8_unit(TByte b) noexcept
100 {
101 return static_cast<unsigned char>(b);
102 }

121 template <typename TByte>
122 requires (std::is_same_v<TByte, char8_t> || std::is_same_v<TByte, char>)
123 constexpr utf8_status utf8_decode_one(const TByte*& from, const TByte* from_end,
124 char32_t& cp) noexcept
125 {
126 const auto c1 = utf8_unit(*from);
127 if (c1 < 0x80U) [[likely]]
128 {
129 cp = c1;
130 from += 1;
131 return utf8_status::ok;
132 }
133 if (c1 < 0xE0U)
134 {
135 if (c1 < 0xC0U) [[unlikely]]
136 return utf8_status::bad;
137 if (from_end - from < 2) return utf8_status::short_input;
138 auto c2 = utf8_unit(from[1]);
139 if ((c2 & 0xC0U) != 0x80U) [[unlikely]]
140 return utf8_status::bad;
141 const auto encoded = c2 + (c1 << 6);
142 constexpr auto prefix_bias = (0xC0U << 6) + 0x80U;
143 const auto c = encoded - prefix_bias;
144 if (c < 0x80U) return utf8_status::bad;
145 cp = c;
146 from += 2;
147 return utf8_status::ok;
148 }
149 if (c1 < 0xF0U)
150 {
151 if (from_end - from < 3) return utf8_status::short_input;
152 auto c2 = utf8_unit(from[1]);
153 auto c3 = utf8_unit(from[2]);
154 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U)) [[unlikely]]
155 return utf8_status::bad;
156 const auto encoded = c3 + (c2 << 6) + (c1 << 12);
157 constexpr auto prefix_bias = (0xE0U << 12) + (0x80U << 6) + 0x80U;
158 const auto c = encoded - prefix_bias;
159 if (c < 0x800U) return utf8_status::bad;
160 if (c >= 0xD800U && c <= 0xDFFFU) [[unlikely]]
161 return utf8_status::bad;
162 cp = c;
163 from += 3;
164 return utf8_status::ok;
165 }
166 if (c1 < 0xF8U)
167 {
168 if (from_end - from < 4) return utf8_status::short_input;
169 auto c2 = utf8_unit(from[1]);
170 auto c3 = utf8_unit(from[2]);
171 auto c4 = utf8_unit(from[3]);
172 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U) || ((c4 & 0xC0U) != 0x80U)) [[unlikely]]
173 return utf8_status::bad;
174 const auto encoded = c4 + (c3 << 6) + (c2 << 12) + (c1 << 18);
175 constexpr auto prefix_bias =
176 (0xF0U << 18) + (0x80U << 12) + (0x80U << 6) + 0x80U;
177 const auto c = encoded - prefix_bias;
178 if (c < 0x10000U || c > 0x10FFFFU) return utf8_status::bad;
179 cp = c;
180 from += 4;
181 return utf8_status::ok;
182 }
183 return utf8_status::bad;
184 }

198 template <typename TByte>
199 requires (std::is_same_v<TByte, char8_t> || std::is_same_v<TByte, char>)
200 constexpr bool utf8_encode_one(char32_t cp, TByte*& to) noexcept
201 {
202 const auto c = static_cast<uint32_t>(cp);
203 if (0xD800U <= c && c <= 0xDFFFU) [[unlikely]]
204 return false;
205
206 if (c < 0x80U)
207 *to++ = static_cast<TByte>(c);
208 else if (c <= 0x7ffU)
209 {
210 *to++ = static_cast<TByte>((c >> 6) + 0xC0U);
211 *to++ = static_cast<TByte>((c & 0x3FU) + 0x80U);
212 }
213 else if (c <= 0xFFFFU)
214 {
215 *to++ = static_cast<TByte>((c >> 12) + 0xE0U);
216 *to++ = static_cast<TByte>(((c >> 6) & 0x3FU) + 0x80U);
217 *to++ = static_cast<TByte>((c & 0x3FU) + 0x80U);
218 }
219 else if (c <= 0x10FFFFU)
220 {
221 *to++ = static_cast<TByte>((c >> 18) + 0xF0U);
222 *to++ = static_cast<TByte>(((c >> 12) & 0x3FU) + 0x80U);
223 *to++ = static_cast<TByte>(((c >> 6) & 0x3FU) + 0x80U);
224 *to++ = static_cast<TByte>((c & 0x3FU) + 0x80U);
225 }
226 else [[unlikely]]
227 return false;
228 return true;
229 }

322 bool out_helper(const wchar_t*& from, const wchar_t* from_end, TByte*& to, TByte* to_end)
323 {
324 while (from != from_end && to_end - to >= 4)
325 {
326 const auto unit = static_cast<char16_t>(*from);
327 char32_t cp = unit;
328 if (m_held != 0)
329 {
330 // The high surrogate ended the previous run.
331 const char16_t high = std::exchange(m_held, char16_t{});
332 if (!is_low(unit))
333 return false;
334 cp = combine(high, unit);
335 }
336 else if (is_high(unit))
337 {
338 if (from + 1 == from_end)
339 {
340 // The low surrogate comes with the next run.
341 m_held = unit;
342 ++from;
343 continue;
344 }
345 const auto low = static_cast<char16_t>(from[1]);
346 if (!is_low(low))
347 return false;
348 cp = combine(unit, low);
349 ++from;
350 }
351 else if (is_low(unit))
352 return false;
353
354 // A BMP character or a whole pair: never a surrogate, never above 0x10FFFF.
355 (void)utf8_encode_one(cp, to);
356 ++from;
357 }
358 return true;
359 }

382 bool in_helper(const TByte*& from, const TByte* from_end, wchar_t*& to, wchar_t* to_end)
383 {
384 if (m_held != 0 && to != to_end)
385 *to++ = static_cast<wchar_t>(std::exchange(m_held, char16_t{}));
386
387 while (from != from_end && to != to_end)
388 {
389 char32_t cp{};
390 const auto status = utf8_decode_one(from, from_end, cp);
391 if (status != utf8_status::ok)
392 return status != utf8_status::bad;
393
394 if (cp < 0x10000U)
395 {
396 *to++ = static_cast<wchar_t>(cp);
397 continue;
398 }
399 const auto v = static_cast<std::uint32_t>(cp) - 0x10000U;
400 *to++ = static_cast<wchar_t>(0xD800U + (v >> 10));
401 const auto low = static_cast<char16_t>(0xDC00U + (v & 0x3FFU));
402 if (to == to_end)
403 m_held = low; // the next call hands it over
404 else
405 *to++ = static_cast<wchar_t>(low);
406 }
407 return true;
408 }
409
410 private:
411 static constexpr bool is_high(char16_t u) noexcept { return u >= 0xD800U && u <= 0xDBFFU; }
412 static constexpr bool is_low(char16_t u) noexcept { return u >= 0xDC00U && u <= 0xDFFFU; }
413 static constexpr char32_t combine(char16_t high, char16_t low) noexcept
414 {
415 return 0x10000U + ((static_cast<char32_t>(high) - 0xD800U) << 10)
416 + (static_cast<char32_t>(low) - 0xDC00U);
417 }
418
419 char16_t m_held = 0;
420 };

546 // Defensive check: the C standard requires MB_CUR_MAX to be a positive
547 // integer, but a misconfigured CRT / exotic libc could report 0, which
548 // would lead to division-by-zero UB in seek / rseek. Reject it here so
549 // m_epc >= 1 holds as a class invariant for any constructed instance.

555 // Note: mbtowc uses internal static state, so concurrent construction of
556 // codecvt_kernel instances is NOT thread-safe. However, this is acceptable
557 // because this class does not support multi-threaded construction.
558 // Once constructed, instance-level operations use explicit mbstate_t (m_state)
559 // and are safe for single-instance usage.

561 m_is_state_dep = (std::mbtowc(nullptr, nullptr, MB_CUR_MAX) != 0);

787 bool out_helper(const TInt*& from, const TInt* from_end, char*& to, char* to_end)
788 {
789 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
790 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
791 if constexpr (s_has_utf16)

875 bool in_helper(const char*& from, const char* from_end, TInt*& to, TInt* to_end)
876 {
877 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
878 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
879 if constexpr (s_has_utf16)
880 {
881 if (m_is_utf8)
882 return m_utf16.in_helper(from, from_end, to, to_end);
883 }
884 clocale_user guard(m_inter_locale);
885 // No wchar_t value glibc decodes to: it tells a shift-only run from a character.
886 constexpr auto no_char = static_cast<wchar_t>(-1);
887
888 const auto find_nul = [&]
889 {
890 if (!m_is_state_dep || from >= from_end)
891 return from_end;
892 const void* p = std::memchr(from, 0, static_cast<std::size_t>(from_end - from));
893 return p != nullptr ? static_cast<const char*>(p) : from_end;
894 };
895 const char* nul = find_nul();
896
897 while (to < to_end && from < from_end)
898 {
899 if (nul < from)
900 nul = find_nul();
901 if (*from == '\0')
902 {
903 ++from;
904 m_state = std::mbstate_t{};
905 *to++ = TInt{};
906 continue;
907 }
908
909 const auto avail = static_cast<std::size_t>(nul - from);
910 auto tmp_state = m_state;
911 wchar_t wch = no_char;
912 std::size_t conv = mbrtowc(&wch, from, avail, &tmp_state);
913 if (m_is_state_dep && (conv == static_cast<std::size_t>(-1) || conv == static_cast<std::size_t>(-2))) // NOLINT(modernize-use-integer-sign-comparison)
914 {
915 // glibc judges a run that opens with a complete shift sequence as a
916 // whole; take that sequence off first, so the error or the incomplete
917 // sequence starts behind it.
918 for (std::size_t k = 1; k < avail; ++k)
919 {
920 auto st = m_state;
921 wchar_t w = no_char;
922 const std::size_t r = mbrtowc(&w, from, k, &st);
923 if (r == static_cast<std::size_t>(-2)) continue; // NOLINT(modernize-use-integer-sign-comparison)
924 if (r != static_cast<std::size_t>(-1) && r != 0 && w == no_char) // NOLINT(modernize-use-integer-sign-comparison)
925 {
926 conv = r;
927 tmp_state = st;
928 }
929 break;
930 }
931 }
932 if (conv == static_cast<std::size_t>(-1) || conv == 0) // NOLINT(modernize-use-integer-sign-comparison)
933 {
934 // Only tmp_state is unspecified after EILSEQ (C11 7.29.6.3.2); m_state
935 // still holds the shift state before the bad byte. conv == 0 (a NUL
936 // decoded) is rejected the same way.
937 return false;
938 }
939 if (conv == static_cast<std::size_t>(-2)) // NOLINT(modernize-use-integer-sign-comparison)
940 {
941 // A NUL inside, or longer than any character: not the start of one.
942 if (nul != from_end || avail > MB_LEN_MAX)
943 return false;
944 break; // incomplete: left at from for the caller to keep
945 }
946
947 m_state = tmp_state;
948 from += conv;
949 if (wch == no_char)
950 continue; // a shift sequence alone
951 *to++ = static_cast<TInt>(wch);
952 }
953
954 return true;
955 }
956
957private:
958 clocale_wrapper m_inter_locale;
959 unsigned m_epc = 0;
960 std::mbstate_t m_state{};
961 bool m_is_state_dep = false;
962
963 // Windows' UTF-8 locales only: whether this is one, and the conversion used there in
964 // place of mbrtowc/wcrtomb_s. Takes no room elsewhere. Where wchar_t_is_utf16 holds TInt
965 // can only be wchar_t; testing TInt anyway makes the condition depend on it, so the
966 // discarded `if constexpr` branches that call m_utf16 are not checked elsewhere.
967 static constexpr bool s_has_utf16 = wchar_t_is_utf16 && std::is_same_v<TInt, wchar_t>;
968 IOV2_NO_UNIQUE_ADDRESS std::conditional_t<s_has_utf16, detail::utf8_utf16_codec<char>, std::monostate> m_utf16{};
969 bool m_is_utf8 = false;
970};

1738 void put_main(const internal_type* to, std::size_t to_size)
1739 requires (cvt_cpt::support_put<KernelType>)
1740 {
1741 auto writer = BT::kernel_writer();
1742 writer.reset(s_max_buf_size);
1743 const std::size_t epc = m_cvt_kernel.epc();
1744 const internal_type* cur = to;
1745 const internal_type* const end = to + to_size;
1746 while (cur != end)
1747 {
1748 const std::size_t n = std::min<std::size_t>(static_cast<std::size_t>(end - cur),
1749 s_max_buf_size / epc);
1750 const std::size_t buf_len = n * epc;
1751 external_type* out_beg = writer.put_buf(buf_len);
1752 external_type* out_next = out_beg;
1753 put_buf_guard guard{writer, buf_len};
1754
1755 // Room for n * epc bytes: the kernel stops early only at a character it
1756 // cannot encode.
1757 const bool ok = m_cvt_kernel.out_helper(cur, cur + n, out_next, out_beg + buf_len);
1758 guard.used(static_cast<std::size_t>(out_next - out_beg));
1759 if (!ok)
1760 throw cvt_error("code_cvt::put fail: input character cannot be encoded");
1761 }
1762 // commit() is the responsibility of abs_cvt::put; m_accu_len is
1763 // updated only on success so that a thrown encoding error leaves
1764 // it untouched (the converter will be tainted by abs_cvt::put,
1765 // so further reads of m_accu_len are blocked anyway).
1766 m_accu_len += to_size;
1767 }

1798 std::pair<std::size_t, std::exception_ptr>
1799 get_main(internal_type* to, std::size_t to_max) noexcept
1800 requires (cvt_cpt::support_get<KernelType>)
1801 {
1802 std::size_t total_size = 0;
1803 try
1804 {
1805 if (to_max == 0) return {0, nullptr};
1806 if (m_cvt_kernel.has_pending_get())
1807 {
1808 // A character the kernel fully decoded last time but had no room for (the
1809 // low surrogate of a pair) needs no bytes, only room: an empty input hands
1810 // it over. Without this, at EOF the loop below returns before asking.
1811 const external_type* none = nullptr;
1812 const internal_type* const to_before = to;
1813 (void)m_cvt_kernel.in_helper(none, none, to, to + to_max);
1814 total_size = static_cast<std::size_t>(to - to_before);
1815 m_accu_len += total_size;
1816 }
1817 auto reader = BT::kernel_reader();
1818 reader.reset(s_max_buf_size);
1819
1820 std::size_t prev_rollback = 0;
1821 while (total_size < to_max)
1822 {
1823 std::size_t dest_size = std::min<std::size_t>(to_max - total_size, s_max_buf_size);
1824 dest_size = std::max(dest_size, prev_rollback + 1);
1825
1826 if (dest_size > s_max_buf_size) [[unlikely]]
1827 throw cvt_error("code_cvt::get fail: input sequence too long");
1828
1829 auto [ptr, cur_size] = reader.get_buf(dest_size);
1830 if (cur_size == prev_rollback)
1831 {
1832 // Nothing more below. An incomplete sequence goes back where it was:
1833 // every later get finds it again, until a reposition drops it.
1834 if (prev_rollback != 0)
1835 reader.rollback(prev_rollback);
1836 if (prev_rollback == 0 || total_size != 0)
1837 return {total_size, nullptr};
1838 throw cvt_error("code_cvt::get fail: partial input sequence");
1839 }
1840
1841 auto ext_cur = ptr;
1842 const internal_type* const to_before = to;
1843 const bool succ = m_cvt_kernel.in_helper(ext_cur, ptr + cur_size, to, to + to_max - total_size);
1844 const auto int_len = static_cast<std::size_t>(to - to_before);
1845
1846 // in_helper may have written `int_len` chars into `to` before hitting an
1847 // invalid byte: count them before anything can throw.
1848 m_accu_len += int_len;
1849 total_size += int_len;
1850
1851 if (!succ)
1852 {
1853 // ext_cur is at the start of the bad sequence. With characters to hand
1854 // over, leave it there for the next get to find and report. Otherwise
1855 // report it now and skip its first byte, as block reads always did.
1856 if (total_size != 0)
1857 {
1858 if (ext_cur != ptr + cur_size)
1859 reader.rollback(ptr + cur_size - ext_cur);
1860 return {total_size, nullptr};
1861 }
1862 const auto* resume = std::min(ext_cur + 1, ptr + cur_size);
1863 if (resume != ptr + cur_size)
1864 reader.rollback(ptr + cur_size - resume);
1865 throw cvt_error("code_cvt::get fail: invalid external sequence");
1866 }
1867
1868 if (ext_cur == ptr + cur_size)
1869 prev_rollback = 0;
1870 else
1871 {
1872 prev_rollback = ptr + cur_size - ext_cur;
1873 reader.rollback(prev_rollback);
1874 }
1875 }
1876 return {total_size, nullptr};
1877 }
1878 catch (...)
1879 {
1880 return {total_size, std::current_exception()};
1881 }
1882 }

2025 void seek_impl(std::size_t pos)
2026 requires (cvt_cpt::support_positioning<KernelType>)
2027 {

2029 m_cvt_kernel.is_var_length() || m_cvt_kernel.is_state_dep();
2030
2031 if (needs_state_reset && (pos != 0 || BT::m_io_status == io_status::output))
2032 throw cvt_error("code_cvt::seek fail: cannot seek with dependent converter");
2033

2036 throw cvt_error("code_cvt::seek fail: position overflow");
