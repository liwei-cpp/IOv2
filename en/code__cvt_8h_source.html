93 // The byte's value, 0 to 255 whether or not char is signed. The UTF-8 code below takes
94 // char8_t (the UTF-8 kernels) or char (the locale kernel under a UTF-8 locale on Windows):
95 // one cannot be read through a pointer to the other, char8_t lacking char's aliasing
96 // privilege, so it is instantiated for each.
97 template <typename TByte>
98 requires (std::is_same_v<TByte, char8_t> || std::is_same_v<TByte, char>)
99 constexpr uint32_t utf8_unit(TByte b) noexcept
100 {
101 return static_cast<unsigned char>(b);
102 }

121 template <typename TByte>
122 requires (std::is_same_v<TByte, char8_t> || std::is_same_v<TByte, char>)
123 constexpr utf8_status utf8_decode_one(const TByte*& from, const TByte* from_end,
124 char32_t& cp) noexcept
125 {
126 const auto c1 = utf8_unit(*from);
127 if (c1 < 0x80U) [[likely]]
128 {
129 cp = c1;
130 from += 1;
131 return utf8_status::ok;
132 }
133 if (c1 < 0xE0U)
134 {
135 if (c1 < 0xC0U) [[unlikely]]
136 return utf8_status::bad;
137 if (from_end - from < 2) return utf8_status::short_input;
138 auto c2 = utf8_unit(from[1]);
139 if ((c2 & 0xC0U) != 0x80U) [[unlikely]]
140 return utf8_status::bad;
141 const auto encoded = c2 + (c1 << 6);
142 constexpr auto prefix_bias = (0xC0U << 6) + 0x80U;
143 const auto c = encoded - prefix_bias;
144 if (c < 0x80U) return utf8_status::bad;
145 cp = c;
146 from += 2;
147 return utf8_status::ok;
148 }
149 if (c1 < 0xF0U)
150 {
151 if (from_end - from < 3) return utf8_status::short_input;
152 auto c2 = utf8_unit(from[1]);
153 auto c3 = utf8_unit(from[2]);
154 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U)) [[unlikely]]
155 return utf8_status::bad;
156 const auto encoded = c3 + (c2 << 6) + (c1 << 12);
157 constexpr auto prefix_bias = (0xE0U << 12) + (0x80U << 6) + 0x80U;
158 const auto c = encoded - prefix_bias;
159 if (c < 0x800U) return utf8_status::bad;
160 if (c >= 0xD800U && c <= 0xDFFFU) [[unlikely]]
161 return utf8_status::bad;
162 cp = c;
163 from += 3;
164 return utf8_status::ok;
165 }
166 if (c1 < 0xF8U)
167 {
168 if (from_end - from < 4) return utf8_status::short_input;
169 auto c2 = utf8_unit(from[1]);
170 auto c3 = utf8_unit(from[2]);
171 auto c4 = utf8_unit(from[3]);
172 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U) || ((c4 & 0xC0U) != 0x80U)) [[unlikely]]
173 return utf8_status::bad;
174 const auto encoded = c4 + (c3 << 6) + (c2 << 12) + (c1 << 18);
175 constexpr auto prefix_bias =
176 (0xF0U << 18) + (0x80U << 12) + (0x80U << 6) + 0x80U;
177 const auto c = encoded - prefix_bias;
178 if (c < 0x10000U || c > 0x10FFFFU) return utf8_status::bad;
179 cp = c;
180 from += 4;
181 return utf8_status::ok;
182 }
183 return utf8_status::bad;
184 }

198 template <typename TByte>
199 requires (std::is_same_v<TByte, char8_t> || std::is_same_v<TByte, char>)
200 constexpr bool utf8_encode_one(char32_t cp, TByte*& to) noexcept
201 {
202 const auto c = static_cast<uint32_t>(cp);
203 if (0xD800U <= c && c <= 0xDFFFU) [[unlikely]]
204 return false;
205
206 if (c < 0x80U)
207 *to++ = static_cast<TByte>(c);
208 else if (c <= 0x7ffU)
209 {
210 *to++ = static_cast<TByte>((c >> 6) + 0xC0U);
211 *to++ = static_cast<TByte>((c & 0x3FU) + 0x80U);
212 }
213 else if (c <= 0xFFFFU)
214 {
215 *to++ = static_cast<TByte>((c >> 12) + 0xE0U);
216 *to++ = static_cast<TByte>(((c >> 6) & 0x3FU) + 0x80U);
217 *to++ = static_cast<TByte>((c & 0x3FU) + 0x80U);
218 }
219 else if (c <= 0x10FFFFU)
220 {
221 *to++ = static_cast<TByte>((c >> 18) + 0xF0U);
222 *to++ = static_cast<TByte>(((c >> 12) & 0x3FU) + 0x80U);
223 *to++ = static_cast<TByte>(((c >> 6) & 0x3FU) + 0x80U);
224 *to++ = static_cast<TByte>((c & 0x3FU) + 0x80U);
225 }
226 else [[unlikely]]
227 return false;
228 return true;
229 }

322 bool out_helper(const wchar_t*& from, const wchar_t* from_end, TByte*& to, TByte* to_end)
323 {
324 while (from != from_end && to_end - to >= 4)
325 {
326 const auto unit = static_cast<char16_t>(*from);
327 char32_t cp = unit;
328 if (m_held != 0)
329 {
330 // The high surrogate ended the previous run.
331 const char16_t high = std::exchange(m_held, char16_t{});
332 if (!is_low(unit))
333 return false;
334 cp = combine(high, unit);
335 }
336 else if (is_high(unit))
337 {
338 if (from + 1 == from_end)
339 {
340 // The low surrogate comes with the next run.
341 m_held = unit;
342 ++from;
343 continue;
344 }
345 const auto low = static_cast<char16_t>(from[1]);
346 if (!is_low(low))
347 return false;
348 cp = combine(unit, low);
349 ++from;
350 }
351 else if (is_low(unit))
352 return false;
353
354 // A BMP character or a whole pair: never a surrogate, never above 0x10FFFF.
355 (void)utf8_encode_one(cp, to);
356 ++from;
357 }
358 return true;
359 }

382 bool in_helper(const TByte*& from, const TByte* from_end, wchar_t*& to, wchar_t* to_end)
383 {
384 if (m_held != 0 && to != to_end)
385 *to++ = static_cast<wchar_t>(std::exchange(m_held, char16_t{}));
386
387 while (from != from_end && to != to_end)
388 {
389 char32_t cp{};
390 const auto status = utf8_decode_one(from, from_end, cp);
391 if (status != utf8_status::ok)
392 return status != utf8_status::bad;
393
394 if (cp < 0x10000U)
395 {
396 *to++ = static_cast<wchar_t>(cp);
397 continue;
398 }
399 const auto v = static_cast<std::uint32_t>(cp) - 0x10000U;
400 *to++ = static_cast<wchar_t>(0xD800U + (v >> 10));
401 const auto low = static_cast<char16_t>(0xDC00U + (v & 0x3FFU));
402 if (to == to_end)
403 m_held = low; // the next call hands it over
404 else
405 *to++ = static_cast<wchar_t>(low);
406 }
407 return true;
408 }
409
410 private:
411 static constexpr bool is_high(char16_t u) noexcept { return u >= 0xD800U && u <= 0xDBFFU; }
412 static constexpr bool is_low(char16_t u) noexcept { return u >= 0xDC00U && u <= 0xDFFFU; }
413 static constexpr char32_t combine(char16_t high, char16_t low) noexcept
414 {
415 return 0x10000U + ((static_cast<char32_t>(high) - 0xD800U) << 10)
416 + (static_cast<char32_t>(low) - 0xDC00U);
417 }
418
419 char16_t m_held = 0;
420 };

544 // integer, but a misconfigured CRT / exotic libc could report 0, which
545 // would lead to division-by-zero UB in seek / rseek. Reject it here so
546 // m_epc >= 1 holds as a class invariant for any constructed instance.

553 // codecvt_kernel instances is NOT thread-safe. However, this is acceptable
554 // because this class does not support multi-threaded construction.

556 // and are safe for single-instance usage.

558 m_is_state_dep = (std::mbtowc(nullptr, nullptr, MB_CUR_MAX) != 0);

784 bool out_helper(const TInt*& from, const TInt* from_end, char*& to, char* to_end)
785 {
786 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
787 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
788 if constexpr (s_has_utf16)
789 {
790 if (m_is_utf8)
791 return m_utf16.out_helper(from, from_end, to, to_end);

793 clocale_user guard(m_inter_locale);

795 std::array<char, MB_LEN_MAX> local{};
796 while (from != from_end && static_cast<std::size_t>(to_end - to) >= m_epc) // NOLINT(modernize-use-integer-sign-comparison)

799 const std::size_t conv = detail::crt_wcrtomb(local, *from, m_state);

803 return false;

808 throw cvt_error("codecvt_kernel::out_helper fail: encoding exceeds MB_CUR_MAX");

872 bool in_helper(const char*& from, const char* from_end, TInt*& to, TInt* to_end)
873 {
874 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
875 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
876 if constexpr (s_has_utf16)
877 {
878 if (m_is_utf8)
879 return m_utf16.in_helper(from, from_end, to, to_end);
880 }
881 clocale_user guard(m_inter_locale);
882 // No wchar_t value glibc decodes to: it tells a shift-only run from a character.
883 constexpr auto no_char = static_cast<wchar_t>(-1);
884
885 const auto find_nul = [&]
886 {
887 if (!m_is_state_dep || from >= from_end)
888 return from_end;
889 const void* p = std::memchr(from, 0, static_cast<std::size_t>(from_end - from));
890 return p != nullptr ? static_cast<const char*>(p) : from_end;
891 };
892 const char* nul = find_nul();
893
894 while (to < to_end && from < from_end)
895 {
896 if (nul < from)
897 nul = find_nul();
898 if (*from == '\0')
899 {
900 ++from;
901 m_state = std::mbstate_t{};
902 *to++ = TInt{};
903 continue;
904 }
905
906 const auto avail = static_cast<std::size_t>(nul - from);
907 auto tmp_state = m_state;
908 wchar_t wch = no_char;
909 std::size_t conv = mbrtowc(&wch, from, avail, &tmp_state);
910 if (m_is_state_dep && (conv == static_cast<std::size_t>(-1) || conv == static_cast<std::size_t>(-2))) // NOLINT(modernize-use-integer-sign-comparison)
911 {
912 // glibc judges a run that opens with a complete shift sequence as a
913 // whole; take that sequence off first, so the error or the incomplete
914 // sequence starts behind it.
915 for (std::size_t k = 1; k < avail; ++k)
916 {
917 auto st = m_state;
918 wchar_t w = no_char;
919 const std::size_t r = mbrtowc(&w, from, k, &st);
920 if (r == static_cast<std::size_t>(-2)) continue; // NOLINT(modernize-use-integer-sign-comparison)
921 if (r != static_cast<std::size_t>(-1) && r != 0 && w == no_char) // NOLINT(modernize-use-integer-sign-comparison)
922 {
923 conv = r;
924 tmp_state = st;
925 }
926 break;
927 }
928 }
929 if (conv == static_cast<std::size_t>(-1) || conv == 0) // NOLINT(modernize-use-integer-sign-comparison)
930 {
931 // Only tmp_state is unspecified after EILSEQ (C11 7.29.6.3.2); m_state
932 // still holds the shift state before the bad byte. conv == 0 (a NUL
933 // decoded) is rejected the same way.
934 return false;
935 }
936 if (conv == static_cast<std::size_t>(-2)) // NOLINT(modernize-use-integer-sign-comparison)
937 {
938 // A NUL inside, or longer than any character: not the start of one.
939 if (nul != from_end || avail > MB_LEN_MAX)
940 return false;
941 break; // incomplete: left at from for the caller to keep
942 }
943
944 m_state = tmp_state;
945 from += conv;
946 if (wch == no_char)
947 continue; // a shift sequence alone
948 *to++ = static_cast<TInt>(wch);
949 }
950
951 return true;
952 }
953
954private:
955 clocale_wrapper m_inter_locale;
956 unsigned m_epc = 0;
957 std::mbstate_t m_state{};
958 bool m_is_state_dep = false;
959
960 // Windows' UTF-8 locales only: whether this is one, and the conversion used there in
961 // place of mbrtowc/wcrtomb_s. Takes no room elsewhere. Where wchar_t_is_utf16 holds TInt
962 // can only be wchar_t; testing TInt anyway makes the condition depend on it, so the
963 // discarded `if constexpr` branches that call m_utf16 are not checked elsewhere.
964 static constexpr bool s_has_utf16 = wchar_t_is_utf16 && std::is_same_v<TInt, wchar_t>;
965 IOV2_NO_UNIQUE_ADDRESS std::conditional_t<s_has_utf16, detail::utf8_utf16_codec<char>, std::monostate> m_utf16{};
966 bool m_is_utf8 = false;
967};

1731 void put_main(const internal_type* to, std::size_t to_size)
1732 requires (cvt_cpt::support_put<KernelType>)
1733 {
1734 auto writer = BT::kernel_writer();
1735 writer.reset(s_max_buf_size);
1736 const std::size_t epc = m_cvt_kernel.epc();
1737 const internal_type* cur = to;
1738 const internal_type* const end = to + to_size;
1739 while (cur != end)
1740 {
1741 const std::size_t n = std::min<std::size_t>(static_cast<std::size_t>(end - cur),
1742 s_max_buf_size / epc);
1743 const std::size_t buf_len = n * epc;
1744 external_type* out_beg = writer.put_buf(buf_len);
1745 external_type* out_next = out_beg;
1746 put_buf_guard guard{writer, buf_len};
1747
1748 // Room for n * epc bytes: the kernel stops early only at a character it
1749 // cannot encode.
1750 const bool ok = m_cvt_kernel.out_helper(cur, cur + n, out_next, out_beg + buf_len);
1751 guard.used(static_cast<std::size_t>(out_next - out_beg));
1752 if (!ok)
1753 throw cvt_error("code_cvt::put fail: input character cannot be encoded");
1754 }
1755 // commit() is the responsibility of abs_cvt::put; m_accu_len is
1756 // updated only on success so that a thrown encoding error leaves
1757 // it untouched (the converter will be tainted by abs_cvt::put,
1758 // so further reads of m_accu_len are blocked anyway).
1759 m_accu_len += to_size;
1760 }

1791 std::pair<std::size_t, std::exception_ptr>
1792 get_main(internal_type* to, std::size_t to_max) noexcept
1793 requires (cvt_cpt::support_get<KernelType>)
1794 {
1795 std::size_t total_size = 0;
1796 try
1797 {
1798 if (to_max == 0) return {0, nullptr};
1799 if (m_cvt_kernel.has_pending_get())
1800 {
1801 // A character the kernel fully decoded last time but had no room for (the
1802 // low surrogate of a pair) needs no bytes, only room: an empty input hands
1803 // it over. Without this, at EOF the loop below returns before asking.
1804 const external_type* none = nullptr;
1805 const internal_type* const to_before = to;
1806 (void)m_cvt_kernel.in_helper(none, none, to, to + to_max);
1807 total_size = static_cast<std::size_t>(to - to_before);
1808 m_accu_len += total_size;
1809 }
1810 auto reader = BT::kernel_reader();
1811 reader.reset(s_max_buf_size);
1812
1813 std::size_t prev_rollback = 0;
1814 while (total_size < to_max)
1815 {
1816 std::size_t dest_size = std::min<std::size_t>(to_max - total_size, s_max_buf_size);
1817 dest_size = std::max(dest_size, prev_rollback + 1);
1818
1819 if (dest_size > s_max_buf_size) [[unlikely]]
1820 throw cvt_error("code_cvt::get fail: input sequence too long");
1821
1822 auto [ptr, cur_size] = reader.get_buf(dest_size);
1823 if (cur_size == prev_rollback)
1824 {
1825 // Nothing more below. An incomplete sequence goes back where it was:
1826 // every later get finds it again, until a reposition drops it.
1827 if (prev_rollback != 0)
1828 reader.rollback(prev_rollback);
1829 if (prev_rollback == 0 || total_size != 0)
1830 return {total_size, nullptr};
1831 throw cvt_error("code_cvt::get fail: partial input sequence");
1832 }
1833
1834 auto ext_cur = ptr;
1835 const internal_type* const to_before = to;
1836 const bool succ = m_cvt_kernel.in_helper(ext_cur, ptr + cur_size, to, to + to_max - total_size);
1837 const auto int_len = static_cast<std::size_t>(to - to_before);
1838
1839 // in_helper may have written `int_len` chars into `to` before hitting an
1840 // invalid byte: count them before anything can throw.
1841 m_accu_len += int_len;
1842 total_size += int_len;
1843
1844 if (!succ)
1845 {
1846 // ext_cur is at the start of the bad sequence. With characters to hand
1847 // over, leave it there for the next get to find and report. Otherwise
1848 // report it now and skip its first byte, as block reads always did.
1849 if (total_size != 0)
1850 {
1851 if (ext_cur != ptr + cur_size)
1852 reader.rollback(ptr + cur_size - ext_cur);
1853 return {total_size, nullptr};
1854 }
1855 const auto* resume = std::min(ext_cur + 1, ptr + cur_size);
1856 if (resume != ptr + cur_size)
1857 reader.rollback(ptr + cur_size - resume);
1858 throw cvt_error("code_cvt::get fail: invalid external sequence");
1859 }
1860
1861 if (ext_cur == ptr + cur_size)
1862 prev_rollback = 0;
1863 else
1864 {
1865 prev_rollback = ptr + cur_size - ext_cur;
1866 reader.rollback(prev_rollback);
1867 }
1868 }
1869 return {total_size, nullptr};
1870 }
1871 catch (...)
1872 {
1873 return {total_size, std::current_exception()};
1874 }
1875 }

2018 void seek_impl(std::size_t pos)
2019 requires (cvt_cpt::support_positioning<KernelType>)
2020 {
2021 const bool needs_state_reset =
2022 m_cvt_kernel.is_var_length() || m_cvt_kernel.is_state_dep();
2023
2024 if (needs_state_reset && (pos != 0 || BT::m_io_status == io_status::output))
2025 throw cvt_error("code_cvt::seek fail: cannot seek with dependent converter");
2026
2027 const unsigned epc = m_cvt_kernel.epc();
2028 if (pos > std::numeric_limits<std::size_t>::max() / epc)
2029 throw cvt_error("code_cvt::seek fail: position overflow");
2030
2031 // Commit lower layer first. If it throws, this object is unchanged
2032 // (strong exception guarantee). After this point the remaining
2033 // mutations are noexcept (mbstate_t reset / scalar assignment), so
2034 // we cannot end up half-updated.
2035 BT::kernel_seek(pos * epc);
2036
2037 if (needs_state_reset)
2038 m_cvt_kernel.init_state();
2039 m_accu_len = pos;
