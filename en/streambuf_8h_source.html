365 std::size_t sgetn(char_type* s, std::size_t n, std::size_t* got = nullptr) requires (IsIn)
366 {
367 if (got) *got = 0;
368 if ((s == nullptr) || (n == 0)) return 0;
369
370 if constexpr (IsOut)
371 switch_to_get();
372
373 std::size_t res = 0;
374 while (!m_read_buf.empty())
375 {
376 *s++ = m_read_buf.back();
377 m_read_buf.pop_back();
378 if (++res == n) break;
379 }
380 if (got) *got = res;
381
382 while (res < n)
383 {
384 std::size_t c = m_cvt.get(s, n - res);
385 assert(c <= n - res);
386 if (c == 0) break;
387 res += c;
388 s += c;
389 if (got) *got = res;
390 }
391 return res;
392 }

721 void switch_to_put() requires (IsIn && IsOut)
722 {
723 if (!m_read_buf.empty())
724 {
725 try
726 {
727 const std::size_t pos = tell();
728 m_cvt.seek(pos);
729 }
730 catch (const cvt_error& e)
731 {
732 throw cvt_error("base_streambuf::switch_to_put fails: cannot reposition "
733 + std::to_string(m_read_buf.size())
734 + " buffered/put-back character(s) before switching to output mode: "
735 + e.what());
736 }
737 // Must immediately follow a *successful* seek: the seek consumed the buffer, and
738 // tell() subtracts m_read_buf.size(), so keeping it would subtract twice. A failed
739 // seek skips this and leaves the buffer intact; a throw from the switch below
740 // therefore finds it already cleared. Both are by design -- see iostream's @warning.
741 m_read_buf.clear();
742 }
743 m_cvt.switch_to_put();
744 }

806 std::pair<device_type, std::exception_ptr> detach() noexcept
807 {
808 if constexpr (IsIn)
809 {
810 if (!m_read_buf.empty())
811 {
812 try
813 {
814 // On over-putback (more put-back than actually read), tell()
815 // saturates at 0, so the device is handed back positioned at the
816 // stream origin. That is consistent with the logical read cursor
817 // being at 0; see tell()/sputbackc() and switch_to_put().
818 const std::size_t pos = tell();
819 m_cvt.seek(pos);
820 }
821 catch (...) // NOLINT(bugprone-empty-catch)
822 {
823 // Swallowed on purpose: a device that doesn't support positioning
824 // (e.g. a pipe/tty-backed stdin) can never honor this reposition,
825 // and that is an inherent property of the device, not an
826 // actionable failure - the caller could not have repositioned it
827 // either after getting it back. Reporting this error would make
828 // routine detach()/attach() cycles on such devices start throwing
829 // (e.g. sync_with_stdio() right after a formatted read has left
830 // one buffered lookahead character), even though nothing is
831 // actually broken; losing that lookahead character is the
832 // accepted, unavoidable cost for non-positionable devices.
833 }
834 m_read_buf.clear();
835 }
836 }
837 return m_cvt.detach();
838 }

916 void init_cvt()
917 {
918 auto res = m_cvt.bos();
919 m_cvt.main_cont_beg();
920 if constexpr (IsIn && !IsOut)
921 {
922 if (res != io_status::input)
923 m_cvt.switch_to_get();
924 }

927 if (res != io_status::output)
928 m_cvt.switch_to_put();
