103template <>
104struct hash<IOv2::detail::msg_key>
105{
106 std::size_t operator()(const IOv2::detail::msg_key& k) const noexcept
107 {
108 // Golden-ratio constant sized to std::size_t (2^bits / phi). The 64-bit
109 // 0x9e3779b97f4a7c15 mixes the high half of a 64-bit seed that the 32-bit
110 // 0x9e3779b9 leaves largely untouched; chosen at compile time so the mixer
111 // matches the actual width of std::size_t on every platform. (The unselected
112 // ternary branch is a well-defined explicit truncation, valid on either width.)
113 constexpr std::size_t golden = sizeof(std::size_t) >= 8
114 ? static_cast<std::size_t>(0x9e3779b97f4a7c15ULL)
115 : static_cast<std::size_t>(0x9e3779b9UL);
116
117 const std::hash<std::string> h;
118 std::size_t seed = h(k.domain);
119 seed ^= h(k.lang) + golden + (seed << 6) + (seed >> 2);
120 seed ^= std::filesystem::hash_value(k.dirname) + golden + (seed << 6) + (seed >> 2);
121 seed ^= h(k.cvt) + golden + (seed << 6) + (seed >> 2);
122 return seed;
123 }
124};
125} // namespace std

199 template <typename TF>
200 std::shared_ptr<abs_ft> try_get(const std::string& name)
201 {
202 const facet_id_t id = TF::id();
203
204 {
205 std::scoped_lock guard(m_mutex);
206 if (auto it = m_cache.find(id); it != m_cache.end())
207 {
208 if (auto cached = it->second.get(name))
209 return *cached;
210 }
211 }
212
213 // Construct the facet *outside* the lock: a facet constructor may be
214 // expensive and -- crucially -- may itself call back into ori_facet_buf, so
215 // building under m_mutex would both serialize all construction and risk a
216 // self-deadlock on this non-recursive mutex. Re-lock only to insert, and
217 // double-check: if another thread (or a re-entrant call) inserted the same
218 // (id, name) meanwhile, return that entry and discard the object built here.
219 auto obj = std::make_shared<TF>(name);
220
221 std::scoped_lock guard(m_mutex);
222 auto& sub_cache = m_cache[id];
223 if (auto cached = sub_cache.get(name))
224 return *cached;
225 // The get() above missed under this same lock, so the key is absent and put()
226 // simply inserts (no existing value to overwrite).
227 sub_cache.put(name, obj);
228 return obj;
229 }

474 static std::string resolve_locale(const char* category_var)
475 {
476 // get_env gives "" for unset and empty alike: the first one set and non-empty wins.
477 std::string name = detail::get_env("LC_ALL");
478 if (name.empty())
479 name = detail::get_env(category_var);
480 if (name.empty())
481 name = detail::get_env("LANG");
482 if (name.empty())
483 name = "C";
484
485 // Validate the resolved name against the C library: clocale_wrapper builds a
486 // locale_t via newlocale and throws if the name cannot be instantiated. An
487 // unusable name (e.g. a bogus LANG) falls back to the neutral "C". Done here,
488 // once per category at construction, so locale_name stays a plain O(1) read.
489 try
490 {
491 clocale_wrapper probe(name);
492 return name;
493 }
494 catch (const io_error&)
495 {
496 return "C";
497 }
498 }
