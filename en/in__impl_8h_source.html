392 bool sync_with_stdio(bool sync = true)
393 {
394 // Nothing to switch: return before the lock, which a blocked read may hold for long.
395 if (detail::stdin_shared_imp.synced() == sync)
396 return sync;
397
398 std::lock_guard guard(this->io_mutex());
399 const bool old_sync_state = detail::stdin_shared_imp.synced();
400 if (old_sync_state == sync)
401 return old_sync_state;
402
403 // Only the shared flag flips, which the root does at the end of the adjust, for every
404 // standard input stream: buffers, decoder and device all stay.
405 // m_channel's adjust, not this stream's: that one comes back here.
406 m_channel.adjust(stdin_sync{sync});
407 return old_sync_state;
408 }

729 {
730 // One lock over the query and the switch, so concurrent switches each return the
