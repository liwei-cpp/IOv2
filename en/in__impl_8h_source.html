283 bool sync_with_stdio(bool sync = true)
284 {
285 // Nothing to switch: return before the lock, which a blocked read may hold for long.
286 if (m_sync_with_stdio.load() == sync)
287 return sync;
288
289 std::lock_guard guard(this->io_mutex());
290 auto old_sync_state = m_sync_with_stdio.load();
291 if (old_sync_state == sync)
292 return old_sync_state;
293
294 // Only the root's flag flips: buffers, decoder and device all stay. A tainted wide
295 // converter recovers first, and that can fail -- before the flag is touched.
296 // m_channel's adjust, not this stream's: that one comes back here.
297 try {
298 m_channel.adjust(stdin_sync{sync});
299 } catch (...) {
300 this->handle_exception(std::current_exception());
301 return old_sync_state;
302 }
303 m_sync_with_stdio.store(sync);
304 return old_sync_state;
305 }

326 [[nodiscard]] bool synced_with_stdio() const noexcept

329 }

605 {
606 // One lock over the query and the switch, so concurrent switches each return the
607 // code the one before them set.

609 auto res = code();
