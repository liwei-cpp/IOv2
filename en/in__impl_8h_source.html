87 bool sync_with_stdio(bool sync = true) noexcept
88 {
89 std::lock_guard guard(this->io_mutex());
90 auto old_sync_state = m_sync_with_stdio;
91 if (old_sync_state == sync)
92 return old_sync_state;
93
94 try {
95 auto [dev, err] = m_streambuf.detach();
96 if (err) std::abort();
97 if constexpr (std::is_same_v<char_type, char>)
98 m_streambuf = istreambuf<device_type, char_type>(std::move(dev), !sync);
99 else if constexpr (std::is_same_v<char_type, wchar_t>)
100 m_streambuf = istreambuf<device_type, wchar_t>(std::move(dev), code_cvt_stdio_creator(code()), !sync);
101 else
102 static_assert(dependent_false_v<char_type>, "invalid character type");
103 } catch (...) {
104 // Memory exhaustion or a vanished locale database: no usable stream to
105 // hand back. Same policy as sing_temp::init.
106 std::abort();
107 }
108 m_sync_with_stdio = sync;
109 return old_sync_state;
110 }
