214 bool sync_with_stdio(bool sync = true)
215 {
216 std::lock_guard guard(this->io_mutex());
217 auto old_sync_state = m_sync_with_stdio.load();
218 if (old_sync_state == sync)
219 return old_sync_state;
220
221 // The decoder goes over to the new iochannel with its shift state and any
222 // half character; a fresh one would start from the initial state.
223 code_cvt_stdio_state state;
224 if constexpr (std::is_same_v<char_type, wchar_t>)
225 {
226 try {
227 m_channel.retrieve(state);
228 } catch (...) {
229 this->handle_exception(std::current_exception());
230 return old_sync_state;
231 }
232 }
233
234 auto [dev, err] = m_channel.detach();
235 try {
236 if constexpr (std::is_same_v<char_type, char>)
237 m_channel = ichannel<device_type, char_type>(std::move(dev), !sync);
238 else if constexpr (std::is_same_v<char_type, wchar_t>)
239 {
240 // Straight from the iochannel, not through code(): that wrapper reports a
241 // failure as a state bit and an empty name, which would rebuild on the
242 // environment's encoding instead of the current one.
243 code_cvt_access acc;
244 m_channel.retrieve(acc);
245 m_channel = ichannel<device_type, wchar_t>(std::move(dev), code_cvt_stdio_creator(acc.code), !sync);
246 m_channel.adjust(state);
247 }
248 else
249 static_assert(dependent_false_v<char_type>, "invalid character type");
250 } catch (...) {
251 // The iochannel was detached and the new one was never built: every
252 // operation that reads the device now fails until reset() attaches a
253 // fresh device. The flag stays where it was, so what the stream reports
254 // and how it actually reads still agree.
255 this->handle_exception(std::current_exception());
256 return old_sync_state;
257 }
258 m_sync_with_stdio.store(sync);
259 if (err) this->handle_exception(err);
260 return old_sync_state;
261 }
