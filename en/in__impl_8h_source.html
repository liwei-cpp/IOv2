200 bool sync_with_stdio(bool sync = true)
201 {
202 std::lock_guard guard(this->io_mutex());
203 auto old_sync_state = m_sync_with_stdio.load();
204 if (old_sync_state == sync)
205 return old_sync_state;
206
207 auto [dev, err] = m_streambuf.detach();
208 try {
209 if constexpr (std::is_same_v<char_type, char>)
210 m_streambuf = istreambuf<device_type, char_type>(std::move(dev), !sync);
211 else if constexpr (std::is_same_v<char_type, wchar_t>)
212 {
213 // Straight from the streambuf, not through code(): that wrapper reports a
214 // failure as a state bit and an empty name, which would rebuild on the
215 // environment's encoding instead of the current one.
216 code_cvt_access acc;
217 m_streambuf.retrieve(acc);
218 m_streambuf = istreambuf<device_type, wchar_t>(std::move(dev), code_cvt_stdio_creator(acc.code), !sync);
219 }
220 else
221 static_assert(dependent_false_v<char_type>, "invalid character type");
222 } catch (...) {
223 // The streambuf was detached and the new one was never built: every
224 // operation now fails through the state bits until reset() attaches a
225 // fresh device. The flag stays where it was, so what the stream reports
226 // and how it actually reads still agree.
227 this->handle_exception(std::current_exception());
228 return old_sync_state;
229 }
230 m_sync_with_stdio.store(sync);
231 if (err) this->handle_exception(err);
232 return old_sync_state;
233 }
