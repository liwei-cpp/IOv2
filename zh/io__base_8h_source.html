748 std::shared_ptr<void> set_pword(std::size_t id, std::shared_ptr<void> pword)
749 {
750 std::lock_guard guard(io_mutex());
751 if (auto it = m_pwords.find(id); it == m_pwords.end())

843 void access_callbacks(const locale<TChar>& new_loc)
844 {
845 std::lock_guard guard(io_mutex());
846 std::exception_ptr throw_exception = nullptr;
847
848 for (const auto& [cb, id] : m_callbacks)
849 {
850 try
851 {
852 auto pre = m_pwords.find(id);
853 std::shared_ptr<void> old_data =
854 (pre != m_pwords.end()) ? pre->second : nullptr;
855
856 std::shared_ptr<void> new_data = cb(new_loc, old_data);
857
858 // Re-locate after the callback: cb may reentrantly mutate
859 // m_pwords (via set_pword), which can rehash/erase and would
860 // invalidate an iterator held across the call.
861 auto it = m_pwords.find(id);
862 if (new_data)
863 {
864 if (it != m_pwords.end()) it->second = std::move(new_data);
865 else m_pwords.insert({id, std::move(new_data)});
866 }
867 else if (it != m_pwords.end())
868 {
869 m_pwords.erase(it);
870 }
871 }
872 catch(...)
873 {
874 if (!throw_exception)
875 throw_exception = std::current_exception();
876 }
877 }
878
879 if (throw_exception)
880 std::rethrow_exception(throw_exception);
881 }

1060 template <bool ignore_exception_mask = false>
1061 void clear(ios_defs::iostate s = ios_defs::goodbit) noexcept(ignore_exception_mask)
1062 {
1063 std::lock_guard guard(this->io_mutex());
1064 m_stream_state.store(s);
1065 if ((s & ios_defs::devfailbit) == ios_defs::goodbit) m_exp_dev_fail = std::exception_ptr{};
1066 if ((s & ios_defs::cvtfailbit) == ios_defs::goodbit) m_exp_cvt_fail = std::exception_ptr{};
1067 if ((s & ios_defs::strfailbit) == ios_defs::goodbit) m_exp_str_fail = std::exception_ptr{};
1068 if ((s & ios_defs::otherfailbit) == ios_defs::goodbit) m_exp_other_fail = std::exception_ptr{};
1069
1070 if constexpr(!ignore_exception_mask)
1071 {
1072 ios_defs::iostate state_in_exp = m_exception & s;
1073 if (state_in_exp & ios_defs::devfailbit)
1074 {
1075 if (m_exp_dev_fail)
1076 std::rethrow_exception(std::exchange(m_exp_dev_fail, nullptr));
1077 else
1078 throw device_error("device failure bit has been set");
1079 }
1080 else if (state_in_exp & ios_defs::cvtfailbit)
1081 {
1082 if (m_exp_cvt_fail)
1083 std::rethrow_exception(std::exchange(m_exp_cvt_fail, nullptr));
1084 else
1085 throw cvt_error("converter failure bit has been set");
1086 }
1087 else if (state_in_exp & ios_defs::strfailbit)
1088 {
1089 if (m_exp_str_fail)
1090 std::rethrow_exception(std::exchange(m_exp_str_fail, nullptr));
1091 else
1092 throw stream_error("stream failure bit has been set");
1093 }
1094 else if (state_in_exp & ios_defs::otherfailbit)
1095 {
1096 if (m_exp_other_fail)
1097 std::rethrow_exception(std::exchange(m_exp_other_fail, nullptr));
1098 else
1099 throw stream_error("other failure bit has been set");
1100 }
1101 else if (state_in_exp & ios_defs::eofbit)
1102 {
1103 throw eof_error{};
1104 }
1105 }

1348 template <bool ignore_exception_mask = false>
1349 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
1350 noexcept(ignore_exception_mask)
1351 {
1352 if (!ex) return;
1353 if constexpr (!ignore_exception_mask)
1354 {
1355 if (std::uncaught_exceptions() != 0)
1356 {
1357 handle_exception<true>(ex, at_eof);
1358 return;
1359 }
1360 }
1361 std::lock_guard guard(this->io_mutex());
1362 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
1363 try
1364 {
1365 std::rethrow_exception(ex);
1366 }
1367 catch (device_error&)
1368 {
1369 if (!m_exp_dev_fail)
1370 m_exp_dev_fail = std::current_exception();
1371 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
1372 }
1373 catch (cvt_error&)
1374 {
1375 if (!m_exp_cvt_fail)
1376 m_exp_cvt_fail = std::current_exception();
1377 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
1378 }
1379 catch (stream_error&)
1380 {
1381 if (!m_exp_str_fail)
1382 m_exp_str_fail = std::current_exception();
1383 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
1384 }
1385 catch (eof_error&)
1386 {
1387 setstate<ignore_exception_mask>(ios_defs::eofbit);
1388 }
1389 catch(...)
1390 {
1391 if (!m_exp_other_fail)
1392 m_exp_other_fail = std::current_exception();
1393 setstate<ignore_exception_mask>(ios_defs::otherfailbit | eof);
1394 }
1395 }
