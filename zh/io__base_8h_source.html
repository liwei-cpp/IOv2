709 void access_callbacks(const locale<TChar>& new_loc)
710 {
711 std::lock_guard guard(m_io_mutex);
712 std::exception_ptr throw_exception = nullptr;
713
714 for (const auto& [cb, id] : m_callbacks)
715 {
716 try
717 {
718 auto pre = m_pwords.find(id);
719 std::shared_ptr<void> old_data =
720 (pre != m_pwords.end()) ? pre->second : nullptr;
721
722 std::shared_ptr<void> new_data = cb(new_loc, old_data);
723
724 // Re-locate after the callback: cb may reentrantly mutate
725 // m_pwords (via set_pword), which can rehash/erase and would
726 // invalidate an iterator held across the call.
727 auto it = m_pwords.find(id);
728 if (new_data)
729 {
730 if (it != m_pwords.end()) it->second = std::move(new_data);
731 else m_pwords.insert({id, std::move(new_data)});
732 }
733 else if (it != m_pwords.end())
734 {
735 m_pwords.erase(it);
736 }
737 }
738 catch(...)
739 {
740 if (!throw_exception)
741 throw_exception = std::current_exception();
742 }
743 }
744
745 if (throw_exception)
746 std::rethrow_exception(throw_exception);
747 }

1202 template <bool ignore_exception_mask = false>
1203 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
1204 {
1205 if (!ex) return;
1206 if constexpr (!ignore_exception_mask)
1207 {
1208 if (std::uncaught_exceptions() != 0)
1209 {
1210 try { handle_exception<true>(ex, at_eof); }
1211 catch (...) {} // NOLINT(bugprone-empty-catch)
1212 return;
1213 }
1214 }
1215 std::lock_guard guard(this->io_mutex());
1216 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
1217 try
1218 {
1219 std::rethrow_exception(ex);
1220 }
1221 catch (device_error&)
1222 {
1223 if (!m_exp_dev_fail)
1224 m_exp_dev_fail = std::current_exception();
1225 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
1226 }
1227 catch (cvt_error&)
1228 {
1229 if (!m_exp_cvt_fail)
1230 m_exp_cvt_fail = std::current_exception();
1231 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
1232 }
1233 catch (stream_error&)
1234 {
1235 if (!m_exp_str_fail)
1236 m_exp_str_fail = std::current_exception();
1237 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
1238 }
1239 catch (eof_error&)
1240 {
1241 setstate<ignore_exception_mask>(ios_defs::eofbit);
1242 }
1243 catch(...)
1244 {
1245 if (!m_exp_other_fail)
1246 m_exp_other_fail = std::current_exception();
1247 setstate<ignore_exception_mask>(ios_defs::otherfailbit | eof);
1248 }
1249 }
