714 void access_callbacks(const locale<TChar>& new_loc)
715 {
716 std::lock_guard guard(m_io_mutex);
717 std::exception_ptr throw_exception = nullptr;
718
719 for (const auto& [cb, id] : m_callbacks)
720 {
721 try
722 {
723 auto pre = m_pwords.find(id);
724 std::shared_ptr<void> old_data =
725 (pre != m_pwords.end()) ? pre->second : nullptr;
726
727 std::shared_ptr<void> new_data = cb(new_loc, old_data);
728
729 // Re-locate after the callback: cb may reentrantly mutate
730 // m_pwords (via set_pword), which can rehash/erase and would
731 // invalidate an iterator held across the call.
732 auto it = m_pwords.find(id);
733 if (new_data)
734 {
735 if (it != m_pwords.end()) it->second = std::move(new_data);
736 else m_pwords.insert({id, std::move(new_data)});
737 }
738 else if (it != m_pwords.end())
739 {
740 m_pwords.erase(it);
741 }
742 }
743 catch(...)
744 {
745 if (!throw_exception)
746 throw_exception = std::current_exception();
747 }
748 }
749
750 if (throw_exception)
751 std::rethrow_exception(throw_exception);
752 }

923 template <bool ignore_exception_mask = false>
924 void clear(ios_defs::iostate s = ios_defs::goodbit)
925 {
926 std::lock_guard guard(this->io_mutex());
927 m_stream_state.store(s);
928 if ((s & ios_defs::devfailbit) == ios_defs::goodbit) m_exp_dev_fail = std::exception_ptr{};
929 if ((s & ios_defs::cvtfailbit) == ios_defs::goodbit) m_exp_cvt_fail = std::exception_ptr{};
930 if ((s & ios_defs::strfailbit) == ios_defs::goodbit) m_exp_str_fail = std::exception_ptr{};
931 if ((s & ios_defs::otherfailbit) == ios_defs::goodbit) m_exp_other_fail = std::exception_ptr{};
932
933 if constexpr(!ignore_exception_mask)
934 {
935 ios_defs::iostate state_in_exp = m_exception & s;
936 if (state_in_exp & ios_defs::devfailbit)
937 {
938 if (m_exp_dev_fail)
939 std::rethrow_exception(std::exchange(m_exp_dev_fail, nullptr));
940 else
941 throw device_error("device failure bit has been set");
942 }
943 else if (state_in_exp & ios_defs::cvtfailbit)
944 {
945 if (m_exp_cvt_fail)
946 std::rethrow_exception(std::exchange(m_exp_cvt_fail, nullptr));
947 else
948 throw cvt_error("converter failure bit has been set");
949 }
950 else if (state_in_exp & ios_defs::strfailbit)
951 {
952 if (m_exp_str_fail)
953 std::rethrow_exception(std::exchange(m_exp_str_fail, nullptr));
954 else
955 throw stream_error("stream failure bit has been set");
956 }
957 else if (state_in_exp & ios_defs::otherfailbit)
958 {
959 if (m_exp_other_fail)
960 std::rethrow_exception(std::exchange(m_exp_other_fail, nullptr));
961 else
962 throw stream_error("other failure bit has been set");
963 }
964 else if (state_in_exp & ios_defs::eofbit)
965 {
966 throw eof_error{};
967 }

1209 template <bool ignore_exception_mask = false>
1210 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
1211 {
1212 if (!ex) return;
1213 if constexpr (!ignore_exception_mask)
1214 {
1215 if (std::uncaught_exceptions() != 0)
1216 {
1217 try { handle_exception<true>(ex, at_eof); }
1218 catch (...) {} // NOLINT(bugprone-empty-catch)
1219 return;
1220 }
1221 }
1222 std::lock_guard guard(this->io_mutex());
1223 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
1224 try
1225 {
1226 std::rethrow_exception(ex);
1227 }
1228 catch (device_error&)
1229 {
1230 if (!m_exp_dev_fail)
1231 m_exp_dev_fail = std::current_exception();
1232 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
1233 }
1234 catch (cvt_error&)
1235 {
1236 if (!m_exp_cvt_fail)
1237 m_exp_cvt_fail = std::current_exception();
1238 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
1239 }
1240 catch (stream_error&)
1241 {
1242 if (!m_exp_str_fail)
1243 m_exp_str_fail = std::current_exception();
1244 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
1245 }
1246 catch (eof_error&)
1247 {
1248 setstate<ignore_exception_mask>(ios_defs::eofbit);
1249 }
1250 catch(...)
1251 {
1252 if (!m_exp_other_fail)
1253 m_exp_other_fail = std::current_exception();
1254 setstate<ignore_exception_mask>(ios_defs::otherfailbit | eof);
1255 }
1256 }
