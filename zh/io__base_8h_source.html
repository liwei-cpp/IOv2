752 void access_callbacks(const locale<TChar>& new_loc)
753 {
754 std::lock_guard guard(m_io_mutex);
755 std::exception_ptr throw_exception = nullptr;
756
757 for (const auto& [cb, id] : m_callbacks)
758 {
759 try
760 {
761 auto pre = m_pwords.find(id);
762 std::shared_ptr<void> old_data =
763 (pre != m_pwords.end()) ? pre->second : nullptr;
764
765 std::shared_ptr<void> new_data = cb(new_loc, old_data);
766
767 // Re-locate after the callback: cb may reentrantly mutate
768 // m_pwords (via set_pword), which can rehash/erase and would
769 // invalidate an iterator held across the call.
770 auto it = m_pwords.find(id);
771 if (new_data)
772 {
773 if (it != m_pwords.end()) it->second = std::move(new_data);
774 else m_pwords.insert({id, std::move(new_data)});
775 }
776 else if (it != m_pwords.end())
777 {
778 m_pwords.erase(it);
779 }
780 }
781 catch(...)
782 {
783 if (!throw_exception)
784 throw_exception = std::current_exception();
785 }
786 }
787
788 if (throw_exception)
789 std::rethrow_exception(throw_exception);
790 }

961 template <bool ignore_exception_mask = false>
962 void clear(ios_defs::iostate s = ios_defs::goodbit)
963 {
964 std::lock_guard guard(this->io_mutex());
965 m_stream_state.store(s);
966 if ((s & ios_defs::devfailbit) == ios_defs::goodbit) m_exp_dev_fail = std::exception_ptr{};
967 if ((s & ios_defs::cvtfailbit) == ios_defs::goodbit) m_exp_cvt_fail = std::exception_ptr{};
968 if ((s & ios_defs::strfailbit) == ios_defs::goodbit) m_exp_str_fail = std::exception_ptr{};
969 if ((s & ios_defs::otherfailbit) == ios_defs::goodbit) m_exp_other_fail = std::exception_ptr{};
970
971 if constexpr(!ignore_exception_mask)
972 {
973 ios_defs::iostate state_in_exp = m_exception & s;
974 if (state_in_exp & ios_defs::devfailbit)
975 {
976 if (m_exp_dev_fail)
977 std::rethrow_exception(std::exchange(m_exp_dev_fail, nullptr));
978 else
979 throw device_error("device failure bit has been set");
980 }
981 else if (state_in_exp & ios_defs::cvtfailbit)
982 {
983 if (m_exp_cvt_fail)
984 std::rethrow_exception(std::exchange(m_exp_cvt_fail, nullptr));
985 else
986 throw cvt_error("converter failure bit has been set");
987 }
988 else if (state_in_exp & ios_defs::strfailbit)
989 {
990 if (m_exp_str_fail)
991 std::rethrow_exception(std::exchange(m_exp_str_fail, nullptr));
992 else
993 throw stream_error("stream failure bit has been set");
994 }
995 else if (state_in_exp & ios_defs::otherfailbit)
996 {
997 if (m_exp_other_fail)
998 std::rethrow_exception(std::exchange(m_exp_other_fail, nullptr));
999 else
1000 throw stream_error("other failure bit has been set");
1001 }
1002 else if (state_in_exp & ios_defs::eofbit)
1003 {
1004 throw eof_error{};

1233 template <bool ignore_exception_mask = false>
1234 void handle_exception(const std::exception_ptr& ex, bool at_eof = false)
1235 {
1236 if (!ex) return;
1237 if constexpr (!ignore_exception_mask)
1238 {
1239 if (std::uncaught_exceptions() != 0)
1240 {
1241 try { handle_exception<true>(ex, at_eof); }
1242 catch (...) {} // NOLINT(bugprone-empty-catch)
1243 return;
1244 }
1245 }
1246 std::lock_guard guard(this->io_mutex());
1247 const ios_defs::iostate eof = at_eof ? ios_defs::eofbit : ios_defs::goodbit;
1248 try
1249 {
1250 std::rethrow_exception(ex);
1251 }
1252 catch (device_error&)
1253 {
1254 if (!m_exp_dev_fail)
1255 m_exp_dev_fail = std::current_exception();
1256 setstate<ignore_exception_mask>(ios_defs::devfailbit | eof);
1257 }
1258 catch (cvt_error&)
1259 {
1260 if (!m_exp_cvt_fail)
1261 m_exp_cvt_fail = std::current_exception();
1262 setstate<ignore_exception_mask>(ios_defs::cvtfailbit | eof);
1263 }
1264 catch (stream_error&)
1265 {
1266 if (!m_exp_str_fail)
1267 m_exp_str_fail = std::current_exception();
1268 setstate<ignore_exception_mask>(ios_defs::strfailbit | eof);
1269 }
1270 catch (eof_error&)
1271 {
1272 setstate<ignore_exception_mask>(ios_defs::eofbit);
1273 }
1274 catch(...)
1275 {
1276 if (!m_exp_other_fail)
1277 m_exp_other_fail = std::current_exception();
1278 setstate<ignore_exception_mask>(ios_defs::otherfailbit | eof);
1279 }
1280 }
