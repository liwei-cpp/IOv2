266 bool sync_with_stdio(bool sync = true)
267 {
268 if (!sync)
269 return m_sync_with_stdio.exchange(false);
270
271 // Flag and hand-over under one lock, so a caller that returns can rely on the
272 // bytes buffered before the call being in stdio's hands.
273 std::lock_guard guard(this->io_mutex());
274 if (m_sync_with_stdio.exchange(true))
275 return true;
276
277 try
278 {
279 // The sentry's operation, not the stream-level flush(): no fflush of bytes
280 // that are not this stream's. No good() gate: a stream in a failed state may
281 // still hold bytes (root_cvt::flush keeps them for a retry), and leaving
282 // them behind would put them after the caller's next printf. A retry that
283 // fails again only re-reports the bit that is already set.
284 m_channel.flush();
285 }
286 catch (...)
287 {
288 this->handle_exception(std::current_exception());
289 }
290 return false;
291 }

537 void reset()
538 {
539 std::lock_guard guard(this->io_mutex());
540 this->clear();
541 this->exceptions(ios_defs::goodbit);
542
543 auto detached = m_channel.detach();
544
545 try { detached.first.dflush(); }
546 catch (...)
547 {
548 if (!detached.second)
549 detached.second = std::current_exception();
550 }
551
552 try { m_channel.attach(); }
553 catch (...) { this->handle_exception(std::current_exception()); }
554
555 if (detached.second) this->handle_exception(detached.second);
556 }
