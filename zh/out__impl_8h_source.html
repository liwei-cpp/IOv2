399 bool sync_with_stdio(bool sync = true)
400 {
401 if (!sync)
402 return m_sync_with_stdio.exchange(false);
403
404 // Flag and hand-over under one lock, so a caller that returns can rely on the
405 // bytes buffered before the call being in stdio's hands. The hand-over runs even
406 // when already synchronized: a failed write leaves bytes behind in that mode too.
407 std::lock_guard guard(this->io_mutex());
408 std::exception_ptr hand_over_error;
409 try
410 {
411 // The sentry's operation, not the stream-level flush(): no fflush of bytes
412 // that are not this stream's, and nothing at all on an empty buffer. No good()
413 // gate: a stream in a failed state may still hold bytes (root_cvt::flush keeps
414 // them for a retry). A retry that fails again re-reports the bit already set.
415 m_channel.flush();
416 }
417 catch (...)
418 {
419 hand_over_error = std::current_exception();
420 }
421
422 // Only now true: synced_with_stdio() reads without the lock, and a thread that sees
423 // true may printf at once. Reported after the flip, which an armed mask would skip.
424 const bool old_sync_state = m_sync_with_stdio.exchange(true);
425 if (hand_over_error)
426 this->handle_exception(hand_over_error);
427 return old_sync_state;
428 }

566 requires std::is_same_v<TChar, wchar_t>
567 {
568 // One lock over the query and the switch, so concurrent switches each return the

570 std::lock_guard guard(this->io_mutex());
571 auto res = code();

738 void reset()
739 {
740 std::lock_guard guard(this->io_mutex());
741 this->clear();
742 this->exceptions(ios_defs::goodbit);
743
744 auto detached = m_channel.detach();
745
746 try { detached.first.dflush(); }
747 catch (...)
748 {
749 if (!detached.second)
750 detached.second = std::current_exception();
751 }
752
753 try { m_channel.attach(); }
754 catch (...) { this->handle_exception(std::current_exception()); }
755
756 if (detached.second) this->handle_exception(detached.second);
757 }
