394 bool sync_with_stdio(bool sync = true)
395 {
396 if (!sync)
397 return m_sync_with_stdio.exchange(false);
398
399 // Flag and hand-over under one lock, so a caller that returns can rely on the
400 // bytes buffered before the call being in stdio's hands. The hand-over runs even
401 // when already synchronized: a failed write leaves bytes behind in that mode too.
402 std::lock_guard guard(this->io_mutex());
403 std::exception_ptr hand_over_error;
404 try
405 {
406 // The sentry's operation, not the stream-level flush(): no fflush of bytes
407 // that are not this stream's, and nothing at all on an empty buffer. No good()
408 // gate: a stream in a failed state may still hold bytes (root_cvt::flush keeps
409 // them for a retry). A retry that fails again re-reports the bit already set.
410 m_channel.flush();
411 }
412 catch (...)
413 {
414 hand_over_error = std::current_exception();
415 }
416
417 // Only now true: synced_with_stdio() reads without the lock, and a thread that sees
418 // true may printf at once. Reported after the flip, which an armed mask would skip.
419 const bool old_sync_state = m_sync_with_stdio.exchange(true);
420 if (hand_over_error)
421 this->handle_exception(hand_over_error);
422 return old_sync_state;
423 }

560 std::string switch_code(const std::string& new_code)
561 requires std::is_same_v<TChar, wchar_t>
562 {
563 // One lock over the query and the switch, so concurrent switches each return the
564 // code the one before them set.
565 std::lock_guard guard(this->io_mutex());
566 auto res = code();
567 if (res != new_code)

569 code_cvt_switch acc(new_code);
570 this->adjust(acc);
571 }

733 void reset()
734 {
735 std::lock_guard guard(this->io_mutex());
736 this->clear();
737 this->exceptions(ios_defs::goodbit);
738
739 auto detached = m_channel.detach();
740
741 try { detached.first.dflush(); }
742 catch (...)
743 {
744 if (!detached.second)
745 detached.second = std::current_exception();
746 }
747
748 try { m_channel.attach(); }
749 catch (...) { this->handle_exception(std::current_exception()); }
750
751 if (detached.second) this->handle_exception(detached.second);
752 }
