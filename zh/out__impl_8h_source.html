29template <typename T, typename TDevice, typename TChar>
30class stdout_api : public ios_state<TChar>
31 , public out_flusher<T>
32 , public ostream_operators<TChar>
33 , public stream_common_operators
34{
35public:
36 using device_type = TDevice;
37 using char_type = TChar;
38 using out_sentry_type = out_sentry<T, false, true>;
39 using out_iter_type = ostreambuf_iterator<ostreambuf<device_type, char_type>>;
40
41 friend out_sentry_type;
42 friend out_flusher<T>;
43 friend ostream_operators<TChar>;
44 friend stream_common_operators;
45
46public:
47 stdout_api()
48 : m_streambuf(device_type{}) {}
49
50 template <cvt_creator TCreator>
51 stdout_api(const TCreator& creator)
52 : m_streambuf(device_type{}, creator) {}
53
54public:
78 bool sync_with_stdio(bool sync = true) noexcept
79 {
80 return m_sync_with_stdio.exchange(sync);
81 }
82
83 std::string code() const
84 requires std::is_same_v<TChar, wchar_t>
85 {
86 std::lock_guard guard(this->io_mutex());
87 code_cvt_access acc;
88 m_streambuf.retrieve(acc);
89 return acc.code;
90 }
91
92 std::string switch_code(const std::string& new_code)
93 requires std::is_same_v<TChar, wchar_t>
94 {
95 std::lock_guard guard(this->io_mutex());
96 auto res = code();
97 if (res != new_code)
98 {
99 code_cvt_switch acc(new_code);
100 m_streambuf.adjust(acc);
101 }
102 return res;
103 }
104
105 std::pair<device_type, std::exception_ptr> detach() = delete;
106 void attach(device_type&& dev = device_type{}) = delete;
107
108 void reset() // mainly used for unit-test
109 {
110 std::lock_guard guard(this->io_mutex());
111 this->clear();
112 this->exceptions(ios_defs::goodbit);
113 m_streambuf.attach();
114 }
115
116protected:
117 ostreambuf<device_type, char_type> m_streambuf;
118 IOv2::locale<char_type> m_locale;
119 copyable_atomic<bool> m_sync_with_stdio{true};
120};
