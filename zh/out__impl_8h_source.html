278 bool sync_with_stdio(bool sync = true)
279 {
280 if (!sync)
281 return m_sync_with_stdio.exchange(false);
282
283 // Flag and hand-over under one lock, so a caller that returns can rely on the
284 // bytes buffered before the call being in stdio's hands.
285 std::lock_guard guard(this->io_mutex());
286 if (m_sync_with_stdio.exchange(true))
287 return true;
288
289 try
290 {
291 // The sentry's operation, not the stream-level flush(): no fflush of bytes
292 // that are not this stream's. No good() gate: a stream in a failed state may
293 // still hold bytes (root_cvt::flush keeps them for a retry), and leaving
294 // them behind would put them after the caller's next printf. A retry that
295 // fails again only re-reports the bit that is already set.
296 m_channel.flush();
297 }
298 catch (...)
299 {
300 this->handle_exception(std::current_exception());
301 }
302 return false;
303 }
304
324 [[nodiscard]] bool synced_with_stdio() const noexcept
325 {
326 return m_sync_with_stdio.load();
327 }

547 void reset()
548 {
549 std::lock_guard guard(this->io_mutex());
550 this->clear();
551 this->exceptions(ios_defs::goodbit);
552
553 auto detached = m_channel.detach();
554
555 try { detached.first.dflush(); }
556 catch (...)
557 {
558 if (!detached.second)
559 detached.second = std::current_exception();
560 }
561
562 try { m_channel.attach(); }
563 catch (...) { this->handle_exception(std::current_exception()); }
564
565 if (detached.second) this->handle_exception(detached.second);
566 }
