332 bool sync_with_stdio(bool sync = true)
333 {
334 if (!sync)
335 return m_sync_with_stdio.exchange(false);
336
337 // Flag and hand-over under one lock, so a caller that returns can rely on the
338 // bytes buffered before the call being in stdio's hands. The hand-over runs even
339 // when already synchronized: a failed write leaves bytes behind in that mode too.
340 std::lock_guard guard(this->io_mutex());
341 std::exception_ptr hand_over_error;
342 try
343 {
344 // The sentry's operation, not the stream-level flush(): no fflush of bytes
345 // that are not this stream's, and nothing at all on an empty buffer. No good()
346 // gate: a stream in a failed state may still hold bytes (root_cvt::flush keeps
347 // them for a retry). A retry that fails again re-reports the bit already set.
348 m_channel.flush();
349 }
350 catch (...)
351 {
352 hand_over_error = std::current_exception();
353 }
354
355 // Only now true: synced_with_stdio() reads without the lock, and a thread that sees
356 // true may printf at once. Reported after the flip, which an armed mask would skip.
357 const bool old_sync_state = m_sync_with_stdio.exchange(true);
358 if (hand_over_error)
359 this->handle_exception(hand_over_error);
360 return old_sync_state;
361 }

389 [[nodiscard]] bool synced_with_stdio() const noexcept

391 return m_sync_with_stdio.load();
392 }

498 std::string switch_code(const std::string& new_code)
499 requires std::is_same_v<TChar, wchar_t>
500 {
501 // One lock over the query and the switch, so concurrent switches each return the
502 // code the one before them set.
503 std::lock_guard guard(this->io_mutex());
504 auto res = code();
505 if (res != new_code)
506 {
507 code_cvt_switch acc(new_code);
508 this->adjust(acc);
509 }
510 return res;
511 }

558 void adjust(const cvt_behavior& acc)
559 {
560 // Through T& so the call depends on T: gcc 15 resolves a this-> call at the definition
561 // and, not applying the constraints there yet, calls the two overloads ambiguous.
562 static_cast<T&>(*this).std_stream_common_operators::adjust(acc);
563 }
564
654 void reset()
655 {
656 std::lock_guard guard(this->io_mutex());
657 this->clear();
658 this->exceptions(ios_defs::goodbit);
659
660 auto detached = m_channel.detach();
661
662 try { detached.first.dflush(); }
663 catch (...)
664 {
665 if (!detached.second)
666 detached.second = std::current_exception();
667 }
668
669 try { m_channel.attach(); }
670 catch (...) { this->handle_exception(std::current_exception()); }
671
672 if (detached.second) this->handle_exception(detached.second);
673 }
674
675protected:
676 // The exit hook. Marked first, so that an insertion after this hook (a destructor or atexit
677 // function registered earlier, or another thread) hands its bytes to stdio and flushes it. Not
678 // m_sync_with_stdio: try_flush() may move nothing, and sync_with_stdio(true) reads a true
679 // flag as "already handed over".
680 void flush_at_exit() noexcept
681 {
682 m_exited.store(true);
683 this->try_flush();
684 }

687 IOv2::locale<char_type> m_locale;
688 copyable_atomic<bool> m_sync_with_stdio{true};
689 copyable_atomic<bool> m_exited{false};
