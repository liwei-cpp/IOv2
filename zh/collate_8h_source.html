158 template <std::input_iterator TIter1, std::input_iterator TIter2>
159 [[nodiscard]] std::weak_ordering compare(TIter1 low1, TIter1 high1, TIter2 low2, TIter2 high2) const
160 {
161 std::vector<CharT> buf1;
162 std::vector<CharT> buf2;
163
164 while ((low1 != high1) && (low2 != high2))
165 {
166 auto [seg1, null1] = next_segment(low1, high1, buf1);
167 auto [seg2, null2] = next_segment(low2, high2, buf2);
168
169 if (auto res = m_obj->compare(seg1, seg2); res != 0)
170 return res;
171 // The side whose segment did not end at a null is out of input, so it is less.
172 if (null1 != null2)
173 return null1 <=> null2;
174 }
175
176 if (low1 != high1) return std::weak_ordering::greater;
177 if (low2 != high2) return std::weak_ordering::less;
178 return std::weak_ordering::equivalent;
179 }

311 template <std::input_iterator TIter, std::output_iterator<CharT> TOut>
312 std::pair<TOut, std::size_t> transform(TIter low, TIter high, TOut dest, std::size_t n) const
313 {
314 // 0 used to mean "unlimited"; reject it so old-style calls fail loudly.
315 if (n == 0)
316 throw io_error("collate::transform: n must be greater than 0");
317
318 std::size_t written = 0;
319 std::vector<CharT> buf; // input staging for next_segment
320 std::vector<CharT> key; // one segment's key, plus the '\0' strxfrm appends
321
322 while ((low != high) && (written < n))
323 {
324 auto [seg, ended_by_null] = next_segment(low, high, buf);
325 std::size_t len = m_obj->transform_length(seg);
326 key.resize(len + 1);
327 m_obj->transform(seg, key.data(), key.size());
328
329 std::size_t cnt = std::min(len, n - written);
330 dest = std::copy_n(key.data(), cnt, dest);
331 written += cnt;
332
333 if (ended_by_null && (written < n))
334 {
335 *dest++ = static_cast<CharT>(0);
336 ++written;
337 }
338 }
339 return {dest, written};
340 }

381 template <typename TIter>
382 static std::pair<const CharT*, bool> next_segment(TIter& low, TIter high, std::vector<CharT>& buf)
383 {
384 // Terminating null, then zero padding so glibc's 32-byte SIMD reads stay inside buf.
385 auto terminate_and_pad = [&buf]
386 {
387 buf.resize(buf.size() + 1 + SIMD_PADDING_BYTES / sizeof(CharT), static_cast<CharT>(0));
388 };
389
390 if constexpr (std::contiguous_iterator<TIter> &&
391 std::is_same_v<std::iter_value_t<TIter>, CharT>)
392 {
393 const CharT* first = std::to_address(low);
394 const CharT* last = first + (high - low);
395 if (const CharT* eos = std::find(first, last, static_cast<CharT>(0)); eos != last)
396 {
397 low += (eos - first) + 1;
398 return {first, true};
399 }
400 buf.assign(first, last);
401 low = high;
402 terminate_and_pad();
403 return {buf.data(), false};
404 }
405 else
406 {
407 buf.clear();
408 bool ended_by_null = false;
409 while (low != high)
410 {
411 CharT ch = *low++;
412 if (ch == static_cast<CharT>(0))
413 {
414 ended_by_null = true;
415 break;
416 }
417 buf.push_back(ch);
418 }
419 terminate_and_pad();
420 return {buf.data(), ended_by_null};
421 }
422 }
