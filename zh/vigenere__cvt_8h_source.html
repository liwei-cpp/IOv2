282 std::size_t get_main(cvt_reader<KernelType>& reader, internal_type* to, std::size_t to_max)
283 requires (cvt_cpt::support_get<KernelType>)
284 {
285 if (to_max == 0) return 0;
286 if (m_key.empty())
287 throw cvt_error("vigenere_cvt: key not initialized (moved-from object?)");
288
289 std::size_t total_count = 0;
290 reader.reset(s_buf_len);
291 while (total_count != to_max)
292 {
293 const std::size_t dest_size = std::min(s_buf_len, to_max - total_count);
294 auto [ptr, cur_size] = reader.get_buf(dest_size);
295 if (cur_size == 0)
296 return total_count;
297
298 for (std::size_t i = 0; i < cur_size; ++i)
299 {
300 std::size_t pos = (m_pos++) % m_key.size();
301 const internal_type src = *ptr++;
302 *to++ = sub_wrap(src, m_key[pos]);
303 }
304 total_count += cur_size;
305 }
306 return total_count;
307 }
