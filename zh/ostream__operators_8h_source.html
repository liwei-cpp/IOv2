132 out_sentry(TStream& os, bool is_unit_buf, bool is_app_mode)
133 : m_os(os)
134 , m_is_unit_buf(is_unit_buf)
135 {
136 if (!static_cast<bool>(m_os))
137 throw stream_error("ostream_sentry create fail: Invalid ostream");
138
139 if (auto* tied = m_os.tie())
140 tied->try_flush();
141
142 if constexpr (is_std)
143 m_sync_with_stdio = os.m_sync_with_stdio.load();
144
145 if constexpr (!is_std)
146 {
147 if (is_app_mode)
148 {
149 try
150 {
151 m_os.m_channel.rseek(0);
152 }
153 catch (const cvt_error& e)
154 {
155 throw cvt_error(std::string("ostream_sentry create fail: appmode cannot "
156 "reposition to the end; appmode requires a "
157 "fixed-length, state-independent encoding: ")
158 + e.what());
159 }
160 }
161 }
162
163 if constexpr (involve_input)
164 os.m_channel.switch_to_put();
165
166 if (!static_cast<bool>(m_os))
167 throw stream_error("ostream_sentry create fail: Invalid ostream");
168 }

204 ~out_sentry() noexcept(false)
205 {
206 try
207 {
208 if (m_os)
209 {
210 auto exited = [&]() noexcept {
211 if constexpr (is_std) return m_os.m_exited.load();
212 else return false;
213 };
214 if (m_is_unit_buf || m_sync_with_stdio || exited())
215 m_os.m_channel.flush();
216
217 // After the exit hook, stdio's own last flush may already be behind us:
218 // glibc leaves a stdout it never used fully buffered, so push the bytes out.
219 if (m_is_unit_buf || exited())
220 m_os.m_channel.device().dflush();
221 }
222 }
223 catch (...)
224 {
225 m_os.handle_exception(std::current_exception());
226 }
227 }

230 out_sentry& operator=(const out_sentry&) = delete;
231 out_sentry(out_sentry&&) = delete;
232 out_sentry& operator=(out_sentry&&) = delete;

424template <typename T>
425struct out_tie_target : public tie_target
426{
486 void try_flush() noexcept final
487 {
488 // Downstream first, before taking our own lock; tie() keeps the chain acyclic.
489 if (auto ptr = tied_to(); ptr != nullptr)
490 ptr->try_flush();
491
492 constexpr int attempts = 3;
493
494 T& obj = static_cast<T&>(*this);
495 for (int i = 0; i < attempts; ++i)
496 {
497 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
498 if (lk)
499 {
500 try
501 {
502 obj.flush();
503 }
504 catch (...)
505 {
506 // Record on the tie target, never on the initiator, and never rethrow.
507 // <true> keeps the stored exception alive; see the note above.
508 obj.template handle_exception<true>(std::current_exception());
509 }
510 return;
511 }
512 if (i + 1 < attempts)
513 std::this_thread::yield();
514 }
515 }
516
522 [[nodiscard]] tie_target* tied_to() const noexcept final
523 {
524 return static_cast<const T&>(*this).tie();
525 }

591template <typename T, typename TKey, typename TArg = TKey>
592concept insertable_with_stream = ostream_type<T> &&
593 requires(T& obj, const TArg& value) { io_traits<typename T::char_type, TKey>::swrite(obj, value); };

610template <typename T, typename TKey, typename TArg = TKey>
611concept insertable_with_iter = ostream_type<T> &&

636 ostream_type<T> &&

639 insertable_with_iter<T, TValue> ||

968template <ostream_type T, typename TValue>
969 requires detail::insertable<T, TValue>
970T& operator<<(T& obj, const TValue& value)
971{
972 using TChar = typename T::char_type;
973 using TDecay = std::decay_t<const TValue&>;
974
975 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
976 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
977 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay, TValue>;
978 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay, TValue>;
979
980 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
981 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
982 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
983 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
984 "See io/traits/traits_base.h.");
985
986 if constexpr (stream_v || stream_d)
987 {
988 try
989 {
990 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
991 }
992 catch (...)
993 {
994 obj.handle_exception(std::current_exception());
995 }
996 }
997 else if constexpr (iter_v || iter_d)
998 {
999 using sentry_type = typename T::out_sentry_type;
1000 std::lock_guard guard(obj.io_mutex());
1001 try
1002 {
1003 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
1004 bool(obj.flags() & ios_defs::appmode));
1005
1006 auto iter = obj.o_iter();
1007 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
1008 iter, obj, obj.locale(), value);
1009 }
1010 catch (...)
1011 {
1012 obj.handle_exception(std::current_exception());
1013 }
1014 }
1015 else
1016 static_assert(dependent_false_v<TValue>,
1017 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
1018 "branch for it. The concept and the chain must stay in step -- see "
1019 "IOv2/io/utilities/ostream_operators.h.");
1020
1021 return obj;
1022}
