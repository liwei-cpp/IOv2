402 void try_flush() override
403 {
404 constexpr int attempts = 3;
405
406 T& obj = static_cast<T&>(*this);
407 for (int i = 0; i < attempts; ++i)
408 {
409 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
410 if (lk)
411 {
412 obj.flush();
413 return;
414 }
415 if (i + 1 < attempts)
416 std::this_thread::yield();
417 }
418 }

851template <ostream_type T, typename TValue>
852 requires detail::insertable<T, TValue>
853T& operator<<(T& obj, const TValue& value)
854{
855 using TChar = typename T::char_type;
856 using TDecay = std::decay_t<TValue>;
857
858 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
859 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
860 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay>;
861 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay>;
862
863 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
864 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
865 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
866 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
867 "See io/traits/traits_base.h.");
868
869 if constexpr (stream_v || stream_d)
870 {
871 try
872 {
873 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
874 }
875 catch (...)
876 {
877 obj.handle_exception(std::current_exception());
878 }
879 }
880 else if constexpr (iter_v || iter_d)
881 {
882 using sentry_type = typename T::out_sentry_type;
883 std::lock_guard guard(obj.io_mutex());
884 try
885 {
886 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
887 bool(obj.flags() & ios_defs::appmode));
888
889 auto iter = obj.o_iter();
890 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
891 iter, obj, obj.locale(), value);
892 }
893 catch (...)
894 {
895 obj.handle_exception(std::current_exception());
896 }
897 }
898 else
899 static_assert(dependent_false_v<TValue>,
900 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
901 "branch for it. The concept and the chain must stay in step -- see "
902 "io/utilities/ostream_operators.h.");
903
904 return obj;
905}
