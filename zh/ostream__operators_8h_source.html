128 out_sentry(TStream& os, bool is_unit_buf, bool is_app_mode)
129 : m_os(os)
130 , m_is_unit_buf(is_unit_buf)
131 {
132 if (!static_cast<bool>(m_os))
133 throw stream_error("ostream_sentry create fail: Invalid ostream");
134
135 if (auto* tied = m_os.tie())
136 tied->try_flush();
137

143

157 + e.what());
158 }
159 }
160 }
161

163 throw stream_error("ostream_sentry create fail: Invalid ostream");
164 }

371template <typename T>
372struct out_flusher : public abs_flusher
373{
434 void try_flush() noexcept override
435 {
436 constexpr int attempts = 3;
437
438 T& obj = static_cast<T&>(*this);
439 for (int i = 0; i < attempts; ++i)
440 {
441 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
442 if (lk)
443 {
444 try
445 {
446 obj.flush();
447 }
448 catch (...)
449 {
450 // Record on the tie target, never on the initiator, and never rethrow.
451 // <true> keeps the stored exception alive; see the note above.
452 obj.template handle_exception<true>(std::current_exception());
453 }
454 return;
455 }
456 if (i + 1 < attempts)
457 std::this_thread::yield();
458 }
459 }

547template <typename T, typename TKey, typename TArg = TKey>
548concept insertable_with_stream = ostream_type<T> &&
549 requires(T& obj, const TArg& value) { io_traits<typename T::char_type, TKey>::swrite(obj, value); };

566template <typename T, typename TKey, typename TArg = TKey>
567concept insertable_with_iter = ostream_type<T> &&
568 requires(T& obj, typename T::out_iter_type& iter, const TArg& value)
569 { io_traits<typename T::char_type, TKey>::swrite(iter, obj, obj.locale(), value); };

590template <typename T, typename TValue>
591concept insertable =
592 ostream_type<T> &&
593 (insertable_with_stream<T, TValue> ||
594 insertable_with_stream<T, std::decay_t<const TValue&>, TValue> ||
595 insertable_with_iter<T, TValue> ||
596 insertable_with_iter<T, std::decay_t<const TValue&>, TValue>);

950template <ostream_type T, typename TValue>
951 requires detail::insertable<T, TValue>
952T& operator<<(T& obj, const TValue& value)
953{
954 using TChar = typename T::char_type;
955 using TDecay = std::decay_t<const TValue&>;
956
957 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
958 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
959 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay, TValue>;
960 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay, TValue>;
961
962 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
963 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
964 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
965 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
966 "See io/traits/traits_base.h.");
967
968 if constexpr (stream_v || stream_d)
969 {
970 try
971 {
972 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
973 }
974 catch (...)
975 {
976 obj.handle_exception(std::current_exception());
977 }
978 }
979 else if constexpr (iter_v || iter_d)
980 {
981 using sentry_type = typename T::out_sentry_type;
982 std::lock_guard guard(obj.io_mutex());
983 try
984 {
985 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
986 bool(obj.flags() & ios_defs::appmode));
987
988 auto iter = obj.o_iter();
989 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
990 iter, obj, obj.locale(), value);
991 }
992 catch (...)
993 {
994 obj.handle_exception(std::current_exception());
995 }
996 }
997 else
998 static_assert(dependent_false_v<TValue>,
999 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
1000 "branch for it. The concept and the chain must stay in step -- see "
1001 "IOv2/io/utilities/ostream_operators.h.");
1002
1003 return obj;
1004}
