125 out_sentry(TStream& os, bool is_unit_buf, bool is_app_mode)
126 : m_os(os)
127 , m_is_unit_buf(is_unit_buf)
128 {
129 if (!static_cast<bool>(m_os))
130 throw stream_error("ostream_sentry create fail: Invalid ostream");
131
132 if (auto* tied = m_os.tie())
133 tied->try_flush();
134
135 if constexpr (is_std)
136 m_sync_with_stdio = os.m_sync_with_stdio.load();
137
138 if constexpr (involve_input)
139 os.m_channel.switch_to_put();
140
141 if constexpr (!is_std)
142 {
143 if (is_app_mode)
144 {
145 try
146 {
147 m_os.m_channel.rseek(0);
148 }
149 catch (const cvt_error& e)
150 {
151 throw cvt_error(std::string("ostream_sentry create fail: appmode cannot "
152 "reposition to the end; appmode requires a "
153 "fixed-length, state-independent encoding: ")
154 + e.what());
155 }
156 }
157 }
158
159 if (!static_cast<bool>(m_os))
160 throw stream_error("ostream_sentry create fail: Invalid ostream");
161 }
162
197 ~out_sentry() noexcept(false)
198 {
199 try
200 {
201 if (m_os)
202 {
203 auto exited = [&]() noexcept {
204 if constexpr (is_std) return m_os.m_exited.load();
205 else return false;
206 };
207 if (m_is_unit_buf || m_sync_with_stdio || exited())
208 m_os.m_channel.flush();
209
210 // After the exit hook, stdio's own last flush may already be behind us:
211 // glibc leaves a stdout it never used fully buffered, so push the bytes out.
212 if (m_is_unit_buf || exited())
213 m_os.m_channel.device().dflush();
214 }
215 }
216 catch (...)
217 {
218 m_os.handle_exception(std::current_exception());
219 }
220 }

413template <typename T>
414struct out_tie_target : public tie_target
415{
461 void try_flush() noexcept final
462 {
463 constexpr int attempts = 3;
464
465 T& obj = static_cast<T&>(*this);
466 for (int i = 0; i < attempts; ++i)
467 {
468 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
469 if (lk)
470 {
471 try
472 {
473 obj.flush();
474 }
475 catch (...)
476 {
477 // Record on the tie target, never on the initiator, and never rethrow.
478 // <true> keeps the stored exception alive; see the note above.
479 obj.template handle_exception<true>(std::current_exception());
480 }
481 return;
482 }
483 if (i + 1 < attempts)
484 std::this_thread::yield();
485 }
486 }
487
493 [[nodiscard]] tie_target* tied_to() const noexcept final
494 {
495 return static_cast<const T&>(*this).tie();
496 }

562template <typename T, typename TKey, typename TArg = TKey>
563concept insertable_with_stream = ostream_type<T> &&
564 requires(T& obj, const TArg& value) { io_traits<typename T::char_type, TKey>::swrite(obj, value); };

581template <typename T, typename TKey, typename TArg = TKey>
582concept insertable_with_iter = ostream_type<T> &&
583 requires(T& obj, typename T::out_iter_type& iter, const TArg& value)
584 { io_traits<typename T::char_type, TKey>::swrite(iter, obj, obj.locale(), value); };

605template <typename T, typename TValue>
606concept insertable =
607 ostream_type<T> &&
608 (insertable_with_stream<T, TValue> ||
609 insertable_with_stream<T, std::decay_t<const TValue&>, TValue> ||
610 insertable_with_iter<T, TValue> ||
611 insertable_with_iter<T, std::decay_t<const TValue&>, TValue>);

939template <ostream_type T, typename TValue>
940 requires detail::insertable<T, TValue>
941T& operator<<(T& obj, const TValue& value)
942{
943 using TChar = typename T::char_type;
944 using TDecay = std::decay_t<const TValue&>;
945
946 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
947 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
948 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay, TValue>;
949 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay, TValue>;
950
951 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
952 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
953 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
954 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
955 "See io/traits/traits_base.h.");
956
957 if constexpr (stream_v || stream_d)
958 {
959 try
960 {
961 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
962 }
963 catch (...)
964 {
965 obj.handle_exception(std::current_exception());
966 }
967 }
968 else if constexpr (iter_v || iter_d)
969 {
970 using sentry_type = typename T::out_sentry_type;
971 std::lock_guard guard(obj.io_mutex());
972 try
973 {
974 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
975 bool(obj.flags() & ios_defs::appmode));
976
977 auto iter = obj.o_iter();
978 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
979 iter, obj, obj.locale(), value);
980 }
981 catch (...)
982 {
983 obj.handle_exception(std::current_exception());
984 }
985 }
986 else
987 static_assert(dependent_false_v<TValue>,
988 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
989 "branch for it. The concept and the chain must stay in step -- see "
990 "IOv2/io/utilities/ostream_operators.h.");
991
992 return obj;
993}
