886template <ostream_type T, typename TValue>
887 requires (is_writer_def<typename T::char_type, TValue>
888 || is_writer_def<typename T::char_type, std::decay_t<TValue>>)
889T& operator<<(T& obj, const TValue& value)
890{
891 using TDecay = std::decay_t<TValue>;
892 using TChar = typename T::char_type;
893 std::unique_lock lk(obj.io_mutex(), std::defer_lock);
894 try
895 {
896 using sentry_type = typename T::out_sentry_type;
897 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf), bool(obj.flags() & ios_defs::appmode), lk);
898
899 if constexpr (is_writer_def<TChar, TValue>)
900 writer<TChar, TValue>::swrite(obj.o_iter(), obj, obj.locale(), value);
901 else if constexpr (is_writer_def<TChar, TDecay>)
902 writer<TChar, TDecay>::swrite(obj.o_iter(), obj, obj.locale(), value);
903 else
904 static_assert(dependent_false_v<TValue>, "No format method provided");
905 }
906 catch(...)
907 {
908 obj.handle_exception(std::current_exception());
909 }
910 return obj;
911}
