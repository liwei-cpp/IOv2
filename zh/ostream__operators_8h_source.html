124 out_sentry(TStream& os, bool is_unit_buf, bool is_app_mode)
125 : m_os(os)
126 , m_is_unit_buf(is_unit_buf)
127 {
128 if (!static_cast<bool>(m_os))
129 throw stream_error("ostream_sentry create fail: Invalid ostream");
130
131 if (auto* tied = m_os.tie())
132 tied->try_flush();
133
134 if constexpr (is_std)
135 m_sync_with_stdio = os.m_sync_with_stdio.load();
136
137 if constexpr (involve_input)
138 os.m_channel.switch_to_put();
139
140 if constexpr (!is_std)
141 {
142 if (is_app_mode)
143 {
144 try
145 {
146 m_os.m_channel.rseek(0);
147 }
148 catch (const cvt_error& e)
149 {
150 throw cvt_error(std::string("ostream_sentry create fail: appmode cannot "
151 "reposition to the end; appmode requires a "
152 "fixed-length, state-independent encoding: ")
153 + e.what());
154 }
155 }
156 }
157
158 if (!static_cast<bool>(m_os))
159 throw stream_error("ostream_sentry create fail: Invalid ostream");
160 }
161
191 ~out_sentry() noexcept(false)
192 {
193 try
194 {
195 if (m_os)
196 {
197 if (m_is_unit_buf || m_sync_with_stdio)
198 m_os.m_channel.flush();
199
200 if (m_is_unit_buf)
201 m_os.m_channel.device().dflush();
202 }
203 }
204 catch (...)
205 {
206 m_os.handle_exception(std::current_exception());
207 }
208 }

355template <typename T>
356struct out_flusher : public abs_flusher
357{
403 void try_flush() noexcept override
404 {
405 constexpr int attempts = 3;
406
407 T& obj = static_cast<T&>(*this);
408 for (int i = 0; i < attempts; ++i)
409 {
410 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
411 if (lk)
412 {
413 try
414 {
415 obj.flush();
416 }
417 catch (...)
418 {
419 // Record on the tie target, never on the initiator, and never rethrow.
420 // <true> keeps the stored exception alive; see the note above.
421 obj.template handle_exception<true>(std::current_exception());
422 }
423 return;
424 }
425 if (i + 1 < attempts)
426 std::this_thread::yield();
427 }
428 }

871template <ostream_type T, typename TValue>
872 requires detail::insertable<T, TValue>
873T& operator<<(T& obj, const TValue& value)
874{
875 using TChar = typename T::char_type;
876 using TDecay = std::decay_t<const TValue&>;
877
878 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
879 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
880 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay, TValue>;
881 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay, TValue>;
882
883 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
884 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
885 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
886 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
887 "See io/traits/traits_base.h.");
888
889 if constexpr (stream_v || stream_d)
890 {
891 try
892 {
893 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
894 }
895 catch (...)
896 {
897 obj.handle_exception(std::current_exception());
898 }
899 }
900 else if constexpr (iter_v || iter_d)
901 {
902 using sentry_type = typename T::out_sentry_type;
903 std::lock_guard guard(obj.io_mutex());
904 try
905 {
906 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
907 bool(obj.flags() & ios_defs::appmode));
908
909 auto iter = obj.o_iter();
910 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
911 iter, obj, obj.locale(), value);
912 }
913 catch (...)
914 {
915 obj.handle_exception(std::current_exception());
916 }
917 }
918 else
919 static_assert(dependent_false_v<TValue>,
920 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
921 "branch for it. The concept and the chain must stay in step -- see "
922 "IOv2/io/utilities/ostream_operators.h.");
923
924 return obj;
925}
