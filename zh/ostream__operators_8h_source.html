368template <typename T>
369struct out_flusher : public abs_flusher

431 void try_flush() noexcept override
432 {
433 constexpr int attempts = 3;
434
435 T& obj = static_cast<T&>(*this);
436 for (int i = 0; i < attempts; ++i)
437 {
438 std::unique_lock lk(obj.io_mutex(), std::try_to_lock);
439 if (lk)
440 {
441 try
442 {
443 obj.flush();
444 }
445 catch (...)
446 {
447 // Record on the tie target, never on the initiator, and never rethrow.
448 // <true> keeps the stored exception alive; see the note above.
449 obj.template handle_exception<true>(std::current_exception());
450 }
451 return;
452 }
453 if (i + 1 < attempts)
454 std::this_thread::yield();
455 }
456 }

947template <ostream_type T, typename TValue>
948 requires detail::insertable<T, TValue>
949T& operator<<(T& obj, const TValue& value)
950{
951 using TChar = typename T::char_type;
952 using TDecay = std::decay_t<const TValue&>;
953
954 constexpr bool stream_v = detail::insertable_with_stream<T, TValue>;
955 constexpr bool iter_v = detail::insertable_with_iter<T, TValue>;
956 constexpr bool stream_d = detail::insertable_with_stream<T, TDecay, TValue>;
957 constexpr bool iter_d = detail::insertable_with_iter<T, TDecay, TValue>;
958
959 static_assert(!(stream_v && iter_v) && !(stream_d && iter_d),
960 "IOv2: this io_traits specialization provides both forms of swrite() -- the stream form "
961 "swrite(stream, value) and the iterator form swrite(iter, io, loc, value). Provide "
962 "exactly one: the iterator form for formatted output, the stream form for a manipulator. "
963 "See io/traits/traits_base.h.");
964
965 if constexpr (stream_v || stream_d)
966 {
967 try
968 {
969 io_traits<TChar, std::conditional_t<stream_v, TValue, TDecay>>::swrite(obj, value);
970 }
971 catch (...)
972 {
973 obj.handle_exception(std::current_exception());
974 }
975 }
976 else if constexpr (iter_v || iter_d)
977 {
978 using sentry_type = typename T::out_sentry_type;
979 std::lock_guard guard(obj.io_mutex());
980 try
981 {
982 sentry_type cerb(obj, bool(obj.flags() & ios_defs::unitbuf),
983 bool(obj.flags() & ios_defs::appmode));
984
985 auto iter = obj.o_iter();
986 io_traits<TChar, std::conditional_t<iter_v, TValue, TDecay>>::swrite(
987 iter, obj, obj.locale(), value);
988 }
989 catch (...)
990 {
991 obj.handle_exception(std::current_exception());
992 }
993 }
994 else
995 static_assert(dependent_false_v<TValue>,
996 "IOv2 internal: detail::insertable admitted this type but the dispatch chain has no "
997 "branch for it. The concept and the chain must stay in step -- see "
998 "io/utilities/ostream_operators.h.");
999
1000 return obj;
1001}
