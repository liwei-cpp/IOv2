426 iostream& operator=(iostream&& other) noexcept
427 {
428 if (this == &other) return *this;
429
430 std::lock_guard guard(this->io_mutex());
431 // NOLINTBEGIN(bugprone-use-after-move): each `std::move(other)` below binds to a
432 // *different* base subobject, and every one of those base assignments touches only its
433 // own members; `m_streambuf` and `m_locale` belong to no base at all. Nothing is read
434 // after being moved from -- clang-tidy just cannot see that the operands are disjoint.
435 ios_state<TChar>::operator=(std::move(other));
436 istream_operators<TChar>::operator=(std::move(other));
437 out_flusher<iostream<TDevice, TChar>>::operator=(std::move(other));
438 ostream_operators<TChar>::operator=(std::move(other));
439 stream_common_operators::operator=(std::move(other));
440 m_streambuf = std::move(other.m_streambuf);
441 m_locale = std::move(other.m_locale);
442 // NOLINTEND(bugprone-use-after-move)
443 return *this;
444 }
