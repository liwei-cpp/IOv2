236 template <typename TIter>
237 void add(TIter b, TIter e, TValue v)
238 {
239 node* node_ptr = &m_root;
240 while (b != e)
241 {
242 CharT ch = *b++;
243
244 auto child = node_ptr->children.find(ch);
245 if (child == node_ptr->children.end())
246 {
247 bool insert_suc = false;
248 auto c = std::make_unique<node>(std::nullopt, node_ptr->depth + 1);
249 std::tie(child, insert_suc) = node_ptr->children.insert(std::pair(ch, std::move(c)));
250 if (!insert_suc)
251 throw std::runtime_error("prefix_tree insert fails");
252 }
253 node_ptr = child->second.get();
254 }
255
256 if (node_ptr->val.has_value() && *node_ptr->val != v)
257 throw std::runtime_error("duplicate items in prefix_tree");
258 node_ptr->val = v;
259 }

296 template <steppable_back TIter, std::sentinel_for<TIter> TSent>
297 [[nodiscard]] TIter max_match(TIter b, TSent e, match_out_type& out) const
298 {
299 // Debug assertion: check for valid range (only for random access iterators)
300 if constexpr (std::random_access_iterator<TIter> && std::same_as<TIter, TSent>)
301 assert(b <= e && "max_match: invalid iterator range (b > e)");
302
303 if constexpr (is_small_type_v<TValue>)
304 out = std::nullopt;
305 else
306 out = nullptr;
307
308 if (m_root.val.has_value())
309 {
310 if constexpr (is_small_type_v<TValue>)
311 out = m_root.val;
312 else
313 out = &(*m_root.val);
314 }
315
316 std::size_t found_depth = 0;
317
318 const node* node_ptr = &m_root;
319 for (; b != e; ++b)
320 {
321 auto ch_real = *b;
322 CharT ch = static_cast<CharT>(ch_real);
323 // The input range may be spelled in a wider character type than the one the tree
324 // is keyed on (see the note on TIter): a character that does not survive the round
325 // trip is not one of the tree's keys, so it can only end the match. Note the
326 // comparison is written as a round trip rather than as `ch_real == ch`, which
327 // would promote both to int and so declare char(-1) and its own byte unequal.
328 if constexpr (!std::same_as<CharT, std::iter_value_t<TIter>>)
329 {
330 if (static_cast<std::iter_value_t<TIter>>(ch) != ch_real) break;
331 }
332 auto it = node_ptr->children.find(ch);
333 if (it == node_ptr->children.end()) break;
334 node_ptr = it->second.get();
335 if (node_ptr->val.has_value())
336 {
337 if constexpr (is_small_type_v<TValue>)
338 out = node_ptr->val;
339 else
340 out = &(*node_ptr->val);
341 found_depth = node_ptr->depth;
342 }
343 }
344
345 if (node_ptr->depth != found_depth)
346 {
347 const std::size_t steps_back = node_ptr->depth - found_depth;
348 // Not std::advance / std::ranges::advance: the former dispatches on the C++17
349 // iterator_category, the latter requires the bidirectional_iterator concept, and a
350 // negative distance violates the precondition of both for a single-pass iterator
351 // that merely offers operator-- (see steppable_back). `--` is the primitive every
352 // type admitted here provides; random access keeps its O(1) path explicitly.
353 if constexpr (std::random_access_iterator<TIter>)
354 b -= static_cast<std::iter_difference_t<TIter>>(steps_back);
355 else
356 for (std::size_t k = 0; k < steps_back; ++k) --b;
357 }
358 return b;
359 }

390 template <is_istreambuf_iterator TIter, std::sentinel_for<TIter> TSent>
391 [[nodiscard]] TIter max_match(TIter b, TSent e, match_out_type& out) const
392 {
393 if constexpr (is_small_type_v<TValue>)
394 out = std::nullopt;
395 else
396 out = nullptr;
397
398 std::forward_list<CharT> checking_chars;
399
400 if (m_root.val.has_value())
401 {
402 if constexpr (is_small_type_v<TValue>)
403 out = m_root.val;
404 else
405 out = &(*m_root.val);
406 }
407
408 std::size_t found_depth = 0;
409
410 const node* node_ptr = &m_root;
411 for (; b != e;)
412 {
413 auto ch_real = *b;
414 CharT ch = static_cast<CharT>(ch_real);
415 // Same round-trip guard as the steppable_back overload; see the note there.
416 // Breaking here leaves the offending character unread, since `++b` is below.
417 if constexpr (!std::same_as<CharT, std::iter_value_t<TIter>>)
418 {
419 if (static_cast<std::iter_value_t<TIter>>(ch) != ch_real) break;
420 }
421 auto it = node_ptr->children.find(ch);
422 if (it == node_ptr->children.end()) break;
423 node_ptr = it->second.get();
424 if (node_ptr->val.has_value())
425 {
426 if constexpr (is_small_type_v<TValue>)
427 out = node_ptr->val;
428 else
429 out = &(*node_ptr->val);
430 found_depth = node_ptr->depth;
431 }
432 checking_chars.push_front(ch);
433 ++b;
434 }
435
436 if (node_ptr->depth != found_depth)
437 {
438 std::size_t count = node_ptr->depth - found_depth;
439 for (std::size_t i = 0; i < count; ++i)
440 {
441 b.sputbackc(checking_chars.front());
442 checking_chars.pop_front();
443 }
444 }
445 return b;
446 }
