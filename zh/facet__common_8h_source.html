53// ─── Facet type-identity key (single source of truth) ───────────────────────
54//
55// Every facet type gets a process-stable key of type `facet_id_t`, used to key
56// facets into the locale caches (`locale::m_facet_confs` / `m_facets`) and into
57// `ori_facet_buf`. Two modes:
58//
59// - Header-only (single binary): the key is the *address* of a per-type static
60// (`type_id<T>::s_id`). Zero collision, constant-initialised (safe to read
61// during static init), and effectively free. Within one binary the linker
62// merges all copies of that static, so every TU agrees on the address.
63//
64// - Shared library (`IOV2_SHARED`): the same facet type is instantiated in
65// several DSOs, and `-fvisibility=hidden` keeps each DSO's `s_id` local -- so
66// an address is per-DSO and breaks cross-DSO facet identity (that was the
67// "cannot get numeric facet" bug). Use `std::type_index` instead: identity is
68// by *type* (`type_info::operator==`), not by object address, so it is stable
69// across DSO / `dlopen(RTLD_LOCAL)` / DLL boundaries and collision-free
70// (`hash_code` only buckets; `operator==` decides the match).
71//
72// This relies on `type_info` comparison being correct across DSOs, which every
73// honest platform provides: on ELF/dlopen targets comparison is name-string
74// based (the default), and targets that instead compare type_info by address
75// do so only where the ABI guarantees a single merged type_info per type. The
76// only way to break it is to force pointer-only type_info comparison (e.g.
77// `-D__GXX_MERGED_TYPEINFO_NAMES=1`) on a platform that does not actually merge
78// names -- a deliberate misconfiguration that also breaks any other
79// `type_index`/RTTI-based code, so IOv2 does not police it here.
80#if defined(IOV2_SHARED)
81using facet_id_t = std::type_index;
82#else
83using facet_id_t = std::size_t;
84
85// Per-type storage whose *address* is the (header-only) type key. Only needed
86// when the key is an address; in shared mode the key comes from `typeid` instead.
87template <typename T>
88struct type_id
89{
90 inline static const void* s_id = nullptr;
91};
92
93// The address key is std::bit_cast<size_t>(pointer); bit_cast is well-formed only
94// when sizeof(To) == sizeof(From), so the pointer and size_t must be equal in size.
95static_assert(sizeof(void*) == sizeof(std::size_t),
96 "facet_id_t address keying uses std::bit_cast<size_t>(pointer), which "
97 "is well-formed only when sizeof(void*) == sizeof(size_t)");
98#endif

112template <typename T>
113inline facet_id_t type_id_v() noexcept
114{

117#else
118 return std::bit_cast<std::size_t>(&type_id<T>::s_id);
119#endif
120}
