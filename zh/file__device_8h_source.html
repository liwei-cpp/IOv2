164 explicit basic_file_device(const std::string& file_name, file_open_flag flags = file_open_flag::none)
165 {
166 const char* mode = fopen_mode(flags);
167 m_file.reset(std::fopen(file_name.c_str(), mode));
168 if (!m_file)
169 throw device_error("cannot open file \"" + file_name + "\" (errno=" + std::to_string(errno) + ")");
170
171 if (fseek_64(m_file.get(), 0, SEEK_END) != 0)
172 throw device_error("cannot get file length for \"" + file_name + "\"");
173
174 int64_t len = ftell_64(m_file.get());
175 if (len < 0)
176 throw device_error("cannot get file length for \"" + file_name + "\"");
177 else if (!std::in_range<std::size_t>(len))
178 throw device_error("file too large for this platform");
179
180 if (fseek_64(m_file.get(), 0, SEEK_SET) != 0)
181 throw device_error("cannot reset the file for \"" + file_name + "\"");
182
183 m_file_len = static_cast<std::size_t>(len);
184 }

430 void dseek(std::size_t v)
431 {
432 if (!is_open())
433 throw device_error("file_device::dseek fail: file is closed");
434
435 // In input-enabled modes, seeking is restricted to the current file length.
436 // This ensures that any subsequent read operation (dget) starts from a valid
437 // position within the existing data bounds.
438 if constexpr (IsIn)
439 {
440 if (v > m_file_len)
441 throw device_error("file_device::dseek fail: invalid parameter");
442 }
443
444 if (v > std::numeric_limits<int64_t>::max())
445 throw device_error("file_device::dseek fail: position exceeds maximum seekable range");
446
447 if (fseek_64(m_file.get(), static_cast<int64_t>(v), SEEK_SET) != 0)
448 throw device_error("file_device::dseek fail: fseek invalid response");
449 }

493 void dput(const CharType* ch, std::size_t n)
494 requires (IsOut)
495 {
496 if (n == 0) return;
497 if (ch == nullptr)
498 throw device_error("file_device::dput fail: null buffer");
499 if (!is_open())
500 throw device_error("file_device::dput fail: file is closed");
501 std::size_t written = std::fwrite(ch, sizeof(CharType), n, m_file.get());
502
503 auto res = ftell_64(m_file.get());
504 if (res >= 0)
505 {
506 if (!std::in_range<std::size_t>(res))
507 throw device_error("file_device::dput fail: file size overflow");
508 if (static_cast<std::size_t>(res) > m_file_len)
509 m_file_len = static_cast<std::size_t>(res);
510 }
511 else
512 throw device_error("file_device::dput fail: cannot determine file position");
513
514 if (written != n)
515 throw device_error("file_device::dput fail: partial write");
516 }

541private:
542 static const char* fopen_mode(file_open_flag flags)
543 {
544 using enum file_open_flag;
545
546 file_open_flag checkres = flags & (trunc | binary | noreplace);
547
548 if constexpr (IsIn && IsOut)
549 {
550 if (checkres == none)
551 return "r+";
552 else if (checkres == trunc)
553 return "w+";
554 else if (checkres == binary)
555 return "r+b";
556 else if (checkres == (trunc | binary))
557 return "w+b";
558 // noreplace (O_EXCL) implies creating a new file, so it's always empty.
559 // Therefore, trunc is implicit in this mode.
560 else if (checkres == noreplace || checkres == (trunc | noreplace))
561 return "w+x";
562 else if (checkres == (binary | noreplace) || checkres == (binary | trunc | noreplace))
563 return "w+bx";
564 else
565 throw device_error("Invalid open mode combination for read-write file");
566 }
567 else if constexpr (IsIn)
568 {
569 if (checkres == none)
570 return "r";
571 else if (checkres == binary)
572 return "rb";
573 else
574 throw device_error("Invalid open mode combination for read-only file");
575 }
576 else if constexpr (IsOut)
577 {
578 // In write-only mode, "w" already truncates by default if noreplace is not present.
579 if (checkres == none || checkres == trunc)
580 return "w";
581 else if (checkres == binary || checkres == (trunc | binary))
582 return "wb";
583 // noreplace implies creation of a new file.
584 else if (checkres == noreplace || checkres == (trunc | noreplace))
585 return "wx";
586 else if (checkres == (binary | noreplace) || checkres == (binary | trunc | noreplace))
587 return "wbx";
588 else
589 throw device_error("Invalid open mode combination for write-only file");
590 }
591 else
592 {
593 static_assert(dependent_false_nttp_v<IsIn, IsOut>, "file device cannot open with neither in nor out mode");
594 }
595 }

616private:
617 std::unique_ptr<FILE, file_deleter> m_file;
618
619 // Cached file length, in bytes.
620 //
621 // Type is `size_t` (not `uint64_t`) because the public API — `dsize()`,
622 // `dseek(size_t)`, `drseek(size_t)` — is `size_t`-typed; using `size_t`
623 // here keeps the value lossless across the boundary on every platform
624 // (including 32-bit, where a `uint64_t` field could hold values that
625 // `dsize()` cannot return without truncation).
626 //
627 // Invariant: `m_file_len <= INT64_MAX`. All writes to this field source
628 // from a non-negative `int64_t` (the result of `ftell_64()`), narrowed
629 // through `std::in_range<size_t>` in the constructor and `dput()`.
630 // This invariant is what makes `-static_cast<int64_t>(offset)` in
631 // `drseek()` safe: any `offset <= m_file_len` fits in `int64_t`'s
632 // positive range, so the negation cannot overflow. Do not introduce
633 // setters or alternate sources that bypass the `int64_t`-derived path.
634 std::size_t m_file_len = 0;
