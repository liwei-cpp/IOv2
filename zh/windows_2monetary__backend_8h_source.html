90// The monetary data of the calling thread's current locale. Call it under a clocale_user.
91inline monetary_raw_punct monetary_read_punct()
92{
93 const auto str = [](const char* s) { return s ? std::string(s) : std::string(); };
94 const auto wstr = [](const wchar_t* s) { return s ? std::wstring(s) : std::wstring(); };
95 const lconv* lc = std::localeconv();
96 monetary_raw_punct res;
97 res.int_curr_symbol = str(lc->int_curr_symbol);
98 res.currency_symbol = str(lc->currency_symbol);
99 res.mon_decimal_point = str(lc->mon_decimal_point);
100 res.mon_thousands_sep = str(lc->mon_thousands_sep);
101 res.mon_grouping = str(lc->mon_grouping);
102 res.positive_sign = str(lc->positive_sign);
103 res.negative_sign = str(lc->negative_sign);
104 res.int_frac_digits = lc->int_frac_digits;
105 res.frac_digits = lc->frac_digits;
106 res.nat_pos = {.cs_precedes = lc->p_cs_precedes, .sep_by_space = lc->p_sep_by_space,
107 .sign_posn = lc->p_sign_posn};
108 res.nat_neg = {.cs_precedes = lc->n_cs_precedes, .sep_by_space = lc->n_sep_by_space,
109 .sign_posn = lc->n_sign_posn};
110 res.int_pos = res.nat_pos;
111 res.int_neg = res.nat_neg;
112 res.wint_curr_symbol = wstr(lc->_W_int_curr_symbol);
113 res.wcurrency_symbol = wstr(lc->_W_currency_symbol);
114 res.wmon_decimal_point = wstr(lc->_W_mon_decimal_point);
115 res.wmon_thousands_sep = wstr(lc->_W_mon_thousands_sep);
116 res.wpositive_sign = wstr(lc->_W_positive_sign);
117 res.wnegative_sign = wstr(lc->_W_negative_sign);
118 return res;
119}
