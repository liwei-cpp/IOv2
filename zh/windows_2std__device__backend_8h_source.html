54// Whether a read would return at once. A disk file never blocks; a pipe does
55// unless it holds data or its write end is gone (the read then returns 0).
56// A console cannot be asked without consuming input events, so the answer
57// there is "no", which costs only a short read.
58inline bool std_fd_readable_now(int fd) noexcept
59{
60 // -1 is INVALID_HANDLE_VALUE; -2 means fd 0 has no stream (a GUI process).
61 const std::intptr_t raw = _get_osfhandle(fd);
62 if (raw == -1 || raw == -2)
63 return false;
64 void* const h = reinterpret_cast<void*>(raw);
65
66 switch (GetFileType(h))
67 {
68 case win_file_type_disk:
69 return true;
70 case win_file_type_pipe:
71 {
72 unsigned long avail = 0;
73 if (PeekNamedPipe(h, nullptr, 0, nullptr, &avail, nullptr))
74 return avail != 0;
75 return GetLastError() == win_error_broken_pipe;
76 }
77 default:
78 return false;
79 }
80}

82inline std::size_t std_fd_read(int fd, char* s, std::size_t n)
83{
84 assert((s != nullptr) && (n != 0));
85
86 constexpr auto max_read = static_cast<std::size_t>(INT_MAX);
87 const int ret = _read(fd, s, static_cast<unsigned int>((std::min)(n, max_read)));
88 if (ret == -1)
89 throw device_error("std_device::fd_read fail: read error");
90 return static_cast<std::size_t>(ret);
91}

93// Holds the lock of stdin's FILE, a CRITICAL_SECTION and so recursive: stdio
94// calls made while it is held, locking ones included, do not deadlock.
95class stdin_file_lock
96{
97public:
98 stdin_file_lock() noexcept { _lock_file(stdin); }
99 ~stdin_file_lock() { _unlock_file(stdin); }
100 stdin_file_lock(const stdin_file_lock&) = delete;
101 stdin_file_lock& operator=(const stdin_file_lock&) = delete;
102};

104// Reads n bytes through stdin's FILE and, like fread(), waits until all have
105// come or the stream ends. 0 means end of file, FILE's sticky flag: until a
106// clearerr(stdin) every call returns 0 at once. On an error the bytes already
107// read are returned first, and the next call meets the error again and throws.
108inline std::size_t std_stdin_file_read(char* s, std::size_t n)
109{
110 assert((s != nullptr) && (n != 0));
111
112 stdin_file_lock guard;
113 if (std::feof(stdin)) return 0;
114 // A stale error flag must not pass for one raised by this read.
115 std::clearerr(stdin);
116
117 const std::size_t got = std::fread(s, 1, n, stdin);
118 if (got == n || !std::ferror(stdin) || got != 0) return got;
119 throw device_error("std_device stdin FILE read fail: read error");
120}

122// Whether stdin is at end of file, waiting for a byte if need be. A byte read
123// to find out goes back with ungetc(), which guarantees one byte of pushback;
124// the lock keeps another thread from reading in between.
125inline bool std_stdin_file_eof()
126{
127 stdin_file_lock guard;
128 char c = 0;
129 if (std_stdin_file_read(&c, 1) == 0) return true;
130 if (std::ungetc(static_cast<unsigned char>(c), stdin) == EOF)
131 throw device_error("std_device::deof fail: ungetc error");
132 return false;
133}

135// Pushing bytes back is refused. The UCRT keeps no pushback area: ungetc() steps
136// back over what was already read from the current buffer, and at its start it
137// either fails or, with unread bytes there, overwrites the next one and still
138// reports success. Whether there is room cannot be asked -- FILE is opaque --
139// and the deof() probe above is safe only because it pushes back the byte it
140// has just read under the same lock.
141template <std::bidirectional_iterator It>
142bool std_stdin_file_unget(It, It)
143{
144 return false;
145}
