508template <istream_type T, typename TMoney>
509inline T& operator>>(T& is, _Get_money<TMoney> f)
510{
511 // Delegate to the generic formatted-extraction operator: it owns the sentry, the
512 // skipws handling, the stream iterator (i_iter is private and befriends only that
513 // template) and the eofbit bookkeeping. The explicit template argument list is
514 // required rather than stylistic -- `is >> f` would re-select *this* overload,
515 // because f is a named lvalue and the by-value candidate is the more specialized
516 // one, and recurse forever.
517 return IOv2::operator>><T, _Get_money<TMoney>>(is, f);
518}

663template <typename TChar>
664struct reader<TChar, _Get_time<TChar>>
665{
693 template <typename TIter, std::sentinel_for<TIter> TSent>
694 requires (std::is_same_v<TChar, typename TIter::value_type>)
695 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc, _Get_time<TChar>& f)
696 {
697 if (f.tmb == nullptr || f.fmt == nullptr)
698 throw stream_error("get_time fail: null tm or format pointer");
699
700 auto mp = loc.template get<timeio<TChar>>();
701 if (!mp)
702 throw stream_error("cannot get timeio facet");
703
704 // The context is date+time without a time zone, matching
705 // parse_context_type<TChar, std::tm>; that is also the combination whose
706 // explicit operator std::tm() is available.
707 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
708 auto res = mp->get(s, s_end, tmp, f.fmt);
709 *(f.tmb) = static_cast<std::tm>(tmp);
710
711 return res;
712 }
713};
