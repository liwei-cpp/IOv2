1259 requires (std::is_same_v<TChar, typename TIter::value_type>
1260 && (steppable_back<TIter> || is_istreambuf_iterator<TIter>))
1261 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc,
1262 const get_time_t<TChar>& f)
1263 {
1264 if (f.tmb == nullptr || f.fmt == nullptr)
1265 throw stream_error("get_time fail: null tm or format pointer");
1266
1267 auto mp = loc.template get<timeio<TChar>>();
1268 if (!mp)
1269 throw stream_error("cannot get timeio facet");
1270
1271 // The context is date+time at the offset tier, matching
1272 // parse_context_type<TChar, std::tm>; that is also the combination whose
1273 // convert_to(std::tm&) is available.
1274 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
1275 auto res = mp->get(s, s_end, tmp, f.fmt);
1276
1277 tmp.convert_to(*(f.tmb));
1278
1279 return res;
1280 }
