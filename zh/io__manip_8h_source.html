999 && (steppable_back<TIter> || is_istreambuf_iterator<TIter>))
1000 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc,
1001 const get_time_t<TChar>& f)

1003 if (f.tmb == nullptr || f.fmt == nullptr)
1004 throw stream_error("get_time fail: null tm or format pointer");
1005
1006 auto mp = loc.template get<timeio<TChar>>();
1007 if (!mp)
1008 throw stream_error("cannot get timeio facet");
1009
1010 // The context is date+time without a time zone, matching
1011 // parse_context_type<TChar, std::tm>; that is also the combination whose
1012 // convert_to(std::tm&) is available.
1013 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
1014 auto res = mp->get(s, s_end, tmp, f.fmt);
1015
1016 tmp.convert_to(*(f.tmb));
1017
1018 return res;
