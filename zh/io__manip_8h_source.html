1076 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc,

1083 if (!mp)

1085
1086 // The context is date+time without a time zone, matching
1087 // parse_context_type<TChar, std::tm>; that is also the combination whose

1090 auto res = mp->get(s, s_end, tmp, f.fmt);

1092 tmp.convert_to(*(f.tmb));
