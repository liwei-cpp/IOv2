839{
875 template <typename TIter, std::sentinel_for<TIter> TSent>
876 requires (std::is_same_v<TChar, typename TIter::value_type>)
877 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc,
878 const _Get_time<TChar>& f)
879 {
880 if (f.tmb == nullptr || f.fmt == nullptr)
881 throw stream_error("get_time fail: null tm or format pointer");
882
883 auto mp = loc.template get<timeio<TChar>>();
884 if (!mp)
885 throw stream_error("cannot get timeio facet");
886
887 // The context is date+time without a time zone, matching
888 // parse_context_type<TChar, std::tm>; that is also the combination whose
889 // explicit operator std::tm() is available.
890 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
891 auto res = mp->get(s, s_end, tmp, f.fmt);
892 *(f.tmb) = static_cast<std::tm>(tmp);
893
894 return res;
895 }
