265template <istream_type T, typename TMoney>
266inline T& operator>>(T& is, _Get_money<TMoney> f)
267{
268 // Delegate to the generic formatted-extraction operator: it owns the sentry, the
269 // skipws handling, the stream iterator (i_iter is private and befriends only that
270 // template) and the eofbit bookkeeping. The explicit template argument list is
271 // required rather than stylistic -- `is >> f` would re-select *this* overload,
272 // because f is a named lvalue and the by-value candidate is the more specialized
273 // one, and recurse forever.
274 return operator>><T, _Get_money<TMoney>>(is, f);
275}
