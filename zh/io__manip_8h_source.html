950struct io_traits<TChar, get_time_t<TChar>>
951{
987 template <typename TIter, std::sentinel_for<TIter> TSent>
988 requires (std::is_same_v<TChar, typename TIter::value_type>)
989 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc,
990 const get_time_t<TChar>& f)
991 {
992 if (f.tmb == nullptr || f.fmt == nullptr)
993 throw stream_error("get_time fail: null tm or format pointer");
994
995 auto mp = loc.template get<timeio<TChar>>();
996 if (!mp)
997 throw stream_error("cannot get timeio facet");
998
999 // The context is date+time without a time zone, matching
1000 // parse_context_type<TChar, std::tm>; that is also the combination whose
1001 // convert_to(std::tm&) is available.
1002 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
1003 auto res = mp->get(s, s_end, tmp, f.fmt);
1004
1005 tmp.convert_to(*(f.tmb));
1006
1007 return res;
1008 }
