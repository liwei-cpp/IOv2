422template <istream_type T, typename TMoney>
423inline T& operator>>(T& is, _Get_money<TMoney> f)
424{
425 // Delegate to the generic formatted-extraction operator: it owns the sentry, the
426 // skipws handling, the stream iterator (i_iter is private and befriends only that
427 // template) and the eofbit bookkeeping. The explicit template argument list is
428 // required rather than stylistic -- `is >> f` would re-select *this* overload,
429 // because f is a named lvalue and the by-value candidate is the more specialized
430 // one, and recurse forever.
431 return IOv2::operator>><T, _Get_money<TMoney>>(is, f);
432}

607 template <typename TIter, std::sentinel_for<TIter> TSent>
608 requires (std::is_same_v<TChar, typename TIter::value_type>)
609 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc, _Get_time<TChar>& f)
610 {
611 if (f.tmb == nullptr || f.fmt == nullptr)
612 throw stream_error("get_time fail: null tm or format pointer");
613
614 auto mp = loc.template get<timeio<TChar>>();
615 if (!mp)
616 throw stream_error("cannot get timeio facet");
617
618 // The context is date+time without a time zone, matching
619 // parse_context_type<TChar, std::tm>; that is also the combination whose
620 // explicit operator std::tm() is available.
621 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
622 auto res = mp->get(s, s_end, tmp, f.fmt);
623 *(f.tmb) = static_cast<std::tm>(tmp);
624
625 return res;
626 }
