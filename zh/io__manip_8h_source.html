1613 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc,
1614 const get_time_t<TChar>& f)

1618
1619 auto mp = loc.template get<timeio<TChar>>();
1620 if (!mp)
1621 throw stream_error("cannot get timeio facet");
1622
1623 // The context is date+time at the offset tier, matching
1624 // parse_context_type<TChar, std::tm>; that is also the combination whose
1625 // convert_to(std::tm&) is available.

1627 auto res = mp->get(s, s_end, tmp, f.fmt);
1628
1629 tmp.convert_to(*(f.tmb));
1630
1631 return res;
