1207 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc,
1208 const get_time_t<TChar>& f)

1210 if (f.tmb == nullptr || f.fmt == nullptr)
1211 throw stream_error("get_time fail: null tm or format pointer");
1212
1213 auto mp = loc.template get<timeio<TChar>>();
1214 if (!mp)
1215 throw stream_error("cannot get timeio facet");
1216
1217 // The context is date+time without a time zone, matching
1218 // parse_context_type<TChar, std::tm>; that is also the combination whose
1219 // convert_to(std::tm&) is available.
1220 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
1221 auto res = mp->get(s, s_end, tmp, f.fmt);
1222
1223 tmp.convert_to(*(f.tmb));
1224
1225 return res;
