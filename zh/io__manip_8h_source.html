1564{
1615 template <typename TIter, std::sentinel_for<TIter> TSent>
1616 requires (std::is_same_v<TChar, typename TIter::value_type>
1617 && (steppable_back<TIter> || is_ichannel_iterator<TIter>))
1618 static TIter sread(TIter s, TSent s_end, ios_base<TChar>&, const locale<TChar>& loc,
1619 const get_time_t<TChar>& f)
1620 {
1621 if (f.tmb == nullptr || f.fmt == nullptr)
1622 throw stream_error("get_time fail: null tm or format pointer");
1623
1624 auto mp = loc.template get<timeio<TChar>>();
1625 if (!mp)
1626 throw stream_error("cannot get timeio facet");
1627
1628 // The context is date+time at the offset tier, matching
1629 // parse_context_type<TChar, std::tm>; that is also the combination whose
1630 // convert_to(std::tm&) is available.
1631 auto tmp = parse_context_type<TChar, std::tm>::make_parse_context(*(f.tmb));
1632 auto res = mp->get(s, s_end, tmp, f.fmt);
1633
1634 tmp.convert_to(*(f.tmb));
1635
1636 return res;
1637 }
