253 void adjust(const cvt_behavior& acc)
254 {
255 if (this->is_tainted()) recover();
256 if (const auto* ptr = dynamic_cast<const code_cvt_switch*>(&acc); ptr)
257 {
258 if (!this->m_cvt_kernel.is_init_state())
259 throw cvt_error("code_cvt_stdio::adjust fail: invalid state");
260 // Build the new kernel before committing: the commit move below MUST be
261 // noexcept, otherwise a partial throw would leave the converter without a
262 // usable kernel.
263 static_assert(
264 std::is_nothrow_move_assignable_v<codecvt_kernel<char, wchar_t>>,
265 "codecvt_kernel<char, wchar_t> move-assign must be noexcept; "
266 "otherwise code_cvt_stdio::adjust loses its strong exception guarantee");
267 codecvt_kernel<char, wchar_t> new_kernel(ptr->code);
268 this->m_cvt_kernel = std::move(new_kernel);
269 }
270 return BT::adjust(acc);
271 }
