311 void adjust(const cvt_behavior& acc)
312 {
313 if (this->is_tainted()) recover();
314 if (const auto* ptr = dynamic_cast<const code_cvt_switch*>(&acc); ptr)
315 {
316 if (!this->m_cvt_kernel.is_init_state() || this->m_cvt_kernel.is_mid_seq())
317 throw cvt_error("code_cvt_stdio::adjust fail: invalid state");
318 // Build the new kernel before committing: the commit move below MUST be
319 // noexcept, otherwise a partial throw would leave the converter without a
320 // usable kernel.
321 static_assert(
322 std::is_nothrow_move_assignable_v<codecvt_kernel<char, wchar_t>>,
323 "codecvt_kernel<char, wchar_t> move-assign must be noexcept; "
324 "otherwise code_cvt_stdio::adjust loses its strong exception guarantee");
325 codecvt_kernel<char, wchar_t> new_kernel(ptr->code);
326 this->m_cvt_kernel = std::move(new_kernel);
327 }
328 else if (const auto* st = dynamic_cast<const code_cvt_stdio_state*>(&acc); st && st->kernel)
329 {
330 codecvt_kernel<char, wchar_t> kernel(*st->kernel);
331 this->m_cvt_kernel = std::move(kernel);
