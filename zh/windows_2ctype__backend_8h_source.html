67// Not _isprint_l: it is any of _BLANK|_PUNCT|_ALPHA|_DIGIT, and a named locale's table,
68// built by GetStringTypeA, gives tab C1_BLANK, so tab would print there though not in "C".
69// C makes printing and control characters disjoint (C11 7.4/3); a blank prints only if
70// it is not a control character.
71inline bool ctype_isprint(int c, clocale_handle loc) noexcept
72{
73 return _isctype_l(c, _PUNCT | _ALPHA | _DIGIT, loc->loc) != 0
74 || (_isctype_l(c, _BLANK, loc->loc) != 0 && _iscntrl_l(c, loc->loc) == 0);
75}

110// The signature every backend shares; only here are wint_t and wctype_t both unsigned short.
111// NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
112inline bool ctype_iswctype(wint_t c, ctype_wclass cls, clocale_handle /*loc*/) noexcept
113{
114 const auto wc = static_cast<wchar_t>(c);
115 unsigned short type = 0;
116 if (GetStringTypeW(win_ct_ctype1, &wc, 1, &type) == 0)
117 return false;
118
119 constexpr ctype_wclass print_class = _BLANK | _PUNCT | _ALPHA | _DIGIT;
120 if (cls == print_class)
121 return (type & (_PUNCT | _ALPHA | _DIGIT)) != 0
122 || ((type & _BLANK) != 0 && (type & _CONTROL) == 0);
123 return (type & cls) != 0;
124}

155inline wint_t ctype_map_case(wint_t c, unsigned long flags, clocale_handle loc) noexcept
156{
157 if (loc->ctype_bcp47.empty())
158 {
159 if (flags == win_lcmap_uppercase && c >= L'a' && c <= L'z') return static_cast<wint_t>(c - L'a' + L'A');
160 if (flags == win_lcmap_lowercase && c >= L'A' && c <= L'Z') return static_cast<wint_t>(c - L'A' + L'a');
161 return c;
162 }
163 const auto src = static_cast<wchar_t>(c);
164 wchar_t dst = 0;
165 if (LCMapStringEx(loc->ctype_bcp47.c_str(), flags | win_lcmap_linguistic_casing,
166 &src, 1, &dst, 1, nullptr, nullptr, 0) != 1)
167 return c;
168 return dst;
169}
