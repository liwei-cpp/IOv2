198 template <typename CharT>
199 inline CharT* add_grouping(CharT* out, std::basic_string_view<CharT> sep,
200 const std::vector<uint8_t>& grouping,
201 const CharT* first, const CharT* last)
202 {
203 assert(!grouping.empty());
204 assert(first <= last);
205
206 const std::size_t last_rule = grouping.size() - 1;
207
208 // Pass 1 -- measure. Peel whole groups off the low-order end for as long
209 // as what is left over can still fill one more. `rule` records how far
210 // the walk got into `grouping`; once it reaches the final entry that
211 // entry keeps applying, and `extra` counts how many further times it did.
212 // `head` is left holding the width of the leading, possibly short, group.
213 std::size_t rule = 0;
214 std::size_t extra = 0;
215 auto head = static_cast<std::size_t>(last - first);
216 while (true)
217 {
218 const std::size_t size = grouping[rule];
219 if (size == 0 || head <= size)
220 break;
221 head -= size;
222 if (rule < last_rule)
223 ++rule;
224 else
225 ++extra;
226 }
227
228 // One separator per peeled group, so the finished width is known up front.
229 const std::size_t groups = rule + extra;
230 const std::size_t width = static_cast<std::size_t>(last - first) + groups * sep.size();
231
232 // Pass 2 -- emit. Fill the buffer from its high-order end, which lets the
233 // groups be replayed in the very order pass 1 peeled them off; rule k
234 // governs the k-th group up from the bottom, saturating at `last_rule`.
235 // `head >= 1` whenever `groups > 0`, so the leading separator write can
236 // never run off the front of the buffer.
237 CharT* dst = out + width;
238 const CharT* src = last;
239 for (std::size_t k = 0; k < groups; ++k)
240 {
241 const std::size_t size = grouping[std::min(k, last_rule)];
242 src -= size;
243 dst -= size;
244 std::copy_n(src, size, dst);
245 dst -= sep.size();
246 std::ranges::copy(sep, dst);
247 }
248 std::copy_n(first, head, out);
249 return out + width;
250 }

310 template <typename TIter, typename CharT>
311 inline bool step_back(TIter& beg, std::basic_string_view<CharT> units, std::size_t k)
312 {
313 if constexpr (std::bidirectional_iterator<TIter>)
314 {
315 std::ranges::advance(beg, -static_cast<std::iter_difference_t<TIter>>(k));
316 return true;
317 }
318 else if constexpr (requires(TIter& it, CharT c) { it.putbackc(c); })
319 {
320 while (k != 0)
321 beg.putbackc(units[--k]);
322 return true;
323 }
324 else
325 return k == 0;
326 }

359 template <typename TIter, std::sentinel_for<TIter> TSent, typename CharT>
360 inline punct_kind take_punct(TIter& beg, TSent end, std::basic_string_view<CharT> sep,
361 std::basic_string_view<CharT> point, bool consume_point)
362 {
363 bool sep_alive = !sep.empty();
364 bool point_alive = !point.empty();
365 std::size_t k = 0;
366 while (beg != end)
367 {
368 const CharT c = *beg;
369 const bool s = sep_alive && sep[k] == c;
370 const bool p = point_alive && point[k] == c;
371 if (!s && !p)
372 break;
373 sep_alive = s;
374 point_alive = p;
375
376 // The point is settled at its last character, before that one is read.
377 if (p && k + 1 == point.size())
378 {
379 if (consume_point)
380 ++beg;
381 else
382 step_back(beg, point, k);
383 return punct_kind::point;
384 }
385 ++beg;
386 ++k;
387 if (s && k == sep.size())
388 return punct_kind::separator;
389 }
390
391 const std::basic_string_view<CharT> read = sep_alive ? sep : point;
392 return step_back(beg, read, k) ? punct_kind::none : punct_kind::torn;
393 }

442 inline void adjust_grouping(std::vector<uint8_t>& grouping)
443 {
444 // Step 1: POSIX "0 = repeat previous" → drop the 0 and the tail.
445 auto zero = std::ranges::find(grouping, uint8_t{0});
446 grouping.erase(zero, grouping.end());
447
448 // Step 2: POSIX CHAR_MAX (= stop) → internal 0 sentinel, plus
449 // canonical-form cleanup. After step 1 there are no 0s in the vector,
450 // so the FIRST CHAR_MAX is the only point at which we introduce one,
451 // and three cases fully cover the outcome:
452 // - at begin(): grouping would start with the stop sentinel,
453 // i.e. no usable leading group ⇒ discard.
454 // (This also subsumes the empty-vector case left
455 // by step 1, since find on an empty range returns
456 // end() == begin().)
457 // - in the tail: rewrite to 0 and drop everything after it; those
458 // bytes are dead code anyway because add_grouping
459 // short-circuits on 0.
460 // - not found: nothing to do.
461 constexpr auto posix_sentinel =
462 static_cast<uint8_t>(std::numeric_limits<char>::max());
463 auto stop = std::ranges::find(grouping, posix_sentinel);
464 if (stop == grouping.begin())
465 grouping.clear();
466 else if (stop != grouping.end())
467 {
468 *stop = 0;
469 grouping.erase(std::next(stop), grouping.end());
470 }
471 }

486 template <typename CharT>
487 inline std::vector<uint8_t> lconv_grouping(std::string_view raw, std::basic_string_view<CharT> sep,
488 std::basic_string_view<CharT> point)
489 {
490 std::vector<uint8_t> res;
491 if (raw.empty() || !punct_distinct(sep, point))
492 return res;
493 res.reserve(raw.size());
494 for (const char c : raw)
495 res.push_back(static_cast<uint8_t>(c));
496 adjust_grouping(res);
497 return res;
498 }

546 inline bool verify_grouping(const std::vector<uint8_t>& grouping,
547 const std::vector<uint8_t>& grouping_tmp)
548 {
549 assert(!grouping.empty());
550 assert(!grouping_tmp.empty());
551
552 const std::size_t last_rule = grouping.size() - 1;
553
554 // Walk the observed groups from the low-order end upwards, advancing a
555 // cursor through the rules alongside. The cursor stalls on the final
556 // rule, which is what makes that rule govern every group beyond the
557 // explicit ones. Every group but the leading one has to be exactly as
558 // wide as the rule facing it.
559 std::size_t rule = 0;
560 for (std::size_t k = grouping_tmp.size() - 1; k > 0; --k)
561 {
562 if (grouping_tmp[k] != grouping[rule])
563 return false;
564 if (rule < last_rule)
565 ++rule;
566 }
567
568 // The cursor now faces the leading group, which is allowed to fall short
569 // of its rule, and is unbounded above when that rule is the stop sentinel.
570 return grouping[rule] == 0 || grouping_tmp[0] <= grouping[rule];
571 }
