179 inline char string_to_char_convert(const std::string& narrow,
180 const std::string& locale_name)
181 {
182 if (narrow.empty()) return '\0';
183 if (narrow.size() == 1) return narrow[0];
184
185 // Convert (multi-byte) narrow to wchar_t then narrow down
186 std::wstring wide_str = detail::to_wstring(narrow.c_str(), locale_name);
187 if (wide_str.empty()) return '\0';
188
189 // Narrow the first wide character back to a single byte in the target
190 // locale. This reproduces ctype_conf<wchar_t>::narrow() (wctob() under
191 // a thread-local locale switch) directly, instead of building a whole
192 // ctype_conf<wchar_t> facet whose 256-entry widen table and wctype
193 // masks this path never reads. The out_of_wchar_range guard used there
194 // is unnecessary here: the value is a wchar_t and is therefore always
195 // representable as wchar_t.
196 clocale_wrapper loc(locale_name.c_str());
197 clocale_user guard(loc);
198 const int c = wctob(static_cast<wint_t>(wide_str[0]));
199 return c == EOF ? '\0' : static_cast<char>(c);
200 }

374 template <typename CharT>
375 inline CharT* add_grouping(CharT* out, CharT sep, const std::vector<uint8_t>& grouping,
376 const CharT* first, const CharT* last)
377 {
378 assert(!grouping.empty());
379 assert(first <= last);
380
381 const std::size_t last_rule = grouping.size() - 1;
382
383 // Pass 1 -- measure. Peel whole groups off the low-order end for as long
384 // as what is left over can still fill one more. `rule` records how far
385 // the walk got into `grouping`; once it reaches the final entry that
386 // entry keeps applying, and `extra` counts how many further times it did.
387 // `head` is left holding the width of the leading, possibly short, group.
388 std::size_t rule = 0;
389 std::size_t extra = 0;
390 auto head = static_cast<std::size_t>(last - first);
391 while (true)
392 {
393 const std::size_t size = grouping[rule];
394 if (size == 0 || head <= size)
395 break;
396 head -= size;
397 if (rule < last_rule)
398 ++rule;
399 else
400 ++extra;
401 }
402
403 // One separator per peeled group, so the finished width is known up front.
404 const std::size_t groups = rule + extra;
405 const std::size_t width = static_cast<std::size_t>(last - first) + groups;
406
407 // Pass 2 -- emit. Fill the buffer from its high-order end, which lets the
408 // groups be replayed in the very order pass 1 peeled them off; rule k
409 // governs the k-th group up from the bottom, saturating at `last_rule`.
410 // `head >= 1` whenever `groups > 0`, so the leading separator write can
411 // never run off the front of the buffer.
412 CharT* dst = out + width;
413 const CharT* src = last;
414 for (std::size_t k = 0; k < groups; ++k)
415 {
416 const std::size_t size = grouping[std::min(k, last_rule)];
417 src -= size;
418 dst -= size;
419 std::copy_n(src, size, dst);
420 *--dst = sep;
421 }
422 std::copy_n(first, head, out);
423 return out + width;
424 }

473 inline void adjust_grouping(std::vector<uint8_t>& grouping)
474 {
475 // Step 1: POSIX "0 = repeat previous" → drop the 0 and the tail.
476 auto zero = std::ranges::find(grouping, uint8_t{0});
477 grouping.erase(zero, grouping.end());
478
479 // Step 2: POSIX CHAR_MAX (= stop) → internal 0 sentinel, plus
480 // canonical-form cleanup. After step 1 there are no 0s in the vector,
481 // so the FIRST CHAR_MAX is the only point at which we introduce one,
482 // and three cases fully cover the outcome:
483 // - at begin(): grouping would start with the stop sentinel,
484 // i.e. no usable leading group ⇒ discard.
485 // (This also subsumes the empty-vector case left
486 // by step 1, since find on an empty range returns
487 // end() == begin().)
488 // - in the tail: rewrite to 0 and drop everything after it; those
489 // bytes are dead code anyway because add_grouping
490 // short-circuits on 0.
491 // - not found: nothing to do.
492 constexpr auto posix_sentinel =
493 static_cast<uint8_t>(std::numeric_limits<char>::max());
494 auto stop = std::ranges::find(grouping, posix_sentinel);
495 if (stop == grouping.begin())
496 grouping.clear();
497 else if (stop != grouping.end())
498 {
499 *stop = 0;
500 grouping.erase(std::next(stop), grouping.end());
501 }
502 }

550 inline bool verify_grouping(const std::vector<uint8_t>& grouping,
551 const std::vector<uint8_t>& grouping_tmp)
552 {
553 assert(!grouping.empty());
554 assert(!grouping_tmp.empty());
555
556 const std::size_t last_rule = grouping.size() - 1;
557
558 // Walk the observed groups from the low-order end upwards, advancing a
559 // cursor through the rules alongside. The cursor stalls on the final
560 // rule, which is what makes that rule govern every group beyond the
561 // explicit ones. Every group but the leading one has to be exactly as
562 // wide as the rule facing it.
563 std::size_t rule = 0;
564 for (std::size_t k = grouping_tmp.size() - 1; k > 0; --k)
565 {
566 if (grouping_tmp[k] != grouping[rule])
567 return false;
568 if (rule < last_rule)
569 ++rule;
570 }
571
572 // The cursor now faces the leading group, which is allowed to fall short
573 // of its rule, and is unbounded above when that rule is the stop sentinel.
574 return grouping[rule] == 0 || grouping_tmp[0] <= grouping[rule];
575 }
