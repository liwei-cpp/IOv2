176 inline char string_to_char_convert(const std::string& narrow,
177 const std::string& locale_name)
178 {
179 if (narrow.empty()) return '\0';
180 if (narrow.size() == 1) return narrow[0];
181
182 // Convert (multi-byte) narrow to wchar_t then narrow down
183 std::wstring wide_str = detail::to_wstring(narrow.c_str(), locale_name);
184 if (wide_str.empty()) return '\0';
185
186 // Narrow the first wide character back to a single byte in the target
187 // locale. This reproduces ctype_conf<wchar_t>::narrow() (wctob() under
188 // a thread-local locale switch) directly, instead of building a whole
189 // ctype_conf<wchar_t> facet whose 256-entry widen table and wctype
190 // masks this path never reads. The out_of_wchar_range guard used there
191 // is unnecessary here: the value is a wchar_t and is therefore always
192 // representable as wchar_t.
193 clocale_wrapper loc(locale_name.c_str());
194 clocale_user guard(loc);
195 const int c = wctob(static_cast<wint_t>(wide_str[0]));
196 return c == EOF ? '\0' : static_cast<char>(c);
197 }

372 inline CharT* add_grouping(CharT* s, CharT sep, const std::vector<uint8_t>& grouping,
373 const CharT* first, const CharT* last)
374 {
375 assert(!grouping.empty());
376 assert(first <= last);
377
378 std::size_t idx = 0;
379 std::size_t ctr = 0;
380 const std::size_t max_idx = grouping.size() - 1;
381
382 while (last - first > grouping[idx]
383 && (grouping[idx] > 0))
384 {
385 last -= grouping[idx];
386 idx < max_idx ? ++idx : ++ctr;
387 }
388
389 s = std::copy(first, last, s);
390 first = last;
391
392 const uint8_t n = grouping[idx];
393 while (ctr--)
394 {
395 *s++ = sep;
396 s = std::copy_n(first, n, s);
397 first += n;
398 }
399
400 while (idx--)
401 {
402 *s++ = sep;
403 const uint8_t m = grouping[idx];
404 s = std::copy_n(first, m, s);
405 first += m;
406 }
407 return s;
408 }

457 inline void adjust_grouping(std::vector<uint8_t>& grouping)
458 {
459 // Step 1: POSIX "0 = repeat previous" → drop the 0 and the tail.
460 auto zero = std::ranges::find(grouping, uint8_t{0});
461 grouping.erase(zero, grouping.end());
462
463 // Step 2: POSIX CHAR_MAX (= stop) → internal 0 sentinel, plus
464 // canonical-form cleanup. After step 1 there are no 0s in the vector,
465 // so the FIRST CHAR_MAX is the only point at which we introduce one,
466 // and three cases fully cover the outcome:
467 // - at begin(): grouping would start with the stop sentinel,
468 // i.e. no usable leading group ⇒ discard.
469 // (This also subsumes the empty-vector case left
470 // by step 1, since find on an empty range returns
471 // end() == begin().)
472 // - in the tail: rewrite to 0 and drop everything after it; those
473 // bytes are dead code anyway because add_grouping
474 // short-circuits on 0.
475 // - not found: nothing to do.
476 constexpr auto posix_sentinel =
477 static_cast<uint8_t>(std::numeric_limits<char>::max());
478 auto stop = std::ranges::find(grouping, posix_sentinel);
479 if (stop == grouping.begin())
480 grouping.clear();
481 else if (stop != grouping.end())
482 {
483 *stop = 0;
484 grouping.erase(std::next(stop), grouping.end());
485 }
486 }

534 inline bool verify_grouping(const std::vector<uint8_t>& grouping,
535 const std::vector<uint8_t>& grouping_tmp)
536 {
537 assert(!grouping.empty());
538 assert(!grouping_tmp.empty());
539
540 std::size_t i = grouping_tmp.size() - 1;
541 const std::size_t min_val = std::min(i, grouping.size() - 1);
542 bool test = true;
543
544 // Parsed number groupings have to match the
545 // numpunct::grouping string exactly, starting at the
546 // right-most point of the parsed sequence of elements ...
547 for (std::size_t j = 0; j < min_val && test; --i, ++j)
548 test = grouping_tmp[i] == grouping[j];
549 for (; i && test; --i)
550 test = grouping_tmp[i] == grouping[min_val];
551 // ... but the first parsed grouping can be <= numpunct grouping.
552 // Skip this check when grouping[min_val] is 0 (stop sentinel,
553 // no upper bound on the first parsed group).
554 if (grouping[min_val] > 0)
555 test &= grouping_tmp[0] <= grouping[min_val];
556 return test;
557 }
