194 static type make_parse_context(const std::tm& tmb)
195 {
196 using namespace std::chrono;
197
198 std::int64_t total_mon = (static_cast<std::int64_t>(tmb.tm_year) + 1900) * 12 + tmb.tm_mon;
199 std::int64_t norm_year = total_mon / 12;
200 std::int64_t norm_mon = total_mon % 12;
201 if (norm_mon < 0)
202 {
203 norm_mon += 12;
204 --norm_year;
205 }
206
207 norm_year = std::clamp<std::int64_t>(norm_year,
208 static_cast<int>(year::min()),
209 static_cast<int>(year::max()));
210
211 constexpr sys_days cal_min{year::min() / January / 1};
212 constexpr sys_days cal_max{year::max() / December / 31};
213 constexpr std::int64_t secs_per_day = std::int64_t{24} * 60 * 60;
214
215 // The time group is folded into one second-of-day count and normalized with the same
216 // carry the date group uses, so an out-of-range hour/minute/second moves the date
217 // instead of silently wrapping within it. A leap second is the one exception: it
218 // becomes 59 because hh_mm_ss cannot represent it.
219 std::int64_t sec = (tmb.tm_sec == 60) ? 59 : tmb.tm_sec;
220 std::int64_t tod = static_cast<std::int64_t>(tmb.tm_hour) * 3600
221 + static_cast<std::int64_t>(tmb.tm_min) * 60
222 + sec;
223 std::int64_t day_carry = (tod >= 0) ? tod / secs_per_day
224 : -((secs_per_day - 1 - tod) / secs_per_day);
225
226 auto first = sys_days{year{static_cast<int>(norm_year)}
227 / month{static_cast<unsigned>(norm_mon) + 1} / day{1}};
228 auto offset = std::clamp<std::int64_t>(
229 static_cast<std::int64_t>(tmb.tm_mday) - 1 + day_carry, -4'000'000, 4'000'000);
230
231 type ctx;
232 ctx.set_hint(
233 year_month_day{std::clamp(first + days{static_cast<int>(offset)}, cal_min, cal_max)});
234 ctx.set_hint(hh_mm_ss<seconds>{seconds{tod - day_carry * secs_per_day}});
235 return ctx;
236 }
237};

288template <typename TChar>
289std::basic_string<TChar> tm_stream_format(const timeio<TChar>& tio)
290{
291 std::basic_string<TChar> fmt = tio.template expand_format<std::tm>('c');
292
293 if constexpr (time_value_fields<std::tm>::has_offset)
294 {
295 if (!timeio<TChar>::contains_specifier(fmt, 'z'))
296 {
297 constexpr std::array<TChar, 3> tail{ static_cast<TChar>(' '), static_cast<TChar>('%'),
298 static_cast<TChar>('z') };
299 fmt.append(tail.data(), tail.size());
300 }
301 }
302
303 if constexpr (time_value_fields<std::tm>::has_zone)
304 {
305 if (!timeio<TChar>::contains_specifier(fmt, 'Z'))
306 {
307 constexpr std::array<TChar, 5> tail{ static_cast<TChar>(' '), static_cast<TChar>('('),
308 static_cast<TChar>('%'), static_cast<TChar>('Z'),
309 static_cast<TChar>(')') };
310 fmt.append(tail.data(), tail.size());
311 }
312 }
313 return fmt;
314}
315} // namespace detail
