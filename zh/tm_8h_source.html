114 static type make_parse_context(const std::tm& tmb)
115 {
116 using namespace std::chrono;
117
118 std::int64_t total_mon = (static_cast<std::int64_t>(tmb.tm_year) + 1900) * 12 + tmb.tm_mon;
119 std::int64_t norm_year = total_mon / 12;
120 std::int64_t norm_mon = total_mon % 12;
121 if (norm_mon < 0)
122 {
123 norm_mon += 12;
124 --norm_year;
125 }
126
127 norm_year = std::clamp<std::int64_t>(norm_year,
128 static_cast<int>(year::min()),
129 static_cast<int>(year::max()));
130
131 constexpr sys_days cal_min{year::min() / January / 1};
132 constexpr sys_days cal_max{year::max() / December / 31};
133 constexpr std::int64_t secs_per_day = 24 * 60 * 60;
134
135 // The time group is folded into one second-of-day count and normalized with the same
136 // carry the date group uses, so an out-of-range hour/minute/second moves the date
137 // instead of silently wrapping within it. A leap second is the one exception: it
138 // becomes 59 because hh_mm_ss cannot represent it.
139 std::int64_t sec = (tmb.tm_sec == 60) ? 59 : tmb.tm_sec;
140 std::int64_t tod = static_cast<std::int64_t>(tmb.tm_hour) * 3600
141 + static_cast<std::int64_t>(tmb.tm_min) * 60
142 + sec;
143 std::int64_t day_carry = (tod >= 0) ? tod / secs_per_day
144 : -((secs_per_day - 1 - tod) / secs_per_day);
145
146 auto first = sys_days{year{static_cast<int>(norm_year)}
147 / month{static_cast<unsigned>(norm_mon) + 1} / day{1}};
148 auto offset = std::clamp<std::int64_t>(
149 static_cast<std::int64_t>(tmb.tm_mday) - 1 + day_carry, -4'000'000, 4'000'000);
150
151 type ctx;
152 ctx.set_hint(
153 year_month_day{std::clamp(first + days{static_cast<int>(offset)}, cal_min, cal_max)});
154 ctx.set_hint(hh_mm_ss<seconds>{seconds{tod - day_carry * secs_per_day}});
155 return ctx;
156 }
157};
