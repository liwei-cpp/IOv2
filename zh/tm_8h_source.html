132 static type make_parse_context(const std::tm& tmb)
133 {
134 using namespace std::chrono;
135
136 std::int64_t total_mon = (static_cast<std::int64_t>(tmb.tm_year) + 1900) * 12 + tmb.tm_mon;
137 std::int64_t norm_year = total_mon / 12;
138 std::int64_t norm_mon = total_mon % 12;
139 if (norm_mon < 0)
140 {
141 norm_mon += 12;
142 --norm_year;
143 }
144
145 norm_year = std::clamp<std::int64_t>(norm_year,
146 static_cast<int>(year::min()),
147 static_cast<int>(year::max()));
148
149 constexpr sys_days cal_min{year::min() / January / 1};
150 constexpr sys_days cal_max{year::max() / December / 31};
151 constexpr std::int64_t secs_per_day = 24 * 60 * 60;
152
153 // The time group is folded into one second-of-day count and normalized with the same
154 // carry the date group uses, so an out-of-range hour/minute/second moves the date
155 // instead of silently wrapping within it. A leap second is the one exception: it
156 // becomes 59 because hh_mm_ss cannot represent it.
157 std::int64_t sec = (tmb.tm_sec == 60) ? 59 : tmb.tm_sec;
158 std::int64_t tod = static_cast<std::int64_t>(tmb.tm_hour) * 3600
159 + static_cast<std::int64_t>(tmb.tm_min) * 60
160 + sec;
161 std::int64_t day_carry = (tod >= 0) ? tod / secs_per_day
162 : -((secs_per_day - 1 - tod) / secs_per_day);
163
164 auto first = sys_days{year{static_cast<int>(norm_year)}
165 / month{static_cast<unsigned>(norm_mon) + 1} / day{1}};
166 auto offset = std::clamp<std::int64_t>(
167 static_cast<std::int64_t>(tmb.tm_mday) - 1 + day_carry, -4'000'000, 4'000'000);
168
169 type ctx;
170 ctx.set_hint(
171 year_month_day{std::clamp(first + days{static_cast<int>(offset)}, cal_min, cal_max)});
172 ctx.set_hint(hh_mm_ss<seconds>{seconds{tod - day_carry * secs_per_day}});
173 return ctx;
174 }
175};

248template <typename TChar>
249std::basic_string<TChar> tm_stream_format(const timeio<TChar>& tio)
250{
251 std::basic_string<TChar> fmt = tio.template expand_format<std::tm>('c');
252
253 if constexpr (time_value_fields<std::tm>::has_offset)
254 {
255 if (!timeio<TChar>::contains_specifier(fmt, 'z'))
256 {
257 const TChar tail[] = { static_cast<TChar>(' '), static_cast<TChar>('%'),
258 static_cast<TChar>('z'), TChar() };
259 fmt += tail;
260 }
261 }
262
263 if constexpr (time_value_fields<std::tm>::has_zone)
264 {
265 if (!timeio<TChar>::contains_specifier(fmt, 'Z'))
266 {
267 const TChar tail[] = { static_cast<TChar>(' '), static_cast<TChar>('('),
268 static_cast<TChar>('%'), static_cast<TChar>('Z'),
269 static_cast<TChar>(')'), TChar() };
270 fmt += tail;
271 }
272 }
273 return fmt;
274}

304 if (!mp)

306 io.width(0);

309

317 return ostream_insert(s, io, buf.data(), buf.size());
