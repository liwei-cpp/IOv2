64// The name of loc's LC_CTYPE category. glibc reports "POSIX" as "C"; macOS keeps it.
65inline const char* clocale_ctype_name(clocale_handle loc) noexcept
66{
67#if defined(__GLIBC__)
68 return nl_langinfo_l(_NL_LOCALE_NAME(LC_CTYPE), loc);
69#else
70 return querylocale(LC_CTYPE_MASK, loc);
71#endif
72}

74// Whether loc's LC_CTYPE codeset is UTF-8. glibc names it "UTF-8" in every UTF-8 locale,
75// C.UTF-8 included, and so does macOS.
76inline bool clocale_ctype_is_utf8(clocale_handle loc) noexcept
77{
78 return std::strcmp(nl_langinfo_l(CODESET, loc), "UTF-8") == 0;
79}
