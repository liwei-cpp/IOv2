214 template <bool Store>
215 static scan_result translate(const internal_type* ptr, std::size_t len,
216 internal_type* out, std::size_t room) noexcept
217 {
218 std::size_t i = 0;
219 std::size_t produced = 0;
220 while (i < len && produced < room)
221 {
222 internal_type c = ptr[i];
223 if (c == s_cr)
224 {
225 if (i + 1 == len)
226 return {i, produced, true};
227 if (ptr[i + 1] == s_lf)
228 {
229 c = s_lf;
230 ++i;
231 }
232 }
233 if constexpr (Store)
234 out[produced] = c;
235 ++produced;
236 ++i;
237 }
238 return {i, produced, false};
239 }

275 std::pair<std::size_t, std::exception_ptr>
276 get_main(internal_type* to, std::size_t to_max) noexcept
277 requires (cvt_cpt::support_get<KernelType>)
278 {
279 std::size_t total = 0;
280 try
281 {
282 if (to_max == 0) return {0, nullptr};
283 auto reader = BT::kernel_reader();
284 reader.reset(s_buf_len);
285
286 bool held_cr = false;
287 while (total < to_max)
288 {
289 // A held \r needs at least one character behind it to be decided.
290 std::size_t want = std::min(to_max - total, s_buf_len);
291 if (held_cr) want = std::max<std::size_t>(want, 2);
292
293 auto [ptr, len] = reader.get_buf(want);
294 if (len == 0)
295 return {total, nullptr};
296 if (held_cr && len == 1)
297 {
298 // Nothing more below: the \r ends the data and is kept.
299 to[total++] = s_cr;
300 ++m_accu_len;
301 return {total, nullptr};
302 }
303
304 const auto res = translate<true>(ptr, len, to + total, to_max - total);
305 held_cr = res.held_cr;
306 total += res.produced;
307 m_accu_len += res.produced;
308 if (res.consumed != len)
309 reader.rollback(len - res.consumed);
310 if (held_cr && total != 0)
311 return {total, nullptr};
312 }
313 return {total, nullptr};
314 }
315 catch (...)
316 {
317 return {total, std::current_exception()};
318 }

344 void put_main(const internal_type* to, std::size_t to_size)
345 requires (cvt_cpt::support_put<KernelType>)
346 {
347 resolve_at_end();
348 if (m_at_end == at_end::no)
349 throw cvt_error("newline_cvt::put fail: not at the end of the stream");
350 if (to_size == 0) return;
351
352 auto writer = BT::kernel_writer();
353 writer.reset(s_buf_len * 2);
354 std::size_t done = 0;
355 while (done != to_size)
356 {
357 // Each character takes at most two slots below.
358 const std::size_t count = std::min(s_buf_len, to_size - done);
359 internal_type* ptr = writer.put_buf(count * 2);
360 std::size_t used = 0;
361 for (std::size_t i = 0; i < count; ++i)
362 {
363 const internal_type c = to[done + i];
364 if (c == s_lf)
365 ptr[used++] = s_cr;
366 ptr[used++] = c;
367 }
368 if (used != count * 2)
369 writer.rollback(count * 2 - used);
370 done += count;
371 }
372 // commit() is the responsibility of abs_cvt::put.
373 m_accu_len += to_size;
374 }

440 void resolve_at_end()
441 {
442 if (m_at_end != at_end::unknown) return;
443
444 try
445 {
446 if constexpr (cvt_cpt::support_get<KernelType>)
447 m_at_end = BT::kernel_is_eof() ? at_end::yes : at_end::no;
448 else if constexpr (cvt_cpt::support_positioning<KernelType>)
449 {
450 const std::size_t saved = BT::kernel_tell();
451 BT::kernel_rseek(0);
452 if (BT::kernel_tell() == saved)
453 m_at_end = at_end::yes;
454 else
455 {
456 BT::kernel_seek(saved);
457 m_at_end = at_end::no;
458 }
459 }
460 else
461 m_at_end = at_end::yes;
462 }
463 catch (...)
464 {
465 BT::set_tainted();
466 throw;
467 }
468 }

568 void rseek_impl(std::size_t pos)
569 requires (cvt_cpt::support_positioning<KernelType>)
570 {
571 if (pos != 0)
572 throw cvt_error("newline_cvt::rseek fail: only rseek(0) is supported");
573
574 if (BT::m_io_status == io_status::output)
575 {
576 // Writes only happen at the end, so normally there is nothing to do; confirm it.
577 resolve_at_end();
578 // Not at the end: reaching it would need the logical length of what lies
579 // between, which takes reading it -- impossible for the only layer that gets
580 // here (write-only). tell() could not be exact, so refuse.
581 if (m_at_end == at_end::no)
582 throw cvt_error("newline_cvt::rseek fail: not at the end of the stream");
583 return;
584 }
585
586 // Reading (a write-only layer below never is): count the characters up to the end,
587 // storing none of them.
588 if constexpr (cvt_cpt::support_get<KernelType>)
589 {
590 try
591 {
592 auto reader = BT::kernel_reader();
593 reader.reset(s_count_buf_len);
594 bool held_cr = false;
595 while (true)
596 {
597 auto [ptr, len] = reader.get_buf(s_count_buf_len);
598 if (len == 0)
599 break;
600 if (held_cr && len == 1)
601 {
602 ++m_accu_len; // the \r ends the data
603 break;
604 }
605 const auto res = translate<false>(ptr, len, nullptr, len);
606 m_accu_len += res.produced;
607 held_cr = res.held_cr;
608 if (res.consumed != len)
609 reader.rollback(len - res.consumed);
610 }
611 }
612 catch (...)
613 {
614 BT::set_tainted();
615 throw;
616 }
617
618 // A repositioning drops an error a previous get held back; reading did not, so
619 // seek where the layer below already stands. On failure nothing has changed.
620 BT::kernel_seek(BT::kernel_tell());
621 }
622 }
