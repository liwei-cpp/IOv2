344 std::size_t sgetn(char_type* s, std::size_t n, std::size_t* got = nullptr) requires (IsIn)
345 {
346 if (got) *got = 0;
347 if ((s == nullptr) || (n == 0)) return 0;
348
349 if constexpr (IsOut)
350 switch_to_get();
351
352 std::size_t res = 0;
353 while (!m_read_buf.empty())
354 {
355 *s++ = m_read_buf.back();
356 m_read_buf.pop_back();
357 if (++res == n) break;
358 }
359 if (got) *got = res;
360
361 while (res < n)
362 {
363 std::size_t c = m_cvt.get(s, n - res);
364 assert(c <= n - res);
365 if (c == 0) break;
366 res += c;
367 s += c;
368 if (got) *got = res;
369 }
370 return res;
371 }

698 void switch_to_put() requires (IsIn && IsOut)
699 {
700 if (!m_read_buf.empty())
701 {
702 try
703 {
704 const std::size_t pos = tell();
705 m_cvt.seek(pos);
706 }
707 catch (const cvt_error& e)
708 {
709 throw cvt_error("base_streambuf::switch_to_put fails: cannot reposition "
710 + std::to_string(m_read_buf.size())
711 + " buffered/put-back character(s) before switching to output mode: "
712 + e.what());
713 }
714 // Must stay after the seek: tell() subtracts m_read_buf.size(), and a failed seek
715 // leaves the buffer intact -- which is what iostream's @warning promises.
716 m_read_buf.clear();
717 }
718 m_cvt.switch_to_put();

781 std::pair<device_type, std::exception_ptr> detach() noexcept
782 {
783 if constexpr (IsIn)
784 {
785 if (!m_read_buf.empty())
786 {
787 try
788 {
789 // On over-pushback (more put-back than actually read), tell()
790 // saturates at 0, so the device is handed back positioned at the
791 // stream origin. That is consistent with the logical read cursor
792 // being at 0; see tell()/sputbackc() and switch_to_put().
793 const std::size_t pos = tell();
794 m_cvt.seek(pos);
795 }
796 catch (...) // NOLINT(bugprone-empty-catch)
797 {
798 // Swallowed on purpose: a device that doesn't support positioning
799 // (e.g. a pipe/tty-backed stdin) can never honor this reposition,
800 // and that is an inherent property of the device, not an
801 // actionable failure - the caller could not have repositioned it
802 // either after getting it back. Reporting this error would make
803 // routine detach()/attach() cycles on such devices start throwing
804 // (e.g. sync_with_stdio() right after a formatted read has left
805 // one buffered lookahead character), even though nothing is
806 // actually broken; losing that lookahead character is the
807 // accepted, unavoidable cost for non-positionable devices.
808 }
809 m_read_buf.clear();
810 }
811 }
812 return m_cvt.detach();
813 }
