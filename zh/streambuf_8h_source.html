361 std::size_t sgetn(char_type* s, std::size_t n, std::size_t* got = nullptr) requires (IsIn)
362 {
363 if (got) *got = 0;
364 if ((s == nullptr) || (n == 0)) return 0;
365
366 if constexpr (IsOut)
367 switch_to_get();
368
369 std::size_t res = 0;
370 while (!m_read_buf.empty())
371 {
372 *s++ = m_read_buf.back();
373 m_read_buf.pop_back();
374 if (++res == n) break;
375 }
376 if (got) *got = res;
377
378 while (res < n)
379 {
380 std::size_t c = m_cvt.get(s, n - res);
381 assert(c <= n - res);
382 if (c == 0) break;
383 res += c;
384 s += c;
385 if (got) *got = res;
386 }
387 return res;
388 }

715 void switch_to_put() requires (IsIn && IsOut)
716 {
717 if (!m_read_buf.empty())
718 {
719 try
720 {
721 const std::size_t pos = tell();
722 m_cvt.seek(pos);
723 }
724 catch (const cvt_error& e)
725 {
726 throw cvt_error("base_streambuf::switch_to_put fails: cannot reposition "
727 + std::to_string(m_read_buf.size())
728 + " buffered/put-back character(s) before switching to output mode: "
729 + e.what());
730 }
731 // Must stay after the seek: tell() subtracts m_read_buf.size(), and a failed seek
732 // leaves the buffer intact -- which is what iostream's @warning promises.
733 m_read_buf.clear();
734 }
735 m_cvt.switch_to_put();

798 std::pair<device_type, std::exception_ptr> detach() noexcept
799 {
800 if constexpr (IsIn)
801 {
802 if (!m_read_buf.empty())
803 {
804 try
805 {
806 // On over-pushback (more put-back than actually read), tell()
807 // saturates at 0, so the device is handed back positioned at the
808 // stream origin. That is consistent with the logical read cursor
809 // being at 0; see tell()/sputbackc() and switch_to_put().
810 const std::size_t pos = tell();
811 m_cvt.seek(pos);
812 }
813 catch (...) // NOLINT(bugprone-empty-catch)
814 {
815 // Swallowed on purpose: a device that doesn't support positioning
816 // (e.g. a pipe/tty-backed stdin) can never honor this reposition,
817 // and that is an inherent property of the device, not an
818 // actionable failure - the caller could not have repositioned it
819 // either after getting it back. Reporting this error would make
820 // routine detach()/attach() cycles on such devices start throwing
821 // (e.g. sync_with_stdio() right after a formatted read has left
822 // one buffered lookahead character), even though nothing is
823 // actually broken; losing that lookahead character is the
824 // accepted, unavoidable cost for non-positionable devices.
825 }
826 m_read_buf.clear();
827 }
828 }
829 return m_cvt.detach();
830 }
