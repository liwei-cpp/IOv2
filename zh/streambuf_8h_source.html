340 std::size_t sgetn(char_type* s, std::size_t n, std::size_t* got = nullptr) requires (IsIn)
341 {
342 if (got) *got = 0;
343 if ((s == nullptr) || (n == 0)) return 0;
344
345 if constexpr (IsOut)
346 switch_to_get();
347
348 std::size_t res = 0;
349 while (!m_read_buf.empty())
350 {
351 *s++ = m_read_buf.back();
352 m_read_buf.pop_back();
353 if (++res == n) break;
354 }
355 if (got) *got = res;
356
357 while (res < n)
358 {
359 std::size_t c = m_cvt.get(s, n - res);
360 assert(c <= n - res);
361 if (c == 0) break;
362 res += c;
363 s += c;
364 if (got) *got = res;
365 }
366 return res;
367 }

775 std::pair<device_type, std::exception_ptr> detach() noexcept
776 {
777 if constexpr (IsIn)
778 {
779 if (!m_read_buf.empty())
780 {
781 try
782 {
783 // On over-pushback (more put-back than actually read), tell()
784 // saturates at 0, so the device is handed back positioned at the
785 // stream origin. That is consistent with the logical read cursor
786 // being at 0; see tell()/sputbackc() and switch_to_put().
787 const std::size_t pos = tell();
788 m_cvt.seek(pos);
789 }
790 catch (...) // NOLINT(bugprone-empty-catch)
791 {
792 // Swallowed on purpose: a device that doesn't support positioning
793 // (e.g. a pipe/tty-backed stdin) can never honor this reposition,
794 // and that is an inherent property of the device, not an
795 // actionable failure - the caller could not have repositioned it
796 // either after getting it back. Reporting this error would make
797 // routine detach()/attach() cycles on such devices start throwing
798 // (e.g. sync_with_stdio() right after a formatted read has left
799 // one buffered lookahead character), even though nothing is
800 // actually broken; losing that lookahead character is the
801 // accepted, unavoidable cost for non-positionable devices.
802 }
803 m_read_buf.clear();
804 }
805 }
806 return m_cvt.detach();
807 }
