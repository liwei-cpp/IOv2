13template <typename TIter, typename TChar>
14 requires (std::is_same_v<TChar, typename TIter::value_type>)
15TIter ostream_insert(TIter iter, ios_base<TChar>& io, const TChar* s, size_t n)
16{
17 const size_t w = io.width();
18 // Consumed up front, so it cannot leak into the next insertion if a step below throws.
19 io.width(0);
20 if (w > n)
21 {
22 const size_t pad = w - n;
23 if (pad > ios_defs::max_pad_count)
24 throw stream_error("ostream insert fail: fill count exceeds max_pad_count");
25
26 const bool left = ((io.flags() & ios_defs::adjustfield) == ios_defs::left);
27 const TChar f = io.fill();
28 if (!left)
29 iter = std::fill_n(iter, pad, f);
30 iter = std::copy(s, s + n, iter);
31 if (left)
32 iter = std::fill_n(iter, pad, f);
33 }
34 else
35 iter = std::copy(s, s + n, iter);
36 return iter;
37}

40 requires (std::is_same_v<TChar, typename TIter::value_type>)
41TIter istream_extract(TIter iter, TSent iter_end, ios_base<TChar>& io, const locale<TChar>& loc, TChar* s, size_t num)
42{
43 const size_t width = io.width();
44 // Consumed up front, so it cannot leak into the next extraction if a step below throws.
45 io.width(0);
46 if (0 < width && width < num)
47 num = width;
48
49 if (num <= 1)
50 {
51 if (num == 1) *s = TChar{};
52 throw stream_error("DO not have enough buffer to save character");
53 }
54
55 auto ct = loc.template get<ctype<TChar>>();
56 if (!ct)
57 throw stream_error("cannot get ctype facet");
58
59 if (iter == iter_end) throw stream_error("cannot extract character from empty stream");
60
61 size_t extracted = 0;
62
63 while (extracted < num - 1
64 && (iter != iter_end)
65 && !(ct->is_any(base_ft<ctype>::space, *iter)))
66 {
67 *s++ = *iter;
68 ++extracted;
69 ++iter;
70 }
71
72 *s = TChar{};
73
74 if (extracted == 0)
75 throw stream_error("istream extraction fail: no characters extracted");
76
77 return iter;
78}
