121template <typename TIter, typename TChar>
122 requires (char_sink_for<TIter, TChar>)
123TIter ostream_insert(TIter iter, ios_base<TChar>& io, const TChar* s, std::size_t n)
124{
125 const std::size_t w = io.width(0);
126 if (w > n)
127 {
128 const std::size_t pad = w - n;
129 if (pad > ios_defs::max_pad_count)
130 throw stream_error("ostream insert fail: fill count exceeds max_pad_count");
131
132 const bool left = ((io.flags() & ios_defs::adjustfield) == ios_defs::left);
133 const TChar f = io.fill();
134 if (!left)
135 iter = std::fill_n(iter, pad, f);
136 iter = std::copy(s, s + n, iter);
137 if (left)
138 iter = std::fill_n(iter, pad, f);
139 }
140 else
141 iter = std::copy(s, s + n, iter);
142 return iter;
143}

211template <typename TIter, std::sentinel_for<TIter> TSent, typename TChar>
212 requires (std::is_same_v<TChar, typename TIter::value_type>)
213TIter istream_extract(TIter iter, TSent iter_end, ios_base<TChar>& io, const locale<TChar>& loc, TChar* s, std::size_t num)
214{
215 const std::size_t width = io.width(0);
216 if (0 < width && width < num)
217 num = width;
218
219 if (num <= 1)
220 {
221 if (num == 1) *s = TChar{};
222 throw stream_error("Do not have enough buffer to save character");
223 }
224
225 auto ct = loc.template get<ctype<TChar>>();
226 if (!ct)
227 {
228 *s = TChar{};
229 throw stream_error("cannot get ctype facet");
230 }
231
232 std::size_t extracted = 0;
233
234 try
235 {
236 while (extracted < num - 1 && (iter != iter_end))
237 {
238 const TChar c = *iter;
239 if (ct->is_any(base_ft<ctype>::space, c))
240 break;
241
242 *s++ = c;
243 ++extracted;
244 ++iter;
245 }
246
247 *s = TChar{};
248 }
249 catch (...)
250 {
251 *s = TChar{};
252 throw;
253 }
254
255 if (extracted == 0)
256 throw stream_error("istream extraction fail: no characters extracted");
257
258 return iter;
259}

1508 template <typename TIter, std::sentinel_for<TIter> TSent>
1509 requires (std::is_same_v<TChar, typename TIter::value_type>)
1510 static TIter sread(TIter iter, TSent iter_end, ios_base<TChar>& io, const locale<TChar>& loc, std::basic_string<TChar, TTraits, TAlloc>& str)
1511 {
1512 str.erase();
1513 constexpr std::size_t buf_size = 128;
1514 std::array<TChar, buf_size> buf;
1515 std::size_t len = 0;
1516 const std::size_t w = io.width(0);
1517 const std::size_t n = w > 0 ? w : str.max_size();
1518 std::size_t extracted = 0;
1519
1520 auto ct = loc.template get<ctype<TChar>>();
1521 if (!ct)
1522 throw stream_error("cannot get ctype facet");
1523 try
1524 {
1525 while (extracted < n && (iter != iter_end))
1526 {
1527 const TChar c = *iter;
1528 if (ct->is_any(base_ft<ctype>::space, c))
1529 break;
1530
1531 if (len == buf_size)
1532 {
1533 str.append(buf.data(), buf_size);
1534 len = 0;
1535 }
1536 buf[len++] = c;
1537 ++extracted;
1538 ++iter;
1539 }
1540 str.append(buf.data(), len);
1541 }
1542 catch (...)
1543 {
1544 // Retried rather than skipped: append is strongly exception-safe, so a failed one
1545 // left nothing behind and this is the only chance to keep the staged characters.
1546 try
1547 {
1548 str.append(buf.data(), len);
1549 }
1550 catch (...) {} // NOLINT(bugprone-empty-catch)
1551 throw;
1552 }
1553
1554 if (extracted == 0)
1555 throw stream_error("istream extraction fail: no characters extracted");
1556
1557 return iter;
1558 }
