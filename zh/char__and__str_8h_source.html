14TIter ostream_insert(TIter iter, ios_base<TChar>& io, const TChar* s, size_t n)
15{
16 const size_t w = io.width();
17 if (w > n)
18 {
19 const bool left = ((io.flags() & ios_defs::adjustfield) == ios_defs::left);
20 const TChar f = io.fill();

22 iter = std::fill_n(iter, w - n, f);
23 iter = std::copy(s, s + n, iter);
24 if (left)
25 iter = std::fill_n(iter, w - n, f);
26 }

35TIter istream_extract(TIter iter, TSent iter_end, ios_base<TChar>& io, const locale<TChar>& loc, TChar* s, size_t num)
36{
37 const size_t width = io.width();
38 if (0 < width && width < num)
39 num = width;
40
41 if (num <= 1)
42 {
43 if (num == 1) *s = TChar{};
44 io.width(0);
45 throw stream_error("DO not have enough buffer to save character");
46 }
47
48 auto ct = loc.template get<ctype<TChar>>();
49 if (!ct)
50 throw stream_error("cannot get ctype facet");
51
52 if (iter == iter_end) throw stream_error("cannot extract character from empty stream");
53
54 size_t extracted = 0;
55
56 while (extracted < num - 1
57 && (iter != iter_end)
58 && !(ct->is_any(base_ft<ctype>::space, *iter)))

60 *s++ = *iter;
61 ++extracted;
62 ++iter;
63 }
64
65 *s = TChar{};
66
67 io.width(0);
68
69 if (extracted == 0)
70 throw stream_error("istream extraction fail: no characters extracted");
71
72 return iter;
