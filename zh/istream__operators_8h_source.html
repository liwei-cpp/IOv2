116 in_sentry(TStream& is, bool noskip)
117 : m_is(is)
118 {
119 if (!m_is)
120 throw stream_error("istream_sentry create fail: Invalid istream");
121
122 if (auto* tied = is.tie())
123 {
124 try { tied->try_flush(); }
125 catch (...) {} // NOLINT(bugprone-empty-catch)
126 }
127
128 if constexpr (involve_output)
129 is.m_streambuf.switch_to_get();
130
131 if (!noskip)
132 {
133 try
134 {
135 auto ct = is.m_locale.template get<IOv2::ctype<typename TStream::char_type>>();
136 if (!ct)
137 throw stream_error{"istream ignore_ws fail: no ctype facet"};

147 }

152 }

611 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
612 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
613 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
614 TOut get(this TSelf& self, TOut s, size_t n, TChar delim)
615 {
616 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
617
618 size_t gcount = 0;
619 bool at_eof = false;
620 std::lock_guard guard(self.io_mutex());
621 try
622 {
623 using sentry_type = typename TSelf::in_sentry_type;
624 sentry_type cerb(self, true);
625 if constexpr (std::is_pointer_v<TOut>)
626 {
627 if (s == nullptr)
628 throw stream_error("istream get fail: null character sequence");
629 }
630 if (n == 0)
631 throw stream_error("istream get fail: zero buffer size");
632 auto c = self.m_streambuf.sgetc();
633 while ((gcount + is_cstr < n) &&
634 (c.has_value()) &&
635 (c.value() != delim))
636 {
637 *s++ = c.value();
638 ++gcount;
639 c = self.m_streambuf.snextc();
640 }
641
642 at_eof = (gcount + is_cstr < n) && (!c.has_value());
643
644 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
645 {
646 if (c.has_value())
647 {
648 if (c.value() == delim)
649 {
650 self.m_streambuf.sbumpc();
651 ++gcount;
652 }
653 else
654 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
655 }
656 }
657
658 if (gcount == 0)
659 throw stream_error{"istream get fail: no character extracted"};
660
661 if (at_eof)
662 self.setstate(ios_defs::eofbit);
663 }
664 catch(...)
665 {
666 if constexpr (is_cstr)
667 {
668 if constexpr (std::is_pointer_v<TOut>)
669 {
670 if (s != nullptr && n != 0)
671 *s++ = TChar{};
672 }
673 else if (n != 0)
674 *s++ = TChar{};
675 }
676 self.handle_exception(std::current_exception(), at_eof);
677 return s;
678 }
679
680 if constexpr (is_cstr)
681 *s++ = TChar{};
682 return s;
683 }

1122 using TTarget = detail::in_target_t<TValue>;
1123
1124 constexpr bool stream_r = detail::extractable_with_stream<T, TValue>;
1125 constexpr bool iter_v = detail::extractable_with_iter<T, TValue>;
1126 constexpr bool iter_c = detail::extractable_with_ctx<T, TValue>;
1127
1128 static_assert(!(stream_r && (iter_v || iter_c)),
1129 "IOv2: this io_traits specialization provides both forms of sread() -- the stream form "
1130 "sread(stream, value) and the iterator form sread(iter, end, io, loc, value). Provide "
1131 "exactly one: the iterator form for formatted extraction, the stream form for a "
1132 "manipulator. See io/traits/traits_base.h.");
1133
1134 if constexpr (stream_r)
1135 {
1136 try
1137 {
1138 io_traits<TChar, TV>::sread(obj, static_cast<TTarget&>(value));
1139 }
1140 catch (...)
1141 {
1142 obj.handle_exception(std::current_exception());
1143 }
1144 }
1145 else if constexpr (iter_v || iter_c)
1146 {
1147 using sentry_type = typename T::in_sentry_type;
1148
1149 bool saw_eof = false;
1150 std::lock_guard guard(obj.io_mutex());
1151 try
1152 {
1153 auto iter = obj.i_iter(&saw_eof);
1154 bool skip = bool(obj.flags() & ios_defs::skipws);
1155 sentry_type cerb(obj, !skip);
1156
1157 if (obj.eof())
1158 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1159
1160 if constexpr (iter_v)
1161 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(),
1162 static_cast<TTarget&>(value));
1163 else
1164 {
1165 TCtx tmp = [&value]() -> TCtx {
1166 if constexpr (requires (const TV& v)
1167 { parse_context_type<TChar, TV>::make_parse_context(v); })
1168 return parse_context_type<TChar, TV>::make_parse_context(value);
1169 else
1170 return TCtx{};
1171 }();
1172 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1173 static_cast<TTarget&>(value) = static_cast<TV>(tmp);
1174 }
1175
1176 if (saw_eof) obj.setstate(ios_defs::eofbit);
1177 }
1178 catch(...)
1179 {
1180 obj.handle_exception(std::current_exception(), saw_eof);
1181 }
1182 }
1183 else
1184 static_assert(dependent_false_v<TValue>,

1186 "branch for it. The concept and the chain must stay in step -- see "
