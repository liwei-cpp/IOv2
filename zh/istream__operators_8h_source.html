606 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
607 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
608 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
609 TOut get(this TSelf& self, TOut s, size_t n, TChar delim)
610 {
611 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
612
613 size_t gcount = 0;
614 bool at_eof = false;
615 std::lock_guard guard(self.io_mutex());
616 try
617 {
618 using sentry_type = typename TSelf::in_sentry_type;
619 sentry_type cerb(self, true);
620 if constexpr (std::is_pointer_v<TOut>)
621 {
622 if (s == nullptr)
623 throw stream_error("istream get fail: null character sequence");
624 }
625 if (n == 0)
626 throw stream_error("istream get fail: zero buffer size");
627 auto c = self.m_streambuf.sgetc();
628 while ((gcount + is_cstr < n) &&
629 (c.has_value()) &&
630 (c.value() != delim))
631 {
632 *s++ = c.value();
633 ++gcount;
634 c = self.m_streambuf.snextc();
635 }
636
637 at_eof = (gcount + is_cstr < n) && (!c.has_value());
638
639 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
640 {
641 if (c.has_value())
642 {
643 if (c.value() == delim)
644 {
645 self.m_streambuf.sbumpc();
646 ++gcount;
647 }
648 else
649 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
650 }
651 }
652
653 if (gcount == 0)
654 throw stream_error{"istream get fail: no character extracted"};
655
656 if (at_eof)
657 self.setstate(ios_defs::eofbit);
658 }
659 catch(...)
660 {
661 if constexpr (is_cstr)
662 {
663 if constexpr (std::is_pointer_v<TOut>)
664 {
665 if (s != nullptr && n != 0)
666 *s++ = TChar{};
667 }
668 else if (n != 0)
669 *s++ = TChar{};
670 }
671 self.handle_exception(std::current_exception(), at_eof);
672 return s;
673 }
674
675 if constexpr (is_cstr)
676 *s++ = TChar{};
677 return s;
678 }

1113{
1114 using TChar = typename T::char_type;
1115 using TV = std::remove_cvref_t<TValue>;
1116 using TCtx = detail::in_ctx_t<TChar, TValue>;
1117 using TTarget = detail::in_target_t<TValue>;
1118
1119 constexpr bool stream_r = detail::extractable_with_stream<T, TValue>;
1120 constexpr bool iter_v = detail::extractable_with_iter<T, TValue>;
1121 constexpr bool iter_c = detail::extractable_with_ctx<T, TValue>;
1122
1123 static_assert(!(stream_r && (iter_v || iter_c)),
1124 "IOv2: this io_traits specialization provides both forms of sread() -- the stream form "
1125 "sread(stream, value) and the iterator form sread(iter, end, io, loc, value). Provide "
1126 "exactly one: the iterator form for formatted extraction, the stream form for a "
1127 "manipulator. See io/traits/traits_base.h.");
1128
1129 if constexpr (stream_r)
1130 {
1131 try
1132 {
1133 io_traits<TChar, TV>::sread(obj, static_cast<TTarget&>(value));
1134 }
1135 catch (...)
1136 {
1137 obj.handle_exception(std::current_exception());
1138 }
1139 }
1140 else if constexpr (iter_v || iter_c)
1141 {
1142 using sentry_type = typename T::in_sentry_type;
1143
1144 bool saw_eof = false;
1145 std::lock_guard guard(obj.io_mutex());
1146 try
1147 {
1148 auto iter = obj.i_iter(&saw_eof);
1149 bool skip = bool(obj.flags() & ios_defs::skipws);
1150 sentry_type cerb(obj, !skip);
1151
1152 if (obj.eof())
1153 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1154
1155 if constexpr (iter_v)
1156 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(),
1157 static_cast<TTarget&>(value));
1158 else
1159 {
1160 TCtx tmp = [&value]() -> TCtx {
1161 if constexpr (requires (const TV& v)
1162 { parse_context_type<TChar, TV>::make_parse_context(v); })
1163 return parse_context_type<TChar, TV>::make_parse_context(value);
1164 else
1165 return TCtx{};
1166 }();
1167 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1168 static_cast<TTarget&>(value) = static_cast<TV>(tmp);
1169 }
1170
1171 if (saw_eof) obj.setstate(ios_defs::eofbit);
1172 }
1173 catch(...)
1174 {
1175 obj.handle_exception(std::current_exception(), saw_eof);
1176 }
1177 }
1178 else
1179 static_assert(dependent_false_v<TValue>,
1180 "IOv2 internal: detail::extractable admitted this type but the dispatch chain has no "
1181 "branch for it. The concept and the chain must stay in step -- see "
1182 "io/utilities/istream_operators.h.");

1185}
