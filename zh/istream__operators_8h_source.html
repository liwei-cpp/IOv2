67template <typename TStream, bool involve_output>
68struct in_sentry
69{
128 in_sentry(TStream& is, bool noskip)
129 : m_is(is)
130 {
131 // good(), not operator bool: a sticky eofbit stops the work here, as the standard's sentry
132 // does. The check at the end stays !m_is -- a skip that reaches EOF sets eofbit only.
133 if (!m_is.good())
134 throw stream_error("istream_sentry create fail: Invalid istream");
135
136 if (auto* tied = is.tie())
137 tied->try_flush();
138
139 if constexpr (involve_output)
140 is.m_channel.switch_to_get();
141
142 if (!noskip)
143 {
144 try
145 {
146 auto ct = is.m_locale.template get<IOv2::ctype<typename TStream::char_type>>();
147 if (!ct)
148 throw stream_error{"istream ignore_ws fail: no ctype facet"};
149 auto c = is.m_channel.getc();
150 while (c.has_value() &&
151 ct->is_any(base_ft<ctype>::space, c.value()))
152 {
153 c = is.m_channel.nextc();
154 }
155
156 if (!c.has_value())
157 throw eof_error{};
158 }
159 catch(...)
160 {
161 is.handle_exception(std::current_exception());
162 }
163 }
164
165 if (!m_is)
166 throw stream_error("istream_sentry create fail: Invalid istream");
167 }
168
169 ~in_sentry() = default;
170
171 in_sentry(const in_sentry&) = delete;
172 in_sentry& operator=(const in_sentry&) = delete;
173 in_sentry(in_sentry&&) = delete;
174 in_sentry& operator=(in_sentry&&) = delete;
175
176private:
177 TStream& m_is;
178};

741 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
742 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
743 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
744 TOut get(this TSelf& self, TOut s, std::ptrdiff_t n, TChar delim)
745 {
746 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
747
748 std::size_t gcount = 0;
749 bool at_eof = false;
750 std::lock_guard guard(self.io_mutex());
751 try
752 {
753 using sentry_type = typename TSelf::in_sentry_type;
754 sentry_type cerb(self, true);
755 if constexpr (std::is_pointer_v<TOut>)
756 {
757 if (s == nullptr)
758 throw stream_error("istream get fail: null character sequence");
759 }
760 if (n < 0)
761 throw stream_error("istream get fail: negative buffer size");
762 if (n == 0)
763 throw stream_error("istream get fail: zero buffer size");
764 const auto cap = static_cast<std::size_t>(n);
765 // Room for no character: fail before asking the device for one. getline still
766 // looks, since a delimiter there is an empty line it consumes.
767 if constexpr (std::is_same_v<DelimPolicy, keep_sep>)
768 {
769 if (cap - is_cstr == 0)
770 throw stream_error{"istream get fail: no character extracted"};
771 }
772 auto c = self.m_channel.getc();
773 while ((gcount + is_cstr < cap) &&
774 (c.has_value()) &&
775 (c.value() != delim))
776 {
777 *s++ = c.value();
778 ++gcount;
779 // Full: consume this one and stop. A peek at the next would read a byte this
780 // call does not need. getline must peek, to tell a delimiter from overflow.
781 if constexpr (std::is_same_v<DelimPolicy, keep_sep>)
782 {
783 if (gcount + is_cstr == cap)
784 {
785 self.m_channel.bumpc();
786 break;
787 }
788 }
789 c = self.m_channel.nextc();
790 }
791
792 at_eof = (gcount + is_cstr < cap) && (!c.has_value());
793
794 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
795 {
796 if (c.has_value())
797 {
798 if (c.value() == delim)
799 {
800 self.m_channel.bumpc();
801 ++gcount;
802 }
803 else
804 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
805 }
806 }
807
808 if (gcount == 0)
809 throw stream_error{"istream get fail: no character extracted"};
810
811 if (at_eof)
812 self.setstate(ios_defs::eofbit);
813 }
814 catch(...)
815 {
816 if constexpr (is_cstr)
817 {
818 if constexpr (std::is_pointer_v<TOut>)
819 {
820 if (s != nullptr && n > 0)
821 *s++ = TChar{};
822 }
823 else if (n > 0)
824 *s++ = TChar{};
825 }
826 self.handle_exception(std::current_exception(), at_eof);
827 return s;
828 }
829
830 if constexpr (is_cstr)
831 *s++ = TChar{};
832 return s;
833 }

1129 self.m_channel.bumpc();

1309template <istream_type T, typename TValue>
1310 requires detail::extractable<T, TValue>
1311// The forwarding reference below is a binder plus a value-category carrier, never a forwarder;
1312// see the @note above for why `std::forward` here would be wrong rather than merely redundant.
1313// NOLINTNEXTLINE(cppcoreguidelines-missing-std-forward)
1314T& operator>>(T& obj, TValue&& value)
1315{
1316 using TChar = typename T::char_type;
1317 using TV = std::remove_cvref_t<TValue>;
1318 using TCtx = detail::in_ctx_t<TChar, TValue>;
1319 using TTarget = detail::in_target_t<TValue>;
1320
1321 constexpr bool stream_r = detail::extractable_with_stream<T, TValue>;
1322 constexpr bool iter_v = detail::extractable_with_iter<T, TValue>;
1323 constexpr bool iter_c = detail::extractable_with_ctx<T, TValue>;
1324
1325 static_assert(!(stream_r && (iter_v || iter_c)),
1326 "IOv2: this io_traits specialization provides both forms of sread() -- the stream form "
1327 "sread(stream, value) and the iterator form sread(iter, end, io, loc, value). Provide "
1328 "exactly one: the iterator form for formatted extraction, the stream form for a "
1329 "manipulator. See io/traits/traits_base.h.");
1330
1331 if constexpr (stream_r)
1332 {
1333 try
1334 {
1335 io_traits<TChar, TV>::sread(obj, static_cast<TTarget&>(value));
1336 }
1337 catch (...)
1338 {
1339 obj.handle_exception(std::current_exception());
1340 }
1341 }
1342 else if constexpr (iter_v || iter_c)
1343 {
1344 using sentry_type = typename T::in_sentry_type;
1345
1346 bool saw_eof = false;
1347 std::lock_guard guard(obj.io_mutex());
1348 try
1349 {
1350 auto iter = obj.i_iter(&saw_eof);
1351 bool skip = bool(obj.flags() & ios_defs::skipws);
1352 sentry_type cerb(obj, !skip);
1353
1354 // ws is the one extraction that may read nothing: like std::ws it reports the end
1355 // with eofbit alone, so a skip that ran off the end is not a failure for it.
1356 if constexpr (!std::is_same_v<TV, ws_t>)
1357 {
1358 if (obj.eof())
1359 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1360 }
1361
1362 if constexpr (iter_v)
1363 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(),
1364 static_cast<TTarget&>(value));
1365 else
1366 {
1367 using TPct = parse_context_type<TChar, TV>;
1368 TCtx tmp = [&value]() -> TCtx {
1369 // Detect the member by name only, so a mis-written one reaches a static_assert
1370 // instead of being read as absent and silently default constructed.
1371 if constexpr (detail::declares_maker<TPct, TTarget>)
1372 {
1373 if constexpr (requires { &TPct::make_parse_context; })
1374 {
1375 static_assert(
1376 detail::maker_signature_v<TCtx, TV,
1377 decltype(&TPct::make_parse_context)>,
1378 "IOv2: this parse_context_type's make_parse_context() must be "
1379 "declared exactly `static type make_parse_context(const T&)` -- it "
1380 "seeds the context from the target, so it takes the target by const "
1381 "reference and returns parse_context_type<TChar, T>::type by value. "
1382 "See io/traits/traits_base.h.");
1383 return TPct::make_parse_context(static_cast<const TV&>(value));
1384 }
1385 else
1386 static_assert(dependent_false_v<TCtx>,
1387 "IOv2: this parse_context_type's make_parse_context() must be a "
1388 "single non-template, non-overloaded static member function, and "
1389 "must be public and not deleted. See io/traits/traits_base.h.");
1390 }
1391 else if constexpr (std::default_initializable<TCtx>)
1392 return TCtx{};
1393 else
1394 static_assert(dependent_false_v<TCtx>,
1395 "IOv2: this parse_context_type's context has no make_parse_context() "
1396 "and is not default constructible, so there is no way to obtain one. "
1397 "Give the context a default constructor, or give parse_context_type a "
1398 "static make_parse_context(const T&). See io/traits/traits_base.h.");
1399 }();
1400 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1401 tmp.convert_to(static_cast<TTarget&>(value));
1402 }
1403
1404 if (saw_eof) obj.setstate(ios_defs::eofbit);
1405 }
1406 catch(...)
1407 {
1408 obj.handle_exception(std::current_exception(), saw_eof);
1409 }
1410 }
1411 else
1412 static_assert(dependent_false_v<TValue>,
1413 "IOv2 internal: detail::extractable admitted this type but the dispatch chain has no "
1414 "branch for it. The concept and the chain must stay in step -- see "
1415 "IOv2/io/utilities/istream_operators.h.");
1416
1417 return obj;
