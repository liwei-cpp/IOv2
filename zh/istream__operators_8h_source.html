116 in_sentry(TStream& is, bool noskip)
117 : m_is(is)
118 {
119 if (!m_is)
120 throw stream_error("istream_sentry create fail: Invalid istream");
121
122 if (auto* tied = is.tie())
123 {
124 try { tied->try_flush(); }
125 catch (...) {} // NOLINT(bugprone-empty-catch)
126 }
127
128 if constexpr (involve_output)
129 is.m_streambuf.switch_to_get();
130
131 if (!noskip)
132 {
133 try
134 {
135 auto ct = is.m_locale.template get<IOv2::ctype<typename TStream::char_type>>();

137 throw stream_error{"istream ignore_ws fail: no ctype facet"};

616 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
617 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
618 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
619 TOut get(this TSelf& self, TOut s, std::size_t n, TChar delim)
620 {
621 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
622
623 std::size_t gcount = 0;
624 bool at_eof = false;
625 std::lock_guard guard(self.io_mutex());
626 try
627 {
628 using sentry_type = typename TSelf::in_sentry_type;
629 sentry_type cerb(self, true);
630 if constexpr (std::is_pointer_v<TOut>)
631 {
632 if (s == nullptr)
633 throw stream_error("istream get fail: null character sequence");
634 }
635 if (n == 0)
636 throw stream_error("istream get fail: zero buffer size");
637 auto c = self.m_streambuf.sgetc();
638 while ((gcount + is_cstr < n) &&
639 (c.has_value()) &&
640 (c.value() != delim))
641 {
642 *s++ = c.value();
643 ++gcount;
644 c = self.m_streambuf.snextc();
645 }
646
647 at_eof = (gcount + is_cstr < n) && (!c.has_value());
648
649 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
650 {
651 if (c.has_value())
652 {
653 if (c.value() == delim)
654 {
655 self.m_streambuf.sbumpc();
656 ++gcount;
657 }
658 else
659 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
660 }
661 }
662
663 if (gcount == 0)
664 throw stream_error{"istream get fail: no character extracted"};
665
666 if (at_eof)
667 self.setstate(ios_defs::eofbit);
668 }
669 catch(...)
670 {
671 if constexpr (is_cstr)
672 {
673 if constexpr (std::is_pointer_v<TOut>)
674 {
675 if (s != nullptr && n != 0)
676 *s++ = TChar{};
677 }
678 else if (n != 0)
679 *s++ = TChar{};
680 }
681 self.handle_exception(std::current_exception(), at_eof);
682 return s;
683 }
684
685 if constexpr (is_cstr)
686 *s++ = TChar{};
687 return s;
688 }

1140 {
1141 try

1144 }
1145 catch (...)
1146 {
1147 obj.handle_exception(std::current_exception());
1148 }
1149 }

1167 static_cast<TTarget&>(value));

1173 return parse_context_type<TChar, TV>::make_parse_context(value);
1174 else
1175 return TCtx{};
1176 }();
1177 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);

1179 }

1182 }
