466 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
467 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
468 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
469 TOut get(this TSelf& self, TOut s, size_t n, TChar delim)
470 {
471 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
472
473 size_t gcount = 0;
474 bool at_eof = false;
475 std::unique_lock lk(self.io_mutex(), std::defer_lock);
476 try
477 {
478 using sentry_type = typename TSelf::in_sentry_type;
479 sentry_type cerb(self, true, lk);
480 if constexpr (std::is_pointer_v<TOut>)
481 {
482 if (s == nullptr)
483 throw stream_error("istream get fail: null character sequence");
484 }
485 if (n == 0)
486 throw stream_error("istream get fail: zero buffer size");
487 auto c = self.m_streambuf.sgetc();
488 while ((gcount + is_cstr < n) &&
489 (c.has_value()) &&
490 (c.value() != delim))
491 {
492 *s++ = c.value();
493 ++gcount;
494 c = self.m_streambuf.snextc();
495 }
496
497 at_eof = (gcount + is_cstr < n) && (!c.has_value());
498
499 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
500 {
501 if (c.has_value())
502 {
503 if (c.value() == delim)
504 {
505 self.m_streambuf.sbumpc();
506 ++gcount;
507 }
508 else
509 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
510 }
511 }
512
513 if (gcount == 0)
514 throw stream_error{"istream get fail: no character extracted"};
515
516 if (at_eof)
517 self.setstate(ios_defs::eofbit);
518 }
519 catch(...)
520 {
521 if constexpr (is_cstr)
522 {
523 if constexpr (std::is_pointer_v<TOut>)
524 {
525 if (s != nullptr && n != 0)
526 *s++ = TChar{};
527 }
528 else if (n != 0)
529 *s++ = TChar{};
530 }
531 self.handle_exception(std::current_exception(), at_eof);
532 return s;
533 }
534
535 if constexpr (is_cstr)
536 *s++ = TChar{};
537 return s;
538 }

1150template <istream_type T, typename TValue>
1151 requires is_reader_def<typename T::char_type,
1152 typename parse_context_type<typename T::char_type, TValue>::type>
1153T& operator>>(T& obj, TValue& value)
1154{
1155 using TChar = typename T::char_type;
1156 using TCtx = typename parse_context_type<TChar, TValue>::type;
1157 using sentry_type = typename T::in_sentry_type;
1158
1159 bool saw_eof = false;
1160 std::unique_lock lk(obj.io_mutex(), std::defer_lock);
1161 try
1162 {
1163 auto iter = obj.i_iter(&saw_eof);
1164 bool skip = bool(obj.flags() & ios_defs::skipws);
1165 sentry_type cerb(obj, !skip, lk);
1166
1167 if (obj.eof())
1168 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1169
1170 if constexpr (std::is_same_v<TCtx, TValue>)
1171 reader<TChar, TValue>::sread(iter, std::default_sentinel, obj, obj.locale(), value);
1172 else
1173 {
1174 TCtx tmp;
1175 reader<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1176 value = static_cast<TValue>(tmp);
1177 }
1178
1179 if (saw_eof) obj.setstate(ios_defs::eofbit);
1180 }
1181 catch(...)
1182 {
1183 obj.handle_exception(std::current_exception(), saw_eof);
1184 }
1185
1186 return obj;
1187}
