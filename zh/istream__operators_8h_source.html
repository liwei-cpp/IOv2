697 template <typename DelimPolicy, typename CStrPolicy, typename TOut, typename TSelf>
698 requires ((std::is_same_v<DelimPolicy, cons_sep> || std::is_same_v<DelimPolicy, keep_sep>) &&
699 (std::is_same_v<CStrPolicy, app_zt> || std::is_same_v<CStrPolicy, no_zt>))
700 TOut get(this TSelf& self, TOut s, std::ptrdiff_t n, TChar delim)
701 {
702 constexpr bool is_cstr = std::is_same_v<CStrPolicy, app_zt>;
703
704 std::size_t gcount = 0;
705 bool at_eof = false;
706 std::lock_guard guard(self.io_mutex());
707 try
708 {
709 using sentry_type = typename TSelf::in_sentry_type;
710 sentry_type cerb(self, true);
711 if constexpr (std::is_pointer_v<TOut>)
712 {
713 if (s == nullptr)
714 throw stream_error("istream get fail: null character sequence");
715 }
716 if (n < 0)
717 throw stream_error("istream get fail: negative buffer size");
718 if (n == 0)
719 throw stream_error("istream get fail: zero buffer size");
720 const auto cap = static_cast<std::size_t>(n);
721 auto c = self.m_streambuf.sgetc();
722 while ((gcount + is_cstr < cap) &&
723 (c.has_value()) &&
724 (c.value() != delim))
725 {
726 *s++ = c.value();
727 ++gcount;
728 c = self.m_streambuf.snextc();
729 }
730
731 at_eof = (gcount + is_cstr < cap) && (!c.has_value());
732
733 if constexpr (std::is_same_v<DelimPolicy, cons_sep>)
734 {
735 if (c.has_value())
736 {
737 if (c.value() == delim)
738 {
739 self.m_streambuf.sbumpc();
740 ++gcount;
741 }
742 else
743 throw stream_error("istream getline fail: delimiter not found within buffer capacity");
744 }
745 }
746
747 if (gcount == 0)
748 throw stream_error{"istream get fail: no character extracted"};
749
750 if (at_eof)
751 self.setstate(ios_defs::eofbit);
752 }
753 catch(...)
754 {
755 if constexpr (is_cstr)
756 {
757 if constexpr (std::is_pointer_v<TOut>)
758 {
759 if (s != nullptr && n > 0)
760 *s++ = TChar{};
761 }
762 else if (n > 0)
763 *s++ = TChar{};
764 }
765 self.handle_exception(std::current_exception(), at_eof);
766 return s;
767 }
768
769 if constexpr (is_cstr)
770 *s++ = TChar{};
771 return s;
772 }

1218template <istream_type T, typename TValue>
1219 requires detail::extractable<T, TValue>
1220// The forwarding reference below is a binder plus a value-category carrier, never a forwarder;
1221// see the @note above for why `std::forward` here would be wrong rather than merely redundant.
1222// NOLINTNEXTLINE(cppcoreguidelines-missing-std-forward)
1223T& operator>>(T& obj, TValue&& value)
1224{
1225 using TChar = typename T::char_type;
1226 using TV = std::remove_cvref_t<TValue>;
1227 using TCtx = detail::in_ctx_t<TChar, TValue>;
1228 using TTarget = detail::in_target_t<TValue>;
1229
1230 constexpr bool stream_r = detail::extractable_with_stream<T, TValue>;
1231 constexpr bool iter_v = detail::extractable_with_iter<T, TValue>;
1232 constexpr bool iter_c = detail::extractable_with_ctx<T, TValue>;
1233
1234 static_assert(!(stream_r && (iter_v || iter_c)),
1235 "IOv2: this io_traits specialization provides both forms of sread() -- the stream form "
1236 "sread(stream, value) and the iterator form sread(iter, end, io, loc, value). Provide "
1237 "exactly one: the iterator form for formatted extraction, the stream form for a "
1238 "manipulator. See io/traits/traits_base.h.");
1239
1240 if constexpr (stream_r)
1241 {
1242 try
1243 {
1244 io_traits<TChar, TV>::sread(obj, static_cast<TTarget&>(value));
1245 }
1246 catch (...)
1247 {
1248 obj.handle_exception(std::current_exception());
1249 }
1250 }
1251 else if constexpr (iter_v || iter_c)
1252 {
1253 using sentry_type = typename T::in_sentry_type;
1254
1255 bool saw_eof = false;
1256 std::lock_guard guard(obj.io_mutex());
1257 try
1258 {
1259 auto iter = obj.i_iter(&saw_eof);
1260 bool skip = bool(obj.flags() & ios_defs::skipws);
1261 sentry_type cerb(obj, !skip);
1262
1263 if (obj.eof())
1264 throw stream_error("istream extraction fail: reached EOF with no value extracted");
1265
1266 if constexpr (iter_v)
1267 io_traits<TChar, TV>::sread(iter, std::default_sentinel, obj, obj.locale(),
1268 static_cast<TTarget&>(value));
1269 else
1270 {
1271 using TPct = parse_context_type<TChar, TV>;
1272 TCtx tmp = [&value]() -> TCtx {
1273 // Detect the member by name only, so a mis-written one reaches a static_assert
1274 // instead of being read as absent and silently default constructed.
1275 if constexpr (detail::declares_maker<TPct, TTarget>)
1276 {
1277 if constexpr (requires { &TPct::make_parse_context; })
1278 {
1279 static_assert(
1280 detail::maker_signature_v<TCtx, TV,
1281 decltype(&TPct::make_parse_context)>,
1282 "IOv2: this parse_context_type's make_parse_context() must be "
1283 "declared exactly `static type make_parse_context(const T&)` -- it "
1284 "seeds the context from the target, so it takes the target by const "
1285 "reference and returns parse_context_type<TChar, T>::type by value. "
1286 "See io/traits/traits_base.h.");
1287 return TPct::make_parse_context(static_cast<const TV&>(value));
1288 }
1289 else
1290 static_assert(dependent_false_v<TCtx>,
1291 "IOv2: this parse_context_type's make_parse_context() must be a "
1292 "single non-template, non-overloaded static member function, and "
1293 "must be public and not deleted. See io/traits/traits_base.h.");
1294 }
1295 else if constexpr (std::default_initializable<TCtx>)
1296 return TCtx{};
1297 else
1298 static_assert(dependent_false_v<TCtx>,
1299 "IOv2: this parse_context_type's context has no make_parse_context() "
1300 "and is not default constructible, so there is no way to obtain one. "
1301 "Give the context a default constructor, or give parse_context_type a "
1302 "static make_parse_context(const T&). See io/traits/traits_base.h.");
1303 }();
1304 io_traits<TChar, TCtx>::sread(iter, std::default_sentinel, obj, obj.locale(), tmp);
1305 tmp.convert_to(static_cast<TTarget&>(value));
1306 }
1307
1308 if (saw_eof) obj.setstate(ios_defs::eofbit);
1309 }
1310 catch(...)
1311 {
1312 obj.handle_exception(std::current_exception(), saw_eof);
1313 }
1314 }
1315 else
1316 static_assert(dependent_false_v<TValue>,
1317 "IOv2 internal: detail::extractable admitted this type but the dispatch chain has no "
1318 "branch for it. The concept and the chain must stay in step -- see "
1319 "IOv2/io/utilities/istream_operators.h.");
1320
1321 return obj;
1322}
