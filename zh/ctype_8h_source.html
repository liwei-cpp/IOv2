481template <typename CharT>
482 requires (sizeof(CharT) == 1)
483class ctype<CharT> : public detail::ctype_ops
484{
485 // Note: we have to use unsigned char here to avoid negative value.
486 // Sizing is in units of "values representable by unsigned char", not a
487 // hard-coded 256. The CHAR_BIT == 8 assumption is enforced by a
488 // static_assert in ctype_details.h, so on every supported platform
489 // s_len == 256.
493 constexpr static unsigned s_len = std::numeric_limits<unsigned char>::max() + 1;
494
495public:
499 using create_rules = facet_create_rule<ctype_conf<CharT>>;
500
503 using char_type = CharT;
504
508 using mask = typename ctype_conf<CharT>::mask;
509
530 template <shared_ptr_to<ctype_conf<CharT>> TConfPtr>
531 ctype(TConfPtr p_obj)
532 {
533 if (!p_obj) throw io_error("shared_ptr is empty");
534 for (unsigned i = 0; i < s_len; ++i)
535 {
536 m_table[i] = p_obj->is(static_cast<CharT>(i));
537 m_toupper[i] = p_obj->toupper(static_cast<CharT>(i));
538 m_tolower[i] = p_obj->tolower(static_cast<CharT>(i));
539 m_widen[i] = p_obj->widen(static_cast<CharT>(i));
540 m_narrow[i] = p_obj->narrow(static_cast<CharT>(i));
541 }
542 }
