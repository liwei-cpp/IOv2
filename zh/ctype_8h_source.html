480template <typename CharT>

482class ctype<CharT> : public detail::ctype_ops
483{
484 // Note: we have to use unsigned char here to avoid negative value.
485 // Sizing is in units of "values representable by unsigned char", not a
486 // hard-coded 256. The CHAR_BIT == 8 assumption is enforced by a
487 // static_assert in ctype_details.h, so on every supported platform
488 // s_len == 256.
492 constexpr static unsigned s_len = std::numeric_limits<unsigned char>::max() + 1;
493
494public:
498 using create_rules = facet_create_rule<ctype_conf<CharT>>;
499
502 using char_type = CharT;
503
507 using mask = typename ctype_conf<CharT>::mask;
508
529 template <shared_ptr_to<ctype_conf<CharT>> TConfPtr>
530 ctype(TConfPtr p_obj)
531 {
532 if (!p_obj) throw io_error("shared_ptr is empty");
533 for (unsigned i = 0; i < s_len; ++i)
534 {
535 m_table[i] = p_obj->is(static_cast<CharT>(i));
536 m_toupper[i] = p_obj->toupper(static_cast<CharT>(i));
537 m_tolower[i] = p_obj->tolower(static_cast<CharT>(i));
538 m_widen[i] = p_obj->widen(static_cast<CharT>(i));
