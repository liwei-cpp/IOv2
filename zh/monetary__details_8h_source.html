222 static pattern s_construct_pattern(pattern_spec spec)
223 {
224 using enum part;
225 const auto [precedes, sp, posn] = spec;
226
227 // POSIX fills sign_posn with CHAR_MAX when the locale has nothing to say; a
228 // value outside either range leaves nothing to build a layout from.
229 if (posn < 0 || posn > 4 || sp < 0 || sp > 2)
230 return s_default_pattern;
231
232 // Two things get laid out: the amount, and the currency symbol -- which for
233 // sign_posn 3 and 4 has the sign glued to its front or its back, and so
234 // travels with it. Whichever of the two goes first is what `precedes` says.
235 std::array<part, 2> group{};
236 std::size_t group_len = 0;
237 if (posn == 3)
238 group[group_len++] = sign;
239 group[group_len++] = symbol;
240 if (posn == 4)
241 group[group_len++] = sign;
242
243 pattern ret{};
244 std::size_t at = 0;
245
246 // A sign that was not glued to the symbol brackets the whole field instead:
247 // in front for sign_posn 0 and 1, behind for 2.
248 if (posn <= 1)
249 ret[at++] = sign;
250
251 // sep_by_space 1 puts the separator on the seam between the symbol and the
252 // amount.
253 if (precedes)
254 {
255 for (std::size_t k = 0; k < group_len; ++k)
256 ret[at++] = group[k];
257 if (sp == 1)
258 ret[at++] = space;
259 ret[at++] = value;
260 }
261 else
262 {
263 ret[at++] = value;
264 if (sp == 1)
265 ret[at++] = space;
266 for (std::size_t k = 0; k < group_len; ++k)
267 ret[at++] = group[k];
268 }
269
270 if (posn == 2)
271 ret[at++] = sign;
272
273 // sep_by_space 2 puts it next to the sign instead: after a leading sign, before
274 // a trailing one, and between a sign and the symbol it is glued to.
275 if (sp == 2 && posn != 0)
276 {
277 std::size_t gap = 2;
278 if (ret[0] == sign || (ret[1] == sign && ret[0] == symbol))
279 gap = 1;
280 for (std::size_t k = at; k > gap; --k)
281 ret[k] = ret[k - 1];
282 ret[gap] = space;
283 ++at;
284 }
285
286 // Without a space the field is one slot short. `none` takes up the slack, and
287 // being last is the one position it is allowed to hold.
288 if (at < ret.size())
289 ret[at++] = none;
290
291 return ret;
292 }
