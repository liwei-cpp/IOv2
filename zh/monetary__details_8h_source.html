187 static pattern s_construct_pattern(pattern_spec spec)
188 {
189 using enum part;
190 const auto [precedes, sp, posn] = spec;
191 pattern ret;
192
193 // This insanely complicated routine attempts to construct a valid
194 // pattern for use with moneypunct. A couple of invariants:
195
196 // if (precedes) symbol -> value
197 // else value -> symbol
198
199 // if (sp) space
200 // else none
201
202 // none == never first
203 // space never first or last
204 switch (posn)
205 {
206 case 0:
207 case 1:
208 // 1 The sign precedes the value and symbol.
209 ret[0] = sign;
210 if (sp)
211 {
212 // Pattern starts with sign.
213 if (precedes)
214 {
215 ret[1] = symbol;
216 ret[3] = value;
217 }
218 else
219 {
220 ret[1] = value;
221 ret[3] = symbol;
222 }
223 ret[2] = space;
224 }
225 else
226 {
227 // Pattern starts with sign and ends with none.
228 if (precedes)
229 {
230 ret[1] = symbol;
231 ret[2] = value;
232 }
233 else
234 {
235 ret[1] = value;
236 ret[2] = symbol;
237 }
238 ret[3] = none;
239 }
240 break;
241 case 2:
242 // 2 The sign follows the value and symbol.
243 if (sp)
244 {
245 // Pattern either ends with sign.
246 if (precedes)
247 {
248 ret[0] = symbol;
249 ret[2] = value;
250 }
251 else
252 {
253 ret[0] = value;
254 ret[2] = symbol;
255 }
256 ret[1] = space;
257 ret[3] = sign;
258 }
259 else
260 {
261 // Pattern ends with sign then none.
262 if (precedes)
263 {
264 ret[0] = symbol;
265 ret[1] = value;
266 }
267 else
268 {
269 ret[0] = value;
270 ret[1] = symbol;
271 }
272 ret[2] = sign;
273 ret[3] = none;
274 }
275 break;
276 case 3:
277 // 3 The sign immediately precedes the symbol.
278 if (precedes)
279 {
280 ret[0] = sign;
281 ret[1] = symbol;
282 if (sp)
283 {
284 ret[2] = space;
285 ret[3] = value;
286 }
287 else
288 {
289 ret[2] = value;
290 ret[3] = none;
291 }
292 }
293 else
294 {
295 ret[0] = value;
296 if (sp)
297 {
298 ret[1] = space;
299 ret[2] = sign;
300 ret[3] = symbol;
301 }
302 else
303 {
304 ret[1] = sign;
305 ret[2] = symbol;
306 ret[3] = none;
307 }
308 }
309 break;
310 case 4:
311 // 4 The sign immediately follows the symbol.
312 if (precedes)
313 {
314 ret[0] = symbol;
315 ret[1] = sign;
316 if (sp)
317 {
318 ret[2] = space;
319 ret[3] = value;
320 }
321 else
322 {
323 ret[2] = value;
324 ret[3] = none;
325 }
326 }
327 else
328 {
329 ret[0] = value;
330 if (sp)
331 {
332 ret[1] = space;
333 ret[2] = symbol;
334 ret[3] = sign;
335 }
336 else
337 {
338 ret[1] = symbol;
339 ret[2] = sign;
340 ret[3] = none;
341 }
342 }
343 break;
344 default:
345 ret = s_default_pattern;
346 }
347 return ret;
