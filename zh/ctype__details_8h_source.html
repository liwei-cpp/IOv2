76// The ctype lookup tables in this header and in ctype.h are sized as
77// std::numeric_limits<unsigned char>::max() + 1, on the assumption that a
78// byte is 8 bits and that the table covers all values an unsigned char can
79// hold. Exotic platforms with CHAR_BIT != 8 (some DSPs, historical mainframes)
80// would silently grow every per-byte table by 2^(CHAR_BIT-8)x. Reject them at
81// compile time rather than ship a non-obviously-broken binary.
82static_assert(CHAR_BIT == 8,
83 "IOv2/facet/ctype tables assume an 8-bit byte; see ctype.h::s_len");

216template <>
217class ctype_conf<char> : public ft_basic<ctype<char>>
218{
219public:
235 ctype_conf(const std::string& name)
236 : ft_basic<ctype<char>>()
237 , m_inter_locale(name)
238 {
239 // Build lookup tables with the C library's locale-taking functions (see ctype_backend.h)
240 const detail::clocale_handle loc = m_inter_locale.c_locale;
241 for (unsigned u = 0; u <= std::numeric_limits<unsigned char>::max(); ++u)
242 {
243 const auto c = static_cast<int>(u);
244 m_toupper_table[u] = detail::ctype_toupper(c, loc);
245 m_tolower_table[u] = detail::ctype_tolower(c, loc);
246
247 mask m = 0;
248 if (detail::ctype_isupper(c, loc)) m |= base_ft<ctype>::upper;
249 if (detail::ctype_islower(c, loc)) m |= base_ft<ctype>::lower;
250 if (detail::ctype_isalpha(c, loc)) m |= base_ft<ctype>::alpha;
251 if (detail::ctype_isdigit(c, loc)) m |= base_ft<ctype>::digit;
252 if (detail::ctype_isxdigit(c, loc)) m |= base_ft<ctype>::xdigit;
253 if (detail::ctype_isspace(c, loc)) m |= base_ft<ctype>::space;
254 if (detail::ctype_isprint(c, loc)) m |= base_ft<ctype>::print;
255 if (detail::ctype_iscntrl(c, loc)) m |= base_ft<ctype>::cntrl;
256 if (detail::ctype_ispunct(c, loc)) m |= base_ft<ctype>::punct;
257 m_table[u] = m;
258 }
259 }
260
261public:
280 [[nodiscard]] virtual mask is(char c) const
281 {
282 return m_table[static_cast<unsigned char>(c)];
283 }
284
303 [[nodiscard]] virtual char toupper(char c) const
304 {
305 return static_cast<char>(m_toupper_table[static_cast<unsigned char>(c)]);
306 }
307
326 [[nodiscard]] virtual char tolower(char c) const
327 {
328 return static_cast<char>(m_tolower_table[static_cast<unsigned char>(c)]);
329 }
330
348 [[nodiscard]] virtual char widen(char c) const { return c; }
349
367 [[nodiscard]] virtual std::optional<char> narrow(char c) const { return c; }
368
369private:
372 clocale_wrapper m_inter_locale;
373
377 std::array<int, std::numeric_limits<unsigned char>::max() + 1> m_toupper_table {};
378
382 std::array<int, std::numeric_limits<unsigned char>::max() + 1> m_tolower_table {};
383
387 std::array<mask, std::numeric_limits<unsigned char>::max() + 1> m_table {};
388};

513 [[nodiscard]] virtual base_ft<ctype>::mask is(CharT _c) const
514 {
515 if (out_of_wchar_range(_c)) return 0;
516
517 base_ft<ctype>::mask res = 0;
518 const auto c = static_cast<wchar_t>(_c);
519 const auto in = [&](detail::ctype_wclass cls)
520 { return detail::ctype_iswctype(static_cast<wint_t>(c), cls, m_inter_locale.c_locale); };
521 if (in(m_wmask_upper)) res |= base_ft<ctype>::upper;
522 if (in(m_wmask_lower)) res |= base_ft<ctype>::lower;
523 if (in(m_wmask_alpha)) res |= base_ft<ctype>::alpha;
524 if (in(m_wmask_digit)) res |= base_ft<ctype>::digit;
525 if (in(m_wmask_xdigit)) res |= base_ft<ctype>::xdigit;
526 if (in(m_wmask_space)) res |= base_ft<ctype>::space;
527 if (in(m_wmask_print)) res |= base_ft<ctype>::print;
528 if (in(m_wmask_cntrl)) res |= base_ft<ctype>::cntrl;
529 if (in(m_wmask_punct)) res |= base_ft<ctype>::punct;
530 return res;
