380 ostream& operator=(ostream&& other) noexcept
381 {
382 if (this == &other) return *this;
383
384 std::lock_guard guard(this->io_mutex());
385 // NOLINTBEGIN(bugprone-use-after-move): each `std::move(other)` below binds to a
386 // *different* base subobject, and every one of those base assignments touches only its
387 // own members; `m_streambuf` and `m_locale` belong to no base at all. Nothing is read
388 // after being moved from -- clang-tidy just cannot see that the operands are disjoint.
389 ios_state<TChar>::operator=(std::move(other));
390 out_flusher<ostream<TDevice, TChar>>::operator=(std::move(other));
391 ostream_operators<TChar>::operator=(std::move(other));
392 stream_common_operators::operator=(std::move(other));
393 m_streambuf = std::move(other.m_streambuf);
394 m_locale = std::move(other.m_locale);
395 // NOLINTEND(bugprone-use-after-move)
396 return *this;
397 }
398
421 // `override` because the `abs_flusher` base (reached through `out_flusher`) has a virtual
422 // destructor, so this one is virtual whether or not it says so.
423 ~ostream() override = default;

460 ostream& operator=(const ostream& other)
461 {
462 static_assert(std::is_nothrow_move_assignable_v<ostream<TDevice, TChar>>,
463 "the noexcept on move assignment must stay: it is what keeps the commit "
464 "step below from leaving *this half-updated");
465 if (this != &other)
466 {
467 ostream tmp(other);
468 *this = std::move(tmp);
469 }

471 }

612 requires (std::is_same_v<typename T::char_type, TChar>)
613 static void swrite(T& os, const endl_t&)

615 std::lock_guard guard(os.io_mutex());

618 auto mp = os.locale().template get<ctype<TChar>>();

620 throw stream_error("endl fail: cannot get ctype facet");

622 os.put(mp->widen('\n'), /*force_flush=*/true);
623 }
624 catch(...)
