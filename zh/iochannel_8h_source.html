392 std::size_t getn(char_type* s, std::size_t n, std::size_t* got = nullptr) requires (IsIn)
393 {
394 if (got) *got = 0;
395 if ((s == nullptr) || (n == 0)) return 0;
396
397 switch_to_get();
398
399 std::size_t res = 0;
400 while (!m_read_buf.empty())
401 {
402 *s++ = m_read_buf.back();
403 m_read_buf.pop_back();
404 if (++res == n) break;
405 }
406 if (got) *got = res;
407
408 while (res < n)
409 {
410 std::size_t c = m_cvt.get(s, n - res);
411 assert(c <= n - res);
412 if (c == 0) break;
413 res += c;
414 s += c;
415 if (got) *got = res;
416 }
417 return res;
418 }

788 void switch_to_put() requires (IsOut)
789 {
790 if constexpr (IsIn)
791 {
792 if (!m_read_buf.empty())
793 {
794 try
795 {
796 const std::size_t pos = tell();
797 m_cvt.seek(pos);
798 }
799 catch (const cvt_error& e)
800 {
801 throw cvt_error("base_channel::switch_to_put fails: cannot reposition "
802 + std::to_string(m_read_buf.size())
803 + " buffered/put-back character(s) before switching to output mode: "
804 + e.what());
805 }
806 // Must immediately follow a *successful* seek: the seek consumed the buffer, and
807 // tell() subtracts m_read_buf.size(), so keeping it would subtract twice. A failed
808 // seek skips this and leaves the buffer intact; a throw from the switch below
809 // therefore finds it already cleared. Both are by design -- see iostream's @warning.
810 m_read_buf.clear();
811 }
812 }
813 m_cvt.switch_to_put();
814 }

883 std::pair<device_type, std::exception_ptr> detach() noexcept
884 {
885 if constexpr (IsIn)
886 {
887 if (!m_read_buf.empty())
888 {
889 try
890 {
891 // On over-putback (more put-back than actually read), tell()
892 // saturates at 0, so the device is handed back positioned at the
893 // stream origin. That is consistent with the logical read cursor
894 // being at 0; see tell()/putbackc() and switch_to_put().
895 const std::size_t pos = tell();
896 m_cvt.seek(pos);
897 }
898 catch (...) // NOLINT(bugprone-empty-catch)
899 {
900 // Swallowed on purpose: a device that doesn't support positioning
901 // (e.g. a pipe/tty-backed stdin) can never honor this reposition,
902 // and that is an inherent property of the device, not an
903 // actionable failure - the caller could not have repositioned it
904 // either after getting it back. Reporting this error would make
905 // routine detach()/attach() cycles on such devices start throwing
906 // (e.g. reset() right after a formatted read has left
907 // one buffered lookahead character), even though nothing is
908 // actually broken; losing that lookahead (and, below, the root
909 // converter's read buffer) is the
910 // accepted, unavoidable cost for non-positionable devices.
911 }
912 m_read_buf.clear();
913 }
914 }
915 return m_cvt.detach();
916 }
