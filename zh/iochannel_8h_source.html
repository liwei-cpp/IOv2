392 std::size_t getn(char_type* s, std::size_t n, std::size_t* got = nullptr) requires (IsIn)
393 {
394 if (got) *got = 0;
395 if ((s == nullptr) || (n == 0)) return 0;
396
397 if constexpr (IsOut)
398 switch_to_get();
399
400 std::size_t res = 0;
401 while (!m_read_buf.empty())
402 {
403 *s++ = m_read_buf.back();
404 m_read_buf.pop_back();
405 if (++res == n) break;
406 }
407 if (got) *got = res;
408
409 while (res < n)
410 {
411 std::size_t c = m_cvt.get(s, n - res);
412 assert(c <= n - res);
413 if (c == 0) break;
414 res += c;
415 s += c;
416 if (got) *got = res;
417 }
418 return res;
419 }

728 void switch_to_put() requires (IsIn && IsOut)
729 {
730 if (!m_read_buf.empty())
731 {
732 try
733 {
734 const std::size_t pos = tell();
735 m_cvt.seek(pos);
736 }
737 catch (const cvt_error& e)
738 {
739 throw cvt_error("base_channel::switch_to_put fails: cannot reposition "
740 + std::to_string(m_read_buf.size())
741 + " buffered/put-back character(s) before switching to output mode: "
742 + e.what());
743 }
744 // Must immediately follow a *successful* seek: the seek consumed the buffer, and
745 // tell() subtracts m_read_buf.size(), so keeping it would subtract twice. A failed
746 // seek skips this and leaves the buffer intact; a throw from the switch below
747 // therefore finds it already cleared. Both are by design -- see iostream's @warning.
748 m_read_buf.clear();
749 }
750 m_cvt.switch_to_put();
751 }

820 std::pair<device_type, std::exception_ptr> detach() noexcept
821 {
822 if constexpr (IsIn)
823 {
824 if (!m_read_buf.empty())
825 {
826 try
827 {
828 // On over-putback (more put-back than actually read), tell()
829 // saturates at 0, so the device is handed back positioned at the
830 // stream origin. That is consistent with the logical read cursor
831 // being at 0; see tell()/putbackc() and switch_to_put().
832 const std::size_t pos = tell();
833 m_cvt.seek(pos);
834 }
835 catch (...) // NOLINT(bugprone-empty-catch)
836 {
837 // Swallowed on purpose: a device that doesn't support positioning
838 // (e.g. a pipe/tty-backed stdin) can never honor this reposition,
839 // and that is an inherent property of the device, not an
840 // actionable failure - the caller could not have repositioned it
841 // either after getting it back. Reporting this error would make
842 // routine detach()/attach() cycles on such devices start throwing
843 // (e.g. reset() right after a formatted read has left
844 // one buffered lookahead character), even though nothing is
845 // actually broken; losing that lookahead (and, below, the root
846 // converter's read buffer) is the
847 // accepted, unavoidable cost for non-positionable devices.
848 }
849 m_read_buf.clear();
850 }
851 }
852 return m_cvt.detach();
853 }
