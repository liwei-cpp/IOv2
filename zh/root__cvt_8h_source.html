486 void attach(device_type&& dev = device_type{})
487 {
488 const bool was_output = (m_io_status == io_status::output);
489 auto detach_res = detach();
490
491 if constexpr (dev_cpt::support_put<device_type>)
492 {
493 if (was_output)
494 {
495 try
496 {
497 detach_res.first.dflush();
498 }
499 catch (...)
500 {
501 // detach() performs the converter-level flush first, so its
502 // error has priority over a later device-level flush error.
503 if (!detach_res.second)
504 detach_res.second = std::current_exception();
505 }
506 }
507 }
508
509 m_device = std::move(dev);
510 m_bos_len = 0;
511
512 m_buffer.resize(s_buffer_length);
513 m_buf_cur = m_buffer.data();
514 m_buf_end = m_buffer.data();
515 m_io_status = io_status::neutral;
516 if (detach_res.second)
517 std::rethrow_exception(detach_res.second);
518 }

566 void main_cont_beg()
567 {
568 // bos() must be called before main_cont_beg() to establish io_status
569 assert(m_io_status != io_status::neutral);
570
571 if constexpr (dev_cpt::support_put<device_type>)
572 {
573 if (m_io_status == io_status::output)
574 flush();
575 }
576
577 if constexpr (dev_cpt::support_positioning<device_type>)
578 {
579 // Invariant: dtell() >= buffered data. The buffer is filled via dget(),
580 // which advances the device position by the amount read. Consuming data
581 // from the buffer only moves m_buf_cur forward, never increasing the
582 // buffered amount beyond what was read from the device.
583 const auto buffered = static_cast<std::size_t>(m_buf_end - m_buf_cur);
584 assert(m_device.dtell() >= buffered);
585 m_bos_len = m_device.dtell() - buffered;
586 }

728 std::size_t get(internal_type* to, std::size_t to_max)
729 requires (dev_cpt::support_get<device_type>)
730 {
731 if constexpr (dev_cpt::support_put<device_type>)
732 {
733 if (m_io_status != io_status::input)
734 switch_to_get();

737 throw cvt_error("root_cvt::get fails: invalid io status");

746 m_buf_cur = m_buffer.data();
747 m_buf_end = m_buf_cur + act_len;
748 }

788 void put(const internal_type* to, std::size_t to_size)
789 requires (dev_cpt::support_put<device_type>)
790 {
791 if constexpr (dev_cpt::support_get<device_type>)
792 {
793 if (m_io_status != io_status::output)
794 switch_to_put();
795 }
796 if (m_io_status != io_status::output)
797 throw cvt_error("root_cvt::put fails: invalid io status");
798
799 const std::size_t buf_used = m_buf_cur - m_buffer.data();
800 const std::size_t remain = s_buffer_length - buf_used;
801 if (to_size < remain)
802 {
803 m_buf_cur = std::copy(to, to + to_size, m_buf_cur);
804 return;
805 }
806
807 if (to_size == remain)
808 {
809 std::copy(to, to + to_size, m_buf_cur);
810 m_device.dput(m_buffer.data(), s_buffer_length);
811 m_buf_cur = m_buffer.data();
812 return;
813 }
814
815 if (buf_used > 0)
816 m_device.dput(m_buffer.data(), buf_used);
817 if (to_size < s_buffer_length / 2)
818 m_buf_cur = std::copy(to, to + to_size, m_buffer.data());
819 else
820 {
821 m_buf_cur = m_buffer.data();
822 m_device.dput(to, to_size);
823 }
824 }

838 void flush()

843 if ((m_buf_cur != nullptr) && (m_buf_cur != m_buffer.data()))

872 [[nodiscard]] std::size_t tell() const
873 requires (dev_cpt::support_positioning<device_type>)
874 {
875 const std::size_t device_tell = m_device.dtell();
876
877 switch(m_io_status)
878 {
879 case io_status::output:
880 {
881 // Invariant: device_tell >= m_bos_len. The device position only advances
882 // via flush operations, never moving backward from the stream start.
883 assert(device_tell >= m_bos_len);
884 const auto buf_used = static_cast<std::size_t>(m_buf_cur - m_buffer.data());
885 return device_tell - m_bos_len + buf_used;
886 }
887 case io_status::input:
888 {
889 // Invariant: device_tell - m_bos_len >= buffered. Same invariant as
890 // main_cont_beg(): buffer data comes from dget(), which advances device
891 // position by the amount read.
892 const auto buffered = static_cast<std::size_t>(m_buf_end - m_buf_cur);
893 assert(device_tell >= m_bos_len);
894 assert(device_tell - m_bos_len >= buffered);
895 return device_tell - m_bos_len - buffered;
896 }
897 default:
898 throw cvt_error("root_cvt::tell fails: invalid io status");
899 }
900 }

921 requires (dev_cpt::support_positioning<device_type>)

1782 template <bool Saturate = false>
1783 auto get_buf(std::size_t to_max)
1784 {
1785 if (to_max == 0)
1786 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
1787 if (to_max > KernelType::s_buffer_length)
1788 throw cvt_error("cvt_reader::get_buf fail, read size too large");
1789
1790 if constexpr (dev_cpt::support_put<device_type>)
1791 {
1792 if (m_kernel.m_io_status != io_status::input)
1793 m_kernel.switch_to_get();
1794 }
1795 if (m_kernel.m_io_status != io_status::input)
1796 throw cvt_error("cvt_reader::get_buf fail, invalid io status");
1797
1798 const std::size_t remain = m_kernel.m_buf_end - m_kernel.m_buf_cur;
1799 // need to read, then read at most as it can
1800 if (remain < to_max)
1801 {
1802 if (remain != 0)
1803 m_kernel.m_buf_end = std::copy(m_kernel.m_buf_cur, m_kernel.m_buf_end, m_kernel.m_buffer.data());
1804 else
1805 m_kernel.m_buf_end = m_kernel.m_buffer.data();
