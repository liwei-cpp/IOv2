1503 template <typename OutIt>
1504 OutIt put(OutIt out, const std::tm& t, std::basic_string_view<CharT> fmt) const
1505 {
1506 using namespace std::chrono;
1507
1508 // Validate raw tm fields before constructing chrono types: year/month
1509 // truncate out-of-range values (short / unsigned char) and would silently
1510 // wrap, so range-check the integers first, then reject field combinations
1511 // that are individually in range but not a real calendar date (e.g. Feb 30).
1512 // tm_sec is intentionally limited to [0,59], NOT C's [0,60]: this facet
1513 // routes time-of-day through std::chrono::hh_mm_ss<seconds>, which cannot
1514 // represent a leap second (23:59:60 would normalise to 24:00:00 and be
1515 // mis-formatted), so a leap-second tm is rejected up front here rather
1516 // than silently corrupted on output.
1517 const int y = t.tm_year + 1900;
1518 if (t.tm_mon < 0 || t.tm_mon > 11 ||
1519 t.tm_mday < 1 || t.tm_mday > 31 ||
1520 t.tm_hour < 0 || t.tm_hour > 23 ||
1521 t.tm_min < 0 || t.tm_min > 59 ||
1522 t.tm_sec < 0 || t.tm_sec > 59 ||
1523 y < static_cast<int>(year::min()) || y > static_cast<int>(year::max()))
1524 throw stream_error("timeio put error: std::tm field out of range");
1525
1526 year_month_day ymd{ year{y}, month{static_cast<unsigned>(t.tm_mon) + 1}, day{static_cast<unsigned>(t.tm_mday)} };
1527 if (!ymd.ok())
1528 throw stream_error("timeio put error: std::tm is not a valid calendar date");
1529
1530 weekday wd{ymd};
1531
1532 seconds sec = hours{t.tm_hour} + minutes{t.tm_min} + seconds{t.tm_sec};
1533 std::chrono::hh_mm_ss hms{sec};
1534 return do_put(out, fmt, &ymd, &wd, &hms, nullptr);
1535 }
