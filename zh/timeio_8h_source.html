405 // week is known. Resolve it as the first matching weekday on or after the
406 // fallback date, so the result stays within six days of the fallback
407 // instead of jumping to whichever week of the year happens to match.

417 deduced_year = int(snapped.year());
418 deduced_month = static_cast<uint8_t>(static_cast<unsigned>(snapped.month()));
419 deduced_mday = static_cast<uint8_t>(static_cast<unsigned>(snapped.day()));
420 }

431 }

434 {

436 {

444 }

4454 "%m/%d/%Y %T" on a time-only value "%T" rather than "// %T". Trim backwards when
4455 anything precedes, forwards only at the very start. A closing bracket is never a
