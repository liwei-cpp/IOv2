230 [[nodiscard]] std::chrono::year_month_day compute_ymd() const
231 {
232 using namespace std::chrono;
233 if (m_have_year && m_have_mon && m_have_mday)
234 return year_month_day{ year{m_year}, month{static_cast<uint8_t>(m_month)}, day{m_mday} };
235 if (m_have_year && m_have_yday)
236 {
237 sys_days sd = sys_days{ year{m_year} / 1 / 1 } + days{ m_yday };
238 return year_month_day{sd};
239 }
240
241 auto deduced_year = m_year;
242 // Deduce year
243 if (!m_have_year)
244 {
245 if (m_have_century && m_have_year_in_century)
246 deduced_year = deduced_year % 100 + m_century * 100; // NOLINT(bugprone-branch-clone)
247 else if ((m_have_year_of_era) && (!m_era_items.empty()))
248 {
249 using namespace TimeioHelper;
250
251 // use month and day to decide
252 if (m_have_mon && m_have_mday)
253 {
254 auto it = m_era_items.begin();
255 for (; it != m_era_items.end(); ++it)
256 {
257 int64_t est_year_64 = static_cast<int64_t>(it->from_year)
258 + (static_cast<int64_t>(m_year_of_era) - static_cast<int64_t>(it->offset)) * it->direction;
259 int est_year = static_cast<int>(std::clamp<int64_t>(est_year_64,
260 std::numeric_limits<int>::min(), std::numeric_limits<int>::max()));
261 bool cmp1 = era_small_or_equal(it->from_year, it->from_month, it->from_day,
262 est_year, m_month, m_mday) &&
263 era_small_or_equal(est_year, m_month, m_mday,
264 it->to_year, it->to_month, it->to_day);
265 bool cmp2 = era_small_or_equal(est_year, m_month, m_mday,
266 it->from_year, it->from_month, it->from_day) &&
267 era_small_or_equal(it->to_year, it->to_month, it->to_day,
268 est_year, m_month, m_mday);
269 if (!cmp1 && !cmp2) continue;
270 deduced_year = est_year;
271 break;
272 }
273
274 // nothing matches, choose the first item.
275 if (it == m_era_items.end())
276 deduced_year = m_era_items.begin()->from_year;
277 }
278 else if (m_have_mon)
279 {
280 auto it = m_era_items.begin();
281 for (; it != m_era_items.end(); ++it)
282 {
283 int64_t est_year_64 = static_cast<int64_t>(it->from_year)
284 + (static_cast<int64_t>(m_year_of_era) - static_cast<int64_t>(it->offset)) * it->direction;
285 int est_year = static_cast<int>(std::clamp<int64_t>(est_year_64,
286 std::numeric_limits<int>::min(), std::numeric_limits<int>::max()));
287 bool cmp1 = era_small_or_equal(it->from_year, it->from_month, it->from_day,
288 est_year, m_month, m_mday) &&
289 era_small_or_equal(est_year, m_month, m_mday,
290 it->to_year, it->to_month, it->to_day);
291 bool cmp2 = era_small_or_equal(est_year, m_month, m_mday,
292 it->from_year, it->from_month, it->from_day) &&
293 era_small_or_equal(it->to_year, it->to_month, it->to_day,
294 est_year, m_month, m_mday);
295 if (!cmp1 && !cmp2) continue;
296 deduced_year = est_year;
297 break;
298 }
299
300 // nothing matches, choose the first item.
301 if (it == m_era_items.end())
302 deduced_year = m_era_items.begin()->from_year;
303 }
304 else
305 {
306 auto it = m_era_items.begin();
307 for (; it != m_era_items.end(); ++it)
308 {
309 int64_t est_year_64 = static_cast<int64_t>(it->from_year)
310 + (static_cast<int64_t>(m_year_of_era) - static_cast<int64_t>(it->offset)) * it->direction;
311 int est_year = static_cast<int>(std::clamp<int64_t>(est_year_64,
312 std::numeric_limits<int>::min(), std::numeric_limits<int>::max()));
313 const bool within_forward = it->from_year <= est_year && est_year <= it->to_year;
314 const bool within_backward = it->to_year <= est_year && est_year <= it->from_year;
315 if (!within_forward && !within_backward) continue;
316 deduced_year = est_year;
317 break;
318 }
319
320 // nothing matches, choose the first item.
321 if (it == m_era_items.end())
322 deduced_year = m_era_items.begin()->from_year;
323 }
324 }
325
326 // Fuzzy matching, we do not have enough information, but need to set year.
327 else if (m_have_year_in_century) { /* do nothing */ }
328 else if (m_have_century)
329 // %C with no year within the century: the year within the century is 0,
330 // as in POSIX strptime. Deriving it from m_year would make the result
331 // depend on whatever the fallback happens to be -- the wall-clock year
332 // for a default-constructed context, the caller's old value for a hinted
333 // one -- for a format string that says nothing about it.
334 deduced_year = m_century * 100;
335 else if (m_have_era && !m_era_items.empty())
336 deduced_year = m_era_items.begin()->from_year;
337 }
338
339 // ISO-8601 week date. Prefer the ISO year (%G); otherwise use the year deduced
340 // above -- an explicit %Y, a century plus %y, an era, or the fallback -- instead
341 // of dropping the week number. The two fully-specified branches above
342 // (year+mon+mday, year+yday) already returned, so reaching here means the date is
343 // not pinned down by an explicit month/day or day-of-year.
344 if (m_have_iso_8601_week && m_have_wday)
345 {
346 int iso_year = m_have_iso_8601_year ? m_iso_8601_year : deduced_year;
347 int iso_wd = (m_wday == 0 ? 7 : m_wday);
348 year_month_day jan4 = year{iso_year}/January/4;
349 weekday wd_jan4{sys_days{jan4}};
350 sys_days week1_monday = sys_days{jan4} - (wd_jan4 - Monday);
351 sys_days final = week1_monday + days{7 * (m_iso_8601_week - 1)} + days{iso_wd - 1};
352 return year_month_day{final};
353 }
354
355 auto deduced_month = m_month;
356 auto deduced_mday = m_mday;
357 // True when no deduction path below can touch the day, i.e. the day is whatever
358 // set_date_hint() or the default constructor left behind. Every assignment to
359 // deduced_mday sits inside one of those paths and is guarded by !m_have_mday.
360 // A lone weekday, with no week number and no month or day of its own, is resolved
361 // against the fallback date rather than against January 1: see the branch below.
362 const bool wday_snaps = m_have_wday && !m_have_mon && !m_have_mday
363 && !(m_have_yday || m_have_uweek || m_have_wweek);
364 const bool mday_is_fallback = !m_have_mday
365 && !(m_have_yday || m_have_uweek || m_have_wweek || wday_snaps);
366 int deduced_yday = static_cast<int>(m_yday);
367 bool have_yday = m_have_yday;
368 // Deduce month / mday. When neither is given, both are derived from the
369 // day-of-year. When the month IS given but the day is not, the day-of-month
370 // is computed relative to the *reported* month (deduced_month) rather than
371 // the yday-derived month, so the returned month and day stay mutually
372 // consistent (contradictory month-vs-week input is GIGO and may still yield
373 // an out-of-range day, since this conversion has no error channel).
374 if ((m_have_uweek || m_have_wweek) && m_have_wday && (!have_yday))
375 {
376 int w_offset = m_have_uweek ? 0 : 1;
377
378 // calculate the week of day for Jan 1
379 int wday = day_of_the_week(deduced_year, 1, 1);
380
381 deduced_yday = ((7 - (wday - w_offset)) % 7 + (m_week_no - 1) * 7 + (m_wday - w_offset + 7) % 7);
382 have_yday = true;
383 }
384
385 if (!m_have_mon || !m_have_mday)
386 {
387 if (have_yday)
388 {
389 if (deduced_yday < 0)
390 {
391 deduced_year -= 1;
392 deduced_yday += isleap(deduced_year) ? 366 : 365;
393 }
394 while (deduced_yday >= (isleap(deduced_year) ? 366 : 365))
395 {
396 deduced_yday -= isleap(deduced_year) ? 366 : 365;
397 deduced_year += 1;
398 }
399 int t_mon = 0;
400 while (t_mon < 12 && std::cmp_less_equal(s_mon_yday[isleap(deduced_year)][t_mon], deduced_yday))
401 t_mon++;
402 if (!m_have_mon) deduced_month = static_cast<uint8_t>(t_mon);
403 if (!m_have_mday)
404 deduced_mday = static_cast<uint8_t>(deduced_yday - s_mon_yday[isleap(deduced_year)][deduced_month - 1] + 1);
405 }
406 else if (wday_snaps)
407 {
408 // A weekday on its own cannot locate a date; only the offset within some
409 // week is known. Resolve it as the first matching weekday on or after the
410 // fallback date, so the result stays within six days of the fallback
411 // instead of jumping to whichever week of the year happens to match.
412 auto last_day = static_cast<unsigned>(
413 year_month_day_last{year{deduced_year},
414 month_day_last{month{static_cast<unsigned>(deduced_month)}}}
415 .day());
416 sys_days base{year{deduced_year}
417 / month{static_cast<unsigned>(deduced_month)}
418 / day{std::min<unsigned>(deduced_mday, last_day)}};
419 auto cur = static_cast<int>(weekday{base}.c_encoding());
420 year_month_day snapped{base + days{(static_cast<int>(m_wday) - cur + 7) % 7}};
421 deduced_year = static_cast<int>(snapped.year());
422 deduced_month = static_cast<uint8_t>(static_cast<unsigned>(snapped.month()));
423 deduced_mday = static_cast<uint8_t>(static_cast<unsigned>(snapped.day()));
424 }
425 else if (m_have_uweek || m_have_wweek)
426 {
427 // assume wday is 1
428 int w_offset = m_have_uweek ? 0 : 1;
429 auto j1_wday = day_of_the_week(deduced_year, 1, 1);
430
431 if (!have_yday)
432 {
433 deduced_yday = ((7 - (j1_wday - w_offset)) % 7 + (m_week_no - 1) * 7 + (1 - w_offset + 7) % 7);
434 have_yday = true;
435 }
436
437 if (!m_have_mday || !m_have_mon)
438 {
439 if (deduced_yday < 0)
440 {
441 deduced_year -= 1;
442 deduced_yday += isleap(deduced_year) ? 366 : 365;
443 }
444 while (deduced_yday >= (isleap(deduced_year) ? 366 : 365))
445 {
446 deduced_yday -= isleap(deduced_year) ? 366 : 365;
447 deduced_year += 1;
448 }
449 int t_mon = 0;
450 while (t_mon < 12 && std::cmp_less_equal(s_mon_yday[isleap(deduced_year)][t_mon], deduced_yday))
451 t_mon++;
452 if (!m_have_mon)
453 deduced_month = static_cast<uint8_t>(t_mon);
454 if (!m_have_mday)
455 deduced_mday = static_cast<uint8_t>(deduced_yday - s_mon_yday[isleap(deduced_year)][deduced_month - 1] + 1);
456 }
457 }
458 }
459
460 // A day that came from the fallback rather than from the input must give way when
461 // it cannot exist in the deduced month: the fallback only supplies fields the
462 // format string did not parse, so it must never turn a well-formed parse into an
463 // invalid date. A day the input really did specify is left alone and reported as
464 // invalid.
465 if (mday_is_fallback && deduced_month >= 1 && deduced_month <= 12)
466 {
467 auto last_day = static_cast<unsigned>(
468 year_month_day_last{year{deduced_year},
469 month_day_last{month{static_cast<unsigned>(deduced_month)}}}
470 .day());
471 if (static_cast<unsigned>(deduced_mday) > last_day)
472 deduced_mday = static_cast<uint8_t>(last_day);
473 }
474
475 return year_month_day{ year{deduced_year}, month{static_cast<uint8_t>(deduced_month)}, day{static_cast<uint8_t>(deduced_mday)} };

511private:
512 constexpr static const std::array<std::array<unsigned short int, 13>, 2> s_mon_yday =
513 {{
514 // Row 0: days already gone by when each month opens in a common year.
515 {{ 0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334, 365 }},
516 // Row 1: the same for a leap year. The trailing entry of each row is the
517 // length of the year, so a day-of-year search can walk the row to its end
518 // without a special case.
519 {{ 0, 31, 60, 91, 121, 152, 182, 213, 244, 274, 305, 335, 366 }}
520 }};

522 // The Gregorian rule: every fourth year, except centuries, except every fourth
523 // century.
524 static bool isleap(int year)
525 {
526 return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
527 }
528
529 // Weekday of a date, 0 = Sunday. `month` is 1-based and expected to be a real
530 // month, 1 to 12.
531 //
532 // The count runs from 0001-01-01, which was a Monday in the proleptic Gregorian
533 // calendar, so the number of days elapsed since then is exactly one less than the
534 // weekday index.
535 //
536 // Years at or below zero are reachable here -- %Y accepts 0000 and %C a century
537 // of 0 -- and C++ division truncates towards zero, which would gain a day on every
538 // negative quotient in the leap-day terms below. Rather than special-case the
539 // sign, the year is first lifted above zero by a whole number of 400-year cycles.
540 // A cycle is 146097 days, which is exactly 20871 weeks, so shifting by any
541 // multiple of it moves the date without moving the weekday.
542 //
543 // (This is a deliberate correction, not a translation of what stood here before:
544 // the previous formulation was right for year >= 1 -- verified against this one
545 // over every int -- but reported 0000-01-01 as a Sunday when it was a Saturday.)
546 static int day_of_the_week(int year, int month, int mday)
547 {
548 constexpr int64_t cycles = 400LL * 5368710LL; // enough to lift any int year
549
550 const int64_t y = static_cast<int64_t>(year) - 1 + cycles;
551 const int64_t elapsed = 365 * y + y / 4 - y / 100 + y / 400
552 + s_mon_yday[isleap(year)][month - 1] + mday - 1;
553 return static_cast<int>((elapsed + 1) % 7);
554 }

622 void convert_to(std::chrono::hh_mm_ss<TDuration>& out) const
623 {
624 uint8_t hour_in_24 = m_hour;
625 // When %I sets m_have_I, m_hour was stored as (mem % 12), so 12 AM is
626 // already normalised to 0 at parse time and m_hour is always in [0,11]
627 // here; only the PM (+12) adjustment remains.
628 if (m_have_I && m_is_pm && hour_in_24 < 12)
629 hour_in_24 += 12;
630
631 std::chrono::seconds time_sec =
632 std::chrono::hours{hour_in_24} +
633 std::chrono::minutes{m_minute} +
634 std::chrono::seconds{m_second};
635
636 out = std::chrono::hh_mm_ss<TDuration>{time_sec};
637 }

4078 void validate_format_recursion() const
4079 {
4080 // The node numbering. era_first is both the count of fixed nodes and the number of the
4081 // first era format: how many era formats a locale has varies, so they can only be
4082 // numbered from there on.
4083 // NOLINTNEXTLINE(performance-enum-size)
4084 enum class fmt_node : std::size_t
4085 {
4086 date_time, era_date_time,
4087 date, era_date,
4088 time, era_time,
4089 am_pm,
4090 era_first
4091 };
4092
4093 // The order here IS fmt_node; era formats take the numbers from era_first on.
4094 std::vector<const std::basic_string<CharT>*> nodes{
4095 &m_date_time_format, &m_era_date_time_format,
4096 &m_date_format, &m_era_date_format,
4097 &m_time_format, &m_era_time_format,
4098 &m_am_pm_format};
4099 assert(nodes.size() == std::to_underlying(fmt_node::era_first));
4100 for (const era_entry& entry : m_era_master) nodes.push_back(&entry.format);
4101
4102 // Modifier gating mirrors the three switches: %Oc / %Ox / %OX / %Er / %Or degrade to
4103 // literals there, so no edge here. %% needs no case -- the second % is consumed as the
4104 // specifier character. Brackets need none either: a %c inside a group still recurses.
4105 //
4106 // The graph deliberately over-approximates, since a shade too many edges costs nothing
4107 // while a missing one costs the guarantee: %EY edges to every era node (do_get does try
4108 // them all, do_put expands only the era the year lands in, and expand_and_filter leaves
4109 // %EY unexpanded), and an edge stands even where the value type could not supply the
4110 // specifier and the switch would drop it. A %EY inside an era format re-resolves to that
4111 // same era, so that self-loop is real rather than an approximation.
4112 auto scan_edges = [n_total = nodes.size()]
4113 (std::basic_string_view<CharT> fmt, auto on_edge)
4114 {
4115 for (auto p = fmt.cbegin(); p != fmt.cend(); )
4116 {
4117 if (*p != static_cast<CharT>('%')) { ++p; continue; }
4118 if (++p == fmt.cend()) break;
4119
4120 CharT modifier = 0;
4121 if (*p == static_cast<CharT>('E') || *p == static_cast<CharT>('O'))
4122 {
4123 modifier = *p;
4124 if (++p == fmt.cend()) break;
4125 }
4126
4127 const bool era = (modifier == static_cast<CharT>('E'));
4128 switch (*p)
4129 {
4130 case static_cast<CharT>('c'):
4131 if (modifier != static_cast<CharT>('O'))
4132 on_edge(era ? fmt_node::era_date_time : fmt_node::date_time);
4133 break;
4134
4135 case static_cast<CharT>('x'):
4136 if (modifier != static_cast<CharT>('O'))
4137 on_edge(era ? fmt_node::era_date : fmt_node::date);
4138 break;
4139
4140 case static_cast<CharT>('X'):
4141 if (modifier != static_cast<CharT>('O'))
4142 on_edge(era ? fmt_node::era_time : fmt_node::time);
4143 break;
4144
4145 case static_cast<CharT>('r'):
4146 if (!modifier) on_edge(fmt_node::am_pm);
4147 break;
4148
4149 case static_cast<CharT>('Y'):
4150 if (era)
4151 for (std::size_t i = std::to_underlying(fmt_node::era_first); i < n_total; ++i)
4152 on_edge(static_cast<fmt_node>(i));
4153 break;
4154
4155 default:
4156 break;
4157 }
4158 ++p;
4159 }
4160 };
4161
4162 // Three-colour DFS: 0 unvisited, 1 on the stack, 2 finished. A grey target is a back
4163 // edge, i.e. a cycle; a directly self-referential %c lands here with u == v.
4164 std::vector<char> color(nodes.size(), 0);
4165 auto visit = [&](auto&& self, std::size_t v) -> void
4166 {
4167 color[v] = 1;
4168 scan_edges(*nodes[v], [&](fmt_node node)
4169 {
4170 const std::size_t u = std::to_underlying(node);
4171 if (color[u] == 1)
4172 throw io_error("timeio: self-referential compound format in locale data");
4173 if (color[u] == 0) self(self, u);
4174 });
4175 color[v] = 2;
4176 };
4177
4178 for (std::size_t v = 0; v < nodes.size(); ++v)
4179 if (color[v] == 0) visit(visit, v);
4180 }

4851 template <bool has_date, bool has_time, bool has_offset, bool has_zone>
4852 void expand_and_filter(std::basic_string<CharT>& result, std::basic_string_view<CharT> fmt) const
4853 {
4854 auto closer_for = [](CharT c)
4855 {
4856 if (c == static_cast<CharT>('(')) return static_cast<CharT>(')');
4857 if (c == static_cast<CharT>('[')) return static_cast<CharT>(']');
4858 if (c == static_cast<CharT>('{')) return static_cast<CharT>('}');
4859 return CharT();
4860 };
4861
4862 auto f = fmt.cbegin();
4863
4864 /* Drop one separator adjacent to whatever was just removed, or an orphaned piece of
4865 punctuation is left behind: "%T %Z" has to become "%T" rather than "%T ", and
4866 "%m/%d/%Y %T" on a time-only value "%T" rather than "// %T". Trim backwards when
4867 anything precedes, forwards only at the very start. A closing bracket is never a
4868 separator -- its opener still stands, and eating it would orphan that one -- and
4869 symmetrically an opening bracket is never one going forwards. */
4870 auto trim_separator = [&]
4871 {
4872 if (!result.empty())
4873 {
4874 auto n = result.size();
4875 while (n > 0 && is_space(result[n - 1])) --n;
4876
4877 if (n > 0 && is_punct(result[n - 1]) &&
4878 result[n - 1] != static_cast<CharT>(')') &&
4879 result[n - 1] != static_cast<CharT>(']') &&
4880 result[n - 1] != static_cast<CharT>('}'))
4881 {
4882 --n;
4883 while (n > 0 && is_space(result[n - 1])) --n;
4884 }
4885 result.resize(n);
4886 }
4887
4888 if (result.empty())
4889 {
4890 while (f != fmt.cend() && is_space(*f)) ++f;
4891 if (f != fmt.cend() && is_punct(*f) && closer_for(*f) == CharT())
4892 {
4893 ++f;
4894 while (f != fmt.cend() && is_space(*f)) ++f;
4895 }
4896 }
4897 };
4898

4900 {
4901 if (*f != static_cast<CharT>('%'))
4902 {
4903 /* A bracket group is all-or-nothing: filter its content on its own, and if a
4904 group that held something comes back empty, the brackets go with it. That is
4905 what turns "%T (%Z)" into "%T" and "%T (%Z, %z)" into "%T", while leaving both
4906 halves of a group that keeps something ("%T (%Z, x)" -> "%T (x)") and every

4909 if (closer != CharT())

4930 result.push_back(closer);
4931 f = close_it + 1;
4932 continue;
4933 }
4934 }
4935 result.push_back(*f++);
4936 continue;
4937 }
4938

4941 {
4942 result.push_back(static_cast<CharT>('%'));

5347 template <typename TIter, std::sentinel_for<TIter> TSent>
5348 static TIter scan_int_field(TIter beg, TSent end, int& out, int min_val, int max_val, std::size_t width, bool& succ) // NOLINT(bugprone-easily-swappable-parameters)
5349 {
5350 constexpr auto zero = static_cast<CharT>('0');
5351 constexpr auto nine = static_cast<CharT>('9');
5352
5353 int value = 0;
5354 std::size_t digits = 0;
5355 bool fits = true;
5356
5357 // A digit is committed -- iterator advanced, counter bumped -- only once the
5358 // value it produces has been checked against `max_val`. So the digit that
5359 // pushes the field past its ceiling is left unread for whatever parses next,
5360 // while `value` keeps it: that is what stops the range test below from
5361 // mistaking the truncated prefix for a field that fits. `max_val` is at most
5362 // 9999 over `width` <= 4 digits, so the running value cannot overflow `int`.
5363 while (digits < width && beg != end)
5364 {
5365 const CharT c = *beg;
5366 if (c < zero || c > nine)
5367 break;
5368
5369 value = value * 10 + static_cast<int>(c - zero);
5370 if (value > max_val)
5371 {
5372 fits = false;
5373 break;
5374 }
5375
5376 ++beg;
5377 ++digits;
5378 }
5379
5380 // `fits` already implies `value <= max_val`, so only the lower bound is
5381 // left to check. Failure never clears a `succ` the caller had set.
5382 if (fits && digits != 0 && value >= min_val)
5383 out = value;
5384 else
5385 succ = false;
5386
5387 return beg;
5388 }
