419 deduced_mday = static_cast<uint8_t>(static_cast<unsigned>(snapped.day()));

431 }

1652 template <typename OutIt>
1653 OutIt put(OutIt out, const std::tm& t, std::basic_string_view<CharT> fmt) const
1654 {
1655 using namespace std::chrono;
1656
1657 // Validate raw tm fields before constructing chrono types: year/month
1658 // truncate out-of-range values (short / unsigned char) and would silently
1659 // wrap, so range-check the integers first, then reject field combinations
1660 // that are individually in range but not a real calendar date (e.g. Feb 30).
1661 // tm_sec is intentionally limited to [0,59], NOT C's [0,60]: this facet
1662 // routes time-of-day through std::chrono::hh_mm_ss<seconds>, which cannot
1663 // represent a leap second (23:59:60 would normalise to 24:00:00 and be
1664 // mis-formatted), so a leap-second tm is rejected up front here rather
1665 // than silently corrupted on output.
1666 if (t.tm_mon < 0 || t.tm_mon > 11 ||
1667 t.tm_mday < 1 || t.tm_mday > 31 ||
1668 t.tm_hour < 0 || t.tm_hour > 23 ||
1669 t.tm_min < 0 || t.tm_min > 59 ||
1670 t.tm_sec < 0 || t.tm_sec > 59 ||
1671 t.tm_year > static_cast<int>(year::max()) - 1900 ||
1672 t.tm_year + 1900 < static_cast<int>(year::min()))
1673 throw stream_error("timeio put error: std::tm field out of range");
1674
1675 const int y = t.tm_year + 1900;
1676
1677 year_month_day ymd{ year{y}, month{static_cast<unsigned>(t.tm_mon) + 1}, day{static_cast<unsigned>(t.tm_mday)} };
1678 if (!ymd.ok())
1679 throw stream_error("timeio put error: std::tm is not a valid calendar date");
1680
1681 weekday wd{ymd};
1682
1683 seconds sec = hours{t.tm_hour} + minutes{t.tm_min} + seconds{t.tm_sec};
1684 std::chrono::hh_mm_ss hms{sec};
1685 return do_put(out, fmt, &ymd, &wd, &hms, nullptr);
1686 }

2714 const std::array<std::basic_string<CharT>, N>& abbr,
2715 const char* what)
2716 {

2718 for (size_t i = 0; i < N; ++i)
2719 {
2720 if (full[i].empty() || abbr[i].empty())
2721 throw std::runtime_error(std::string("timeio: empty ") + what + " name in locale data");
2722 names[2 * i] = &full[i];

2725 // names[p] belongs to index p / 2.
2726 for (size_t a = 0; a < names.size(); ++a)
2727 for (size_t b = a + 1; b < names.size(); ++b)
2728 if ((a / 2) != (b / 2) && *names[a] == *names[b])
2729 throw std::runtime_error(std::string("timeio: duplicate ") + what + " name in locale data");
2730 }
