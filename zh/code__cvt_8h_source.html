168 explicit codecvt_kernel(const std::string& name)
169 : m_inter_locale([&] {
170 // A name the C library rejects is a converter that cannot be built.
171 try { return clocale_wrapper(name); }
172 catch (const io_error& e) { throw cvt_error(e.what()); }
173 }())
174 {
175 clocale_user guard(m_inter_locale);

177 // there are no known constant-length encodings
178 // m_epc == 1 means fixed length
179 // Defensive check: the C standard requires MB_CUR_MAX to be a positive
180 // integer, but a misconfigured CRT / exotic libc could report 0, which
181 // would lead to division-by-zero UB in seek / rseek. Reject it here so
182 // m_epc >= 1 holds as a class invariant for any constructed instance.
183 const auto cur_max = MB_CUR_MAX;
184 if (cur_max == 0) [[unlikely]]
185 throw cvt_error("codecvt_kernel: locale reports MB_CUR_MAX == 0");
186 m_epc = static_cast<unsigned>(cur_max);
187
188 // Note: mbtowc uses internal static state, so concurrent construction of
189 // codecvt_kernel instances is NOT thread-safe. However, this is acceptable
190 // because this class does not support multi-threaded construction.
191 // Once constructed, instance-level operations use explicit mbstate_t (m_state)
192 // and are safe for single-instance usage.
193 // NOLINTNEXTLINE(cppcoreguidelines-prefer-member-initializer)
194 m_is_state_dep = (std::mbtowc(nullptr, nullptr, MB_CUR_MAX) != 0);
195 init_state();
196 }

351 bool out_helper(TInt ch, char*& to, char* to_end)
352 {
353 if (std::greater<>{}(to, to_end)) [[unlikely]]
354 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");

431 std::pair<bool, std::size_t> in_helper(const char*& from, const char* from_end,
432 TInt*& to, TInt* to_end)
433 {
434 if (std::greater<>{}(from, from_end) || std::greater<>{}(to, to_end)) [[unlikely]]
435 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
436 clocale_user guard(m_inter_locale);
437 // No wchar_t value glibc decodes to: it tells a shift-only run from a character.
438 constexpr auto no_char = static_cast<wchar_t>(-1);
439 std::size_t i_count = 0;
440
441 const auto find_nul = [&]
442 {
443 if (!m_is_state_dep || from >= from_end)
444 return from_end;
445 const void* p = std::memchr(from, 0, static_cast<std::size_t>(from_end - from));
446 return p != nullptr ? static_cast<const char*>(p) : from_end;
447 };
448 const char* nul = find_nul();
449
450 const std::size_t to_max = to_end - to;
451 while (i_count < to_max && from < from_end)
452 {
453 if (nul < from)
454 nul = find_nul();
455 if (*from == '\0')
456 {
457 ++from;
458 m_state = std::mbstate_t{};
459 *to++ = TInt{};
460 ++i_count;
461 continue;
462 }
463
464 const auto avail = static_cast<std::size_t>(nul - from);
465 auto tmp_state = m_state;
466 wchar_t wch = no_char;
467 std::size_t conv = mbrtowc(&wch, from, avail, &tmp_state);
468 if (m_is_state_dep && (conv == static_cast<std::size_t>(-1) || conv == static_cast<std::size_t>(-2))) // NOLINT(modernize-use-integer-sign-comparison)
469 {
470 // glibc judges a run that opens with a complete shift sequence as a
471 // whole; take that sequence off first, so the error or the incomplete
472 // sequence starts behind it.
473 for (std::size_t k = 1; k < avail; ++k)
474 {
475 auto st = m_state;
476 wchar_t w = no_char;
477 const std::size_t r = mbrtowc(&w, from, k, &st);
478 if (r == static_cast<std::size_t>(-2)) continue; // NOLINT(modernize-use-integer-sign-comparison)
479 if (r != static_cast<std::size_t>(-1) && r != 0 && w == no_char) // NOLINT(modernize-use-integer-sign-comparison)
480 {
481 conv = r;
482 tmp_state = st;
483 }
484 break;
485 }
486 }
487 if (conv == static_cast<std::size_t>(-1) || conv == 0) // NOLINT(modernize-use-integer-sign-comparison)
488 {
489 // Only tmp_state is unspecified after EILSEQ (C11 7.29.6.3.2); m_state
490 // still holds the shift state before the bad byte. conv == 0 (a NUL
491 // decoded) is rejected the same way.
492 return std::pair{false, i_count};
493 }
494 if (conv == static_cast<std::size_t>(-2)) // NOLINT(modernize-use-integer-sign-comparison)
495 {
496 // A NUL inside, or longer than any character: not the start of one.
497 if (nul != from_end || avail > MB_LEN_MAX)
498 return std::pair{false, i_count};
499 break; // incomplete: left at from for the caller to keep
500 }
501
502 m_state = tmp_state;
503 from += conv;
504 if (wch == no_char)
505 continue; // a shift sequence alone
506 *to++ = static_cast<TInt>(wch);
507 ++i_count;
508 }
509
510 return std::pair{true, i_count};
511 }
512
513private:
514 clocale_wrapper m_inter_locale;
515 unsigned m_epc = 0;
516 std::mbstate_t m_state{};
517 bool m_is_state_dep = false;
518};

688 bool out_helper(TInt ch, char8_t*& to, char8_t* to_end)
689 {
690 if (std::greater<>{}(to, to_end)) [[unlikely]]
691 throw cvt_error("codecvt_kernel::out_helper fail: invalid pointer range");
692 // Check for maximum UTF-8 length (4 bytes) upfront
693 if (to_end - to < 4)
694 return false;
695 const auto c = static_cast<uint32_t>(ch);
696 if (0xD800U <= c && c <= 0xDFFFU) [[unlikely]]
697 return false;
698
699 if (c < 0x80U)
700 *to++ = static_cast<char8_t>(c);
701 else if (c <= 0x7ffU)
702 {
703 *to++ = static_cast<char8_t>((c >> 6) + 0xC0U);
704 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
705 }
706 else if (c <= 0xFFFFU)
707 {
708 *to++ = static_cast<char8_t>((c >> 12) + 0xE0U);
709 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);
710 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);
711 }
712 else if (c <= 0x10FFFFU)
713 {
714 *to++ = static_cast<char8_t>((c >> 18) + 0xF0U);
715 *to++ = static_cast<char8_t>(((c >> 12) & 0x3FU) + 0x80U);
716 *to++ = static_cast<char8_t>(((c >> 6) & 0x3FU) + 0x80U);
717 *to++ = static_cast<char8_t>((c & 0x3FU) + 0x80U);

720 return false;
721 return true;

765 std::pair<bool, std::size_t> in_helper(const char8_t*& from, const char8_t* from_end,
766 TInt*& to, TInt* to_end)
767 {

769 throw cvt_error("codecvt_kernel::in_helper fail: invalid pointer range");
770 const TInt* const ori_to = to;
771 const auto written = [&] { return static_cast<std::size_t>(to - ori_to); };
772
773 // An incomplete character at the end (short_input) is left at from, unconsumed,
774 // for the caller to keep; so is an invalid one (bad).

779private:
780 enum class run_status : std::uint8_t { done, short_input, bad };
781
782 // Decodes characters from [from, from_end) into [to, to_end) until either runs
783 // out: done, short_input when the bytes left at from are the start of a
784 // character but too few, or bad at an invalid sequence.
785 static run_status decode_run(const char8_t*& from, const char8_t* from_end, TInt*& to, TInt* to_end)
786 {
787 while ((from != from_end) && (to != to_end))
788 {
789 auto c1 = static_cast<uint32_t>(*from);
790 if (c1 < 0x80U) [[likely]]
791 {
792 ++from;
793 *to++ = static_cast<TInt>(c1);
794 }
795 else if (c1 < 0xE0U)
796 {
797 if (c1 < 0xC0U) [[unlikely]]
798 return run_status::bad;
799 if (from_end - from < 2) return run_status::short_input;
800 auto c2 = static_cast<uint32_t>(from[1]);
801 if ((c2 & 0xC0U) != 0x80U) [[unlikely]]
802 return run_status::bad;
803 const auto encoded = c2 + (c1 << 6);
804 constexpr auto prefix_bias = (0xC0U << 6) + 0x80U;
805 const auto c = encoded - prefix_bias;
806 if (c < 0x80U) return run_status::bad;
807 *to++ = c;
808 from += 2;
809 }
810 else if (c1 < 0xF0U)
811 {
812 if (from_end - from < 3) return run_status::short_input;
813 auto c2 = static_cast<uint32_t>(from[1]);
814 auto c3 = static_cast<uint32_t>(from[2]);
815 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U)) [[unlikely]]
816 return run_status::bad;
817 const auto encoded = c3 + (c2 << 6) + (c1 << 12);
818 constexpr auto prefix_bias = (0xE0U << 12) + (0x80U << 6) + 0x80U;
819 const auto c = encoded - prefix_bias;
820 if (c < 0x800U) return run_status::bad;
821 if (c >= 0xD800U && c <= 0xDFFFU) [[unlikely]]
822 return run_status::bad;
823 *to++ = c;
824 from += 3;
825 }
826 else if (c1 < 0xF8U)
827 {
828 if (from_end - from < 4) return run_status::short_input;
829 auto c2 = static_cast<uint32_t>(from[1]);
830 auto c3 = static_cast<uint32_t>(from[2]);
831 auto c4 = static_cast<uint32_t>(from[3]);
832 if (((c2 & 0xC0U) != 0x80U) || ((c3 & 0xC0U) != 0x80U) || ((c4 & 0xC0U) != 0x80U)) [[unlikely]]
833 return run_status::bad;
834 const auto encoded = c4 + (c3 << 6) + (c2 << 12) + (c1 << 18);
835 constexpr auto prefix_bias =
836 (0xF0U << 18) + (0x80U << 12) + (0x80U << 6) + 0x80U;
837 const auto c = encoded - prefix_bias;
838 if (c < 0x10000U || c > 0x10FFFFU) return run_status::bad;
839 *to++ = c;
840 from += 4;
841 }
842 else
843 return run_status::bad;
844 }
845
846 return run_status::done;
847 }

1154 void put_main(cvt_writer<KernelType>& writer, const internal_type* to, std::size_t to_size)
1155 requires (cvt_cpt::support_put<KernelType>)
1156 {

1204 std::pair<std::size_t, std::exception_ptr>
1205 get_main(cvt_reader<KernelType>& reader, internal_type* to, std::size_t to_max) noexcept
1206 requires (cvt_cpt::support_get<KernelType>)
1207 {
1208 std::size_t total_size = 0;
1209 try
1210 {
1211 if (to_max == 0) return {0, nullptr};
1212 reader.reset(s_max_buf_size);
1213
1214 std::size_t prev_rollback = 0;
1215 while (total_size < to_max)
1216 {
1217 std::size_t dest_size = std::min<std::size_t>(to_max - total_size, s_max_buf_size);
1218 dest_size = std::max(dest_size, prev_rollback + 1);
1219
1220 if (dest_size > s_max_buf_size) [[unlikely]]
1221 throw cvt_error("code_cvt::get fail: input sequence too long");
1222
1223 auto [ptr, cur_size] = reader.get_buf(dest_size);
1224 if (cur_size == prev_rollback)
1225 {
1226 // Nothing more below. An incomplete sequence goes back where it was:
1227 // every later get finds it again, until a reposition drops it.
1228 if (prev_rollback != 0)
1229 reader.rollback(prev_rollback);
1230 if (prev_rollback == 0 || total_size != 0)
1231 return {total_size, nullptr};
1232 throw cvt_error("code_cvt::get fail: partial input sequence");
1233 }
1234
1235 auto ext_cur = ptr;
1236 auto [succ, int_len] = m_cvt_kernel.in_helper(ext_cur, ptr + cur_size, to, to + to_max - total_size);
1237
1238 // in_helper may have written `int_len` chars into `to` before hitting an
1239 // invalid byte: count them before anything can throw.
1240 m_accu_len += int_len;
1241 total_size += int_len;
1242
1243 if (!succ)
1244 {
1245 // ext_cur is at the start of the bad sequence. With characters to hand
1246 // over, leave it there for the next get to find and report. Otherwise
1247 // report it now and skip its first byte, as block reads always did.
1248 if (total_size != 0)
1249 {
1250 if (ext_cur != ptr + cur_size)
1251 reader.rollback(ptr + cur_size - ext_cur);
1252 return {total_size, nullptr};
1253 }
1254 const auto* resume = std::min(ext_cur + 1, ptr + cur_size);
1255 if (resume != ptr + cur_size)
1256 reader.rollback(ptr + cur_size - resume);
1257 throw cvt_error("code_cvt::get fail: invalid external sequence");
1258 }
1259
1260 if (ext_cur == ptr + cur_size)
1261 prev_rollback = 0;
1262 else
1263 {
1264 prev_rollback = ptr + cur_size - ext_cur;
1265 reader.rollback(prev_rollback);
1266 }
1267 }
1268 return {total_size, nullptr};
1269 }
1270 catch (...)
1271 {
1272 return {total_size, std::current_exception()};
1273 }
