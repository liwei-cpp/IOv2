131 template <shared_ptr_to<monetary_conf<CharT>> TConfPtr>
132 monetary(TConfPtr p_obj)
133 {
134 // Validate before any dereference below: a null pointer would be UB.
135 // Mirrors the guard in the messages facet.
136 if (!p_obj)
137 throw io_error("shared_ptr is empty");
138
139 m_grouping = p_obj->grouping();
140 m_nat = {.m_curr_symbol = p_obj->curr_symbol_nat(),
141 .m_positive_sign = p_obj->positive_sign_nat(),
142 .m_negative_sign = p_obj->negative_sign_nat(),
143 .m_pos_format = p_obj->pos_format_nat(),
144 .m_neg_format = p_obj->neg_format_nat(),
145 .m_frac_digits = p_obj->frac_digits_nat()};
146 m_int = {.m_curr_symbol = p_obj->curr_symbol_int(),
147 .m_positive_sign = p_obj->positive_sign_int(),
148 .m_negative_sign = p_obj->negative_sign_int(),
149 .m_pos_format = p_obj->pos_format_int(),
150 .m_neg_format = p_obj->neg_format_int(),
151 .m_frac_digits = p_obj->frac_digits_int()};
152 m_decimal_point = p_obj->decimal_point();
153 m_thousands_sep = p_obj->thousands_sep();
154 if (m_decimal_point.empty())
155 throw io_error("monetary: the decimal point is empty");
156
157 // grouping() is contracted to return the INTERNAL convention
158 // (1–255 = group size, 0 = stop, last element implicitly repeats).
159 // monetary_conf and any user-derived override are both expected to
160 // satisfy this contract — POSIX-style normalisation is done at the
161 // POSIX boundary in monetary_conf, not here. Separators parsing could
162 // not tell apart turn grouping off, as monetary_conf itself does it.
163 if (!FacetHelper::punct_distinct<CharT>(m_thousands_sep, m_decimal_point))
164 m_grouping.clear();
165 }

407 template <typename TIter, std::integral TVal>
408 requires (!std::same_as<TVal, bool>)
409 TIter put(TIter s, bool intl, ios_base<char_type>& io, TVal v) const
410 {
411 constexpr std::size_t buf_size = std::numeric_limits<TVal>::digits10 + 3;
412 std::array<char_type, buf_size> vec;
413
414 char_type* p = vec.data() + buf_size;
415 *--p = '\0';
416
417 using TU = std::make_unsigned_t<TVal>;
418 bool negative = false;
419 TU uv;
420
421 if constexpr(std::is_unsigned_v<TVal>)
422 uv = v;
423 else
424 {
425 negative = (v < 0);
426 uv = negative ? static_cast<TU>(-(v + 1)) + 1
427 : static_cast<TU>(v);
428 }
429
430 do { // NOLINT(cppcoreguidelines-avoid-do-while)
431 *--p = static_cast<char_type>('0' + (uv % 10));
432 uv /= 10;
433 } while (uv != 0);
434
435 if (negative)
436 *--p = '-';
437
438 return intl ? insert<true>(s, io, p)
439 : insert<false>(s, io, p);
440 }

599 template <typename TIter, std::sentinel_for<TIter> TSent>
600 TIter get(TIter beg, TSent end, bool intl, ios_base<char_type>& io, std::basic_string<char_type>& digits) const
601 {
602 bool succ = true;
603
604 std::string str;
605 std::tie(succ, beg) = intl ? extract<true>(beg, end, io, str)
606 : extract<false>(beg, end, io, str);
607 const auto len = str.size();
608 std::basic_string<char_type> tmp;
609 if (len)
610 {
611 tmp.reserve(len);
612 for (auto ch : str)
613 {
614 if (ch == '-')
615 tmp.push_back(s_atoms[0]);
616 else if ((ch >= '0') && (ch <= '9'))
617 tmp.push_back(s_atoms[ch - '0' + 1]);
618 }
619 }
620 if (!succ)
621 throw stream_error("monetary parse fail");
622 if (len)
623 digits.swap(tmp);
624 return beg;
625 }

678 template <typename TLeft, typename TRight>
679 static bool space_kept(const base_ft<monetary>::pattern& order, bool symbol_written,
680 TLeft left_writes, TRight right_writes)
681 {
682 using part = base_ft<monetary>::part;
683 const auto at = std::ranges::find(order, part::space);
684 if (at == order.end())
685 return true;
686
687 if (!symbol_written && ((at != order.begin() && at[-1] == part::symbol)
688 || (at + 1 != order.end() && at[1] == part::symbol)))
689 return false;
690 return std::ranges::any_of(order.begin(), at, left_writes)
691 && std::ranges::any_of(at + 1, order.end(), right_writes);
692 }

786 [[nodiscard]] bool fill_alters_reading(char_type fill, const split_info& info,
787 bool leads_digits, bool trails_value,
788 bool negative) const
789 {
790 const char_type* const digits = s_atoms.data() + s_zero;
791 if (std::find(digits, digits + 10, fill) != digits + 10)
792 return !(fill == digits[0] && leads_digits);
793
794 // Past the amount nothing but a digit can still be read into it.
795 if (trails_value)
796 return false;
797
798 // A fill is one character, so it is held against the first of the point.
799 if (fill == m_decimal_point.front())
800 return true;
801 if (!info.m_negative_sign.empty() && fill == info.m_negative_sign[0])
802 return !negative;
803 if (!info.m_positive_sign.empty() && fill == info.m_positive_sign[0])
804 return negative;
805 return false;
806 }

864 template <bool isIntl, typename TIter>
865 TIter insert(TIter s, ios_base<char_type>& io, const std::basic_string<char_type>& digits) const
866 {
867 const split_info& info = isIntl ? m_int : m_nat;
868 using part = base_ft<monetary>::part;
869
870 const std::size_t width = io.width(0);
871
872 // A leading minus is how the caller says the amount is negative. It selects
873 // this locale's negative pattern and negative sign string and is then dropped:
874 // the sign the field ends up carrying comes from that string, which may spell
875 // it quite differently (or not at all).
876 const char_type* first = digits.data();
877 const char_type* const stop = first + digits.size();
878
879 const bool negative = (*first == s_atoms[s_minus]);
880 // The pattern is four bytes and is read once per slot while the strings below
881 // are being built; taken by value it stays in a register, where a reference
882 // into the facet would have to be reloaded around every write.
883 const base_ft<monetary>::pattern order = negative ? info.m_neg_format : info.m_pos_format;
884 const std::basic_string<char_type>& sign = negative ? info.m_negative_sign : info.m_positive_sign;
885 if (negative)
886 ++first;
887
888 // The amount runs up to the first character that is not one of this facet's
889 // digits; whatever the caller put after that is not ours to format.
890 const char_type* const digit_table = s_atoms.data() + s_zero;
891 std::size_t len = 0;
892 for (const char_type* q = first; q != stop; ++q)
893 {
894 if (std::find(digit_table, digit_table + 10, *q) == digit_table + 10)
895 break;
896 ++len;
897 }
898 if (!len)
899 return s;
900
901 // The digits are the smallest units of the currency, so the amount they spell
902 // is read off their right-hand end: the run splits into an integer head and a
903 // fractional tail with the cut m_frac_digits places from that end. A run too
904 // short to reach the cut has a head of nothing, and the fraction picks up the
905 // shortfall as leading zeros — two fractional places turn 7 into .07, not 7.0.
906 // A negative frac_digits asks for no fraction at all and keeps the whole run.
907 const int frac = info.m_frac_digits;
908 // Widened once, and only from a value the `frac < 0` arm has already ruled
909 // out as negative, so every length comparison below stays unsigned-to-unsigned.
910 const std::size_t fracw = (frac > 0) ? static_cast<std::size_t>(frac) : 0;
911 const std::size_t head = (frac < 0) ? len
912 : (len > fracw) ? len - fracw
913 : 0;
914
915 std::basic_string<char_type> value;
916 value.reserve(2 * len);
917
918 if (head)
919 {
920 if (m_grouping.empty())
921 value.assign(first, head);
922 else
923 {
924 // add_grouping writes at most one separator per digit, which bounds
925 // the room it needs; the slack is trimmed off afterwards.
926 value.assign(head * (1 + m_thousands_sep.size()), char_type());
927 const char_type* const wrote = FacetHelper::add_grouping(
928 value.data(), std::basic_string_view<char_type>(m_thousands_sep), m_grouping,
929 first, first + head);
930 value.erase(static_cast<std::size_t>(wrote - value.data()));
931 }
932 }
933
934 if (frac > 0)
935 {
936 value += m_decimal_point;
937 if (fracw > len)
938 value.append(fracw - len, s_atoms[s_zero]);
939 value.append(first + head, len - head);
940 }
941
942 const ios_defs::fmtflags adjust = io.flags() & ios_defs::adjustfield;
943 const bool show_symbol = (io.flags() & ios_defs::showbase) != 0;
944
945 // What the field is worth before any padding: the amount, the sign string in
946 // full (even the part that trails), and the currency symbol if it is shown.
947 len = value.size() + sign.size() + (show_symbol ? info.m_curr_symbol.size() : 0);
948
949 if (width > len && width - len > ios_defs::max_pad_count)
950 throw stream_error("monetary put fail: fill count exceeds max_pad_count");
951
952 std::basic_string<char_type> res;
953 res.reserve(2 * len);
954
955 // Non-zero only under `internal`, where the shortfall is not tacked onto an
956 // end but poured into whichever pattern slot is spare.
957 const std::size_t spread = (adjust == ios_defs::internal && len < width) ? width - len : 0;
958
959 // Where each run of fill lands is recorded as it is written, so that the
960 // readability check below can be made against the finished field instead of
961 // against the pattern: whether a run abuts the amount depends on which
962 // parts between them turn out to be empty, which the pattern alone does not
963 // say. A pattern writes at most one run per part, plus one final pad.
964 struct fill_run { std::size_t pos; std::size_t len; };
965 std::array<fill_run, 5> runs{};
966 std::size_t run_count = 0;
967 std::size_t value_pos = std::basic_string<char_type>::npos;
968
969 // A space only separates things that are there: it is dropped next to a symbol
970 // left out, and where nothing is written on one side of it (see space_kept).
971 const bool symbol_written = show_symbol && !info.m_curr_symbol.empty();
972 const auto writes = [&](part p) {
973 return p == part::value || (p == part::symbol && symbol_written)
974 || (p == part::sign && !sign.empty());
975 };
976 const bool keep_space = space_kept(order, symbol_written, writes, writes);
977
978 // Lay the parts down in the order this locale asks for.
979 for (const part slot : order)
980 {
981 const part which = (slot == part::space && !keep_space) ? part::none : slot;
982 switch (which)
983 {
984 case part::symbol:
985 if (show_symbol)
986 res += info.m_curr_symbol;
987 break;
988 case part::sign:
989 // Only the sign's first character belongs to this slot; a longer sign
990 // string has its remainder appended once the pattern is fully down.
991 if (!sign.empty())
992 res += sign[0];
993 break;
994 case part::value:
995 value_pos = res.size();
996 res += value;
997 break;
998 case part::space:
999 // This slot owes at least one fill character, and takes the whole
1000 // internal spread when there is one to take.
1001 runs[run_count++] = {res.size(), spread ? spread : 1};
1002 res.append(spread ? spread : 1, io.fill());
1003 break;
1004 case part::none:
1005 // A slot that writes nothing of its own is still somewhere the
1006 // internal spread can go.
1007 if (spread)
1008 {
1009 runs[run_count++] = {res.size(), spread};
1010 res.append(spread, io.fill());
1011 }
1012 break;
1013 }
1014 }
1015
1016 // A sign spelled with more than one character wraps the field: its first
1017 // character sat in the sign slot, the rest trails everything.
1018 if (sign.size() > 1)
1019 res.append(sign, 1);
1020
1021 // Whatever width is still unaccounted for is fill.
1022 len = res.size();
1023 if (width > len)
1024 {
1025 const std::size_t pad = width - len;
1026 if (adjust == ios_defs::left)
1027 {
1028 runs[run_count++] = {len, pad};
1029 res.append(pad, io.fill());
1030 }
1031 else
1032 {
1033 // Padding in front shifts everything already recorded — the amount
1034 // included — right by that much.
1035 res.insert(0, pad, io.fill());
1036 for (std::size_t r = 0; r < run_count; ++r)
1037 runs[r].pos += pad;
1038 if (value_pos != std::basic_string<char_type>::npos)
1039 value_pos += pad;
1040 runs[run_count++] = {0, pad};
1041 }
1042 len = width;
1043 }
1044
1045 // Fill has now been written, so this is where it gets vetted: reject a fill
1046 // character that would change the amount this field reads as. The check sits
1047 // here rather than at the top of the function because `fill` is sticky stream
1048 // state — a stream carrying setfill('1') must keep working for every output
1049 // whose width leaves nothing to pad, and those take no run at all.
1050 for (std::size_t r = 0; r < run_count; ++r)
1051 {
1052 const bool leads_digits = (value_pos != std::basic_string<char_type>::npos)
1053 && (runs[r].pos + runs[r].len == value_pos);
1054 const bool trails_value = (value_pos != std::basic_string<char_type>::npos)
1055 && (runs[r].pos >= value_pos + value.size());
1056 if (fill_alters_reading(io.fill(), info, leads_digits, trails_value, negative))
1057 throw stream_error("monetary put fail: fill would change the value the field reads as");
1058 }
1059
1060 return std::copy(res.data(), res.data() + len, s);
1061 }

1135 template <bool isIntl, typename TIter, std::sentinel_for<TIter> TSent>
1136 std::pair<bool, TIter> extract(TIter beg, TSent end, ios_base<char_type>& io, std::string& units) const
1137 {
1138 const split_info& info = isIntl ? m_int : m_nat;
1139 using part = base_ft<monetary>::part;
1140
1141 // The negative pattern is the one walked, whichever sign the input turns out
1142 // to carry: it is the wider of the two, since a locale may spell the positive
1143 // sign as nothing at all, and the polarity is settled by the sign slot itself.
1144 const base_ft<monetary>::pattern& order = info.m_neg_format;
1145
1146 const char_type* const digit_table = s_atoms.data() + s_zero;
1147 const bool show_symbol = (io.flags() & ios_defs::showbase) != 0;
1148
1149 // A locale that spells both signs cannot have the field leave one out.
1150 const bool sign_required = !info.m_positive_sign.empty() && !info.m_negative_sign.empty();
1151 const bool symbol_written = show_symbol && !info.m_curr_symbol.empty();
1152
1153 bool negative = false; // polarity the sign slot deduced
1154 bool valid = true; // still looking at a well-formed field
1155 bool saw_point = false; // the decimal point has gone by
1156 bool read_any = false; // a symbol, sign or digit has been read
1157 int sign_len = 0; // length of the sign string the input turned out to carry
1158
1159 // Digits seen since the last structural character. Before the decimal point
1160 // that is the current group; when the point arrives the count is handed to
1161 // `int_run` and restarts, so afterwards it counts fractional places.
1162 int run = 0;
1163 int int_run = 0;
1164
1165 // The separator takes part only while grouping is on.
1166 const std::basic_string_view<char_type> sep =
1167 m_grouping.empty() ? std::basic_string_view<char_type>() : m_thousands_sep;
1168
1169 // Group widths in the order met, checked against m_grouping once the whole
1170 // amount is in.
1171 std::vector<uint8_t> grouping_tmp;
1172 if (!m_grouping.empty())
1173 grouping_tmp.reserve(32);
1174
1175 // Digits accumulate here as ASCII, and are handed to the caller only if the
1176 // field turns out to be well-formed all the way to its end.
1177 std::string out;
1178 out.reserve(32);
1179
1180 // Which part carries the amount: a run of fill consumed before that index sits
1181 // in front of the amount, one consumed after it sits behind.
1182 int value_idx = 3;
1183 for (int i = 0; i < 4; ++i)
1184 if (order[i] == part::value)
1185 {
1186 value_idx = i;
1187 break;
1188 }
1189
1190 for (int i = 0; i < 4 && valid; ++i)
1191 {
1192 // Set when this part consumes at least one fill character, which is the
1193 // only case in which fill decides how much of the input counts as the
1194 // amount — and so the only case worth vetting.
1195 bool ate_fill = false;
1196 switch (order[i])
1197 {
1198 case part::symbol:
1199 {
1200 // [locale.money.get.virtuals]: showbase makes the currency symbol
1201 // mandatory. Otherwise it is optional, and is read only where letting
1202 // it go would strand a part that still has to be reached — which
1203 // depends on what this slot is followed by, hence the case analysis.
1204 bool required = show_symbol || sign_len > 1;
1205 if (!required)
1206 switch (i)
1207 {
1208 case 0:
1209 // Nothing has been read yet; parsing has to start somewhere.
1210 required = true;
1211 break;
1212 case 1:
1213 required = sign_required || order[0] == part::sign
1214 || order[2] == part::space;
1215 break;
1216 case 2:
1217 required = order[3] == part::value
1218 || (sign_required && order[3] == part::sign);
1219 break;
1220 default:
1221 break;
1222 }
1223 if (!required)
1224 break;
1225
1226 const std::basic_string<char_type>& symbol = info.m_curr_symbol;
1227 std::size_t taken = 0;
1228 while (taken != symbol.size() && beg != end && *beg == symbol[taken])
1229 {
1230 ++beg;
1231 ++taken;
1232 }
1233 // Stopping part-way through the symbol always makes the field
1234 // malformed; stopping before its first character only does so when
1235 // showbase promised the symbol would be there.
1236 if (taken != symbol.size() && (taken != 0 || show_symbol))
1237 valid = false;
1238 read_any = read_any || taken != 0;
1239 break;
1240 }
1241 case part::sign:
1242 {
1243 // Whichever sign string the input opens with is the one it wears.
1244 const std::basic_string<char_type>& pos = info.m_positive_sign;
1245 const std::basic_string<char_type>& neg = info.m_negative_sign;
1246 const std::basic_string<char_type>* worn = nullptr;
1247 if (beg != end)
1248 {
1249 if (!pos.empty() && *beg == pos[0])
1250 worn = &pos;
1251 else if (!neg.empty() && *beg == neg[0])
1252 worn = &neg;
1253 }
1254
1255 if (worn)
1256 {
1257 negative = (worn == &neg);
1258 sign_len = static_cast<int>(worn->size());
1259 read_any = true;
1260 ++beg;
1261 }
1262 else if (pos.empty() != neg.empty())
1263 // [locale.money.get.virtuals]: where one of the two sign strings
1264 // is empty, a field with no sign in it takes the polarity of that
1265 // empty string. Only an empty negative string makes that negative.
1266 negative = neg.empty();
1267 else if (sign_required)
1268 valid = false;
1269 break;
1270 }
1271 case part::value:
1272 // Digits go straight into the result; the separators between them are
1273 // not kept, only the widths of the groups they mark off.
1274 while (beg != end)
1275 {
1276 const char_type* d = std::find(digit_table, digit_table + 10, *beg);
1277 if (d != digit_table + 10)
1278 {
1279 out += static_cast<char>('0' + (d - digit_table));
1280 ++run;
1281 ++beg;
1282 continue;
1283 }
1284
1285 // Punctuation structures the integer part only. Past the decimal
1286 // point the fraction is a plain run of digits, so a separator or a
1287 // second point there ends the amount instead of shaping it. A locale
1288 // with no fractional places has nothing for a point to introduce.
1289 // Either may be several characters long (see take_punct).
1290 const bool point_on = !saw_point && info.m_frac_digits > 0;
1291 const FacetHelper::punct_kind punct = FacetHelper::take_punct(
1292 beg, end, saw_point ? std::basic_string_view<char_type>() : sep,
1293 point_on ? std::basic_string_view<char_type>(m_decimal_point)
1294 : std::basic_string_view<char_type>(),
1295 true);
1296 if (punct == FacetHelper::punct_kind::point)
1297 {
1298 int_run = run;
1299 run = 0;
1300 saw_point = true;
1301 }
1302 else if (punct == FacetHelper::punct_kind::separator)
1303 {
1304 // A separator with no preceding digits, or a group
1305 // longer than the largest representable group size,
1306 // can never satisfy any grouping spec: reject outright
1307 // rather than truncating the count.
1308 if (run == 0 || std::cmp_greater(run, std::numeric_limits<uint8_t>::max()))
1309 {
1310 // The separator that broke the grouping stays in the input.
1311 FacetHelper::step_back(beg, sep, sep.size());
1312 valid = false;
1313 break;
1314 }
1315 grouping_tmp.push_back(static_cast<uint8_t>(run));
1316 run = 0;
1317 }
1318 else
1319 break;
1320 }
1321 // An amount with no digits at all is not an amount.
1322 if (out.empty())
1323 valid = false;
1324 read_any = true;
1325 break;
1326 case part::space:
1327 // Owed (at least one fill character) only where put is bound to have
1328 // written it; otherwise it is read as a `none`.
1329 if (space_kept(order, symbol_written, [&](part) { return read_any; },
1330 [&](part p) {
1331 return p == part::value
1332 || (p == part::symbol && symbol_written)
1333 || (p == part::sign && sign_required);
1334 }))
1335 {
1336 if (beg != end && (*beg == io.fill()))
1337 {
1338 ++beg;
1339 ate_fill = true;
1340 }
1341 else
1342 valid = false;
1343 }
1344 [[fallthrough]];
1345 case part::none:
1346 // Either slot then soaks up whatever further fill follows — except in
1347 // the last one, where trailing fill belongs to the stream, not to us.
1348 if (i != 3)
1349 for (; beg != end && (*beg == io.fill()); ++beg)
1350 ate_fill = true;
1351 // Reject fill that a reader would have taken for part of the amount:
1352 // `is >> setfill('1') >> get_money(v)` on "112" must fail loudly rather
1353 // than silently hand back 2. Whether the run leads the digits is read
1354 // off the input itself — the character it stopped on — because the
1355 // pattern does not say which parts in between came out empty.
1356 if (ate_fill
1357 && fill_alters_reading(io.fill(), info,
1358 i < value_idx && beg != end
1359 && (*beg == m_decimal_point.front()
1360 || std::find(digit_table, digit_table + 10, *beg)
1361 != digit_table + 10),
1362 i > value_idx, negative))
1363 throw stream_error("monetary get fail: fill would change the value the field reads as");
1364 break;
1365 }
1366 }
1367
1368 // A sign spelled with more than one character wraps the field: its first
1369 // character came out of the sign slot, and the rest has to trail everything.
1370 if (sign_len > 1 && valid)
1371 {
1372 const std::basic_string<char_type>& sign_str = negative ? info.m_negative_sign
1373 : info.m_positive_sign;
1374 std::size_t taken = 1;
1375 while (taken != sign_str.size() && beg != end && *beg == sign_str[taken])
1376 {
1377 ++beg;
1378 ++taken;
1379 }
1380 if (taken != sign_str.size())
1381 valid = false;
1382 }
1383
1384 bool succ = true;
1385 if (valid)
1386 {
1387 // Leading zeros carry no value, but the amount must not be erased along
1388 // with them: a field of nothing but zeros still comes back as one digit.
1389 if (out.size() > 1)
1390 {
1391 const std::size_t keep = out.find_first_not_of('0');
1392 if (keep == std::string::npos)
1393 out.erase(0, out.size() - 1);
1394 else if (keep != 0)
1395 out.erase(0, keep);
1396 }
1397
1398 // [locale.money.get.virtuals]: the deduced sign goes on the result — but
1399 // zero has no sign, so a field that read as zero stays unmarked.
1400 if (negative && !out.empty() && out[0] != '0')
1401 out.insert(out.begin(), '-');
1402
1403 // Whether the separators fell where m_grouping says they should.
1404 if (!grouping_tmp.empty())
1405 {
1406 // The leading group is the one no separator introduced, so it is
1407 // counted here rather than in the loop. A group longer than the
1408 // largest representable group size cannot satisfy any spec: fail
1409 // rather than truncating.
1410 const int last_group = saw_point ? int_run : run;
1411 if (std::cmp_greater(last_group, std::numeric_limits<uint8_t>::max()))
1412 succ = false;
1413 else
1414 {
1415 grouping_tmp.push_back(static_cast<uint8_t>(last_group));
1416 succ = FacetHelper::verify_grouping(m_grouping, grouping_tmp);
1417 }
1418 }
1419
1420 // A decimal point commits the field to exactly as many fractional places
1421 // as the locale declares — no more, and no fewer.
1422 if (saw_point && run != info.m_frac_digits)
1423 valid = false;
1424 }
1425
1426 // The result is handed over only for a field that held up all the way through.
1427 if (!valid)
1428 succ = false;
1429 else
1430 units.swap(out);
1431
1432 return std::pair(succ, beg);
1433 }

1465 template <std::integral TVal>
1466 bool str_to_v(const std::string& s, TVal& value) const
1467 {
1468 if (s.empty()) return false;
1469
1470 std::size_t i = 0;
1471 value = 0;
1472 constexpr TVal max_value = std::numeric_limits<TVal>::max();
1473
1474 if constexpr (std::is_signed_v<TVal>)
1475 {
1476 bool negative = false;
1477 constexpr TVal min_value = std::numeric_limits<TVal>::min();
1478
1479 if (s[i] == '+' || s[i] == '-') {
1480 negative = (s[i] == '-');
1481 ++i;
1482 }
1483
1484 if (i == s.size()) return false;
1485
1486 for (; i < s.size(); ++i)
1487 {
1488 char c = s[i];
1489 if (c < '0' || c > '9') return false;
1490
1491 int digit = c - '0';
1492
1493 if (!negative)
1494 {
1495 if (value > (max_value - digit) / 10)
1496 return false;
1497 }
1498 else if (value < (min_value + digit) / 10)
1499 return false;
1500
1501 value = value * 10 + (negative ? -digit : digit);
1502 }
1503 }
1504 else
1505 {
1506 if (s[i] == '+')
1507 ++i;
1508 else if (s[i] == '-')
1509 return false;
1510
1511 if (i == s.size()) return false;
1512
1513 for (; i < s.size(); ++i)
1514 {
1515 char c = s[i];
1516 if (c < '0' || c > '9') return false;
1517
1518 TVal digit = static_cast<TVal>(c - '0');
1519 if (value > (max_value - digit) / 10)
1520 return false;
1521
1522 value = value * 10 + digit;
1523 }
1524 }
1525 return true;
1526 }
