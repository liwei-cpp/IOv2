390 template <typename TIter, std::integral TVal>
391 requires (!std::same_as<TVal, bool>)
392 TIter put(TIter s, bool intl, ios_base<char_type>& io, TVal v) const
393 {
394 constexpr std::size_t buf_size = std::numeric_limits<TVal>::digits10 + 3;
395 std::array<char_type, buf_size> vec;
396
397 char_type* p = vec.data() + buf_size;
398 *--p = '\0';
399
400 using TU = std::make_unsigned_t<TVal>;
401 bool negative = false;
402 TU uv;
403
404 if constexpr(std::is_unsigned_v<TVal>)
405 uv = v;
406 else
407 {
408 negative = (v < 0);
409 uv = negative ? static_cast<TU>(-(v + 1)) + 1
410 : static_cast<TU>(v);
411 }
412
413 do { // NOLINT(cppcoreguidelines-avoid-do-while)
414 *--p = static_cast<char_type>('0' + (uv % 10));
415 uv /= 10;
416 } while (uv != 0);
417
418 if (negative)
419 *--p = '-';
420
421 return intl ? insert<true>(s, io, p)
422 : insert<false>(s, io, p);
423 }

565 : extract<false>(beg, end, io, str);
566 const auto len = str.size();
567 std::basic_string<char_type> tmp;
568 if (len)
569 {
570 tmp.reserve(len);
571 for (auto ch : str)
572 {
573 if (ch == '-')
574 tmp.push_back(s_atoms[0]);
575 else if ((ch >= '0') && (ch <= '9'))
576 tmp.push_back(s_atoms[ch - '0' + 1]);
577 }
578 }
579 if (!succ)
580 throw stream_error("monetary parse fail");
581 if (len)
582 digits.swap(tmp);
583 return beg;

679 [[nodiscard]] bool fill_alters_reading_(char_type fill, const split_info& info,
680 bool leads_digits, bool trails_value,
681 bool negative) const
682 {
683 const char_type* const digits = s_atoms.data() + s_zero;
684 if (std::find(digits, digits + 10, fill) != digits + 10)
685 return !(fill == digits[0] && leads_digits);
686

688 if (trails_value)

691 if (fill == m_decimal_point)

694 return !negative;
695 if (!info.m_positive_sign.empty() && fill == info.m_positive_sign[0])
696 return negative;

754 template <bool isIntl, typename TIter>
755 TIter insert(TIter s, ios_base<char_type>& io, const std::basic_string<char_type>& digits) const
756 {
757 const split_info& info = isIntl ? m_int : m_nat;
758 using part = base_ft<monetary>::part;
759
760 // Capture and consume the field width up front. width() is one-shot, so
761 // it must be cleared exactly once per put; resetting it here — before any
762 // allocation or copy that could throw — guarantees no leftover width
763 // leaks into the next output operation on whichever path we leave by,
764 // including an exception thrown while formatting. The captured value is
765 // used for padding below.
766 const std::size_t width = io.width();
767 io.width(0);
768
769 // Determine if negative or positive formats are to be used, and
770 // discard leading negative_sign if it is present.
771 const char_type* beg = digits.data();
772
773 base_ft<monetary>::pattern p;
774 const std::basic_string<char_type>* sign_ptr = nullptr;
775 if (!(*beg == s_atoms[s_minus]))
776 {
777 p = info.m_pos_format;
778 sign_ptr = &(info.m_positive_sign);
779 }
780 else
781 {
782 p = info.m_neg_format;
783 sign_ptr = &(info.m_negative_sign);
784 if (digits.size())
785 ++beg;
786 }
787
788 // Look for valid numbers in input digits.
789 std::size_t len = 0;
790 for (auto i = beg; i != digits.data() + digits.size(); ++i)
791 {
792 char_type ch = *i;
793 int j = 1;
794 for (; j < 11; ++j)
795 if (ch == s_atoms[j]) break;
796
797 if (j == 11) break;
798 ++len;
799 }
800
801 if (len)
802 {
803 // Assume valid input, and attempt to format.
804 // Break down input numbers into base components, as follows:
805 // final_value = grouped units + (decimal point) + (digits)
806 std::basic_string<char_type> value;
807 value.reserve(2 * len);
808
809 // Add thousands separators to non-decimal digits, per
810 // grouping rules.
811 // len counts characters of `digits`, so it never exceeds PTRDIFF_MAX and
812 // the conversion is lossless; the subtraction must stay signed because
813 // m_frac_digits may exceed len (and may itself be negative).
814 std::ptrdiff_t paddec = static_cast<std::ptrdiff_t>(len) - info.m_frac_digits;
815 if (paddec > 0)
816 {
817 if (info.m_frac_digits < 0)
818 paddec = static_cast<std::ptrdiff_t>(len);
819 if (!m_grouping.empty())
820 {
821 value.assign(2 * paddec, char_type());
822 char_type* vend = FacetHelper::add_grouping(&value[0], m_thousands_sep, m_grouping, beg, beg + paddec);
823 value.erase(vend - &value[0]);
824 }
825 else
826 value.assign(beg, paddec);
827 }
828
829 // Deal with decimal point, decimal digits.
830 if (info.m_frac_digits > 0)
831 {
832 value += m_decimal_point;
833 if (paddec >= 0)
834 value.append(beg + paddec, info.m_frac_digits);
835 else
836 {
837 // Have to pad zeros in the decimal position.
838 value.append(-paddec, s_atoms[s_zero]);
839 value.append(beg, len);
840 }
841 }
842
843 // Calculate length of resulting string.
844 const ios_defs::fmtflags f = io.flags() & ios_defs::adjustfield;
845 len = value.size() + sign_ptr->size();
846 len += ((io.flags() & ios_defs::showbase) ? info.m_curr_symbol.size() : 0);
847
848 if (width > len && width - len > ios_defs::max_pad_count)
849 throw stream_error("monetary put fail: fill count exceeds max_pad_count");
850
851 std::basic_string<char_type> res;
852 res.reserve(2 * len);
853
854 const bool testipad = (f == ios_defs::internal && len < width);
855
856 // Where each run of fill lands is recorded as it is written, so that the
857 // readability check below can be made against the finished field instead of
858 // against the pattern: whether a run abuts the amount depends on which
859 // parts between them turn out to be empty, which the pattern alone does not
860 // say. A pattern writes at most one run per part, plus one final pad.
861 struct fill_run { std::size_t pos; std::size_t len; };
862 std::array<fill_run, 5> runs{};
863 std::size_t run_count = 0;
864 std::size_t value_pos = std::basic_string<char_type>::npos;
865
866 // Fit formatted digits into the required pattern.
867 for (int i = 0; i < 4; ++i)
868 {
869 const part which = static_cast<part>(p[i]);
870 switch (which)
871 {
872 case part::symbol:
873 if (io.flags() & ios_defs::showbase)
874 res += info.m_curr_symbol;
875 break;
876 case part::sign:
877 // Sign might not exist, or be more than one
878 // character long. In that case, add in the rest
879 // below.
880 if (!sign_ptr->empty())
881 res += (*sign_ptr)[0];
882 break;
883 case part::value:
884 value_pos = res.size();
885 res += value;
886 break;
887 case part::space:
888 // At least one space is required, but if internal
889 // formatting is required, an arbitrary number of
890 // fill spaces will be necessary.
891 runs[run_count++] = {res.size(), testipad ? width - len : 1};
892 if (testipad)
893 res.append(width - len, io.fill());
894 else
895 res += io.fill();
896 break;
897 case part::none:
898 if (testipad)
899 {
900 runs[run_count++] = {res.size(), width - len};
901 res.append(width - len, io.fill());
902 }
903 break;
904 }
905 }
906
907 // Special case of multi-part sign parts.
908 if (sign_ptr->size() > 1)
909 res.append(sign_ptr->c_str() + 1, sign_ptr->size() - 1);
910
911 // Pad, if still necessary.
912 len = res.size();
913 if (width > len)
914 {
915 const std::size_t plen = width - len;
916 if (f == ios_defs::left) // After.
917 {
918 runs[run_count++] = {res.size(), plen};
919 res.append(plen, io.fill());
920 }
921 else // Before.
922 {
923 res.insert(0, plen, io.fill());
924 // Everything recorded so far just moved right by plen.
925 for (std::size_t r = 0; r < run_count; ++r)
926 runs[r].pos += plen;
927 if (value_pos != std::basic_string<char_type>::npos)
928 value_pos += plen;
929 runs[run_count++] = {0, plen};
930 }
931 len = width;
932 }
933
934 // Fill has now been written, so this is where it gets vetted: reject a fill
935 // character that would change the amount this field reads as. The check sits
936 // here rather than at the top of the function because `fill` is sticky stream
937 // state — a stream carrying setfill('1') must keep working for every output
938 // whose width leaves nothing to pad, and those take no run at all.
939 for (std::size_t r = 0; r < run_count; ++r)
940 {
941 const bool leads_digits = (value_pos != std::basic_string<char_type>::npos)
942 && (runs[r].pos + runs[r].len == value_pos);
943 const bool trails_value = (value_pos != std::basic_string<char_type>::npos)
944 && (runs[r].pos >= value_pos + value.size());
945 if (fill_alters_reading_(io.fill(), info, leads_digits, trails_value,
946 sign_ptr == &info.m_negative_sign))
947 throw stream_error("monetary put fail: fill would change the value the field reads as");
948 }
949
950 // Write resulting, fully-formatted string to output iterator.
951 s = std::copy(res.data(), res.data() + len, s);
952 }
953 return s;
954 }

1024 template <bool isIntl, typename TIter, std::sentinel_for<TIter> TSent>
1025 std::pair<bool, TIter> extract(TIter beg, TSent end, ios_base<char_type>& io, std::string& units) const
1026 {
1027 const split_info& info = isIntl ? m_int : m_nat;
1028 using part = base_ft<monetary>::part;
1029
1030 // Deduced sign.
1031 bool negative = false;
1032 // Sign size.
1033 int sign_size = 0;
1034 // True if sign is mandatory.
1035 const bool mandatory_sign = (!info.m_positive_sign.empty() && !info.m_negative_sign.empty());
1036 // Vector of grouping info from thousands_sep plucked from units.
1037 std::vector<uint8_t> grouping_tmp;
1038 if (!m_grouping.empty())
1039 grouping_tmp.reserve(32);
1040 // Last position before the decimal point.
1041 int last_pos = 0;
1042 // Separator positions, then, possibly, fractional digits.
1043 int n = 0;
1044 // If input iterator is in a valid state.
1045 bool testvalid = true;
1046 // Flag marking when a decimal point is found.
1047 bool testdecfound = false;
1048
1049 // The tentative returned string is stored here.
1050 std::string res; res.reserve(32);
1051
1052 const char_type* lit_zero = s_atoms.data() + s_zero;
1053 const base_ft<monetary>::pattern p = info.m_neg_format;
1054
1055 // Which part carries the amount: a run of fill consumed before that index sits
1056 // in front of the amount, one consumed after it sits behind.
1057 int value_idx = 3;
1058 for (int i = 0; i < 4; ++i)
1059 if (static_cast<part>(p[i]) == part::value)
1060 {
1061 value_idx = i;
1062 break;
1063 }
1064
1065 for (int i = 0; i < 4 && testvalid; ++i)
1066 {
1067 const part which = static_cast<part>(p[i]);
1068 // Set when this part consumes at least one fill character, which is the
1069 // only case in which fill decides how much of the input counts as the
1070 // amount — and so the only case worth vetting.
1071 bool ate_fill = false;
1072 switch (which)
1073 {
1074 case part::symbol:
1075 // According to 22.2.6.1.2, p2, symbol is required
1076 // if (io.flags() & ios_base::showbase), otherwise
1077 // is optional and consumed only if other characters
1078 // are needed to complete the format.
1079 if (io.flags() & ios_defs::showbase || sign_size > 1
1080 || i == 0
1081 || (i == 1 && (mandatory_sign
1082 || (static_cast<part>(p[0])
1083 == part::sign)
1084 || (static_cast<part>(p[2])
1085 == part::space)))
1086 || (i == 2 && ((static_cast<part>(p[3])
1087 == part::value)
1088 || (mandatory_sign
1089 && (static_cast<part>(p[3])
1090 == part::sign)))))
1091 {
1092 const int len = info.m_curr_symbol.size();
1093 int j = 0;
1094 for (; beg != end && j < len && *beg == info.m_curr_symbol[j]; ++beg, (void)++j);
1095 if (j != len && (j || io.flags() & ios_defs::showbase))
1096 testvalid = false;
1097 }
1098 break;
1099 case part::sign:
1100 // Sign might not exist, or be more than one character long.
1101 if (!info.m_positive_sign.empty() && beg != end && *beg == info.m_positive_sign[0])
1102 {
1103 sign_size = info.m_positive_sign.size();
1104 ++beg;
1105 }
1106 else if (!info.m_negative_sign.empty() && beg != end && *beg == info.m_negative_sign[0])
1107 {
1108 negative = true;
1109 sign_size = info.m_negative_sign.size();
1110 ++beg;
1111 }
1112 else if (!info.m_positive_sign.empty() && info.m_negative_sign.empty())
1113 // "... if no sign is detected, the result is given the sign
1114 // that corresponds to the source of the empty string"
1115 negative = true;
1116 else if (mandatory_sign)
1117 testvalid = false;
1118 break;
1119 case part::value:
1120 // Extract digits, remove and stash away the
1121 // grouping of found thousands separators.
1122 for (; beg != end; ++beg)
1123 {
1124 const char_type c = *beg;
1125 const char_type* q = std::find(lit_zero, lit_zero + 10, c);
1126 if (q != lit_zero + 10)
1127 {
1128 res += '0' + (q - lit_zero);
1129 ++n;
1130 }
1131 else if (c == m_decimal_point && !testdecfound)
1132 {
1133 if (info.m_frac_digits <= 0)
1134 break;
1135
1136 last_pos = n;
1137 n = 0;
1138 testdecfound = true;
1139 }
1140 else if (!m_grouping.empty() && c == m_thousands_sep && !testdecfound)
1141 {
1142 // A separator with no preceding digits, or a group
1143 // longer than the largest representable group size,
1144 // can never satisfy any grouping spec: reject outright
1145 // rather than truncating the count.
1146 if (n == 0 || std::cmp_greater(n, std::numeric_limits<uint8_t>::max()))
1147 {
1148 testvalid = false;
1149 break;
1150 }
1151 // Mark position for later analysis.
1152 grouping_tmp.push_back(static_cast<uint8_t>(n));
1153 n = 0;
1154 }
1155 else
1156 break;
1157 }
1158 if (res.empty())
1159 testvalid = false;
1160 break;
1161 case part::space:
1162 // At least one space is required.
1163 if (beg != end && (*beg == io.fill()))
1164 {
1165 ++beg;
1166 ate_fill = true;
1167 }
1168 else
1169 testvalid = false;
1170 [[fallthrough]];
1171 case part::none:
1172 // Only if not at the end of the pattern.
1173 if (i != 3)
1174 for (; beg != end && (*beg == io.fill()); ++beg) ate_fill = true;
1175 // Reject fill that a reader would have taken for part of the amount:
1176 // `is >> setfill('1') >> get_money(v)` on "112" must fail loudly rather
1177 // than silently hand back 2. Whether the run leads the digits is read
1178 // off the input itself — the character it stopped on — because the
1179 // pattern does not say which parts in between came out empty.
1180 if (ate_fill
1181 && fill_alters_reading_(io.fill(), info,
1182 i < value_idx && beg != end
1183 && (*beg == m_decimal_point
1184 || std::find(lit_zero, lit_zero + 10, *beg)
1185 != lit_zero + 10),
1186 i > value_idx, negative))
1187 throw stream_error("monetary get fail: fill would change the value the field reads as");
1188 break;
1189 }
1190 }
1191
1192 // Need to get the rest of the sign characters, if they exist.
1193 if (sign_size > 1 && testvalid)
1194 {
1195 const auto& sign_str = negative ? info.m_negative_sign
1196 : info.m_positive_sign;
1197 int i = 1;
1198 for (; beg != end && i < sign_size && *beg == sign_str[i]; ++beg, (void)++i);
1199 if (i != sign_size)
1200 testvalid = false;
1201 }
1202
1203 bool succ = true;
1204 if (testvalid)
1205 {
1206 // Strip leading zeros.
1207 if (res.size() > 1)
1208 {
1209 const auto first = res.find_first_not_of('0');
1210 const bool only_zeros = first == std::string::npos;
1211 if (first)
1212 res.erase(0, only_zeros ? res.size() - 1 : first);
1213 }
1214
1215 // 22.2.6.1.2, p4
1216 if (negative && !res.empty() && res[0] != '0')
1217 res.insert(res.begin(), '-');
1218
1219 // Test for grouping fidelity.
1220 if (!grouping_tmp.empty())
1221 {
1222 // Add the ending grouping. A final group longer than the
1223 // largest representable group size cannot satisfy any spec:
1224 // fail rather than truncating.
1225 const int last_group = testdecfound ? last_pos : n;
1226 if (std::cmp_greater(last_group, std::numeric_limits<uint8_t>::max()))
1227 succ = false;
1228 else
1229 {
1230 grouping_tmp.push_back(static_cast<uint8_t>(last_group));
1231 succ = FacetHelper::verify_grouping(m_grouping, grouping_tmp);
1232 }
1233 }
1234
1235 // Iff not enough digits were supplied after the decimal-point.
1236 if (testdecfound && n != info.m_frac_digits)
1237 testvalid = false;
1238 }
1239
1240 // Iff valid sequence is not recognized.
1241 if (!testvalid)
1242 succ = false;
1243 else
1244 units.swap(res);
1245
1246 return std::pair(succ, beg);
1247 }
