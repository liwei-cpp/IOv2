67// The user's preferred languages, most preferred first, named as the system names them
68// (BCP-47 on macOS: "zh-Hans-CN", "en-US"); empty where there is no such list (Linux).
69inline std::vector<std::string> messages_system_languages()
70{
71#if defined(__APPLE__)
72 // Released however the loop below ends; CFRelease takes no null, unique_ptr passes none.
73 struct cf_release
74 {
75 void operator()(CFArrayRef a) const noexcept { CFRelease(a); }
76 };
77 const std::unique_ptr<std::remove_pointer_t<CFArrayRef>, cf_release> langs(CFLocaleCopyPreferredLanguages());
78
79 std::vector<std::string> res;
80 if (!langs)
81 return res;
82 const CFIndex count = CFArrayGetCount(langs.get());
83 for (CFIndex i = 0; i < count; ++i)
84 {
85 const auto name = static_cast<CFStringRef>(CFArrayGetValueAtIndex(langs.get(), i));
86 std::array<char, 128> buf{}; // a language tag is a few ASCII letters
87 if (name && CFStringGetCString(name, buf.data(), static_cast<CFIndex>(buf.size()), kCFStringEncodingUTF8))
88 res.emplace_back(buf.data());
89 }
90 return res;
91#else
92 return {};
93#endif
94}
