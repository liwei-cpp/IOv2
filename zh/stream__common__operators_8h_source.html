1028 bool check_tie(abs_flusher* str) const noexcept
1029 {
1030 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
1031 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
1032 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
1033 // tie target.
1034 auto next = [](stream_common_operators* p)
1035 { return dynamic_cast<stream_common_operators*>(p->tie()); };
1036
1037 auto slow = dynamic_cast<stream_common_operators*>(str);
1038 auto fast = slow;
1039
1040 while (slow != nullptr)
1041 {
1042 if (slow == this)
1043 return false;
1044
1045 slow = next(slow);
1046 fast = (fast != nullptr) ? next(fast) : nullptr;
1047 fast = (fast != nullptr) ? next(fast) : nullptr;
1048
1049 if (fast != nullptr && fast == slow)
1050 return false;
1051 }
1052
1053 return true;
1054 }
