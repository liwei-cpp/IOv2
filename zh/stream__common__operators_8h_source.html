815private:
862 void ensure_no_cycle(abs_flusher* str) const
863 {
864 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
865 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
866 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
867 // tie target.
868 auto next = [](stream_common_operators* p)
869 { return dynamic_cast<stream_common_operators*>(p->tie()); };
870
871 stream_common_operators* slow = dynamic_cast<stream_common_operators*>(str);
872 stream_common_operators* fast = slow;
873
874 while (slow != nullptr)
875 {
876 if (slow == this)
877 throw stream_error("stream tie fail: the requested tie would form a cycle in the tie chain");
878
879 slow = next(slow);
880 fast = (fast != nullptr) ? next(fast) : nullptr;
881 fast = (fast != nullptr) ? next(fast) : nullptr;
882
883 if (fast != nullptr && fast == slow)
884 throw stream_error("stream tie fail: the tie chain already contains a cycle; "
885 "something other than tie() has written a tie pointer");
886 }
887 }
