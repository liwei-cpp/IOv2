1015 bool check_tie(abs_flusher* str) const noexcept
1016 {
1017 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
1018 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
1019 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
1020 // tie target.
1021 auto next = [](stream_common_operators* p)
1022 { return dynamic_cast<stream_common_operators*>(p->tie()); };
1023
1024 stream_common_operators* slow = dynamic_cast<stream_common_operators*>(str);
1025 stream_common_operators* fast = slow;
1026
1027 while (slow != nullptr)
1028 {
1029 if (slow == this)
1030 return false;
1031
1032 slow = next(slow);
1033 fast = (fast != nullptr) ? next(fast) : nullptr;
1034 fast = (fast != nullptr) ? next(fast) : nullptr;
1035
1036 if (fast != nullptr && fast == slow)
1037 return false;
1038 }
1039
1040 return true;
1041 }
