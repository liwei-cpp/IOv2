583 template <typename TSelf>
584 abs_flusher* tie(this TSelf& self, abs_flusher* str)
585 {
586 std::lock_guard graph_lock(tie_graph_mutex());
587 auto res = self.m_tie_stream.load();
588 if constexpr (std::derived_from<TSelf, abs_flusher>)
589 {
590 abs_flusher* ptr = str;
591 auto check = static_cast<abs_flusher*>(&self);
592 while (ptr != nullptr)
593 {
594 if (ptr == check)
595 throw stream_error("stream tie fail: the requested tie would form a cycle in the tie chain");
596 auto* node = dynamic_cast<stream_common_operators*>(ptr);
597 if (!node)
598 break;
599 ptr = node->tie();
600 }
601 }
602 self.m_tie_stream.store(str);
603 return res;
604 }
