897 {
898 tie_target* res = nullptr;
899 bool ok = true;
900
901 {
902 std::lock_guard graph_lock(tie_graph_mutex());
903 res = self.m_tie_stream.load();
904 if constexpr (std::derived_from<TSelf, tie_target>)

912 "stream tie fail: the requested tie would form a cycle in the tie chain, "
913 "or the chain already contains one")));

1064 static bool check_tie(const tie_target& self, const tie_target* str) noexcept
1065 {
1066 // A node with no outgoing edge (tied_to() is nullptr) ends the chain.
1067 const tie_target* slow = str;
1068 const tie_target* fast = slow;
1069
1070 while (slow != nullptr)
1071 {
1072 if (slow == &self)
1073 return false;
1074
1075 slow = slow->tied_to();
1076 fast = (fast != nullptr) ? fast->tied_to() : nullptr;
1077 fast = (fast != nullptr) ? fast->tied_to() : nullptr;
1078
1079 if (fast != nullptr && fast == slow)
1080 return false;
1081 }
1082
1083 return true;
1084 }
