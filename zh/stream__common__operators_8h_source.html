717private:
764 void ensure_no_cycle(abs_flusher* str) const
765 {
766 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
767 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
768 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
769 // tie target.
770 auto next = [](stream_common_operators* p)
771 { return dynamic_cast<stream_common_operators*>(p->tie()); };
772
773 stream_common_operators* slow = dynamic_cast<stream_common_operators*>(str);
774 stream_common_operators* fast = slow;
775
776 while (slow != nullptr)
777 {
778 if (slow == this)
779 throw stream_error("stream tie fail: the requested tie would form a cycle in the tie chain");
780
781 slow = next(slow);
782 fast = (fast != nullptr) ? next(fast) : nullptr;
783 fast = (fast != nullptr) ? next(fast) : nullptr;
784
785 if (fast != nullptr && fast == slow)
786 throw stream_error("stream tie fail: the tie chain already contains a cycle; "
787 "something other than tie() has written a tie pointer");
788 }
789 }
