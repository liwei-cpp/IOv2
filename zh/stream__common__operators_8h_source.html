870 template <typename TSelf>
871 abs_flusher* tie(this TSelf& self, abs_flusher* str)
872 {
873 abs_flusher* res = nullptr;
874 bool ok = true;
875
876 {
877 std::lock_guard graph_lock(tie_graph_mutex());
878 res = self.m_tie_stream.load();
879 if constexpr (std::derived_from<TSelf, abs_flusher>)
880 ok = self.check_tie(str);
881 if (ok)
882 self.m_tie_stream.store(str);
883 }

887 "stream tie fail: the requested tie would form a cycle in the tie chain, "
888 "or the chain already contains one")));

890 return res;

913private:
965 bool check_tie(abs_flusher* str) const noexcept
966 {
967 // Edges are stored as abs_flusher*, but only a stream_common_operators carries an
968 // outgoing edge, so each step has to cross-cast. A node that is a bare abs_flusher
969 // yields nullptr and ends the chain -- which is what keeps a plain flusher usable as a
970 // tie target.
971 auto next = [](stream_common_operators* p)
972 { return dynamic_cast<stream_common_operators*>(p->tie()); };
973
974 auto slow = dynamic_cast<stream_common_operators*>(str);
975 auto fast = slow;
976
977 while (slow != nullptr)
978 {
979 if (slow == this)
980 return false;
981
982 slow = next(slow);
983 fast = (fast != nullptr) ? next(fast) : nullptr;
984 fast = (fast != nullptr) ? next(fast) : nullptr;
985
986 if (fast != nullptr && fast == slow)
987 return false;
988 }
989
990 return true;
991 }
