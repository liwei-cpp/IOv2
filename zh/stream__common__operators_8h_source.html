900 tie_target* tie(this TSelf& self, tie_target* str)
901 {
902 tie_target* res = nullptr;
903 bool ok = true;
904
905 {
906 std::lock_guard graph_lock(tie_graph_mutex());
907 res = self.m_tie_stream.load();
908 if constexpr (std::derived_from<TSelf, tie_target>)
909 ok = check_tie(static_cast<const tie_target&>(self), str);
910 if (ok)
911 self.m_tie_stream.store(str);
912 }
913
914 if (!ok)
915 self.handle_exception(std::make_exception_ptr(stream_error(
916 "stream tie fail: the requested tie would form a cycle in the tie chain, "
917 "or the chain already contains one")));

919 return res;
920 }

1037 static bool check_tie(const tie_target& self, const tie_target* str) noexcept
1038 {
1039 // A node with no outgoing edge (tied_to() is nullptr) ends the chain.
1040 const tie_target* slow = str;
1041 const tie_target* fast = slow;
1042
1043 while (slow != nullptr)
1044 {
1045 if (slow == &self)
1046 return false;
1047
1048 slow = slow->tied_to();
1049 fast = (fast != nullptr) ? fast->tied_to() : nullptr;
1050 fast = (fast != nullptr) ? fast->tied_to() : nullptr;
1051
1052 if (fast != nullptr && fast == slow)
1053 return false;
1054 }
1055
1056 return true;
1057 }
