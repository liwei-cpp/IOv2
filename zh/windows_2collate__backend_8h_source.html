62// src, in loc's LC_COLLATE code page, as UTF-16; nullopt if it is invalid there.
63inline std::optional<std::wstring> collate_widen(const char* src, clocale_handle loc)
64{
65 const int len = MultiByteToWideChar(loc->collate_cp, win_mb_err_invalid_chars, src, -1, nullptr, 0);
66 if (len <= 0)
67 return std::nullopt;
68 std::wstring res(static_cast<std::size_t>(len - 1), L'\0');
69 // len counts the NUL, which lands on res's own terminator.
70 if (MultiByteToWideChar(loc->collate_cp, win_mb_err_invalid_chars, src, -1, res.data(), len) != len)
71 return std::nullopt;
72 return res;
73}

75inline std::optional<int> collate_compare_utf16(const wchar_t* s1, const wchar_t* s2, clocale_handle loc) noexcept
76{
77 const int res = CompareStringEx(loc->collate_bcp47.c_str(), 0, s1, -1, s2, -1, nullptr, nullptr, 0);
78 if (res == 0)
79 return std::nullopt;
80 return res - win_cstr_equal;

83// The sort key of s, without the NUL LCMapStringEx ends it with; nullopt on failure.
84inline std::optional<std::string> collate_sort_key(const wchar_t* s, clocale_handle loc)
85{
86 const wchar_t* name = loc->collate_bcp47.c_str();
87 const int len = LCMapStringEx(name, win_lcmap_sortkey, s, -1, nullptr, 0, nullptr, nullptr, 0);
88 if (len <= 0)
89 return std::nullopt;
90 std::string res(static_cast<std::size_t>(len), '\0');
91 // Under LCMAP_SORTKEY the destination is a buffer of len bytes, whatever its type.
92 if (LCMapStringEx(name, win_lcmap_sortkey, s, -1,
93 reinterpret_cast<wchar_t*>(res.data()), len, // NOLINT(cppcoreguidelines-pro-type-reinterpret-cast)
94 nullptr, nullptr, 0) != len)
95 return std::nullopt;
96 res.pop_back();
97 return res;
98}

100inline std::optional<int> collate_strcoll(const char* s1, const char* s2, clocale_handle loc)
101{
102 if (loc->collate_bcp47.empty())
103 {
104 const int res = std::strcmp(s1, s2);
105 return (res > 0) - (res < 0);
106 }
107 const auto w1 = collate_widen(s1, loc);
108 const auto w2 = collate_widen(s2, loc);
109 if (!w1 || !w2)
110 return std::nullopt;
111 return collate_compare_utf16(w1->c_str(), w2->c_str(), loc);
112}

124inline std::optional<std::size_t> collate_strxfrm(char* dest, const char* src, std::size_t n, clocale_handle loc)
125{
126 if (loc->collate_bcp47.empty())
127 {
128 const std::size_t len = std::strlen(src);
129 if (len < n)
130 std::memcpy(dest, src, len + 1);
131 return len;
132 }
133 const auto wide = collate_widen(src, loc);
134 if (!wide)
135 return std::nullopt;
136 const auto key = collate_sort_key(wide->c_str(), loc);
137 if (!key)
138 return std::nullopt;
139 if (key->size() < n)
140 std::memcpy(dest, key->c_str(), key->size() + 1);
141 return key->size();
142}

144inline std::optional<std::size_t> collate_wcsxfrm(wchar_t* dest, const wchar_t* src, std::size_t n, clocale_handle loc)
145{
146 if (loc->collate_bcp47.empty())
147 {
148 const std::size_t len = std::wcslen(src);
149 if (len < n)
150 std::wmemcpy(dest, src, len + 1);
151 return len;
152 }
153 const auto key = collate_sort_key(src, loc);
154 if (!key)
155 return std::nullopt;
156 if (key->size() < n)
157 {
158 for (std::size_t i = 0; i < key->size(); ++i)
159 dest[i] = static_cast<wchar_t>(static_cast<unsigned char>((*key)[i]));
160 dest[key->size()] = L'\0';
161 }
162 return key->size();
163}
