26// The value of the environment variable name, copied out at once; empty if it is unset.
27// A value that grows between the two calls no longer fits and reads as unset.
28inline std::string get_env(const char* name)
29{
30 std::size_t len = 0; // the size getenv_s wants, the terminating null included
31 if (getenv_s(&len, nullptr, 0, name) != 0 || len == 0)
32 return {};
33 std::string res(len, '\0');
34 if (getenv_s(&len, res.data(), res.size(), name) != 0 || len == 0)
35 return {};
36 res.resize(len - 1);
37 return res;
38}
39}
