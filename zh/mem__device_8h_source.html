333 template <bool Saturate = false>
334 auto get_buf(std::size_t to_max)
335 {
336 assert(m_str.size() >= m_next_pos);
337 const std::size_t remain = m_str.size() - m_next_pos;
338 if constexpr (Saturate)
339 {
340 if (to_max > remain)
341 throw device_error("mem_device::get_buf fail: not enough input");
342 auto res = const_cast<const CharT*>(m_str.c_str() + m_next_pos);
343 m_next_pos += to_max;
344 return res;
345 }
346 else
347 {
348 const std::size_t res_len = std::min(to_max, remain);
349 auto res = std::pair<const CharT*, std::size_t>{m_str.c_str() + m_next_pos, res_len};
350 m_next_pos += res_len;

368 void get_rollback(std::size_t len)

407 CharT* put_buf(std::size_t len)
408 {
409 if (len > m_str.max_size() - m_next_pos)
410 throw device_error("mem_device::put_buf fail: size overflow");
411
412 m_put_buf_checkpoint = m_str.size();
413 const std::size_t needed = m_next_pos + len;
414 if (needed > m_str.size())
415 {
416 if (needed > m_str.capacity())
417 {
418 const std::size_t cap = m_str.capacity();
419 m_str.reserve(std::max(needed, cap + cap / 2));
420 }
421 m_str.resize(needed);
422 }
423 CharT* res = m_str.data() + m_next_pos;
424 m_next_pos = needed;
425 return res;
426 }
427
451 void put_rollback(std::size_t len)
452 {
453 if (!m_put_buf_checkpoint.has_value())
454 throw device_error("mem_device::put_rollback fail: no active put_buf");
455 if (len == 0)
456 throw device_error("mem_device::put_rollback fail: length cannot be zero");
457 if (m_next_pos < len)
458 throw device_error("mem_device::put_rollback fail: rollback length too large");
459
460 const std::size_t saved_size = m_put_buf_checkpoint.value();
461 m_put_buf_checkpoint.reset();
462
463 m_next_pos -= len;
464 m_str.resize(std::max(m_next_pos, saved_size));
465 }

467private:

471 // Checkpoint for put_buf/put_rollback pairing.

473 // consumed and cleared by put_rollback to restore proper size.
474 // If empty (nullopt), put_rollback will throw.
475 std::optional<std::size_t> m_put_buf_checkpoint;
