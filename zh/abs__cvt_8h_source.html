206 auto get_buf(std::size_t to_max)
207 {
208 if (to_max == 0)
209 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
210 if (to_max > m_buffer.size())
211 throw cvt_error("cvt_reader::get_buf fail, read size too large");
212
213 const auto rollback_size = m_end_pos - m_cur_pos;
214 if (to_max <= rollback_size)
215 {
216 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
217 m_cur_pos += to_max;
218 if constexpr (!Saturate)
219 return res;
220 else
221 return res.first;
222 }
223
224 if (to_max > m_buffer.size() - m_cur_pos)
225 {
226 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
227 m_cur_pos = 0;
228 m_end_pos = rollback_size;
229 }
230 const std::size_t aim_size = to_max - rollback_size;
231 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
232 m_end_pos += read_size;
233
234 if constexpr (Saturate)
235 {
236 while (read_size < aim_size)
237 {
238 const std::size_t new_aim_size = aim_size - read_size;
239 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
240 if (new_read_size == 0)
241 throw cvt_error("get_buf<Saturate> fail: meet eos");
242 m_end_pos += new_read_size;
243 read_size += new_read_size;
244 }
245 }
246
247 if constexpr (!Saturate)
248 {
249 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
250 m_cur_pos = m_end_pos;
251 return res;
252 }
253 else
254 {
255 const char_type* res = m_buffer.data() + m_cur_pos;
256 m_cur_pos = m_end_pos;
257 return res;
258 }

1457 template <typename Self>
1458 std::size_t get(this Self& self, internal_type* to, std::size_t to_max)
1459 requires requires(Self& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1460 { t.get_main(r, data, len) } -> std::same_as<std::size_t>;
1461 }
1462 {
1463 // Read failures do not taint: they only advance the kernel cursor
1464 // and possibly leave a per-call mbstate undefined; the underlying
1465 // stream content is unchanged, and callers can reseek to recover.
1466 // We still honour any pre-existing taint so that a writer that
1467 // corrupted the stream cannot then read back a confused image.
1468 self.assert_not_tainted();
1469
1470 cvt_reader<KernelType> reader(self.m_kernel, self.m_tmp_io_buffer);
1471
1472 if (!self.m_is_bos_done)
1473 {
1474 if (to_max == 0) return 0;
1475
1476 constexpr std::size_t ext_size = sizeof(external_type);
1477
1478 reader.reset(s_bos_chunk);
1479 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1480 char* dest_bytes = reinterpret_cast<char*>(to);
1481 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1482 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1483
1484 // BOS phase requires complete data: if the stream ends unexpectedly
1485 // before providing enough bytes, this is a format error and we throw.
1486 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1487
1488 // Read full external_type chunks
1489 while (dest_bytes + ext_size <= dest_bytes_end)
1490 {
1491 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1492
1493 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1494 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1495 dest_bytes += units_to_read_now * ext_size;
1496 }
1497
1498 // Read the final partial internal_type if needed
1499 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1500 if (final_remainder_bytes > 0)
1501 {
1502 const auto* src_buf = reader.template get_buf<true>(1);
1503 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1504 dest_bytes += final_remainder_bytes;
1505 }
1506
1507 return to_max;
1508 }
1509 else
1510 {
1511 if (self.m_io_status != io_status::input)
1512 {
1513 if constexpr (cvt_cpt::support_io_switch<Self>)
1514 self.switch_to_get();
1515 else
1516 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1517 }
1518 return self.get_main(reader, to, to_max);
1519 }

1567 template <typename Self>
1568 void put(this Self& self, const internal_type* to, std::size_t to_size)
1569 requires requires(Self& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1570 { t.put_main(w, data, len) } -> std::same_as<void>;
1571 }
1572 {
1573 self.assert_not_tainted();
1574
1575 // IO-direction switching does not write bytes to the output stream:
1576 // a failed switch_to_put() leaves the stream unchanged, so the
1577 // converter is not corrupted and must not be tainted. Perform the
1578 // switch BEFORE entering the taint-on-exception region so that only
1579 // actual kernel writes can cause tainting.
1580 if (self.m_is_bos_done && self.m_io_status != io_status::output)
1581 {
1582 if constexpr (cvt_cpt::support_io_switch<Self>)
1583 self.switch_to_put();
1584 else
1585 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1586 }
1587
1588 try
1589 {
1590 cvt_writer<KernelType> writer(self.m_kernel, self.m_tmp_io_buffer);
1591 if (!self.m_is_bos_done)
1592 {
1593 if (to_size == 0) return;
1594
1595 constexpr std::size_t ext_size = sizeof(external_type);
1596
1597 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1598 auto to_bytes = reinterpret_cast<const char*>(to);
1599 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1600 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1601
1602 // BOS phase writes complete external_type units. If the input data
1603 // is not a multiple of ext_size, pad the final unit with zeros.
1604
1605 writer.reset(s_bos_chunk);
1606 while (to_bytes + ext_size <= to_bytes_end)
1607 {
1608 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1609 auto ptr = writer.put_buf(dest_count);
1610 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1611 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1612 to_bytes += dest_count * ext_size;
1613 }
1614
1615 if (to_bytes < to_bytes_end)
1616 {
1617 // The while loop above guarantees remaining < ext_size.
1618 // The modulo is logically redundant but helps the compiler's
1619 // static analyzer prove the bound for memset size calculation.
1620 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1621 auto ptr = writer.put_buf(1);
1622 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1623 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1624 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1625 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1626 }
1627
1628 writer.commit();
1629 }
1630 else
1631 {
1632 // switch_to_put() was already called above if needed;
1633 // m_io_status is guaranteed to be output at this point.
1634 self.put_main(writer, to, to_size);
1635 // commit() is the sole responsibility of abs_cvt::put: derived
1636 // put_main implementations must NOT call writer.commit() themselves.
1637 // If put_main throws, this commit is skipped and any uncommitted
1638 // buffer contents are dropped at writer destruction; the catch
1639 // below taints the converter.
1640 writer.commit();
1641 }
1642 }
1643 catch (...)
1644 {
1645 self.m_is_tainted = true;
1646 throw;
1647 }
1648 }
