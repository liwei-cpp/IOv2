202 auto get_buf(std::size_t to_max)
203 {
204 if (to_max == 0)
205 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
206 if (to_max > m_buffer.size())
207 throw cvt_error("cvt_reader::get_buf fail, read size too large");
208
209 const auto rollback_size = m_end_pos - m_cur_pos;
210 if (to_max <= rollback_size)
211 {
212 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
213 m_cur_pos += to_max;
214 if constexpr (!Saturate)
215 return res;
216 else
217 return res.first;
218 }
219
220 if (to_max > m_buffer.size() - m_cur_pos)
221 {
222 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
223 m_cur_pos = 0;
224 m_end_pos = rollback_size;
225 }
226 const std::size_t aim_size = to_max - rollback_size;
227 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
228 m_end_pos += read_size;
229
230 if constexpr (Saturate)
231 {
232 while (read_size < aim_size)
233 {
234 const std::size_t new_aim_size = aim_size - read_size;
235 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
236 if (new_read_size == 0)
237 throw cvt_error("get_buf<Saturate> fail: meet eos");
238 m_end_pos += new_read_size;
239 read_size += new_read_size;
240 }
241 }
242
243 if constexpr (!Saturate)
244 {
245 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
246 m_cur_pos = m_end_pos;
247 return res;
248 }
249 else
250 {
251 const char_type* res = m_buffer.data() + m_cur_pos;
252 m_cur_pos = m_end_pos;
253 return res;
254 }

1418 std::size_t get(internal_type* to, std::size_t to_max)
1419 requires requires(CurrentType& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1420 { t.get_main(r, data, len) } -> std::same_as<std::size_t>;
1421 }
1422 {
1423 // Read failures do not taint: they only advance the kernel cursor
1424 // and possibly leave a per-call mbstate undefined; the underlying
1425 // stream content is unchanged, and callers can reseek to recover.
1426 // We still honour any pre-existing taint so that a writer that
1427 // corrupted the stream cannot then read back a confused image.
1428 assert_not_tainted();
1429
1430 cvt_reader<KernelType> reader(m_kernel, m_tmp_io_buffer);
1431
1432 if (!m_is_bos_done)
1433 {
1434 if (to_max == 0) return 0;
1435
1436 constexpr std::size_t ext_size = sizeof(external_type);
1437
1438 reader.reset(s_bos_chunk);
1439 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1440 char* dest_bytes = reinterpret_cast<char*>(to);
1441 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1442 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1443
1444 // BOS phase requires complete data: if the stream ends unexpectedly
1445 // before providing enough bytes, this is a format error and we throw.
1446 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1447
1448 // Read full external_type chunks
1449 while (dest_bytes + ext_size <= dest_bytes_end)
1450 {
1451 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1452
1453 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1454 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1455 dest_bytes += units_to_read_now * ext_size;
1456 }
1457
1458 // Read the final partial internal_type if needed
1459 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1460 if (final_remainder_bytes > 0)
1461 {
1462 const auto* src_buf = reader.template get_buf<true>(1);
1463 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1464 dest_bytes += final_remainder_bytes;
1465 }
1466
1467 return to_max;
1468 }
1469 else
1470 {
1471 if (m_io_status != io_status::input)
1472 {
1473 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1474 static_cast<CurrentType*>(this)->switch_to_get();
1475 else
1476 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1477 }
1478 return static_cast<CurrentType*>(this)->get_main(reader, to, to_max);
1479 }
1480 }

1527 void put(const internal_type* to, std::size_t to_size)
1528 requires requires(CurrentType& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1529 { t.put_main(w, data, len) } -> std::same_as<void>;
1530 }
1531 {
1532 assert_not_tainted();
1533
1534 // IO-direction switching does not write bytes to the output stream:
1535 // a failed switch_to_put() leaves the stream unchanged, so the
1536 // converter is not corrupted and must not be tainted. Perform the
1537 // switch BEFORE entering the taint-on-exception region so that only
1538 // actual kernel writes can cause tainting.
1539 if (m_is_bos_done && m_io_status != io_status::output)
1540 {
1541 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1542 static_cast<CurrentType*>(this)->switch_to_put();
1543 else
1544 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1545 }
1546
1547 try
1548 {
1549 cvt_writer<KernelType> writer(m_kernel, m_tmp_io_buffer);
1550 if (!m_is_bos_done)
1551 {
1552 if (to_size == 0) return;
1553
1554 constexpr std::size_t ext_size = sizeof(external_type);
1555
1556 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1557 auto to_bytes = reinterpret_cast<const char*>(to);
1558 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1559 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1560
1561 // BOS phase writes complete external_type units. If the input data
1562 // is not a multiple of ext_size, pad the final unit with zeros.
1563
1564 writer.reset(s_bos_chunk);
1565 while (to_bytes + ext_size <= to_bytes_end)
1566 {
1567 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1568 auto ptr = writer.put_buf(dest_count);
1569 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1570 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1571 to_bytes += dest_count * ext_size;
1572 }
1573
1574 if (to_bytes < to_bytes_end)
1575 {
1576 // The while loop above guarantees remaining < ext_size.
1577 // The modulo is logically redundant but helps the compiler's
1578 // static analyzer prove the bound for memset size calculation.
1579 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1580 auto ptr = writer.put_buf(1);
1581 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1582 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1583 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1584 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1585 }
1586
1587 writer.commit();
1588 }
1589 else
1590 {
1591 // switch_to_put() was already called above if needed;
1592 // m_io_status is guaranteed to be output at this point.
1593 static_cast<CurrentType*>(this)->put_main(writer, to, to_size);
1594 // commit() is the sole responsibility of abs_cvt::put: derived
1595 // put_main implementations must NOT call writer.commit() themselves.
1596 // If put_main throws, this commit is skipped and any uncommitted
1597 // buffer contents are dropped at writer destruction; the catch
1598 // below taints the converter.
1599 writer.commit();
1600 }
1601 }
1602 catch (...)
1603 {
1604 m_is_tainted = true;
1605 throw;
1606 }
1607 }
