204 template <bool Saturate = false>
205 auto get_buf(std::size_t to_max)
206 {
207 if (to_max == 0)
208 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
209 if (to_max > m_buffer.size())
210 throw cvt_error("cvt_reader::get_buf fail, read size too large");
211
212 const auto rollback_size = m_end_pos - m_cur_pos;
213 if (to_max <= rollback_size)
214 {
215 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
216 m_cur_pos += to_max;
217 if constexpr (!Saturate)
218 return res;
219 else
220 return res.first;
221 }
222
223 if (to_max > m_buffer.size() - m_cur_pos)
224 {
225 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
226 m_cur_pos = 0;
227 m_end_pos = rollback_size;
228 }
229 const std::size_t aim_size = to_max - rollback_size;
230 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
231 m_end_pos += read_size;
232
233 if constexpr (Saturate)
234 {
235 while (read_size < aim_size)
236 {
237 const std::size_t new_aim_size = aim_size - read_size;
238 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
239 if (new_read_size == 0)
240 throw cvt_error("get_buf<Saturate> fail: meet eos");
241 m_end_pos += new_read_size;
242 read_size += new_read_size;
243 }
244 }
245
246 if constexpr (!Saturate)
247 {
248 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
249 m_cur_pos = m_end_pos;
250 return res;
251 }
252 else
253 {
254 const char_type* res = m_buffer.data() + m_cur_pos;
255 m_cur_pos = m_end_pos;
256 return res;
257 }
258 }

1443 std::size_t get(internal_type* to, std::size_t to_max)
1444 requires requires(CurrentType& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1445 { t.get_main(r, data, len) } -> std::same_as<std::size_t>;
1446 }
1447 {
1448 // Read failures do not taint: they only advance the kernel cursor
1449 // and possibly leave a per-call mbstate undefined; the underlying
1450 // stream content is unchanged, and callers can reseek to recover.
1451 // We still honour any pre-existing taint so that a writer that
1452 // corrupted the stream cannot then read back a confused image.
1453 assert_not_tainted();
1454
1455 cvt_reader<KernelType> reader(m_kernel, m_tmp_io_buffer);
1456
1457 if (!m_is_bos_done)
1458 {
1459 if (to_max == 0) return 0;
1460
1461 constexpr std::size_t ext_size = sizeof(external_type);
1462
1463 reader.reset(s_bos_chunk);
1464 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1465 char* dest_bytes = reinterpret_cast<char*>(to);
1466 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1467 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1468
1469 // BOS phase requires complete data: if the stream ends unexpectedly
1470 // before providing enough bytes, this is a format error and we throw.
1471 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1472
1473 // Read full external_type chunks
1474 while (dest_bytes + ext_size <= dest_bytes_end)
1475 {
1476 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1477
1478 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1479 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1480 dest_bytes += units_to_read_now * ext_size;
1481 }
1482
1483 // Read the final partial internal_type if needed
1484 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1485 if (final_remainder_bytes > 0)
1486 {
1487 const auto* src_buf = reader.template get_buf<true>(1);
1488 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1489 dest_bytes += final_remainder_bytes;
1490 }
1491
1492 return to_max;
1493 }
1494 else
1495 {
1496 if (m_io_status != io_status::input)
1497 {
1498 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1499 static_cast<CurrentType*>(this)->switch_to_get();
1500 else
1501 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1502 }
1503 return static_cast<CurrentType*>(this)->get_main(reader, to, to_max);
1504 }
1505 }

1552 void put(const internal_type* to, std::size_t to_size)
1553 requires requires(CurrentType& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1554 { t.put_main(w, data, len) } -> std::same_as<void>;
1555 }
1556 {
1557 assert_not_tainted();
1558
1559 // IO-direction switching does not write bytes to the output stream:
1560 // a failed switch_to_put() leaves the stream unchanged, so the
1561 // converter is not corrupted and must not be tainted. Perform the
1562 // switch BEFORE entering the taint-on-exception region so that only
1563 // actual kernel writes can cause tainting.
1564 if (m_is_bos_done && m_io_status != io_status::output)
1565 {
1566 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1567 static_cast<CurrentType*>(this)->switch_to_put();
1568 else
1569 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1570 }
1571
1572 try
1573 {
1574 cvt_writer<KernelType> writer(m_kernel, m_tmp_io_buffer);
1575 if (!m_is_bos_done)
1576 {
1577 if (to_size == 0) return;
1578
1579 constexpr std::size_t ext_size = sizeof(external_type);
1580
1581 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1582 auto to_bytes = reinterpret_cast<const char*>(to);
1583 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1584 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1585
1586 // BOS phase writes complete external_type units. If the input data
1587 // is not a multiple of ext_size, pad the final unit with zeros.
1588
1589 writer.reset(s_bos_chunk);
1590 while (to_bytes + ext_size <= to_bytes_end)
1591 {
1592 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1593 auto ptr = writer.put_buf(dest_count);
1594 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1595 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1596 to_bytes += dest_count * ext_size;
1597 }
1598
1599 if (to_bytes < to_bytes_end)
1600 {
1601 // The while loop above guarantees remaining < ext_size.
1602 // The modulo is logically redundant but helps the compiler's
1603 // static analyzer prove the bound for memset size calculation.
1604 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1605 auto ptr = writer.put_buf(1);
1606 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1607 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1608 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1609 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1610 }
1611
1612 writer.commit();
1613 }
1614 else
1615 {
1616 // switch_to_put() was already called above if needed;
1617 // m_io_status is guaranteed to be output at this point.
1618 static_cast<CurrentType*>(this)->put_main(writer, to, to_size);
1619 // commit() is the sole responsibility of abs_cvt::put: derived
1620 // put_main implementations must NOT call writer.commit() themselves.
1621 // If put_main throws, this commit is skipped and any uncommitted
1622 // buffer contents are dropped at writer destruction; the catch
1623 // below taints the converter.
1624 writer.commit();
1625 }
1626 }
1627 catch (...)
1628 {
1629 m_is_tainted = true;
1630 throw;
1631 }
1632 }
