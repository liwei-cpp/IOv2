217 template <bool Saturate = false>
218 auto get_buf(std::size_t to_max)
219 {
220 if (to_max == 0)
221 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
222 if (to_max > m_buffer.size())
223 throw cvt_error("cvt_reader::get_buf fail, read size too large");
224
225 const auto rollback_size = m_end_pos - m_cur_pos;
226 if (to_max <= rollback_size)
227 {
228 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
229 m_cur_pos += to_max;
230 if constexpr (!Saturate)
231 return res;
232 else
233 return res.first;
234 }
235
236 if (to_max > m_buffer.size() - m_cur_pos)
237 {
238 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
239 m_cur_pos = 0;
240 m_end_pos = rollback_size;
241 }
242 const std::size_t aim_size = to_max - rollback_size;
243 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
244 m_end_pos += read_size;
245
246 if constexpr (Saturate)
247 {
248 while (read_size < aim_size)
249 {
250 const std::size_t new_aim_size = aim_size - read_size;
251 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
252 if (new_read_size == 0)
253 throw cvt_error("get_buf<Saturate> fail: meet eos");
254 m_end_pos += new_read_size;
255 read_size += new_read_size;
256 }
257 }
258
259 if constexpr (!Saturate)
260 {
261 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
262 m_cur_pos = m_end_pos;
263 return res;
264 }
265 else
266 {
267 const char_type* res = m_buffer.data() + m_cur_pos;
268 m_cur_pos = m_end_pos;
269 return res;
270 }
271 }

1540 template <typename Self>
1541 std::size_t get(this Self& self, internal_type* to, std::size_t to_max)
1542 requires requires(Self& t, internal_type* data, std::size_t len) {
1543 t.get_main(data, len);
1544 }
1545 {
1546 // An error that get_main() hit after delivering characters was held back so that
1547 // the caller got those characters; it is reported now.
1548 if (self.m_get_error)
1549 std::rethrow_exception(std::exchange(self.m_get_error, nullptr));
1550
1551 // Read failures do not taint: they only advance the kernel cursor
1552 // and possibly leave a per-call mbstate undefined; the underlying
1553 // stream content is unchanged, and callers can reseek to recover.
1554 // We still honour any pre-existing taint so that a writer that
1555 // corrupted the stream cannot then read back a confused image.
1556 self.assert_not_tainted();
1557
1558 if (!self.m_is_bos_done)
1559 {
1560 if (self.m_io_status == io_status::output)
1561 throw cvt_error("abs_cvt::get fail: cannot switch to input mode in the BOS phase");
1562 if (to_max == 0) return 0;
1563
1564 // Not kernel_reader(): the direction may still be neutral here.
1565 cvt_reader<KernelType> reader(self.m_kernel, self.m_tmp_io_buffer, self.m_buf_cur, self.m_buf_end);
1566
1567 constexpr std::size_t ext_size = sizeof(external_type);
1568
1569 reader.reset(s_bos_chunk);
1570 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1571 char* dest_bytes = reinterpret_cast<char*>(to);
1572 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1573 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1574
1575 // BOS phase requires complete data: if the stream ends unexpectedly
1576 // before providing enough bytes, this is a format error and we throw.
1577 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1578
1579 // Read full external_type chunks
1580 while (dest_bytes + ext_size <= dest_bytes_end)
1581 {
1582 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1583
1584 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1585 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1586 dest_bytes += units_to_read_now * ext_size;
1587 }
1588
1589 // Read the final partial internal_type if needed
1590 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1591 if (final_remainder_bytes > 0)
1592 {
1593 const auto* src_buf = reader.template get_buf<true>(1);
1594 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1595 dest_bytes += final_remainder_bytes;
1596 }
1597
1598 return to_max;
1599 }
1600 else
1601 {
1602 if (self.m_io_status != io_status::input)
1603 {
1604 if constexpr (cvt_cpt::support_io_switch<Self>)
1605 self.switch_to_get();
1606 else
1607 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1608 }
1609 static_assert(std::is_same_v<decltype(self.get_main(to, to_max)),
1610 std::pair<std::size_t, std::exception_ptr>>,
1611 "get_main() must return std::pair<std::size_t, std::exception_ptr>");
1612 static_assert(noexcept(self.get_main(to, to_max)),
1613 "get_main() must be noexcept");
1614 auto [count, err] = self.get_main(to, to_max);
1615 if (err)
1616 {
1617 if (count == 0)
1618 std::rethrow_exception(err);
1619 self.m_get_error = std::move(err);
1620 }
1621 return count;
1622 }
1623 }

1672 template <typename Self>
1673 void put(this Self& self, const internal_type* to, std::size_t to_size)
1674 requires requires(Self& t, const internal_type* data, std::size_t len) {
1675 { t.put_main(data, len) } -> std::same_as<void>;
1676 }
1677 {
1678 self.assert_not_tainted();
1679
1680 // IO-direction switching does not write bytes to the output stream:
1681 // a failed switch_to_put() leaves the stream unchanged, so the
1682 // converter is not corrupted and must not be tainted. Perform the
1683 // switch BEFORE entering the taint-on-exception region so that only
1684 // actual kernel writes can cause tainting.
1685 if (self.m_is_bos_done && self.m_io_status != io_status::output)
1686 {
1687 if constexpr (cvt_cpt::support_io_switch<Self>)
1688 self.switch_to_put();
1689 else
1690 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1691 }
1692 if (!self.m_is_bos_done && self.m_io_status == io_status::input)
1693 throw cvt_error("abs_cvt::put fail: cannot switch to output mode in the BOS phase");
1694
1695 try
1696 {
1697 cvt_writer<KernelType> writer(self.m_kernel, self.m_tmp_io_buffer, self.m_buf_end);
1698 if (!self.m_is_bos_done)
1699 {
1700 if (to_size == 0) return;
1701
1702 constexpr std::size_t ext_size = sizeof(external_type);
1703
1704 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1705 auto to_bytes = reinterpret_cast<const char*>(to);
1706 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1707 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1708
1709 // BOS phase writes complete external_type units. If the input data
1710 // is not a multiple of ext_size, pad the final unit with zeros.
1711
1712 writer.reset(s_bos_chunk);
1713 while (to_bytes + ext_size <= to_bytes_end)
1714 {
1715 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1716 auto ptr = writer.put_buf(dest_count);
1717 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1718 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1719 to_bytes += dest_count * ext_size;
1720 }
1721
1722 if (to_bytes < to_bytes_end)
1723 {
1724 // The while loop above guarantees remaining < ext_size.
1725 // The modulo is logically redundant but helps the compiler's
1726 // static analyzer prove the bound for memset size calculation.
1727 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1728 auto ptr = writer.put_buf(1);
1729 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1730 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1731 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1732 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1733 }
1734
1735 writer.commit();
1736 }
1737 else
1738 {
1739 // switch_to_put() was already called above if needed;
1740 // m_io_status is guaranteed to be output at this point.
1741 self.put_main(to, to_size);
1742 // If put_main throws, this commit is skipped; the catch below drops what
1743 // is still staged (output already handed to the kernel stays) and taints
1744 // the converter.
1745 writer.commit();
1746 }
1747 }
1748 catch (...)
1749 {
1750 // Left in place, it would go out with whatever is written next, such as what
1751 // closing the stream writes.
1752 self.m_buf_end = 0;
1753 self.m_is_tainted = true;
1754 throw;
1755 }
1756 }

1863 void seek(this auto& self, std::size_t pos)
1864 requires (enable_positioning && cvt_cpt::support_positioning<KernelType>)
1865 {
1866 // Fast path for no-op self-seek: skip validation and kernel work when
1867 // pos already equals the current position. Intentional even in modes
1868 // where the validation below would otherwise reject — callers using
1869 // seek(saved_pos) for position-restore must not be rejected when
1870 // saved_pos happens to equal tell().
