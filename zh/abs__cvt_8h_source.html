202 auto get_buf(std::size_t to_max)
203 {
204 if (to_max == 0)
205 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
206 if (to_max > m_buffer.size())
207 throw cvt_error("cvt_reader::get_buf fail, read size too large");
208
209 const auto rollback_size = m_end_pos - m_cur_pos;
210 if (to_max <= rollback_size)
211 {
212 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
213 m_cur_pos += to_max;
214 if constexpr (!Saturate)
215 return res;
216 else
217 return res.first;
218 }
219
220 if (to_max > m_buffer.size() - m_cur_pos)
221 {
222 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
223 m_cur_pos = 0;
224 m_end_pos = rollback_size;
225 }
226 const std::size_t aim_size = to_max - rollback_size;
227 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
228 m_end_pos += read_size;
229
230 if constexpr (Saturate)
231 {
232 while (read_size < aim_size)
233 {
234 const std::size_t new_aim_size = aim_size - read_size;
235 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
236 if (new_read_size == 0)
237 throw cvt_error("get_buf<Saturate> fail: meet eos");
238 m_end_pos += new_read_size;
239 read_size += new_read_size;
240 }
241 }
242
243 if constexpr (!Saturate)
244 {
245 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
246 m_cur_pos = m_end_pos;
247 return res;
248 }
249 else
250 {
251 const char_type* res = m_buffer.data() + m_cur_pos;
252 m_cur_pos = m_end_pos;
253 return res;
254 }
255 }

1388 std::size_t get(internal_type* to, std::size_t to_max)
1389 requires requires(CurrentType& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1390 { t.get_main(r, data, len) } -> std::same_as<std::size_t>;
1391 }
1392 {
1393 // Read failures do not taint: they only advance the kernel cursor
1394 // and possibly leave a per-call mbstate undefined; the underlying
1395 // stream content is unchanged, and callers can reseek to recover.
1396 // We still honour any pre-existing taint so that a writer that
1397 // corrupted the stream cannot then read back a confused image.
1398 assert_not_tainted();
1399
1400 cvt_reader<KernelType> reader(m_kernel, m_tmp_io_buffer);
1401
1402 if (!m_is_bos_done)
1403 {
1404 if (to_max == 0) return 0;
1405
1406 constexpr std::size_t ext_size = sizeof(external_type);
1407
1408 reader.reset(s_bos_chunk);
1409 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1410 char* dest_bytes = reinterpret_cast<char*>(to);
1411 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1412 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1413
1414 // BOS phase requires complete data: if the stream ends unexpectedly
1415 // before providing enough bytes, this is a format error and we throw.
1416 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1417
1418 // Read full external_type chunks
1419 while (dest_bytes + ext_size <= dest_bytes_end)
1420 {
1421 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1422
1423 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1424 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1425 dest_bytes += units_to_read_now * ext_size;
1426 }
1427
1428 // Read the final partial internal_type if needed
1429 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1430 if (final_remainder_bytes > 0)
1431 {
1432 const auto* src_buf = reader.template get_buf<true>(1);
1433 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1434 dest_bytes += final_remainder_bytes;
1435 }
1436
1437 return to_max;
1438 }
1439 else
1440 {
1441 if (m_io_status != io_status::input)
1442 {
1443 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1444 static_cast<CurrentType*>(this)->switch_to_get();
1445 else
1446 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1447 }
1448 return static_cast<CurrentType*>(this)->get_main(reader, to, to_max);
1449 }
1450 }

1497 void put(const internal_type* to, std::size_t to_size)
1498 requires requires(CurrentType& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1499 { t.put_main(w, data, len) } -> std::same_as<void>;
1500 }
1501 {
1502 assert_not_tainted();
1503
1504 // IO-direction switching does not write bytes to the output stream:
1505 // a failed switch_to_put() leaves the stream unchanged, so the
1506 // converter is not corrupted and must not be tainted. Perform the
1507 // switch BEFORE entering the taint-on-exception region so that only
1508 // actual kernel writes can cause tainting.
1509 if (m_is_bos_done && m_io_status != io_status::output)
1510 {
1511 if constexpr (cvt_cpt::support_io_switch<CurrentType>)
1512 static_cast<CurrentType*>(this)->switch_to_put();
1513 else
1514 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1515 }
1516
1517 try
1518 {
1519 cvt_writer<KernelType> writer(m_kernel, m_tmp_io_buffer);
1520 if (!m_is_bos_done)
1521 {
1522 if (to_size == 0) return;
1523
1524 constexpr std::size_t ext_size = sizeof(external_type);
1525
1526 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1527 auto to_bytes = reinterpret_cast<const char*>(to);
1528 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1529 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1530
1531 // BOS phase writes complete external_type units. If the input data
1532 // is not a multiple of ext_size, pad the final unit with zeros.
1533
1534 writer.reset(s_bos_chunk);
1535 while (to_bytes + ext_size <= to_bytes_end)
1536 {
1537 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1538 auto ptr = writer.put_buf(dest_count);
1539 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1540 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1541 to_bytes += dest_count * ext_size;
1542 }
1543
1544 if (to_bytes < to_bytes_end)
1545 {
1546 // The while loop above guarantees remaining < ext_size.
1547 // The modulo is logically redundant but helps the compiler's
1548 // static analyzer prove the bound for memset size calculation.
1549 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1550 auto ptr = writer.put_buf(1);
1551 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1552 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1553 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1554 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1555 }
1556
1557 writer.commit();
1558 }
1559 else
1560 {
1561 // switch_to_put() was already called above if needed;
1562 // m_io_status is guaranteed to be output at this point.
1563 static_cast<CurrentType*>(this)->put_main(writer, to, to_size);
1564 // commit() is the sole responsibility of abs_cvt::put: derived
1565 // put_main implementations must NOT call writer.commit() themselves.
1566 // If put_main throws, this commit is skipped and any uncommitted
1567 // buffer contents are dropped at writer destruction; the catch
1568 // below taints the converter.
1569 writer.commit();
1570 }
1571 }
1572 catch (...)
1573 {
1574 m_is_tainted = true;
1575 throw;
1576 }
1577 }
