216 template <bool Saturate = false>
217 auto get_buf(std::size_t to_max)
218 {
219 if (to_max == 0)
220 throw cvt_error("cvt_reader::get_buf fail, read size cannot be zero");
221 if (to_max > m_buffer.size())
222 throw cvt_error("cvt_reader::get_buf fail, read size too large");
223
224 const auto rollback_size = m_end_pos - m_cur_pos;
225 if (to_max <= rollback_size)
226 {
227 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, to_max};
228 m_cur_pos += to_max;
229 if constexpr (!Saturate)
230 return res;
231 else
232 return res.first;
233 }
234
235 if (to_max > m_buffer.size() - m_cur_pos)
236 {
237 std::copy(m_buffer.data() + m_cur_pos, m_buffer.data() + m_end_pos, m_buffer.data());
238 m_cur_pos = 0;
239 m_end_pos = rollback_size;
240 }
241 const std::size_t aim_size = to_max - rollback_size;
242 auto read_size = m_kernel.get(m_buffer.data() + m_end_pos, aim_size);
243 m_end_pos += read_size;
244
245 if constexpr (Saturate)
246 {
247 while (read_size < aim_size)
248 {
249 const std::size_t new_aim_size = aim_size - read_size;
250 const auto new_read_size = m_kernel.get(m_buffer.data() + m_end_pos, new_aim_size);
251 if (new_read_size == 0)
252 throw cvt_error("get_buf<Saturate> fail: meet eos");
253 m_end_pos += new_read_size;
254 read_size += new_read_size;
255 }
256 }
257
258 if constexpr (!Saturate)
259 {
260 std::pair<const char_type*, std::size_t> res{m_buffer.data() + m_cur_pos, rollback_size + read_size};
261 m_cur_pos = m_end_pos;
262 return res;
263 }
264 else
265 {
266 const char_type* res = m_buffer.data() + m_cur_pos;
267 m_cur_pos = m_end_pos;
268 return res;
269 }
270 }

1492 template <typename Self>
1493 std::size_t get(this Self& self, internal_type* to, std::size_t to_max)
1494 requires requires(Self& t, cvt_reader<KernelType>& r, internal_type* data, std::size_t len) {
1495 { t.get_main(r, data, len) } -> std::same_as<std::size_t>;
1496 }
1497 {
1498 // Read failures do not taint: they only advance the kernel cursor
1499 // and possibly leave a per-call mbstate undefined; the underlying
1500 // stream content is unchanged, and callers can reseek to recover.
1501 // We still honour any pre-existing taint so that a writer that
1502 // corrupted the stream cannot then read back a confused image.
1503 self.assert_not_tainted();
1504
1505 cvt_reader<KernelType> reader(self.m_kernel, self.m_tmp_io_buffer, self.m_rd_cur, self.m_rd_end);
1506
1507 if (!self.m_is_bos_done)
1508 {
1509 if (to_max == 0) return 0;
1510
1511 constexpr std::size_t ext_size = sizeof(external_type);
1512
1513 reader.reset(s_bos_chunk);
1514 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1515 char* dest_bytes = reinterpret_cast<char*>(to);
1516 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1517 const char* dest_bytes_end = reinterpret_cast<const char*>(to + to_max);
1518
1519 // BOS phase requires complete data: if the stream ends unexpectedly
1520 // before providing enough bytes, this is a format error and we throw.
1521 // Unlike get_main() which may return partial data, BOS is all-or-nothing.
1522
1523 // Read full external_type chunks
1524 while (dest_bytes + ext_size <= dest_bytes_end)
1525 {
1526 const std::size_t units_to_read_now = std::min((std::size_t)((dest_bytes_end - dest_bytes) / ext_size), s_bos_chunk);
1527
1528 const auto* src_buf = reader.template get_buf<true>(units_to_read_now);
1529 std::memcpy(dest_bytes, src_buf, units_to_read_now * ext_size);
1530 dest_bytes += units_to_read_now * ext_size;
1531 }
1532
1533 // Read the final partial internal_type if needed
1534 const std::size_t final_remainder_bytes = dest_bytes_end - dest_bytes;
1535 if (final_remainder_bytes > 0)
1536 {
1537 const auto* src_buf = reader.template get_buf<true>(1);
1538 std::memcpy(dest_bytes, src_buf, final_remainder_bytes);
1539 dest_bytes += final_remainder_bytes;
1540 }
1541
1542 return to_max;
1543 }
1544 else
1545 {
1546 if (self.m_io_status != io_status::input)
1547 {
1548 if constexpr (cvt_cpt::support_io_switch<Self>)
1549 self.switch_to_get();
1550 else
1551 throw cvt_error("abs_cvt::get fail: cannot switch to input mode");
1552 }
1553 return self.get_main(reader, to, to_max);
1554 }
1555 }

1602 template <typename Self>
1603 void put(this Self& self, const internal_type* to, std::size_t to_size)
1604 requires requires(Self& t, cvt_writer<KernelType>& w, const internal_type* data, std::size_t len) {
1605 { t.put_main(w, data, len) } -> std::same_as<void>;
1606 }
1607 {
1608 self.assert_not_tainted();
1609
1610 // IO-direction switching does not write bytes to the output stream:
1611 // a failed switch_to_put() leaves the stream unchanged, so the
1612 // converter is not corrupted and must not be tainted. Perform the
1613 // switch BEFORE entering the taint-on-exception region so that only
1614 // actual kernel writes can cause tainting.
1615 if (self.m_is_bos_done && self.m_io_status != io_status::output)
1616 {
1617 if constexpr (cvt_cpt::support_io_switch<Self>)
1618 self.switch_to_put();
1619 else
1620 throw cvt_error("abs_cvt::put fail: cannot switch to output mode");
1621 }
1622
1623 try
1624 {
1625 cvt_writer<KernelType> writer(self.m_kernel, self.m_tmp_io_buffer);
1626 if (!self.m_is_bos_done)
1627 {
1628 if (to_size == 0) return;
1629
1630 constexpr std::size_t ext_size = sizeof(external_type);
1631
1632 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1633 auto to_bytes = reinterpret_cast<const char*>(to);
1634 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1635 auto to_bytes_end = reinterpret_cast<const char*>(to + to_size);
1636
1637 // BOS phase writes complete external_type units. If the input data
1638 // is not a multiple of ext_size, pad the final unit with zeros.
1639
1640 writer.reset(s_bos_chunk);
1641 while (to_bytes + ext_size <= to_bytes_end)
1642 {
1643 auto dest_count = std::min<std::size_t>((to_bytes_end - to_bytes) / ext_size, s_bos_chunk);
1644 auto ptr = writer.put_buf(dest_count);
1645 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1646 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, dest_count * ext_size);
1647 to_bytes += dest_count * ext_size;
1648 }
1649
1650 if (to_bytes < to_bytes_end)
1651 {
1652 // The while loop above guarantees remaining < ext_size.
1653 // The modulo is logically redundant but helps the compiler's
1654 // static analyzer prove the bound for memset size calculation.
1655 std::size_t remaining = static_cast<std::size_t>(to_bytes_end - to_bytes) % ext_size;
1656 auto ptr = writer.put_buf(1);
1657 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1658 std::memcpy(reinterpret_cast<char*>(ptr), to_bytes, remaining);
1659 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast)
1660 std::memset(reinterpret_cast<char*>(ptr) + remaining, 0, ext_size - remaining);
1661 }
1662
1663 writer.commit();
1664 }
1665 else
1666 {
1667 // switch_to_put() was already called above if needed;
1668 // m_io_status is guaranteed to be output at this point.
1669 self.put_main(writer, to, to_size);
1670 // commit() is the sole responsibility of abs_cvt::put: derived
1671 // put_main implementations must NOT call writer.commit() themselves.
1672 // If put_main throws, this commit is skipped and any uncommitted
1673 // buffer contents are dropped at writer destruction; the catch
1674 // below taints the converter.
1675 writer.commit();
1676 }
1677 }
1678 catch (...)
1679 {
1680 self.m_is_tainted = true;
1681 throw;
1682 }
1683 }
