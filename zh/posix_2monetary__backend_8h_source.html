74// The monetary data of the calling thread's current locale. Call it under a clocale_user.
75inline monetary_raw_punct monetary_read_punct()
76{
77 const auto str = [](const char* s) { return s ? std::string(s) : std::string(); };
78 const lconv* lc = std::localeconv();
79 monetary_raw_punct res;
80 res.int_curr_symbol = str(lc->int_curr_symbol);
81 res.currency_symbol = str(lc->currency_symbol);
82 res.mon_decimal_point = str(lc->mon_decimal_point);
83 res.mon_thousands_sep = str(lc->mon_thousands_sep);
84 res.mon_grouping = str(lc->mon_grouping);
85 res.positive_sign = str(lc->positive_sign);
86 res.negative_sign = str(lc->negative_sign);
87 res.int_frac_digits = lc->int_frac_digits;
88 res.frac_digits = lc->frac_digits;
89 res.nat_pos = {.cs_precedes = lc->p_cs_precedes, .sep_by_space = lc->p_sep_by_space,
90 .sign_posn = lc->p_sign_posn};
91 res.nat_neg = {.cs_precedes = lc->n_cs_precedes, .sep_by_space = lc->n_sep_by_space,
92 .sign_posn = lc->n_sign_posn};
93 res.int_pos = {.cs_precedes = lc->int_p_cs_precedes, .sep_by_space = lc->int_p_sep_by_space,
94 .sign_posn = lc->int_p_sign_posn};
95 res.int_neg = {.cs_precedes = lc->int_n_cs_precedes, .sep_by_space = lc->int_n_sep_by_space,
96 .sign_posn = lc->int_n_sign_posn};
97 return res;
98}
