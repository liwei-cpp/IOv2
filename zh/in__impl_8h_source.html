222 bool sync_with_stdio(bool sync = true)
223 {
224 // Nothing to switch: return before the lock, which a blocked read may hold for long.
225 if (m_sync_with_stdio.load() == sync)
226 return sync;
227
228 std::lock_guard guard(this->io_mutex());
229 auto old_sync_state = m_sync_with_stdio.load();
230 if (old_sync_state == sync)
231 return old_sync_state;
232
233 // Only the root's flag flips: buffers, decoder and device all stay. A tainted wide
234 // converter recovers first, and that can fail -- before the flag is touched.
235 // m_channel's adjust, not this stream's: that one comes back here.
236 try {
237 m_channel.adjust(stdin_sync{sync});
238 } catch (...) {
239 this->handle_exception(std::current_exception());
240 return old_sync_state;
241 }
242 m_sync_with_stdio.store(sync);
243 return old_sync_state;
244 }
