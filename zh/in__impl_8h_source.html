407 bool sync_with_stdio(bool sync = true)
408 {
409 // Nothing to switch: return before the lock, which a blocked read may hold for long.
410 if (detail::stdin_shared_imp.synced() == sync)
411 return sync;
412
413 std::lock_guard guard(this->io_mutex());
414 const bool old_sync_state = detail::stdin_shared_imp.synced();
415 if (old_sync_state == sync)
416 return old_sync_state;
417
418 // Only the shared flag flips, which the root does at the end of the adjust, for every
419 // standard input stream: buffers, decoder and device all stay. A tainted wide
420 // converter recovers first, and that can fail -- before the flag is touched.
421 // m_channel's adjust, not this stream's: that one comes back here.
422 try {
423 m_channel.adjust(stdin_sync{sync});
424 } catch (...) {
425 this->handle_exception(std::current_exception());
426 }
427 return old_sync_state;
428 }

756 std::string switch_code(const std::string& new_code)
757 requires std::is_same_v<TChar, wchar_t>
758 {
759 // One lock over the query and the switch, so concurrent switches each return the
760 // code the one before them set.
761 std::lock_guard guard(this->io_mutex());
762 auto res = code();
763 if (res != new_code)
764 {
765 code_cvt_switch acc(new_code);
766 this->adjust(acc);
767 }
768 return res;
769 }
