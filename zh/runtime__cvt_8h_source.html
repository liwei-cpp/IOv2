414 void switch_to_get() override
415 {
416 if (m_io_status == io_status::input) return;
417 if constexpr (!cvt_cpt::support_io_switch<KernelType>)
418 throw cvt_error("runtime_cvt fail: kernel does not support I/O switch");
419 else
420 {
421 m_kernel.switch_to_get();
422 m_io_status = io_status::input;
423 }
424 }

437 void switch_to_put() override
438 {

445 m_io_status = io_status::output;
446 }
447 }

451 io_status m_io_status;
