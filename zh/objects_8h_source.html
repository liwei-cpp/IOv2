245 // Each reason is recovered by rethrowing the pointer and catching it -- which leaves the
246 // pointer intact, so `failed()` still hands out every exception afterwards.
247 static std::string describe(const failures& f)
248 {
249 std::string msg = "IOv2::sync_with_stdio failed for";
250 bool first = true;
251
252 const auto add = [&](const char* name, const std::exception_ptr& err)
253 {
254 if (!err) return;
255 msg += first ? " " : "; ";
256 first = false;
257 msg += name;
258 msg += ": ";
259 try { std::rethrow_exception(err); }
260 catch (const std::exception& e) { msg += e.what(); }
261 catch (...) { msg += "unknown exception"; }
262 };
263
264 add("cout", f.cout_err); add("cerr", f.cerr_err); add("clog", f.clog_err);
265 add("wcout", f.wcout_err); add("wcerr", f.wcerr_err); add("wclog", f.wclog_err);
266 add("cin", f.cin_err); add("wcin", f.wcin_err);
267 return msg;
268 }

326inline void sync_with_stdio(bool sync = true)
327{
328 sync_error::failures failed;
329
330 // Every stream is attempted, so one that throws cannot keep the others from
331 // switching. Collecting costs nothing that can fail: assigning an exception_ptr
332 // is noexcept, and the table is a local.
333 const auto one = [sync](auto& stream, std::exception_ptr& slot) noexcept
334 {
335 try { stream.sync_with_stdio(sync); }
336 catch (...) { slot = std::current_exception(); }
337 };
338
339 one(cout, failed.cout_err);
340 one(cerr, failed.cerr_err);
341 one(clog, failed.clog_err);
342
343 one(wcout, failed.wcout_err);
344 one(wcerr, failed.wcerr_err);
345 one(wclog, failed.wclog_err);
346
347 one(cin, failed.cin_err);
348 one(wcin, failed.wcin_err);
349
350 if (failed.any()) throw sync_error(std::move(failed));
351}
352}
