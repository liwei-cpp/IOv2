173 // Each reason is recovered by rethrowing the pointer and catching it -- which leaves the
174 // pointer intact, so `failed()` still hands out every exception afterwards.
175 static std::string describe(const failures& f)
176 {
177 std::string msg = "IOv2::sync_with_stdio failed for";
178 bool first = true;
179
180 const auto add = [&](const char* name, const std::exception_ptr& err)
181 {
182 if (!err) return;
183 msg += first ? " " : "; ";
184 first = false;
185 msg += name;
186 msg += ": ";
187 try { std::rethrow_exception(err); }
188 catch (const std::exception& e) { msg += e.what(); }
189 catch (...) { msg += "unknown exception"; }
190 };
191
192 add("cout", f.cout_err); add("cerr", f.cerr_err); add("clog", f.clog_err);
193 add("wcout", f.wcout_err); add("wcerr", f.wcerr_err); add("wclog", f.wclog_err);
194 add("cin", f.cin_err); add("wcin", f.wcin_err);
195 return msg;
196 }

256inline void sync_with_stdio(bool sync = true)
257{
258 sync_error::failures failed;
259
260 // Every stream is attempted, so one that throws cannot keep the others from
261 // switching. Collecting costs nothing that can fail: assigning an exception_ptr
262 // is noexcept, and the table is a local.
263 const auto one = [sync](auto& stream, std::exception_ptr& slot) noexcept
264 {
265 try { stream.sync_with_stdio(sync); }
266 catch (...) { slot = std::current_exception(); }
267 };
268
269 one(cout, failed.cout_err);
270 one(cerr, failed.cerr_err);
271 one(clog, failed.clog_err);
272
273 one(wcout, failed.wcout_err);
274 one(wcerr, failed.wcerr_err);
275 one(wclog, failed.wclog_err);
276
277 one(cin, failed.cin_err);
278 one(wcin, failed.wcin_err);
279
280 if (failed.any()) throw sync_error(std::move(failed));
281}
282}
