102 template <typename TIter>
103 requires (char_sink_for<TIter, TChar>)
104 static TIter swrite(TIter s, ios_base<TChar>& io, const locale<TChar>& loc, TValue value)
105 {
106 // Extended floating-point types do not promote, and the facet's %g path is varargs;
107 // relay through the smallest standard type that holds them exactly. See the class @note.
108 using TFacet = std::conditional_t<
109 std::is_floating_point_v<TValue> && !std::is_same_v<TValue, float>
110 && !std::is_same_v<TValue, double> && !std::is_same_v<TValue, long double>,
111 std::conditional_t<(std::numeric_limits<double>::digits >= std::numeric_limits<TValue>::digits
112 && std::numeric_limits<double>::max_exponent
113 >= std::numeric_limits<TValue>::max_exponent
114 && std::numeric_limits<double>::min_exponent
115 <= std::numeric_limits<TValue>::min_exponent),
116 double, long double>,
117 TValue>;
118
119 auto mp = loc.template get<numeric<TChar>>();
120 if (!mp)
121 {
122 io.width(0);
123 throw stream_error("cannot get numeric facet");
124 }
125
126 return mp->put(s, io, static_cast<TFacet>(value));
127 }

162 template <typename TIter, std::sentinel_for<TIter> TSent>
163 requires (std::is_same_v<TChar, typename TIter::value_type>)
164 static TIter sread(TIter s, TSent s_end, ios_base<TChar>& io, const locale<TChar>& loc, TValue& value)
165 requires (!std::is_same_v<TValue, signed char>
166 && !std::is_same_v<TValue, unsigned char>)
167 {
168 // Same relay as swrite: parse into the standard type, then narrow back explicitly.
169 // Without it the implicit narrowing happens inside the facet and trips -Wnarrowing.
170 using TFacet = std::conditional_t<
171 std::is_floating_point_v<TValue> && !std::is_same_v<TValue, float>
172 && !std::is_same_v<TValue, double> && !std::is_same_v<TValue, long double>,
173 std::conditional_t<(std::numeric_limits<double>::digits >= std::numeric_limits<TValue>::digits
174 && std::numeric_limits<double>::max_exponent
175 >= std::numeric_limits<TValue>::max_exponent
176 && std::numeric_limits<double>::min_exponent
177 <= std::numeric_limits<TValue>::min_exponent),
178 double, long double>,
179 TValue>;
180
181 auto mp = loc.template get<numeric<TChar>>();
182 if (!mp)
183 throw stream_error("cannot get numeric facet");
184
185 if constexpr (std::is_same_v<TFacet, TValue>)
186 return mp->get(s, s_end, io, value);
187 else
188 {
189 TFacet tmp{};
190 TIter ret = mp->get(s, s_end, io, tmp);
191 value = static_cast<TValue>(tmp);
192 return ret;
193 }
194 }
195};
