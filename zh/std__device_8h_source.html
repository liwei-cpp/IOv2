249 [[nodiscard]] bool deof()
250 requires (ID == detail::std_in_fd)
251 {
252 if (m_eof_hit) return true;
253
254 if (m_c.has_value()) return false;
255 char_type ch = 0;
256 const std::size_t ret = do_get(&ch, 1);
257 if (ret != 0)

302 std::size_t dget(char* s, std::size_t n)
303 requires (ID == detail::std_in_fd)
304 {
305 if (n == 0 || m_eof_hit) return 0;
306 if (s == nullptr)
307 throw device_error("std_device::dget fail: null buffer");
308
309 if (m_c.has_value())
310 {
311 *s = m_c.value();
312 if (n > 1 && detail::std_fd_readable_now(ID))
313 {
314 // Top up with what is already there, so a probe followed by a bulk
315 // read still reads in bulk. Only when data is ready: waiting here
316 // would block on a pipe whose peer sent that one byte, or on a
317 // terminal where the user typed an empty line. If do_get throws,
318 // m_c must still hold the byte so the next dget() redelivers it.
319 const std::size_t got = do_get(s + 1, n - 1);
320 m_c.reset();
321 return 1 + got;
322 }
323 m_c.reset();
324 return 1;
325 }
326 return do_get(s, n);
327 }

349 void dput(const char* ch, std::size_t n)
350 requires ((ID == detail::std_out_fd) || (ID == detail::std_err_fd))
351 {
352 if (n == 0) return;
353 if (ch == nullptr)
354 throw device_error("std_device::dput fail: null buffer");
355
356 std::size_t written = 0;
357 if constexpr (ID == detail::std_out_fd)
358 written = std::fwrite(ch, sizeof(char), n, stdout);
359 else
360 written = std::fwrite(ch, sizeof(char), n, stderr);
361
362 if (written != n)
363 throw dput_error("std_device::dput fail: partial write", written);
364 }

377 void dflush()
378 requires ((ID == detail::std_out_fd) || (ID == detail::std_err_fd))
379 {
380 bool flush_res = false;
381 if constexpr (ID == detail::std_out_fd)
382 flush_res = (std::fflush(stdout) != EOF);
383 else
384 flush_res = (std::fflush(stderr) != EOF);
385
386 if (!flush_res)
387 throw device_error("std_device::dflush fail: fflush error");
388 }
389
390private:
391 std::size_t do_get(char* s, std::size_t n)
392 {
393 const std::size_t ret = detail::std_fd_read(ID, s, n);
394 if (ret == 0)
395 m_eof_hit = true;
396 return ret;
397 }
398private:
399 IOV2_NO_UNIQUE_ADDRESS std::conditional_t<ID == detail::std_in_fd, bool, std::monostate> m_eof_hit{};
400 IOV2_NO_UNIQUE_ADDRESS std::conditional_t<ID == detail::std_in_fd, std::optional<char_type>, std::monostate> m_c;
401 bool m_active = true;
402};
