378 // Builds the trie s_timezone_tree is bound to, at startup: header-only right after
379 // this class, in shared mode in iov2_objects.cpp. The walk catches its own exceptions.
380 static prefix_tree<char, zone_ref> build_timezone_tree()
381 {
382 prefix_tree<char, zone_ref> res;
383
384 // Registered ahead of the tzdb walk so that it is present even when the walk
385 // throws and leaves the rest of the trie empty: the token the put side writes
386 // for a nameless zone has to parse back regardless of the database. Its text is
387 // empty on purpose -- see s_unknown_zone for why that is not the same as absent.
388 res.add(s_unknown_zone.begin(), s_unknown_zone.end(),
389 zone_ref{.text = std::string{}, .is_name = false, .is_abbrev = true});
390
391 constexpr std::string_view utc = "UTC";
392 bool has_utc = false;
393#if IOV2_HAS_CHRONO_TZDB
394 try
395 {
396 const auto& tzdb = std::chrono::get_tzdb();
397
398 // Classify every key before touching the trie. Two keys can name the same
399 // string from different sources -- CET, EET, EST, GMT, HST, MST, UTC and WET
400 // are each both a link name and an abbreviation -- and going through a map
401 // merges those into one entry with both flags set, instead of calling add()
402 // twice on one key and having it throw on the mismatched values.
403 std::map<std::string, zone_ref> entries;
404
405 // Every abbreviation each zone passes through.
406 {
407 using namespace std::chrono;
408 const sys_seconds projected =
409 time_point_cast<seconds>(system_clock::now()) + years{10};
410 const sys_seconds floor{sys_days{2038y / January / 1}};
411 const sys_seconds horizon = (projected > floor) ? projected : floor;
412
413 // Not sys_seconds::min(): libstdc++ starts every zone's first period at
414 // year::min(), and an earlier instant on a period with rules throws
415 // (CST6CDT and co. since tzdata 2026d). Nothing is lost by starting here.
416 const sys_seconds start{sys_days{year::min() / January / 1}};
417
418 for (const auto& zone : tzdb.zones)
419 {
420 try
421 {
422 sys_seconds t = start;
423 while (t < horizon)
424 {
425 const sys_info info = zone.get_info(t);
426 if (!info.abbrev.empty()) entries[info.abbrev].is_abbrev = true;
427 if (info.end <= t) break;
428 t = time_point_cast<seconds>(info.end);
429 }
430 }
431 catch (...) {} // NOLINT(bugprone-empty-catch)
432 }
433 }
434
435 // Every identifier locate_zone() accepts. Links belong here just as much as
436 // zones do: a link is a full name, merely not the canonical one, and nothing
437 // downstream needs the canonical form -- locate_zone normalizes on its own.
438 for (const auto& zone : tzdb.zones)
439 entries[std::string{zone.name()}].is_name = true;
440 for (const auto& link : tzdb.links)
441 entries[std::string{link.name()}].is_name = true;
442
443 // The unknown-zone token has to parse back to "no zone at all", so it cannot
444 // share a key with a real one. That it does not is a documented invariant of
445 // s_unknown_zone; if some future tzdata breaks it, refuse to answer rather
446 // than quietly get that one entry wrong. Thrown here -- before a single tzdb
447 // entry reaches the trie, and inside the catch below -- so the trie ends up
448 // holding only the token and every real %Z reports a catchable stream_error.
449 // Letting it escape the static initializer instead would call std::terminate
450 // in every program that includes this header, %Z user or not.
451 if (entries.contains(std::string{s_unknown_zone}))
452 throw io_error(
453 "timeio: the tz database has a zone or abbreviation named '"
454 + std::string{s_unknown_zone} + "', colliding with the unknown-zone token");
455
456 // An ICU abbreviation (Windows) follows ICU's default locale and may not be ASCII;
457 // the trie matches bytes, so such a key would never match a wide stream.
458 const auto is_ascii = [](const std::string& key) {
459 return std::ranges::all_of(key, [](char c) { return static_cast<unsigned char>(c) < 0x80; });
460 };
461 for (const auto& [key, ident] : entries)
462 {
463 if (!is_ascii(key))
464 continue;
465 res.add(key.begin(), key.end(),
466 zone_ref{.text = key, .is_name = ident.is_name, .is_abbrev = ident.is_abbrev});
467 has_utc = has_utc || key == utc;
468 }
469 }
470 catch (...) // NOLINT(bugprone-empty-catch)
471 {
472 // tz database unavailable, malformed, or colliding with the unknown-zone
473 // token at static-init time: degrade to a trie holding nothing but that
474 // token, instead of letting the exception escape this static initializer
475 // (which would call std::terminate). %Z then simply fails to match and
476 // reports a catchable stream_error at parse time, matching the defensive
477 // behaviour of time_zone_parse_helper.
478 }
479#endif
480 // put writes "UTC" for a sys_time's %Z, and that has to read back with or without a
481 // database. Added last, so a database's own UTC entry (a name too) is not refused.
482 if (!has_utc)
483 res.add(utc.begin(), utc.end(),
484 zone_ref{.text = std::string{utc}, .is_name = false, .is_abbrev = true});
485 return res;
486 }
487 };

904template <typename CharT>
905std::optional<std::basic_string<CharT>> timeio_normalize_format(std::basic_string_view<CharT> fmt)
906{
907 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
908 // A wide character beyond ASCII must not be narrowed onto a specifier first.
909 const auto is_supported = [supported](CharT c) {
910 return static_cast<std::uint32_t>(c) < 0x80
911 && supported.find(static_cast<char>(c)) != std::string_view::npos;
912 };
913
914 std::basic_string<CharT> out;
915 out.reserve(fmt.size());
916
917 for (std::size_t i = 0; i < fmt.size(); ++i)
918 {
919 if (fmt[i] != '%') { out += fmt[i]; continue; }
920
921 std::size_t j = i + 1;
922 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
923 || fmt[j] == '^' || fmt[j] == '#'))
924 ++j;
925 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
926
927 CharT modifier{};
928 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
929
930 if (j >= fmt.size()) { out.append(fmt.substr(i)); break; }
931
932 CharT spec = fmt[j];
933 if (spec == 'l') spec = 'I';
934 else if (spec == 'k') spec = 'H';
935 else if (spec == 'P') spec = 'p';
936
937 if (!is_supported(spec)) return std::nullopt;
938
939 out += static_cast<CharT>('%');
940 if (modifier != CharT{}) out += modifier;
941 out += spec;
942 i = j;
943 }
944
945 return out;
946}

949template <typename CharT>
950timeio_text<CharT> timeio_c_text()
951{
952 static constexpr std::array<std::string_view, 7> days =
953 {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"};
954 static constexpr std::array<std::string_view, 12> months =
955 {"January", "February", "March", "April", "May", "June",
956 "July", "August", "September", "October", "November", "December"};
957
958 timeio_text<CharT> res;
959 for (std::size_t i = 0; i < days.size(); ++i)
960 {
961 res.day[i] = timeio_ascii<CharT>(days[i]);
962 res.abbr_day[i] = timeio_ascii<CharT>(days[i].substr(0, 3));
963 }
964 for (std::size_t i = 0; i < months.size(); ++i)
965 {
966 res.month[i] = timeio_ascii<CharT>(months[i]);
967 res.abbr_month[i] = timeio_ascii<CharT>(months[i].substr(0, 3));
968 }
969 res.am = timeio_ascii<CharT>("AM");
970 res.pm = timeio_ascii<CharT>("PM");
971 res.date_format = timeio_ascii<CharT>(timeio_c_date_format);
972 res.era_date_format = res.date_format;
973 res.time_format = timeio_ascii<CharT>(timeio_c_time_format);
974 res.era_time_format = res.time_format;
975 res.date_time_format = timeio_ascii<CharT>(timeio_c_date_time_format);
976 res.era_date_time_format = res.date_time_format;
977 res.am_pm_format = timeio_ascii<CharT>(timeio_c_am_pm_format);
978 return res;

1003template <typename CharT, typename TRaw, typename TConv>
1004timeio_text<CharT> timeio_text_from(const TRaw& raw, TConv conv)
1005{
1006 timeio_text<CharT> res;
1007 for (std::size_t i = 0; i < res.day.size(); ++i)
1008 {
1009 res.day[i] = conv(raw.day[i]);
1010 res.abbr_day[i] = conv(raw.abbr_day[i]);
1011 }
1012 for (std::size_t i = 0; i < res.month.size(); ++i)
1013 {
1014 res.month[i] = conv(raw.month[i]);
1015 res.abbr_month[i] = conv(raw.abbr_month[i]);
1016 }
1017 res.am = conv(raw.am);
1018 res.pm = conv(raw.pm);
1019
1020 const auto era_or = [&conv](const auto& era, const std::basic_string<CharT>& plain) {
1021 return era.empty() ? plain : conv(era);
1022 };
1023 res.date_format = conv(raw.date_format);
1024 res.era_date_format = era_or(raw.era_date_format, res.date_format);
1025 res.time_format = conv(raw.time_format);
1026 res.era_time_format = era_or(raw.era_time_format, res.time_format);
1027 res.date_time_format = conv(raw.date_time_format);
1028 res.era_date_time_format = era_or(raw.era_date_time_format, res.date_time_format);
1029 res.am_pm_format = raw.am_pm_format.empty() ? timeio_ascii<CharT>(timeio_c_am_pm_format)
1030 : conv(raw.am_pm_format);
1031
1032 const auto normalized = [](const std::basic_string<CharT>& fmt, std::basic_string<CharT> fallback) {
1033 return timeio_normalize_format<CharT>(fmt).value_or(std::move(fallback));
1034 };
1035 res.date_format = normalized(res.date_format, timeio_ascii<CharT>(timeio_c_date_format));
1036 res.era_date_format = normalized(res.era_date_format, res.date_format);
1037 res.time_format = normalized(res.time_format, timeio_ascii<CharT>(timeio_c_time_format));
1038 res.era_time_format = normalized(res.era_time_format, res.time_format);
1039 res.date_time_format = normalized(res.date_time_format,
1040 timeio_ascii<CharT>(timeio_c_date_time_format));
1041 res.era_date_time_format = normalized(res.era_date_time_format, res.date_time_format);
1042 res.am_pm_format = normalized(res.am_pm_format, timeio_ascii<CharT>(timeio_c_am_pm_format));
1043 return res;
1044}

1494 {

1512 return std::basic_string<CharT>(src.begin(), src.end());
1513 };
1514 take_text(detail::timeio_text_from<CharT>(*raw.wide, same));
1515 }
1516 else
1517 take_text(detail::timeio_text_from<CharT>(raw, convert));
1518
