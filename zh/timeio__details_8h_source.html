289 inline static const prefix_tree<CharT, std::string> s_timezone_tree =
290 []()
291 {
292 prefix_tree<CharT, std::string> res;
293
294 try
295 {
296 const auto& tzdb = std::chrono::get_tzdb();
297
298 // Collect all abbreviations (deduplicated across zones).
299 std::set<std::string> abbrevs;
300 for (const auto& zone : tzdb.zones)
301 {
302 std::string abbr = zone.get_info(std::chrono::sys_time<std::chrono::seconds>{}).abbrev;
303 if (!abbr.empty())
304 abbrevs.insert(std::move(abbr));
305 }
306
307 // Resolve each abbreviation: if locate_zone succeeds (e.g. "EST"
308 // is an IANA link), store the canonical name; otherwise mark "*".
309 for (const auto& abbr : abbrevs)
310 {
311 try
312 {
313 std::string canonical{std::chrono::locate_zone(abbr)->name()};
314 res.add(abbr.begin(), abbr.end(), canonical);
315 }
316 catch (...)
317 {
318 res.add(abbr.begin(), abbr.end(), "*" + abbr);
319 }
320 }
321
322 // Add full zone names that differ from their own abbreviation.
323 for (const auto& zone : tzdb.zones)
324 {
325 std::string full_name{zone.name()};
326 std::string abbr_name = zone.get_info(std::chrono::sys_time<std::chrono::seconds>{}).abbrev;
327 if (full_name != abbr_name)
328 res.add(full_name.begin(), full_name.end(), full_name);
329 }
330 }
331 catch (...) // NOLINT(bugprone-empty-catch)
332 {
333 // tz database unavailable or malformed at static-init time:
334 // degrade to an empty/partial tree instead of letting the
335 // exception escape this static initializer (which would call
336 // std::terminate). %Z then simply fails to match and reports a
337 // catchable stream_error at parse time, matching the defensive
338 // behaviour of time_zone_parse_helper.
339 }
340 return res;
341 }();
342 };

962 static std::optional<std::string> normalize_time_format(const std::string& fmt)
963 {
964 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
965
966 std::string out;
967 out.reserve(fmt.size());
968
969 for (std::size_t i = 0; i < fmt.size(); ++i)
970 {
971 if (fmt[i] != '%') { out += fmt[i]; continue; }
972
973 std::size_t j = i + 1;
974 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
975 || fmt[j] == '^' || fmt[j] == '#'))
976 ++j;
977 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
978
979 char modifier = '\0';
980 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
981
982 if (j >= fmt.size()) { out.append(fmt, i, std::string::npos); break; }
983
984 char spec = fmt[j];
985 if (spec == 'l') spec = 'I';
986 else if (spec == 'k') spec = 'H';
987 else if (spec == 'P') spec = 'p';
988
989 if (supported.find(spec) == std::string_view::npos) return std::nullopt;
990
991 out += '%';
992 if (modifier != '\0') out += modifier;
993 out += spec;
994 i = j;
995 }
996
997 return out;
998 }

1096 static std::vector<era_entry> parse_glibc_era_entries()
1097 {
1098 std::vector<era_entry> items;
1099
1100 const int32_t era_item_num = static_cast<int32_t>(
1101 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
1102 if (era_item_num <= 0)
1103 return items;
1104
1105 items.reserve(era_item_num);
1106 const char *ptr = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
1107 for (int32_t cnt = 0; cnt < era_item_num; ++cnt)
1108 {
1109 const char *base_ptr = ptr;
1110 era_entry cur_entry;
1111
1112 int32_t buf[8];
1113 std::memcpy(static_cast<void*>(buf), static_cast<const void*>(ptr), sizeof(int32_t) * 8);
1114 ptr += sizeof(uint32_t) * 8;
1115
1116 if (buf[2] > std::numeric_limits<int32_t>::max() - 1900)
1117 cur_entry.from_year = std::numeric_limits<int32_t>::max();
1118 else
1119 cur_entry.from_year = buf[2] + 1900;
1120 cur_entry.from_month = buf[3] + 1;
1121 cur_entry.from_day = buf[4];
1122 if (buf[5] > std::numeric_limits<int32_t>::max() - 1900)
1123 {
1124 cur_entry.to_year = std::numeric_limits<int32_t>::max();
1125 cur_entry.to_month = 12;
1126 cur_entry.to_day = 31;
1127 }
1128 else
1129 {
1130 cur_entry.to_year = buf[5] + 1900;
1131 cur_entry.to_month = buf[6] + 1;
1132 cur_entry.to_day = buf[7];
1133 }
1134
1135 // Normalise direction to match glibc's absolute_direction (era.c ~line 98).
1136 // Forward era (from <= to): marker maps directly.
1137 // Backward era (from > to): marker is flipped so that the linear formula
1138 // era_year = offset + (calendar - from_year) * direction
1139 // stays consistent; see OUTPUT INVARIANTS above for the full rationale.
1140 if (TimeioHelper::era_small_or_equal(cur_entry.from_year, cur_entry.from_month, cur_entry.from_day,
1141 cur_entry.to_year, cur_entry.to_month, cur_entry.to_day))
1142 {
1143 if (buf[0] == (uint32_t) '+') cur_entry.direction = 1;
1144 else cur_entry.direction = -1;
1145 }
1146 else
1147 {
1148 if (buf[0] == (uint32_t) '+') cur_entry.direction = -1;
1149 else cur_entry.direction = 1;
1150 }
1151 cur_entry.offset = buf[1];
1152
1153 cur_entry.name = ptr; ptr = strchr(ptr, '\0') + 1;
1154 cur_entry.format = ptr; ptr = strchr(ptr, '\0') + 1;
1155
1156 // skip wchar_t name and format
1157 ptr += 3 - (((ptr - base_ptr) + 3) & 3);
1158 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1159 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1160
1161 items.push_back(std::move(cur_entry));
1162 }
1163 return items;
1164 }
