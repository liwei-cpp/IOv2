361 // The walk below catches its own exceptions: only bad_alloc can escape, at startup,
362 // where nothing could handle it anyway.
363 // NOLINTNEXTLINE(bugprone-throwing-static-initialization)
364 inline static const prefix_tree<char, zone_ref>& s_timezone_tree =
365 *new prefix_tree<char, zone_ref>([]()
366 {
367 prefix_tree<char, zone_ref> res;
368
369 // Registered ahead of the tzdb walk so that it is present even when the walk
370 // throws and leaves the rest of the trie empty: the token the put side writes
371 // for a nameless zone has to parse back regardless of the database. Its text is
372 // empty on purpose -- see s_unknown_zone for why that is not the same as absent.
373 res.add(s_unknown_zone.begin(), s_unknown_zone.end(),
374 zone_ref{.text = std::string{}, .is_name = false, .is_abbrev = true});
375
376#if IOV2_HAS_CHRONO_TZDB
377 try
378 {
379 const auto& tzdb = std::chrono::get_tzdb();
380
381 // Classify every key before touching the trie. Two keys can name the same
382 // string from different sources -- CET, EET, EST, GMT, HST, MST, UTC and WET
383 // are each both a link name and an abbreviation -- and going through a map
384 // merges those into one entry with both flags set, instead of calling add()
385 // twice on one key and having it throw on the mismatched values.
386 std::map<std::string, zone_ref> entries;
387
388 // Every abbreviation each zone passes through.
389 {
390 using namespace std::chrono;
391 const sys_seconds projected =
392 time_point_cast<seconds>(system_clock::now()) + years{10};
393 const sys_seconds floor{sys_days{2038y / January / 1}};
394 const sys_seconds horizon = (projected > floor) ? projected : floor;
395
396 // Not sys_seconds::min(): libstdc++ starts every zone's first period at
397 // year::min(), and an earlier instant on a period with rules throws
398 // (CST6CDT and co. since tzdata 2026d). Nothing is lost by starting here.
399 const sys_seconds start{sys_days{year::min() / January / 1}};
400
401 for (const auto& zone : tzdb.zones)
402 {
403 try
404 {
405 sys_seconds t = start;
406 while (t < horizon)
407 {
408 const sys_info info = zone.get_info(t);
409 if (!info.abbrev.empty()) entries[info.abbrev].is_abbrev = true;
410 if (info.end <= t) break;
411 t = time_point_cast<seconds>(info.end);
412 }
413 }
414 catch (...) {} // NOLINT(bugprone-empty-catch)
415 }
416 }
417
418 // Every identifier locate_zone() accepts. Links belong here just as much as
419 // zones do: a link is a full name, merely not the canonical one, and nothing
420 // downstream needs the canonical form -- locate_zone normalizes on its own.
421 for (const auto& zone : tzdb.zones)
422 entries[std::string{zone.name()}].is_name = true;
423 for (const auto& link : tzdb.links)
424 entries[std::string{link.name()}].is_name = true;
425
426 // The unknown-zone token has to parse back to "no zone at all", so it cannot
427 // share a key with a real one. That it does not is a documented invariant of
428 // s_unknown_zone; if some future tzdata breaks it, refuse to answer rather
429 // than quietly get that one entry wrong. Thrown here -- before a single tzdb
430 // entry reaches the trie, and inside the catch below -- so the trie ends up
431 // holding only the token and every real %Z reports a catchable stream_error.
432 // Letting it escape the static initializer instead would call std::terminate
433 // in every program that includes this header, %Z user or not.
434 if (entries.contains(std::string{s_unknown_zone}))
435 throw io_error(
436 "timeio: the tz database has a zone or abbreviation named '"
437 + std::string{s_unknown_zone} + "', colliding with the unknown-zone token");
438
439 for (const auto& [key, ident] : entries)
440 res.add(key.begin(), key.end(),
441 zone_ref{.text = key, .is_name = ident.is_name, .is_abbrev = ident.is_abbrev});
442 }
443 catch (...) // NOLINT(bugprone-empty-catch)
444 {
445 // tz database unavailable, malformed, or colliding with the unknown-zone
446 // token at static-init time: degrade to a trie holding nothing but that
447 // token, instead of letting the exception escape this static initializer
448 // (which would call std::terminate). %Z then simply fails to match and
449 // reports a catchable stream_error at parse time, matching the defensive
450 // behaviour of time_zone_parse_helper.
451 }
452#else
453 // No database to ask, but put writes "UTC" for a sys_time's %Z, and that has to
454 // read back. With a database UTC is among the keys anyway.
455 constexpr std::string_view utc = "UTC";
456 res.add(utc.begin(), utc.end(),
457 zone_ref{.text = std::string{utc}, .is_name = false, .is_abbrev = true});
458#endif
459 return res;
460 }());
461 };

878template <typename CharT>
879std::optional<std::basic_string<CharT>> timeio_normalize_format(std::basic_string_view<CharT> fmt)
880{
881 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
882 // A wide character beyond ASCII must not be narrowed onto a specifier first.
883 const auto is_supported = [supported](CharT c) {
884 return static_cast<std::uint32_t>(c) < 0x80
885 && supported.find(static_cast<char>(c)) != std::string_view::npos;
886 };
887
888 std::basic_string<CharT> out;
889 out.reserve(fmt.size());
890
891 for (std::size_t i = 0; i < fmt.size(); ++i)
892 {
893 if (fmt[i] != '%') { out += fmt[i]; continue; }
894
895 std::size_t j = i + 1;
896 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
897 || fmt[j] == '^' || fmt[j] == '#'))
898 ++j;
899 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
900
901 CharT modifier{};
902 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
903
904 if (j >= fmt.size()) { out.append(fmt.substr(i)); break; }
905
906 CharT spec = fmt[j];
907 if (spec == 'l') spec = 'I';
908 else if (spec == 'k') spec = 'H';
909 else if (spec == 'P') spec = 'p';
910
911 if (!is_supported(spec)) return std::nullopt;
912
913 out += static_cast<CharT>('%');
914 if (modifier != CharT{}) out += modifier;
915 out += spec;
916 i = j;
917 }
918
919 return out;
920}

923template <typename CharT>
924timeio_text<CharT> timeio_c_text()
925{
926 static constexpr std::array<std::string_view, 7> days =
927 {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"};
928 static constexpr std::array<std::string_view, 12> months =
929 {"January", "February", "March", "April", "May", "June",
930 "July", "August", "September", "October", "November", "December"};
931
932 timeio_text<CharT> res;
933 for (std::size_t i = 0; i < days.size(); ++i)
934 {
935 res.day[i] = timeio_ascii<CharT>(days[i]);
936 res.abbr_day[i] = timeio_ascii<CharT>(days[i].substr(0, 3));
937 }
938 for (std::size_t i = 0; i < months.size(); ++i)
939 {
940 res.month[i] = timeio_ascii<CharT>(months[i]);
941 res.abbr_month[i] = timeio_ascii<CharT>(months[i].substr(0, 3));
942 }
943 res.am = timeio_ascii<CharT>("AM");
944 res.pm = timeio_ascii<CharT>("PM");
945 res.date_format = timeio_ascii<CharT>(timeio_c_date_format);
946 res.era_date_format = res.date_format;
947 res.time_format = timeio_ascii<CharT>(timeio_c_time_format);
948 res.era_time_format = res.time_format;
949 res.date_time_format = timeio_ascii<CharT>(timeio_c_date_time_format);
950 res.era_date_time_format = res.date_time_format;
951 res.am_pm_format = timeio_ascii<CharT>(timeio_c_am_pm_format);
952 return res;

977template <typename CharT, typename TRaw, typename TConv>
978timeio_text<CharT> timeio_text_from(const TRaw& raw, TConv conv)
979{
980 timeio_text<CharT> res;
981 for (std::size_t i = 0; i < res.day.size(); ++i)
982 {
983 res.day[i] = conv(raw.day[i]);
984 res.abbr_day[i] = conv(raw.abbr_day[i]);
985 }
986 for (std::size_t i = 0; i < res.month.size(); ++i)
987 {
988 res.month[i] = conv(raw.month[i]);
989 res.abbr_month[i] = conv(raw.abbr_month[i]);
990 }
991 res.am = conv(raw.am);
992 res.pm = conv(raw.pm);
993
994 const auto era_or = [&conv](const auto& era, const std::basic_string<CharT>& plain) {
995 return era.empty() ? plain : conv(era);
996 };
997 res.date_format = conv(raw.date_format);
998 res.era_date_format = era_or(raw.era_date_format, res.date_format);
999 res.time_format = conv(raw.time_format);
1000 res.era_time_format = era_or(raw.era_time_format, res.time_format);
1001 res.date_time_format = conv(raw.date_time_format);
1002 res.era_date_time_format = era_or(raw.era_date_time_format, res.date_time_format);
1003 res.am_pm_format = raw.am_pm_format.empty() ? timeio_ascii<CharT>(timeio_c_am_pm_format)
1004 : conv(raw.am_pm_format);
1005
1006 const auto normalized = [](const std::basic_string<CharT>& fmt, std::basic_string<CharT> fallback) {
1007 return timeio_normalize_format<CharT>(fmt).value_or(std::move(fallback));
1008 };
1009 res.date_format = normalized(res.date_format, timeio_ascii<CharT>(timeio_c_date_format));
1010 res.era_date_format = normalized(res.era_date_format, res.date_format);
1011 res.time_format = normalized(res.time_format, timeio_ascii<CharT>(timeio_c_time_format));
1012 res.era_time_format = normalized(res.era_time_format, res.time_format);
1013 res.date_time_format = normalized(res.date_time_format,
1014 timeio_ascii<CharT>(timeio_c_date_time_format));
1015 res.era_date_time_format = normalized(res.era_date_time_format, res.date_time_format);
1016 res.am_pm_format = normalized(res.am_pm_format, timeio_ascii<CharT>(timeio_c_am_pm_format));
1017 return res;
1018}

1076template <typename TEntry, typename TConv>
1077TEntry timeio_era_from_raw(const timeio_raw_era& raw, TConv conv)
1078{
1079 TEntry value{};
1080 value.name = conv(raw.name);
1081 value.format = conv(raw.format);
1082 value.from_year = raw.from_year;
1083 value.from_month = raw.from_month;
1084 value.from_day = raw.from_day;
1085 value.to_year = raw.to_year;
1086 value.to_month = raw.to_month;
1087 value.to_day = raw.to_day;
1088 value.offset = raw.offset;
1089
1090 const int8_t marked_direction = raw.direction == '+' ? int8_t{1} : int8_t{-1};
1091 const bool runs_forward = TimeioHelper::era_small_or_equal(
1092 value.from_year, value.from_month, value.from_day,
1093 value.to_year, value.to_month, value.to_day);
1094 value.direction = runs_forward ? marked_direction : static_cast<int8_t>(-marked_direction);
1095 return value;
1096}
