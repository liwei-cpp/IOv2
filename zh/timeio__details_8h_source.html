368 // TODO: this is one trie per character type, so the tzdb walk below is repeated for
369 // every CharT the program instantiates. The walk and the locate_zone resolution are
370 // both CharT-independent; hoisting them into a single shared collection and leaving
371 // only the trie insertion per CharT roughly halves the cost for a program that uses
372 // several character types (measured at -O2: 33.8 ms -> 17 ms for all four).
373 inline static const prefix_tree<CharT, std::string> s_timezone_tree =
374 []()
375 {
376 prefix_tree<CharT, std::string> res;
377
378 // Registered ahead of the tzdb walk so that it is present even when the walk
379 // throws and leaves the rest of the trie empty: the token the put side writes
380 // for a nameless zone has to parse back regardless of the database.
381 res.add(s_unknown_zone.begin(), s_unknown_zone.end(), std::string{});
382
383 try
384 {
385 const auto& tzdb = std::chrono::get_tzdb();
386
387 // Collect all abbreviations (deduplicated across zones).
388 std::set<std::string> abbrevs;
389 {
390 using namespace std::chrono;
391 const sys_seconds projected =
392 time_point_cast<seconds>(system_clock::now()) + years{10};
393 const sys_seconds floor{sys_days{2038y / January / 1}};
394 const sys_seconds horizon = (projected > floor) ? projected : floor;
395
396 for (const auto& zone : tzdb.zones)
397 {
398 sys_seconds t = sys_seconds::min();
399 while (t < horizon)
400 {
401 const sys_info info = zone.get_info(t);
402 if (!info.abbrev.empty()) abbrevs.insert(info.abbrev);
403 if (info.end <= t) break;
404 t = time_point_cast<seconds>(info.end);
405 }
406 }
407 }
408
409 // Resolve each abbreviation: if locate_zone succeeds (e.g. "EST"
410 // is an IANA link), store the canonical name; otherwise mark "*".
411 for (const auto& abbr : abbrevs)
412 {
413 try
414 {
415 std::string canonical{std::chrono::locate_zone(abbr)->name()};
416 res.add(abbr.begin(), abbr.end(), canonical);
417 }
418 catch (...)
419 {
420 res.add(abbr.begin(), abbr.end(), "*" + abbr);
421 }
422 }
423
424 // Add full zone names that differ from their own abbreviation.
425 for (const auto& zone : tzdb.zones)
426 {
427 std::string full_name{zone.name()};
428 std::string abbr_name = zone.get_info(std::chrono::sys_time<std::chrono::seconds>{}).abbrev;
429 if (full_name != abbr_name)
430 res.add(full_name.begin(), full_name.end(), full_name);
431 }
432 }
433 catch (...) // NOLINT(bugprone-empty-catch)
434 {
435 // tz database unavailable or malformed at static-init time:
436 // degrade to an empty/partial tree instead of letting the
437 // exception escape this static initializer (which would call
438 // std::terminate). %Z then simply fails to match and reports a
439 // catchable stream_error at parse time, matching the defensive
440 // behaviour of time_zone_parse_helper.
441 }
442 return res;
443 }();
444 };

1011 static std::optional<std::string> normalize_time_format(const std::string& fmt)
1012 {
1013 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
1014
1015 std::string out;
1016 out.reserve(fmt.size());
1017
1018 for (std::size_t i = 0; i < fmt.size(); ++i)
1019 {
1020 if (fmt[i] != '%') { out += fmt[i]; continue; }
1021
1022 std::size_t j = i + 1;
1023 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
1024 || fmt[j] == '^' || fmt[j] == '#'))
1025 ++j;
1026 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
1027
1028 char modifier = '\0';
1029 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
1030
1031 if (j >= fmt.size()) { out.append(fmt, i, std::string::npos); break; }
1032
1033 char spec = fmt[j];
1034 if (spec == 'l') spec = 'I';
1035 else if (spec == 'k') spec = 'H';
1036 else if (spec == 'P') spec = 'p';
1037
1038 if (supported.find(spec) == std::string_view::npos) return std::nullopt;
1039
1040 out += '%';
1041 if (modifier != '\0') out += modifier;
1042 out += spec;
1043 i = j;
1044 }
1045
1046 return out;
1047 }

1145 static std::vector<era_entry> parse_glibc_era_entries()
1146 {
1147 std::vector<era_entry> items;
1148
1149 const int32_t era_item_num = static_cast<int32_t>(
1150 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
1151 if (era_item_num <= 0)
1152 return items;
1153
1154 items.reserve(era_item_num);
1155 const char *ptr = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
1156 for (int32_t cnt = 0; cnt < era_item_num; ++cnt)
1157 {
1158 const char *base_ptr = ptr;
1159 era_entry cur_entry;
1160
1161 int32_t buf[8];
1162 std::memcpy(static_cast<void*>(buf), static_cast<const void*>(ptr), sizeof(int32_t) * 8);
1163 ptr += sizeof(uint32_t) * 8;
1164
1165 if (buf[2] > std::numeric_limits<int32_t>::max() - 1900)
1166 cur_entry.from_year = std::numeric_limits<int32_t>::max();
1167 else
1168 cur_entry.from_year = buf[2] + 1900;
1169 cur_entry.from_month = buf[3] + 1;
1170 cur_entry.from_day = buf[4];
1171 if (buf[5] > std::numeric_limits<int32_t>::max() - 1900)
1172 {
1173 cur_entry.to_year = std::numeric_limits<int32_t>::max();
1174 cur_entry.to_month = 12;
1175 cur_entry.to_day = 31;
1176 }
1177 else
1178 {
1179 cur_entry.to_year = buf[5] + 1900;
1180 cur_entry.to_month = buf[6] + 1;
1181 cur_entry.to_day = buf[7];
1182 }
1183
1184 // Normalise direction to match glibc's absolute_direction (era.c ~line 98).
1185 // Forward era (from <= to): marker maps directly.
1186 // Backward era (from > to): marker is flipped so that the linear formula
1187 // era_year = offset + (calendar - from_year) * direction
1188 // stays consistent; see OUTPUT INVARIANTS above for the full rationale.
1189 if (TimeioHelper::era_small_or_equal(cur_entry.from_year, cur_entry.from_month, cur_entry.from_day,
1190 cur_entry.to_year, cur_entry.to_month, cur_entry.to_day))
1191 {
1192 if (buf[0] == (uint32_t) '+') cur_entry.direction = 1;
1193 else cur_entry.direction = -1;
1194 }
1195 else
1196 {
1197 if (buf[0] == (uint32_t) '+') cur_entry.direction = -1;
1198 else cur_entry.direction = 1;
1199 }
1200 cur_entry.offset = buf[1];
1201
1202 cur_entry.name = ptr; ptr = strchr(ptr, '\0') + 1;
1203 cur_entry.format = ptr; ptr = strchr(ptr, '\0') + 1;
1204
1205 // skip wchar_t name and format
1206 ptr += 3 - (((ptr - base_ptr) + 3) & 3);
1207 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1208 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1209
1210 items.push_back(std::move(cur_entry));
1211 }
1212 return items;
1213 }
