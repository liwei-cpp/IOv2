334 inline static const prefix_tree<char, zone_ref> s_timezone_tree =
335 []()
336 {
337 prefix_tree<char, zone_ref> res;
338
339 // Registered ahead of the tzdb walk so that it is present even when the walk
340 // throws and leaves the rest of the trie empty: the token the put side writes
341 // for a nameless zone has to parse back regardless of the database. Its text is
342 // empty on purpose -- see s_unknown_zone for why that is not the same as absent.
343 res.add(s_unknown_zone.begin(), s_unknown_zone.end(),
344 zone_ref{std::string{}, false, true});
345
346 try
347 {
348 const auto& tzdb = std::chrono::get_tzdb();
349
350 // Classify every key before touching the trie. Two keys can name the same
351 // string from different sources -- CET, EET, EST, GMT, HST, MST, UTC and WET
352 // are each both a link name and an abbreviation -- and going through a map
353 // merges those into one entry with both flags set, instead of calling add()
354 // twice on one key and having it throw on the mismatched values.
355 std::map<std::string, zone_ref> entries;
356
357 // Every abbreviation each zone passes through.
358 {
359 using namespace std::chrono;
360 const sys_seconds projected =
361 time_point_cast<seconds>(system_clock::now()) + years{10};
362 const sys_seconds floor{sys_days{2038y / January / 1}};
363 const sys_seconds horizon = (projected > floor) ? projected : floor;
364
365 for (const auto& zone : tzdb.zones)
366 {
367 sys_seconds t = sys_seconds::min();
368 while (t < horizon)
369 {
370 const sys_info info = zone.get_info(t);
371 if (!info.abbrev.empty()) entries[info.abbrev].is_abbrev = true;
372 if (info.end <= t) break;
373 t = time_point_cast<seconds>(info.end);
374 }
375 }
376 }
377
378 // Every identifier locate_zone() accepts. Links belong here just as much as
379 // zones do: a link is a full name, merely not the canonical one, and nothing
380 // downstream needs the canonical form -- locate_zone normalizes on its own.
381 for (const auto& zone : tzdb.zones)
382 entries[std::string{zone.name()}].is_name = true;
383 for (const auto& link : tzdb.links)
384 entries[std::string{link.name()}].is_name = true;
385
386 // The unknown-zone token has to parse back to "no zone at all", so it cannot
387 // share a key with a real one. That it does not is a documented invariant of
388 // s_unknown_zone; if some future tzdata breaks it, refuse to answer rather
389 // than quietly get that one entry wrong. Thrown here -- before a single tzdb
390 // entry reaches the trie, and inside the catch below -- so the trie ends up
391 // holding only the token and every real %Z reports a catchable stream_error.
392 // Letting it escape the static initializer instead would call std::terminate
393 // in every program that includes this header, %Z user or not.
394 if (entries.contains(std::string{s_unknown_zone}))
395 throw std::runtime_error(
396 "timeio: the tz database has a zone or abbreviation named '"
397 + std::string{s_unknown_zone} + "', colliding with the unknown-zone token");
398
399 for (const auto& [key, ident] : entries)
400 res.add(key.begin(), key.end(),
401 zone_ref{key, ident.is_name, ident.is_abbrev});
402 }
403 catch (...) // NOLINT(bugprone-empty-catch)
404 {
405 // tz database unavailable, malformed, or colliding with the unknown-zone
406 // token at static-init time: degrade to a trie holding nothing but that
407 // token, instead of letting the exception escape this static initializer
408 // (which would call std::terminate). %Z then simply fails to match and
409 // reports a catchable stream_error at parse time, matching the defensive
410 // behaviour of time_zone_parse_helper.
411 }
412 return res;
413 }();
414 };

1192 static std::optional<std::string> normalize_time_format(const std::string& fmt)
1193 {
1194 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
1195
1196 std::string out;
1197 out.reserve(fmt.size());
1198
1199 for (std::size_t i = 0; i < fmt.size(); ++i)
1200 {
1201 if (fmt[i] != '%') { out += fmt[i]; continue; }
1202
1203 std::size_t j = i + 1;
1204 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
1205 || fmt[j] == '^' || fmt[j] == '#'))
1206 ++j;
1207 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
1208
1209 char modifier = '\0';
1210 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
1211
1212 if (j >= fmt.size()) { out.append(fmt, i, std::string::npos); break; }
1213
1214 char spec = fmt[j];
1215 if (spec == 'l') spec = 'I';
1216 else if (spec == 'k') spec = 'H';
1217 else if (spec == 'P') spec = 'p';
1218
1219 if (supported.find(spec) == std::string_view::npos) return std::nullopt;
1220
1221 out += '%';
1222 if (modifier != '\0') out += modifier;
1223 out += spec;
1224 i = j;
1225 }
1226
1227 return out;
1228 }

1326 static std::vector<era_entry> parse_glibc_era_entries()
1327 {
1328 std::vector<era_entry> items;
1329
1330 const int32_t era_item_num = static_cast<int32_t>(
1331 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
1332 if (era_item_num <= 0)
1333 return items;
1334
1335 items.reserve(era_item_num);
1336 const char *ptr = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
1337 for (int32_t cnt = 0; cnt < era_item_num; ++cnt)
1338 {
1339 const char *base_ptr = ptr;
1340 era_entry cur_entry;
1341
1342 int32_t buf[8];
1343 std::memcpy(static_cast<void*>(buf), static_cast<const void*>(ptr), sizeof(int32_t) * 8);
1344 ptr += sizeof(uint32_t) * 8;
1345
1346 if (buf[2] > std::numeric_limits<int32_t>::max() - 1900)
1347 cur_entry.from_year = std::numeric_limits<int32_t>::max();
1348 else
1349 cur_entry.from_year = buf[2] + 1900;
1350 cur_entry.from_month = buf[3] + 1;
1351 cur_entry.from_day = buf[4];
1352 if (buf[5] > std::numeric_limits<int32_t>::max() - 1900)
1353 {
1354 cur_entry.to_year = std::numeric_limits<int32_t>::max();
1355 cur_entry.to_month = 12;
1356 cur_entry.to_day = 31;
1357 }
1358 else
1359 {
1360 cur_entry.to_year = buf[5] + 1900;
1361 cur_entry.to_month = buf[6] + 1;
1362 cur_entry.to_day = buf[7];
1363 }
1364
1365 // Normalise direction to match glibc's absolute_direction (era.c ~line 98).
1366 // Forward era (from <= to): marker maps directly.
1367 // Backward era (from > to): marker is flipped so that the linear formula
1368 // era_year = offset + (calendar - from_year) * direction
1369 // stays consistent; see OUTPUT INVARIANTS above for the full rationale.
1370 if (TimeioHelper::era_small_or_equal(cur_entry.from_year, cur_entry.from_month, cur_entry.from_day,
1371 cur_entry.to_year, cur_entry.to_month, cur_entry.to_day))
1372 {
1373 if (buf[0] == (uint32_t) '+') cur_entry.direction = 1;
1374 else cur_entry.direction = -1;
1375 }
1376 else
1377 {
1378 if (buf[0] == (uint32_t) '+') cur_entry.direction = -1;
1379 else cur_entry.direction = 1;
1380 }
1381 cur_entry.offset = buf[1];
1382
1383 cur_entry.name = ptr; ptr = strchr(ptr, '\0') + 1;
1384 cur_entry.format = ptr; ptr = strchr(ptr, '\0') + 1;
1385
1386 // skip wchar_t name and format
1387 ptr += 3 - (((ptr - base_ptr) + 3) & 3);
1388 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1389 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1390
1391 items.push_back(std::move(cur_entry));
1392 }
1393 return items;
1394 }
