322 // TODO: this is one trie per character type, so the tzdb walk below is repeated for
323 // every CharT the program instantiates. The walk and the locate_zone resolution are
324 // both CharT-independent; hoisting them into a single shared collection and leaving
325 // only the trie insertion per CharT roughly halves the cost for a program that uses
326 // several character types (measured at -O2: 33.8 ms -> 17 ms for all four).
327 inline static const prefix_tree<CharT, std::string> s_timezone_tree =
328 []()
329 {
330 prefix_tree<CharT, std::string> res;
331
332 try
333 {
334 const auto& tzdb = std::chrono::get_tzdb();
335
336 // Collect all abbreviations (deduplicated across zones).
337 std::set<std::string> abbrevs;
338 {
339 using namespace std::chrono;
340 const sys_seconds projected =
341 time_point_cast<seconds>(system_clock::now()) + years{10};
342 const sys_seconds floor{sys_days{2038y / January / 1}};
343 const sys_seconds horizon = (projected > floor) ? projected : floor;
344
345 for (const auto& zone : tzdb.zones)
346 {
347 sys_seconds t = sys_seconds::min();
348 while (t < horizon)
349 {
350 const sys_info info = zone.get_info(t);
351 if (!info.abbrev.empty()) abbrevs.insert(info.abbrev);
352 if (info.end <= t) break;
353 t = time_point_cast<seconds>(info.end);
354 }
355 }
356 }
357
358 // Resolve each abbreviation: if locate_zone succeeds (e.g. "EST"
359 // is an IANA link), store the canonical name; otherwise mark "*".
360 for (const auto& abbr : abbrevs)
361 {
362 try
363 {
364 std::string canonical{std::chrono::locate_zone(abbr)->name()};
365 res.add(abbr.begin(), abbr.end(), canonical);
366 }
367 catch (...)
368 {
369 res.add(abbr.begin(), abbr.end(), "*" + abbr);
370 }
371 }
372
373 // Add full zone names that differ from their own abbreviation.
374 for (const auto& zone : tzdb.zones)
375 {
376 std::string full_name{zone.name()};
377 std::string abbr_name = zone.get_info(std::chrono::sys_time<std::chrono::seconds>{}).abbrev;
378 if (full_name != abbr_name)
379 res.add(full_name.begin(), full_name.end(), full_name);
380 }
381 }
382 catch (...) // NOLINT(bugprone-empty-catch)
383 {
384 // tz database unavailable or malformed at static-init time:
385 // degrade to an empty/partial tree instead of letting the
386 // exception escape this static initializer (which would call
387 // std::terminate). %Z then simply fails to match and reports a
388 // catchable stream_error at parse time, matching the defensive
389 // behaviour of time_zone_parse_helper.
390 }
391 return res;
392 }();
393 };

1013 static std::optional<std::string> normalize_time_format(const std::string& fmt)
1014 {
1015 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
1016
1017 std::string out;
1018 out.reserve(fmt.size());
1019
1020 for (std::size_t i = 0; i < fmt.size(); ++i)
1021 {
1022 if (fmt[i] != '%') { out += fmt[i]; continue; }
1023
1024 std::size_t j = i + 1;
1025 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
1026 || fmt[j] == '^' || fmt[j] == '#'))
1027 ++j;
1028 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
1029
1030 char modifier = '\0';
1031 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
1032
1033 if (j >= fmt.size()) { out.append(fmt, i, std::string::npos); break; }
1034
1035 char spec = fmt[j];
1036 if (spec == 'l') spec = 'I';
1037 else if (spec == 'k') spec = 'H';
1038 else if (spec == 'P') spec = 'p';
1039
1040 if (supported.find(spec) == std::string_view::npos) return std::nullopt;
1041
1042 out += '%';
1043 if (modifier != '\0') out += modifier;
1044 out += spec;
1045 i = j;
1046 }
1047
1048 return out;
1049 }

1147 static std::vector<era_entry> parse_glibc_era_entries()
1148 {
1149 std::vector<era_entry> items;
1150
1151 const int32_t era_item_num = static_cast<int32_t>(
1152 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
1153 if (era_item_num <= 0)
1154 return items;
1155
1156 items.reserve(era_item_num);
1157 const char *ptr = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
1158 for (int32_t cnt = 0; cnt < era_item_num; ++cnt)
1159 {
1160 const char *base_ptr = ptr;
1161 era_entry cur_entry;
1162
1163 int32_t buf[8];
1164 std::memcpy(static_cast<void*>(buf), static_cast<const void*>(ptr), sizeof(int32_t) * 8);
1165 ptr += sizeof(uint32_t) * 8;
1166
1167 if (buf[2] > std::numeric_limits<int32_t>::max() - 1900)
1168 cur_entry.from_year = std::numeric_limits<int32_t>::max();
1169 else
1170 cur_entry.from_year = buf[2] + 1900;
1171 cur_entry.from_month = buf[3] + 1;
1172 cur_entry.from_day = buf[4];
1173 if (buf[5] > std::numeric_limits<int32_t>::max() - 1900)
1174 {
1175 cur_entry.to_year = std::numeric_limits<int32_t>::max();
1176 cur_entry.to_month = 12;
1177 cur_entry.to_day = 31;
1178 }
1179 else
1180 {
1181 cur_entry.to_year = buf[5] + 1900;
1182 cur_entry.to_month = buf[6] + 1;
1183 cur_entry.to_day = buf[7];
1184 }
1185
1186 // Normalise direction to match glibc's absolute_direction (era.c ~line 98).
1187 // Forward era (from <= to): marker maps directly.
1188 // Backward era (from > to): marker is flipped so that the linear formula
1189 // era_year = offset + (calendar - from_year) * direction
1190 // stays consistent; see OUTPUT INVARIANTS above for the full rationale.
1191 if (TimeioHelper::era_small_or_equal(cur_entry.from_year, cur_entry.from_month, cur_entry.from_day,
1192 cur_entry.to_year, cur_entry.to_month, cur_entry.to_day))
1193 {
1194 if (buf[0] == (uint32_t) '+') cur_entry.direction = 1;
1195 else cur_entry.direction = -1;
1196 }
1197 else
1198 {
1199 if (buf[0] == (uint32_t) '+') cur_entry.direction = -1;
1200 else cur_entry.direction = 1;
1201 }
1202 cur_entry.offset = buf[1];
1203
1204 cur_entry.name = ptr; ptr = strchr(ptr, '\0') + 1;
1205 cur_entry.format = ptr; ptr = strchr(ptr, '\0') + 1;
1206
1207 // skip wchar_t name and format
1208 ptr += 3 - (((ptr - base_ptr) + 3) & 3);
1209 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1210 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1211
1212 items.push_back(std::move(cur_entry));
1213 }
1214 return items;
1215 }
