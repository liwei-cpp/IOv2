234 inline static const prefix_tree<char, std::string> s_timezone_tree =
235 []()
236 {
237 prefix_tree<char, std::string> res;
238
239 // Registered ahead of the tzdb walk so that it is present even when the walk
240 // throws and leaves the rest of the trie empty: the token the put side writes
241 // for a nameless zone has to parse back regardless of the database.
242 res.add(s_unknown_zone.begin(), s_unknown_zone.end(), std::string{});
243
244 try
245 {
246 const auto& tzdb = std::chrono::get_tzdb();
247
248 // Collect all abbreviations (deduplicated across zones).
249 std::set<std::string> abbrevs;
250 {
251 using namespace std::chrono;
252 const sys_seconds projected =
253 time_point_cast<seconds>(system_clock::now()) + years{10};
254 const sys_seconds floor{sys_days{2038y / January / 1}};
255 const sys_seconds horizon = (projected > floor) ? projected : floor;
256
257 for (const auto& zone : tzdb.zones)
258 {
259 sys_seconds t = sys_seconds::min();
260 while (t < horizon)
261 {
262 const sys_info info = zone.get_info(t);
263 if (!info.abbrev.empty()) abbrevs.insert(info.abbrev);
264 if (info.end <= t) break;
265 t = time_point_cast<seconds>(info.end);
266 }
267 }
268 }
269
270 // Resolve each abbreviation: if locate_zone succeeds (e.g. "EST"
271 // is an IANA link), store the canonical name; otherwise mark "*".
272 for (const auto& abbr : abbrevs)
273 {
274 try
275 {
276 std::string canonical{std::chrono::locate_zone(abbr)->name()};
277 res.add(abbr.begin(), abbr.end(), canonical);
278 }
279 catch (...)
280 {
281 res.add(abbr.begin(), abbr.end(), "*" + abbr);
282 }
283 }
284
285 // Add full zone names that differ from their own abbreviation.
286 for (const auto& zone : tzdb.zones)
287 {
288 std::string full_name{zone.name()};
289 std::string abbr_name = zone.get_info(std::chrono::sys_time<std::chrono::seconds>{}).abbrev;
290 if (full_name != abbr_name)
291 res.add(full_name.begin(), full_name.end(), full_name);
292 }
293 }
294 catch (...) // NOLINT(bugprone-empty-catch)
295 {
296 // tz database unavailable or malformed at static-init time:
297 // degrade to an empty/partial tree instead of letting the
298 // exception escape this static initializer (which would call
299 // std::terminate). %Z then simply fails to match and reports a
300 // catchable stream_error at parse time, matching the defensive
301 // behaviour of time_zone_parse_helper.
302 }
303 return res;
304 }();

1083 static std::optional<std::string> normalize_time_format(const std::string& fmt)
1084 {
1085 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
1086
1087 std::string out;
1088 out.reserve(fmt.size());
1089
1090 for (std::size_t i = 0; i < fmt.size(); ++i)
1091 {
1092 if (fmt[i] != '%') { out += fmt[i]; continue; }
1093
1094 std::size_t j = i + 1;
1095 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
1096 || fmt[j] == '^' || fmt[j] == '#'))
1097 ++j;
1098 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
1099
1100 char modifier = '\0';
1101 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
1102
1103 if (j >= fmt.size()) { out.append(fmt, i, std::string::npos); break; }
1104
1105 char spec = fmt[j];
1106 if (spec == 'l') spec = 'I';
1107 else if (spec == 'k') spec = 'H';
1108 else if (spec == 'P') spec = 'p';
1109
1110 if (supported.find(spec) == std::string_view::npos) return std::nullopt;
1111
1112 out += '%';
1113 if (modifier != '\0') out += modifier;
1114 out += spec;
1115 i = j;
1116 }
1117
1118 return out;
1119 }

1217 static std::vector<era_entry> parse_glibc_era_entries()
1218 {
1219 std::vector<era_entry> items;
1220
1221 const int32_t era_item_num = static_cast<int32_t>(
1222 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
1223 if (era_item_num <= 0)
1224 return items;
1225
1226 items.reserve(era_item_num);
1227 const char *ptr = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
1228 for (int32_t cnt = 0; cnt < era_item_num; ++cnt)
1229 {
1230 const char *base_ptr = ptr;
1231 era_entry cur_entry;
1232
1233 int32_t buf[8];
1234 std::memcpy(static_cast<void*>(buf), static_cast<const void*>(ptr), sizeof(int32_t) * 8);
1235 ptr += sizeof(uint32_t) * 8;
1236
1237 if (buf[2] > std::numeric_limits<int32_t>::max() - 1900)
1238 cur_entry.from_year = std::numeric_limits<int32_t>::max();
1239 else
1240 cur_entry.from_year = buf[2] + 1900;
1241 cur_entry.from_month = buf[3] + 1;
1242 cur_entry.from_day = buf[4];
1243 if (buf[5] > std::numeric_limits<int32_t>::max() - 1900)
1244 {
1245 cur_entry.to_year = std::numeric_limits<int32_t>::max();
1246 cur_entry.to_month = 12;
1247 cur_entry.to_day = 31;
1248 }
1249 else
1250 {
1251 cur_entry.to_year = buf[5] + 1900;
1252 cur_entry.to_month = buf[6] + 1;
1253 cur_entry.to_day = buf[7];
1254 }
1255
1256 // Normalise direction to match glibc's absolute_direction (era.c ~line 98).
1257 // Forward era (from <= to): marker maps directly.
1258 // Backward era (from > to): marker is flipped so that the linear formula
1259 // era_year = offset + (calendar - from_year) * direction
1260 // stays consistent; see OUTPUT INVARIANTS above for the full rationale.
1261 if (TimeioHelper::era_small_or_equal(cur_entry.from_year, cur_entry.from_month, cur_entry.from_day,
1262 cur_entry.to_year, cur_entry.to_month, cur_entry.to_day))
1263 {
1264 if (buf[0] == (uint32_t) '+') cur_entry.direction = 1;
1265 else cur_entry.direction = -1;
1266 }
1267 else
1268 {
1269 if (buf[0] == (uint32_t) '+') cur_entry.direction = -1;
1270 else cur_entry.direction = 1;
1271 }
1272 cur_entry.offset = buf[1];
1273
1274 cur_entry.name = ptr; ptr = strchr(ptr, '\0') + 1;
1275 cur_entry.format = ptr; ptr = strchr(ptr, '\0') + 1;
1276
1277 // skip wchar_t name and format
1278 ptr += 3 - (((ptr - base_ptr) + 3) & 3);
1279 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1280 ptr = reinterpret_cast<const char*>(wcschr(reinterpret_cast<const wchar_t*>(ptr), L'\0') + 1);
1281
1282 items.push_back(std::move(cur_entry));
1283 }
1284 return items;
1285 }
