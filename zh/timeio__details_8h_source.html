337 inline static const prefix_tree<char, zone_ref> s_timezone_tree =
338 []()
339 {
340 prefix_tree<char, zone_ref> res;
341
342 // Registered ahead of the tzdb walk so that it is present even when the walk
343 // throws and leaves the rest of the trie empty: the token the put side writes
344 // for a nameless zone has to parse back regardless of the database. Its text is
345 // empty on purpose -- see s_unknown_zone for why that is not the same as absent.
346 res.add(s_unknown_zone.begin(), s_unknown_zone.end(),
347 zone_ref{.text = std::string{}, .is_name = false, .is_abbrev = true});
348
349 try
350 {
351 const auto& tzdb = std::chrono::get_tzdb();
352
353 // Classify every key before touching the trie. Two keys can name the same
354 // string from different sources -- CET, EET, EST, GMT, HST, MST, UTC and WET
355 // are each both a link name and an abbreviation -- and going through a map
356 // merges those into one entry with both flags set, instead of calling add()
357 // twice on one key and having it throw on the mismatched values.
358 std::map<std::string, zone_ref> entries;
359
360 // Every abbreviation each zone passes through.
361 {
362 using namespace std::chrono;
363 const sys_seconds projected =
364 time_point_cast<seconds>(system_clock::now()) + years{10};
365 const sys_seconds floor{sys_days{2038y / January / 1}};
366 const sys_seconds horizon = (projected > floor) ? projected : floor;
367
368 for (const auto& zone : tzdb.zones)
369 {
370 sys_seconds t = sys_seconds::min();
371 while (t < horizon)
372 {
373 const sys_info info = zone.get_info(t);
374 if (!info.abbrev.empty()) entries[info.abbrev].is_abbrev = true;
375 if (info.end <= t) break;
376 t = time_point_cast<seconds>(info.end);
377 }
378 }
379 }
380
381 // Every identifier locate_zone() accepts. Links belong here just as much as
382 // zones do: a link is a full name, merely not the canonical one, and nothing
383 // downstream needs the canonical form -- locate_zone normalizes on its own.
384 for (const auto& zone : tzdb.zones)
385 entries[std::string{zone.name()}].is_name = true;
386 for (const auto& link : tzdb.links)
387 entries[std::string{link.name()}].is_name = true;
388
389 // The unknown-zone token has to parse back to "no zone at all", so it cannot
390 // share a key with a real one. That it does not is a documented invariant of
391 // s_unknown_zone; if some future tzdata breaks it, refuse to answer rather
392 // than quietly get that one entry wrong. Thrown here -- before a single tzdb
393 // entry reaches the trie, and inside the catch below -- so the trie ends up
394 // holding only the token and every real %Z reports a catchable stream_error.
395 // Letting it escape the static initializer instead would call std::terminate
396 // in every program that includes this header, %Z user or not.
397 if (entries.contains(std::string{s_unknown_zone}))
398 throw std::runtime_error(
399 "timeio: the tz database has a zone or abbreviation named '"
400 + std::string{s_unknown_zone} + "', colliding with the unknown-zone token");
401
402 for (const auto& [key, ident] : entries)
403 res.add(key.begin(), key.end(),
404 zone_ref{.text = key, .is_name = ident.is_name, .is_abbrev = ident.is_abbrev});
405 }
406 catch (...) // NOLINT(bugprone-empty-catch)
407 {
408 // tz database unavailable, malformed, or colliding with the unknown-zone
409 // token at static-init time: degrade to a trie holding nothing but that
410 // token, instead of letting the exception escape this static initializer
411 // (which would call std::terminate). %Z then simply fails to match and
412 // reports a catchable stream_error at parse time, matching the defensive
413 // behaviour of time_zone_parse_helper.
414 }
415 return res;
416 }();
417 };

1195 static std::optional<std::string> normalize_time_format(const std::string& fmt)
1196 {
1197 constexpr std::string_view supported = "%ABCDFGHIMRSTUVWXYZabcdeghjmnprtuwxyz";
1198
1199 std::string out;
1200 out.reserve(fmt.size());
1201
1202 for (std::size_t i = 0; i < fmt.size(); ++i)
1203 {
1204 if (fmt[i] != '%') { out += fmt[i]; continue; }
1205
1206 std::size_t j = i + 1;
1207 while (j < fmt.size() && (fmt[j] == '-' || fmt[j] == '_' || fmt[j] == '0'
1208 || fmt[j] == '^' || fmt[j] == '#'))
1209 ++j;
1210 while (j < fmt.size() && fmt[j] >= '0' && fmt[j] <= '9') ++j;
1211
1212 char modifier = '\0';
1213 if (j < fmt.size() && (fmt[j] == 'E' || fmt[j] == 'O')) modifier = fmt[j++];
1214
1215 if (j >= fmt.size()) { out.append(fmt, i, std::string::npos); break; }
1216
1217 char spec = fmt[j];
1218 if (spec == 'l') spec = 'I';
1219 else if (spec == 'k') spec = 'H';
1220 else if (spec == 'P') spec = 'p';
1221
1222 if (supported.find(spec) == std::string_view::npos) return std::nullopt;
1223
1224 out += '%';
1225 if (modifier != '\0') out += modifier;
1226 out += spec;
1227 i = j;
1228 }
1229
1230 return out;
1231 }

1336 static std::vector<era_entry> parse_glibc_era_entries()
1337 {
1338 struct era_wire_header
1339 {
1340 int32_t direction_marker;
1341 int32_t offset;
1342 int32_t from_year;
1343 int32_t from_month;
1344 int32_t from_day;
1345 int32_t to_year;
1346 int32_t to_month;
1347 int32_t to_day;
1348 };
1349 static_assert(sizeof(era_wire_header) == 8 * sizeof(int32_t));
1350
1351 std::vector<era_entry> items;
1352
1353 const auto record_count = static_cast<int32_t>(
1354 reinterpret_cast<uintptr_t>(nl_langinfo(_NL_TIME_ERA_NUM_ENTRIES)));
1355 if (record_count <= 0)
1356 return items;
1357
1358 items.reserve(static_cast<std::size_t>(record_count));
1359 const char* cursor = reinterpret_cast<const char*>(nl_langinfo(_NL_TIME_ERA_ENTRIES));
1360 for (int32_t index = 0; index < record_count; ++index)
1361 {
1362 const char* const record = cursor;
1363 era_wire_header header{};
1364 std::memcpy(&header, cursor, sizeof header);
1365 cursor += sizeof header;
1366
1367 era_entry value{};
1368 constexpr int32_t epoch = 1900;
1369 constexpr int32_t last_relative_year = std::numeric_limits<int32_t>::max() - epoch;
1370 value.from_year = header.from_year > last_relative_year
1371 ? std::numeric_limits<int32_t>::max()
1372 : header.from_year + epoch;
1373 value.from_month = static_cast<uint8_t>(header.from_month + 1);
1374 value.from_day = static_cast<uint8_t>(header.from_day);
1375
1376 const bool open_ended = header.to_year > last_relative_year;
1377 value.to_year = open_ended
1378 ? std::numeric_limits<int32_t>::max()
1379 : header.to_year + epoch;
1380 value.to_month = open_ended ? uint8_t{12}
1381 : static_cast<uint8_t>(header.to_month + 1);
1382 value.to_day = open_ended ? uint8_t{31}
1383 : static_cast<uint8_t>(header.to_day);
1384 value.offset = header.offset;
1385
1386 const int8_t marked_direction = header.direction_marker == static_cast<int32_t>('+')
1387 ? int8_t{1} : int8_t{-1};
1388 const bool runs_forward = TimeioHelper::era_small_or_equal(
1389 value.from_year, value.from_month, value.from_day,
1390 value.to_year, value.to_month, value.to_day);
1391 value.direction = runs_forward
1392 ? marked_direction : static_cast<int8_t>(-marked_direction);
1393
1394 value.name = cursor;
1395 cursor += std::strlen(cursor) + 1;
1396 value.format = cursor;
1397 cursor += std::strlen(cursor) + 1;
1398
1399 constexpr std::size_t wire_alignment = sizeof(int32_t);
1400 const auto narrow_size = static_cast<std::size_t>(cursor - record);
1401 cursor += (wire_alignment - (narrow_size % wire_alignment)) % wire_alignment;
1402
1403 const auto* wide_name = reinterpret_cast<const wchar_t*>(cursor);
1404 cursor += (std::wcslen(wide_name) + 1) * sizeof(wchar_t);
1405 const auto* wide_format = reinterpret_cast<const wchar_t*>(cursor);
1406 cursor += (std::wcslen(wide_format) + 1) * sizeof(wchar_t);
1407
1408 items.push_back(std::move(value));
1409 }
1410 return items;
1411 }
